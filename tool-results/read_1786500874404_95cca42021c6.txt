    57→    57→/** Remap a color name through the color-blind palette if applicable */
    58→    58→function getColorBlindSafeColor(colorName: string, mode: string): string {
    59→    59→  if (mode === 'none' || !COLOR_BLIND_PALETTES[mode]) return colorName;
    60→    60→  return COLOR_BLIND_PALETTES[mode][colorName.toLowerCase()] || colorName;
    61→    61→}

   100→   100→// ---- Perfect-Freehand → Fabric.js Path ----
   101→   101→
   102→   102→function createFreehandPath(
   103→   103→  points: FreehandPoint[],
   104→   104→  strokeColor: string,
   105→   105→  strokeWidth: number
   106→   106→): Path | null {
   107→   107→  if (points.length < 2) return null;
   108→   108→
   109→   109→  const outline = getStroke(points, {
   110→   110→    size: strokeWidth * 3,
   111→   111→    thinning: 0.5,
   112→   112→    smoothing: 0.6,
   113→   113→    streamline: 0.5,
   114→   114→    simulatePressure: true,
   115→   115→  });
   116→   116→
   117→   117→  if (outline.length < 2) return null;
   118→   118→
   119→   119→  // Build SVG path string from the outline points
   120→   120→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
   121→   121→  for (let i = 1; i < outline.length; i++) {
   122→   122→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
   123→   123→  }
   124→   124→  d += ' Z';
   125→   125→
   126→   126→  try {
   127→   127→    const path = new Path(d, {
   128→   128→      fill: strokeColor,
   129→   129→      stroke: strokeColor,
   130→   130→      strokeWidth: 0.5,
   131→   131→      strokeLineCap: 'round',
   132→   132→      strokeLineJoin: 'round',
   133→   133→    } as any);
   134→   134→    return path;
   135→   135→  } catch {
   136→   136→    return null;
   137→   137→  }
   138→   138→}

   145→   145→}

   398→   398→

   405→   405→    // ============================================================
   406→   406→    // Sprint 1: Viewport Broadcast (Tutor Side — Focus Mode)
   407→   407→    // Broadcasts viewport (x, y, zoom) via awareness when focus mode is on.
   408→   408→    // ============================================================

   410→   410→      const fc = fcanvasRef.current;

   417→   417→        const vpt = fc.viewportTransform;
   418→   418→        if (!vpt) return;

   421→   421→        const y = Math.round(vpt[5] || 0);
   422→   422→        const key = `${x},${y},${zoom}`;
   423→   423→
   424→   424→        if (key !== lastKey) {
   425→   425→          lastKey = key;
   426→   426→          if (debounceTimer) clearTimeout(debounceTimer);
   427→   427→          debounceTimer = setTimeout(() => {
   428→   428→            awareness.setLocalStateField('viewport', { x, y, zoom });
   429→   429→          }, 60);
   430→   430→        }
   431→   431→      };

   435→   435→        if (isPanning) broadcastCurrentViewport();
   436→   436→      };
   437→   437→
   438→   438→      const handleZoom = () => {
   439→   439→        broadcastCurrentViewport();
   440→   440→      };
   441→   441→
   442→   442→      // Sprint 1: Also broadcast on mouse:up (when pan/zoom gesture ends)
   443→   443→      const handleMouseUp = () => {
   444→   444→        broadcastCurrentViewport();
   445→   445→      };
   446→   446→
   447→   447→      fc.on('after:render', handleAfterRender);
   448→   448→      fc.on('mouse:wheel', handleZoom);
   449→   449→      fc.on('mouse:up', handleMouseUp);
   450→   450→
   451→   451→      // Initial broadcast
   452→   452→      broadcastCurrentViewport();
   453→   453→
   454→   454→      return () => {
   455→   455→        fc.off('after:render', handleAfterRender);
   456→   456→        fc.off('mouse:wheel', handleZoom);
   457→   457→        fc.off('mouse:up', handleMouseUp);
   458→   458→        if (debounceTimer) clearTimeout(debounceTimer);
   459→   459→        // Clear viewport when focus mode changes or component unmounts
   460→   460→        awareness.setLocalStateField('viewport', null);
   461→   461→      };
   462→   462→    }, [awareness, isTutor, focusMode, isReady, isPanning]);
   463→   463→
   464→   464→    // ============================================================
   465→   465→    // Sprint 1: Viewport Receive (Student Side — Focus Mode)
   466→   466→    // Applies viewport broadcasted by tutor.
   467→   467→    // ============================================================
   468→   468→    useEffect(() => {
   469→   469→      const fc = fcanvasRef.current;
   470→   470→      if (!fc || !appliedViewport || isTutor) return;
   471→   471→
   472→   472→      const vpt = [...(fc.viewportTransform || [1, 0, 0, 1, 0, 0])];
   473→   473→      vpt[0] = appliedViewport.zoom;
   474→   474→      vpt[3] = appliedViewport.zoom;
   475→   475→      vpt[4] = appliedViewport.x;
   476→   476→      vpt[5] = appliedViewport.y;
   477→   477→      fc.setViewportTransform(vpt as any);
   478→   478→      requestRenderThrottled();
   479→   479→    }, [appliedViewport, isTutor, isReady]); // eslint-disable-line react-hooks/exhaustive-deps
   480→   480→
   481→   481→    // ============================================================
   482→   482→    // Canvas Interaction — Pan, Zoom, Draw, Erase, Text, Shapes
   483→   483→    // ============================================================
   484→   484→    useEffect(() => {
   485→   485→      const fc = fcanvasRef.current;
   486→   486→      if (!fc || readOnly) return;
   487→   487→
   488→   488→      // ---- Zoom (mouse wheel) ----
   489→   489→      const handleWheel = (opt: any) => {
   490→   490→        const e = opt.e as WheelEvent;
   491→   491→        e.preventDefault();
   492→   492→        e.stopPropagation();
   493→   493→
   494→   494→        const delta = e.deltaY;
   495→   495→        let zoom = fc.getZoom();
   496→   496→        zoom *= 0.999 ** delta;
   497→   497→        zoom = Math.min(Math.max(0.1, zoom), 10);
   498→   498→        fc.zoomToPoint(new FabricPoint(e.offsetX, e.offsetY), zoom);
   499→   499→        requestRenderThrottled();
   500→   500→      };
   501→   501→      fc.on('mouse:wheel', handleWheel);
   502→   502→
   503→   503→      // ---- Mouse interactions ----
   504→   504→      let panningActive = false;
   505→   505→      let lastPosX = 0;
   506→   506→      let lastPosY = 0;
   507→   507→      let beforeState = '';
   508→   508→
   509→   509→      // ---- Mouse Down ----
   510→   510→      const handleMouseDown = (opt: any) => {
   511→   511→        const e = opt.e as MouseEvent;
   512→   512→
   513→   513→        // Hand tool: left click pans
   514→   514→        if (tools.tool === 'hand' && e.button === 0) {
   515→   515→          panningActive = true;
   516→   516→          setIsPanning(true);
   517→   517→          lastPosX = e.clientX;
   518→   518→          lastPosY = e.clientY;
   519→   519→          (fc as any).defaultCursor = 'grabbing';
   520→   520→          return;
   521→   521→        }
   522→   522→
   523→   523→        // Middle click pans
   524→   524→        if (e.button === 1) {
   525→   525→          panningActive = true;
   526→   526→          setIsPanning(true);
   527→   527→          lastPosX = e.clientX;
   528→   528→          lastPosY = e.clientY;
   529→   529→          return;
   530→   530→        }
   531→   531→
   532→   532→        // Eraser: click to delete
   533→   533→        if (tools.tool === 'eraser' && e.button === 0) {
   534→   534→          const target = opt.target as FabricObject | undefined;
   535→   535→          if (target) {
   536→   536→            beforeState = fc.toJSON().toString();
   537→   537→            fc.remove(target);
   538→   538→            fc.renderAll();
   539→   539→            history.pushState('erase', beforeState, fc.toJSON().toString());
   540→   540→          }
   541→   541→          return;
   542→   542→        }
   543→   543→
   544→   544→        // Text: click empty space to place IText
   545→   545→        if (tools.tool === 'text' && e.button === 0 && !opt.target) {
   546→   546→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   547→   547→          beforeState = fc.toJSON().toString();
   548→   548→
   549→   549→          const text = new IText('Type here', {
   550→   550→            left: pointer.x,
   551→   551→            top: pointer.y,
   552→   552→            fontSize: tools.fontSize,
   553→   553→            fill: tools.strokeColor,
   554→   554→            fontFamily: 'Inter, system-ui, sans-serif',
   555→   555→            selectable: true,
   556→   556→            evented: true,
   557→   557→          } as any);
   558→   558→          (text as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   559→   559→
   560→   560→          fc.add(text);
   561→   561→          fc.setActiveObject(text);
   562→   562→          text.enterEditing();
   563→   563→          text.selectAll();
   564→   564→          fc.renderAll();
   565→   565→
   566→   566→          history.pushState('add-text', beforeState, fc.toJSON().toString());
   567→   567→          return;
   568→   568→        }
   569→   569→
   570→   570→        // Freehand (draw tool): start collecting points for perfect-freehand
   571→   571→        if (tools.tool === 'draw' && e.button === 0 && !opt.target) {
   572→   572→          // We use Fabric's built-in freeDrawingMode but with perfect-freehand
   573→   573→          // post-processing. The PencilBrush collects points; on path:created
   574→   574→          // we replace it with a smooth perfect-freehand path.
   575→   575→          // But for real-time preview, we track raw points ourselves.
   576→   576→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   577→   577→          freehandPointsRef.current = [{ x: pointer.x, y: pointer.y, pressure: 0.5 }];
   578→   578→          isDrawingRef.current = true;
   579→   579→          beforeState = fc.toJSON().toString();
   580→   580→          return;
   581→   581→        }
   582→   582→
   583→   583→        // Shape tools: start drawing
   584→   584→        if (['rectangle', 'ellipse', 'line', 'arrow'].includes(tools.tool) && e.button === 0 && !opt.target) {
   585→   585→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;

   587→   587→          startPointRef.current = pointer;
   588→   588→          beforeState = fc.toJSON().toString();
   589→   589→
   590→   590→          const objName = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   591→   591→          let shape: FabricObject;
   592→   592→
   593→   593→          switch (tools.tool) {
   594→   594→            case 'rectangle':
   595→   595→              shape = new Rect({
   596→   596→                left: pointer.x,
   597→   597→                top: pointer.y,
   598→   598→                width: 1,
   599→   599→                height: 1,
   600→   600→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   601→   601→                stroke: tools.strokeColor,
   602→   602→                strokeWidth: tools.strokeWidth,
   603→   603→                selectable: false,
   604→   604→                evented: false,
   605→   605→              } as any);
   606→   606→              break;
   607→   607→
   608→   608→            case 'ellipse':

   610→   610→                left: pointer.x,
   611→   611→                top: pointer.y,
   612→   612→                rx: 1,
   613→   613→                ry: 1,
   614→   614→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   615→   615→                stroke: tools.strokeColor,
   616→   616→                strokeWidth: tools.strokeWidth,
   617→   617→                selectable: false,
   618→   618→                evented: false,

   620→   620→              break;
   621→   621→
   622→   622→            case 'line':
   623→   623→              shape = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   624→   624→                stroke: tools.strokeColor,
   625→   625→                strokeWidth: tools.strokeWidth,
   626→   626→                selectable: false,
   627→   627→                evented: false,
   628→   628→              } as any);
   629→   629→              break;
   630→   630→
   631→   631→            case 'arrow': {
   632→   632→              const line = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   633→   633→                stroke: tools.strokeColor,
   634→   634→                strokeWidth: tools.strokeWidth,
   635→   635→                selectable: false,
   636→   636→                evented: false,
   637→   637→              } as any);
   638→   638→              const headSize = Math.max(10, tools.strokeWidth * 4);

   640→   640→                left: pointer.x,
   641→   641→                top: pointer.y,
   642→   642→                width: headSize,
   643→   643→                height: headSize,
   644→   644→                fill: tools.strokeColor,
   645→   645→                angle: 90,
   646→   646→                originX: 'center',
   647→   647→                originY: 'center',
   648→   648→                selectable: false,
   649→   649→                evented: false,
   650→   650→              } as any);
   651→   651→              shape = new Group([line, arrowhead], {
   652→   652→                selectable: false,
   653→   653→                evented: false,
   654→   654→              } as any);
   655→   655→              break;
   656→   656→            }
   657→   657→
   658→   658→            default:
   659→   659→              return;
   660→   660→          }
   661→   661→
   662→   662→          (shape as any).name = objName;
   663→   663→          fc.add(shape);
   664→   664→          tempShapeRef.current = shape;
   665→   665→          requestRenderThrottled();
   666→   666→          return;
   667→   667→        }
   668→   668→      };
   669→   669→
   670→   670→      // ---- Mouse Move ----
   671→   671→      const handleMouseMove = (opt: any) => {
   672→   672→        const e = opt.e as MouseEvent;
   673→   673→
   674→   674→        // Panning
   675→   675→        if (panningActive) {
   676→   676→          const vpt = fc.viewportTransform!;
   677→   677→          vpt[4] += e.clientX - lastPosX;
   678→   678→          vpt[5] += e.clientY - lastPosY;
   679→   679→          lastPosX = e.clientX;
   680→   680→          lastPosY = e.clientY;
   681→   681→          requestRenderThrottled();
   682→   682→          return;
   683→   683→        }
   684→   684→
   685→   685→        // Freehand drawing — collect points
   686→   686→        if (isDrawingRef.current && tools.tool === 'draw') {
   687→   687→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   688→   688→          freehandPointsRef.current.push({
   689→   689→            x: pointer.x,
   690→   690→            y: pointer.y,
   691→   691→            pressure: 0.5,
   692→   692→          });
   693→   693→
   694→   694→          // Real-time preview with perfect-freehand
   695→   695→          if (freehandTempPathRef.current) {
   696→   696→            fc.remove(freehandTempPathRef.current);
   697→   697→          }
   698→   698→
   699→   699→          const previewPath = createFreehandPath(
   700→   700→            freehandPointsRef.current,
   701→   701→            tools.strokeColor,
   702→   702→            tools.strokeWidth
   703→   703→          );
   704→   704→          if (previewPath) {
   705→   705→            (previewPath as any).name = '__freehand_temp__';
   706→   706→            freehandTempPathRef.current = previewPath;
   707→   707→            fc.add(previewPath);
   708→   708→          }
   709→   709→
   710→   710→          requestRenderThrottled();
   711→   711→          return;
   712→   712→        }
   713→   713→
   714→   714→        // Shape drawing
   715→   715→        if (isDrawingRef.current && tempShapeRef.current && startPointRef.current) {
   716→   716→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   717→   717→          const start = startPointRef.current;
   718→   718→          const t = tempShapeRef.current;
   719→   719→
   720→   720→          if (t instanceof Rect) {
   721→   721→            const left = Math.min(start.x, pointer.x);
   722→   722→            const top = Math.min(start.y, pointer.y);
   723→   723→            t.set({
   724→   724→              left,
   725→   725→              top,
   726→   726→              width: Math.abs(pointer.x - start.x),
   727→   727→              height: Math.abs(pointer.y - start.y),
   728→   728→            });
   729→   729→          } else if (t instanceof Ellipse) {
   730→   730→            const left = Math.min(start.x, pointer.x);
   731→   731→            const top = Math.min(start.y, pointer.y);
   732→   732→            const rx = Math.max(1, Math.abs(pointer.x - start.x) / 2);
   733→   733→            const ry = Math.max(1, Math.abs(pointer.y - start.y) / 2);
   734→   734→            t.set({ left: left + rx, top: top + ry, rx, ry });
   735→   735→          } else if (t instanceof Line) {
   736→   736→            t.set({ x2: pointer.x, y2: pointer.y });
   737→   737→          } else if (t instanceof Group) {
   738→   738→            const objs = t.getObjects();
   739→   739→            const line = objs[0] as Line;
   740→   740→            const head = objs[1] as Triangle;
   741→   741→            if (line && head) {
   742→   742→              line.set({ x2: pointer.x - start.x, y2: pointer.y - start.y });
   743→   743→              head.set({
   744→   744→                left: pointer.x - start.x,
   745→   745→                top: pointer.y - start.y,
   746→   746→                angle:
   747→   747→                  Math.atan2(pointer.y - start.y, pointer.x - start.x) *
   748→   748→                    (180 / Math.PI) +
   749→   749→                  90,
   750→   750→              });
   751→   751→            }
   752→   752→            t.setCoords();
   753→   753→          }
   754→   754→
   755→   755→          requestRenderThrottled();
   756→   756→        }
   757→   757→      };
   758→   758→
   759→   759→      // ---- Mouse Up ----
   760→   760→      const handleMouseUp = () => {
   761→   761→        // Stop panning
   762→   762→        if (panningActive) {
   763→   763→          panningActive = false;
   764→   764→          setIsPanning(false);
   765→   765→          (fc as any).defaultCursor =
   766→   766→            tools.tool === 'hand' ? 'grab' : 'default';
   767→   767→          return;
   768→   768→        }
   769→   769→
   770→   770→        // Finalize freehand
   771→   771→        if (isDrawingRef.current && tools.tool === 'draw') {
   772→   772→          isDrawingRef.current = false;
   773→   773→
   774→   774→          // Remove preview path
   775→   775→          if (freehandTempPathRef.current) {
   776→   776→            fc.remove(freehandTempPathRef.current);
   777→   777→            freehandTempPathRef.current = null;
   778→   778→          }
   779→   779→
   780→   780→          // Create final perfect-freehand path
   781→   781→          const points = freehandPointsRef.current;
   782→   782→          if (points.length >= 2) {
   783→   783→            const finalPath = createFreehandPath(
   784→   784→              points,
   785→   785→              tools.strokeColor,
   786→   786→              tools.strokeWidth
   787→   787→            );
   788→   788→            if (finalPath) {
   789→   789→              (finalPath as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   790→   790→              fc.add(finalPath);
   791→   791→            }
   792→   792→          }
   793→   793→
   794→   794→          freehandPointsRef.current = [];
   795→   795→          history.pushState('draw', beforeState, fc.toJSON().toString());
   796→   796→          fc.renderAll();
   797→   797→          return;
   798→   798→        }
   799→   799→
   800→   800→        // Finalize shape
   801→   801→        if (isDrawingRef.current && tempShapeRef.current) {
   802→   802→          tempShapeRef.current.set({
   803→   803→            selectable: true,
   804→   804→            evented: true,
   805→   805→          });
   806→   806→
   807→   807→          history.pushState(
   808→   808→            `draw-${tools.tool}`,
   809→   809→            beforeState,
   810→   810→            fc.toJSON().toString()
   811→   811→          );
   812→   812→          tempShapeRef.current = null;
   813→   813→          isDrawingRef.current = false;
   814→   814→          startPointRef.current = null;
   815→   815→          fc.renderAll();
   816→   816→        }
   817→   817→      };
   818→   818→
   819→   819→      fc.on('mouse:down', handleMouseDown);
   820→   820→      fc.on('mouse:move', handleMouseMove);
   821→   821→      fc.on('mouse:up', handleMouseUp);
   822→   822→
   823→   823→      // Track undo state: capture before-state when user starts modifying
   824→   824→      const undoBeforeRef = useRef('');
   825→   825→      const captureBefore = () => {
   826→   826→        undoBeforeRef.current = fc.toJSON().toString();
   827→   827→      };
   828→   828→      fc.on('object:moving', captureBefore);
   829→   829→      fc.on('object:scaling', captureBefore);
   830→   830→      fc.on('object:rotating', captureBefore);
   831→   831→      fc.on('object:modified', () => {
   832→   832→        if (undoBeforeRef.current) {
   833→   833→          history.pushState(
   834→   834→            'modify',
   835→   835→            undoBeforeRef.current,
   836→   836→            fc.toJSON().toString()
   837→   837→          );
   838→   838→          undoBeforeRef.current = '';
   839→   839→        }
   840→   840→      });
   841→   841→
   842→   842→      // Filter out temp freehand paths from sync
   843→   843→      const origBeforeAdd = fc.on;
   844→   844→      fc.on('object:added', ((e: any) => {
   845→   845→        const obj = e.target as FabricObject;
   846→   846→        const name = (obj as any).name;
   847→   847→        if (name === '__freehand_temp__') {
   848→   848→          // Don't sync temp paths
   849→   849→          return;
   850→   850→        }
   851→   851→      }) as any);
   852→   852→
   853→   853→      return () => {
   854→   854→        fc.off('mouse:wheel', handleWheel);
   855→   855→        fc.off('mouse:down', handleMouseDown);
   856→   856→        fc.off('mouse:move', handleMouseMove);
   857→   857→        fc.off('mouse:up', handleMouseUp);
   858→   858→      };
   859→   859→    }, [isReady, tools.tool, tools.strokeColor, tools.strokeWidth, tools.fillColor, tools.fontSize, readOnly, history, requestRenderThrottled]);
   860→   860→
   861→   861→    // ============================================================
   862→   862→    // Touch: Pinch-to-zoom via native DOM events on container
   863→   863→    // ============================================================
   864→   864→    useEffect(() => {
   865→   865→      const container = containerRef.current;
   866→   866→      const fc = fcanvasRef.current;
   867→   867→      if (!container || !fc) return;
   868→   868→
   869→   869→      const domTouchStart = (e: TouchEvent) => {
   870→   870→        const touches = e.touches;
   871→   871→        if (touches.length === 2) {
   872→   872→          e.preventDefault();
   873→   873→          e.stopPropagation();
   874→   874→
   875→   875→          const dx = touches[0].clientX - touches[1].clientX;
   876→   876→          const dy = touches[0].clientY - touches[1].clientY;
   877→   877→          const dist = Math.sqrt(dx * dx + dy * dy);
   878→   878→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   879→   879→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   880→   880→
   881→   881→          touchStateRef.current = {
   882→   882→            isPinching: true,
   883→   883→            lastDist: dist,
   884→   884→            lastCenter: { x: centerX, y: centerY },
   885→   885→            touchCount: 2,
   886→   886→          };
   887→   887→
   888→   888→          // Cancel any active drawing
   889→   889→          isDrawingRef.current = false;
   890→   890→          tempShapeRef.current = null;
   891→   891→          freehandPointsRef.current = [];
   892→   892→          if (freehandTempPathRef.current) {
   893→   893→            fc.remove(freehandTempPathRef.current);
   894→   894→            freehandTempPathRef.current = null;
   895→   895→          }
   896→   896→        }
   897→   897→      };
   898→   898→
   899→   899→      const domTouchMove = (e: TouchEvent) => {
   900→   900→        const touches = e.touches;
   901→   901→        if (touches.length === 2 && touchStateRef.current.isPinching) {
   902→   902→          e.preventDefault();
   903→   903→          e.stopPropagation();
   904→   904→
   905→   905→          const dx = touches[0].clientX - touches[1].clientX;
   906→   906→          const dy = touches[0].clientY - touches[1].clientY;
   907→   907→          const dist = Math.sqrt(dx * dx + dy * dy);
   908→   908→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   909→   909→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   910→   910→
   911→   911→          const prev = touchStateRef.current;
   912→   912→
   913→   913→          // Zoom
   914→   914→          const scale = dist / Math.max(prev.lastDist, 1);
   915→   915→          let zoom = fc.getZoom() * scale;
   916→   916→          zoom = Math.min(Math.max(0.1, zoom), 10);
   917→   917→
   918→   918→          // Pan
   919→   919→          const vpt = fc.viewportTransform!;
   920→   920→          vpt[4] += centerX - prev.lastCenter.x;
   921→   921→          vpt[5] += centerY - prev.lastCenter.y;
   922→   922→
   923→   923→          // Apply zoom to center of pinch
   924→   924→          fc.zoomToPoint(new FabricPoint(centerX, centerY), zoom);
   925→   925→
   926→   926→          touchStateRef.current = {
   927→   927→            isPinching: true,
   928→   928→            lastDist: dist,
   929→   929→            lastCenter: { x: centerX, y: centerY },
   930→   930→            touchCount: 2,
   931→   931→          };
   932→   932→
   933→   933→          requestRenderThrottled();
   934→   934→        }
   935→   935→      };
   936→   936→
   937→   937→      const domTouchEnd = (e: TouchEvent) => {
   938→   938→        if (e.touches.length < 2) {
   939→   939→          touchStateRef.current.isPinching = false;
   940→   940→          touchStateRef.current.touchCount = e.touches.length;
   941→   941→        }
   942→   942→      };
   943→   943→
   944→   944→      container.addEventListener('touchstart', domTouchStart, { passive: false });
   945→   945→      container.addEventListener('touchmove', domTouchMove, { passive: false });
   946→   946→      container.addEventListener('touchend', domTouchEnd, { passive: false });
   947→   947→
   948→   948→      return () => {
   949→   949→        container.removeEventListener('touchstart', domTouchStart);
   950→   950→        container.removeEventListener('touchmove', domTouchMove);
   951→   951→        container.removeEventListener('touchend', domTouchEnd);
   952→   952→      };
   953→   953→    }, [isReady, requestRenderThrottled]);

   955→   955→    // ============================================================
   956→   956→    // Keyboard Shortcuts (Phase 4: extended)
   957→   957→    // ============================================================
   958→   958→    useKeyboardShortcuts({
   959→   959→      onUndo: history.undo,
   960→   960→      onRedo: history.redo,
   961→   961→      onDelete: () => {
   962→   962→        const fc = fcanvasRef.current;
   963→   963→        if (!fc || readOnly) return;
   964→   964→        const active = fc.getActiveObject();
   965→   965→        if (active) {
   966→   966→          const before = fc.toJSON().toString();
   967→   967→          fc.remove(active);
   968→   968→          fc.renderAll();
   969→   969→          history.pushState('delete', before, fc.toJSON().toString());
   970→   970→          fc.discardActiveObject();
   971→   971→        }
   972→   972→      },
   973→   973→      onEscape: () => {
   974→   974→        const fc = fcanvasRef.current;
   975→   975→        if (fc) {
   976→   976→          fc.discardActiveObject();
   977→   977→          fc.requestRenderAll();
   978→   978→        }
   979→   979→      },
   980→   980→      onToolChange: (tool: CanvasTool) => {
   981→   981→        tools.setTool(tool);
   982→   982→      },
   983→   983→      onZoomToFit: () => {
   984→   984→        const fc = fcanvasRef.current;
