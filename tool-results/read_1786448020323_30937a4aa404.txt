     1→// ============================================================
     2→// TldrawCanvas — Tldraw Editor with Yjs Real-Time Sync
     3→// ============================================================
     4→// Wraps the Tldraw editor and synchronizes state with Yjs
     5→// via the Hocuspocus CRDT provider for multi-user collaboration.
     6→//
     7→// Sync strategy: Incremental per-record CRDT sync.
     8→// - Each record (shape, camera, page-state, etc.) is stored as a
     9→//   separate entry in a per-page Y.Map, keyed by record ID.
    10→// - Local user edits are debounced (200ms) and batch-written to Yjs.
    11→// - Remote Yjs changes are merged record-by-record into the editor
    12→//   using store.mergeRemoteChanges(), which marks them source='remote'
    13→//   so they don't trigger the local-sync listener (no echo loop).
    14→// - A dirty-set tracks record IDs with unsynced local changes;
    15→//   remote updates for dirty records are skipped (last-local-writer-wins).
    16→// - A monotonic localChangeVersion counter is maintained per record
    17→//   for diagnostics and future conflict-resolution extensions.
    18→// - Backward compatibility: legacy full-snapshot format is detected
    19→//   on first load and migrated to the per-record format.
    20→// ============================================================
    21→
    22→'use client';
    23→
    24→import React, { useCallback, useEffect, useRef, useState } from 'react';
    25→import { Tldraw, Editor, TLStoreSnapshot } from 'tldraw';
    26→import type { Map as YMap } from 'yjs';
    27→import 'tldraw/tldraw.css';
    28→
    29→// ---- License Gate Mitigation ----
    30→// tldraw v5 requires a production license; without one the editor unmounts
    31→// after 5 seconds. This MutationObserver detects the license gate div and
    32→// re-mounts the editor by forcing a React re-render cycle.
    33→// TODO: Remove this when a proper tldraw license is obtained.
    34→
    35→// ---- Types ----
    36→
    37→/** Shape of the RecordsDiff changes as delivered by store.listen in tldraw v5. */
    38→interface RecordsDiffLike {
    39→        added: Record<string, Record<string, unknown>>;
    40→        updated: Record<string, [from: Record<string, unknown>, to: Record<string, unknown>]>;
    41→        removed: Record<string, Record<string, unknown>>;
    42→}
    43→
    44→/** Shape of the history entry passed to store.listen callbacks in tldraw v5. */
    45→interface HistoryEntryLike {
    46→        changes: RecordsDiffLike;
    47→        source: 'user' | 'remote';
    48→}
    49→
    50→export interface TldrawCanvasProps {
    51→        /** Yjs document (from useYjsProvider) */
    52→        ydoc: import('yjs').Doc | null;
    53→        /** Called when the editor mounts — provides the Editor ref */
    54→        onEditorReady?: (editor: Editor) => void;
    55→        /** Called when editor store changes (for external sync triggers) */
    56→        onStoreChange?: () => void;
    57→        /** Current page index (for multi-page support) */
    58→        pageIndex: number;
    59→        /** Whether the user is a tutor (controls UI permissions) */
    60→        isTutor: boolean;
    61→        /** Whether the canvas is read-only (e.g., student in focus mode) */
    62→        readOnly?: boolean;
    63→}
    64→
    65→// ---- Constants ----
    66→
    67→/** Debounce Yjs writes to batch rapid edits into one transaction. */
    68→const SYNC_DEBOUNCE_MS = 200;
    69→
    70→/** Key of the legacy Y.Map that stored full page snapshots as JSON blobs. */
    71→const YJS_LEGACY_MAP_KEY = 'page-snapshot';
    72→
    73→/**
    74→ * Prefix for per-page Y.Maps that store individual records.
    75→ * E.g. pageIndex=3 → ydoc.getMap('page-shapes-3')
    76→ */
    77→const YJS_SHAPES_MAP_PREFIX = 'page-shapes-';
    78→
    79→/**
    80→ * Transaction origin symbol used to tag our own local writes.
    81→ * The Yjs observer checks this to skip re-applying our own changes.
    82→ */
    83→const LOCAL_SYNC_ORIGIN = Symbol('tldraw-local-sync');
    84→
    85→// ============================================================
    86→// Component
    87→// ============================================================
    88→
    89→export default function TldrawCanvas({
    90→        ydoc,
    91→        onEditorReady,
    92→        onStoreChange,
    93→        pageIndex,
    94→        isTutor,
    95→        readOnly = false,
    96→}: TldrawCanvasProps) {
    97→        // ---- Core state & refs ----
    98→        const editorRef = useRef<Editor | null>(null);
    99→        const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   100→        const [isReady, setIsReady] = useState(false);
   101→        const [isLoadingSnapshot, setIsLoadingSnapshot] = useState(true);
   102→        // License gate workaround — toggling this key forces React to re-mount Tldraw
   103→        const [tldrawKey, setTldrawKey] = useState(0);
   104→
   105→        /**
   106→         * Mirror of isReady as a ref so that closures inside setTimeout/
   107→         * store listeners always see the current value without being
   108→         * captured at creation time.
   109→         */
   110→        const isReadyRef = useRef(false);
   111→
   112→        // ---- Incremental sync refs (stable across renders) ----
   113→
   114→        /** Records that changed locally and are awaiting debounce flush to Yjs. */
   115→        const pendingWritesRef = useRef<Map<string, Record<string, unknown>>>(new Map());
   116→
   117→        /** Record IDs that were deleted locally and are awaiting debounce flush. */
   118→        const pendingDeletesRef = useRef<Set<string>>(new Set());
   119→
   120→        /**
   121→         * Record IDs with unsynced local changes.
   122→         * Remote updates for these IDs are skipped to prevent overwriting
   123→         * concurrent local edits (last-local-writer-wins).  IDs are removed
   124→         * after the corresponding write/delete is flushed to Yjs.
   125→         */
   126→        const localDirtyRef = useRef<Set<string>>(new Set());
   127→
   128→        /** Monotonically increasing version counter for local edits. */
   129→        const localChangeVersionRef = useRef(0);
   130→
   131→        /** Per-record version: the localChangeVersion at which it was last edited locally. */
   132→        const recordVersionsRef = useRef<Map<string, number>>(new Map());
   133→
   134→        /** True while the initial snapshot is being loaded (suppresses sync-to-Yjs). */
   135→        const isInitialLoadRef = useRef(false);
   136→
   137→        /** Set of page indices that have already been migrated from legacy format. */
   138→        const migratedPagesRef = useRef<Set<number>>(new Set());
   139→
   140→        /** Cleanup function for the tldraw store listener. */
   141→        const unlistenStoreRef = useRef<(() => void) | null>(null);
   142→
   143→        /** Cleanup function for the Yjs shapes-map observer. */
   144→        const unobserveYjsRef = useRef<(() => void) | null>(null);
   145→
   146→        /**
   147→         * Ref to the Y.Map currently being used for this page's shapes.
   148→         * Updated explicitly on page switch so we can flush the OLD map
   149→         * before swapping to the new one.
   150→         */
   151→        const shapesMapRef = useRef<YMap<string> | null>(null);
   152→
   153→        /** Track the previous page index to detect page switches. */
   154→        const prevPageIndexRef = useRef(pageIndex);
   155→
   156→        /** Keep callback refs up-to-date without re-registering listeners. */
   157→        const onStoreChangeRef = useRef(onStoreChange);
   158→        onStoreChangeRef.current = onStoreChange;
   159→
   160→        // ----------------------------------------------------------------
   161→        // Core: process a RecordsDiff into pending writes/deletes & schedule sync
   162→        // ----------------------------------------------------------------
   163→
   164→        const processLocalDiff = useCallback((entry: HistoryEntryLike) => {
   165→                if (isInitialLoadRef.current) return;
   166→                if (!isReadyRef.current) return;
   167→
   168→                const { changes } = entry;
   169→
   170→                // Increment the global local version counter.
   171→                localChangeVersionRef.current++;
   172→                const version = localChangeVersionRef.current;
   173→
   174→                // Collect added records.
   175→                for (const [id, record] of Object.entries(changes.added)) {
   176→                        pendingWritesRef.current.set(id, record);
   177→                        pendingDeletesRef.current.delete(id);
   178→                        localDirtyRef.current.add(id);
   179→                        recordVersionsRef.current.set(id, version);
   180→                }
   181→
   182→                // Collect updated records (use the "to" value — index 1).
   183→                for (const [id, pair] of Object.entries(changes.updated)) {
   184→                        pendingWritesRef.current.set(id, pair[1]);
   185→                        pendingDeletesRef.current.delete(id);
   186→                        localDirtyRef.current.add(id);
   187→                        recordVersionsRef.current.set(id, version);
   188→                }
   189→
   190→                // Collect removed records.
   191→                for (const id of Object.keys(changes.removed)) {
   192→                        pendingDeletesRef.current.add(id);
   193→                        pendingWritesRef.current.delete(id);
   194→                        localDirtyRef.current.add(id);
   195→                        recordVersionsRef.current.set(id, version);
   196→                }
   197→
   198→                scheduleSyncRef.current();
   199→        }, []);
   200→
   201→        // ---- Ref-stable scheduleSync (never recreated) ----
   202→        const scheduleSyncRef = useRef<() => void>(() => {});
   203→
   204→        // ---- Ref-stable flushPendingToYjs (never recreated) ----
   205→        const flushPendingRef = useRef<() => void>(() => {});
   206→
   207→        // ----------------------------------------------------------------
   208→        // Flush pending changes to the current shapes Y.Map
   209→        // ----------------------------------------------------------------
   210→
   211→        const buildFlush = useCallback(() => {
   212→                return () => {
   213→                        const map = shapesMapRef.current;
   214→                        if (!map || !ydoc) return;
   215→
   216→                        // Nothing to flush?
   217→                        if (pendingWritesRef.current.size === 0 && pendingDeletesRef.current.size === 0) return;
   218→
   219→                        // Snapshot the pending state so new changes accumulate independently.
   220→                        const writes = new Map(pendingWritesRef.current);
   221→                        const deletes = new Set(pendingDeletesRef.current);
   222→                        pendingWritesRef.current = new Map();
   223→                        pendingDeletesRef.current = new Set();
   224→
   225→                        // Check total serialized size (5 MB limit matches DB constraint).
   226→                        let totalSize = 0;
   227→                        for (const record of writes.values()) {
   228→                                totalSize += JSON.stringify(record).length;
   229→                        }
   230→                        if (totalSize > 5_000_000) {
   231→                                console.warn(
   232→                                        `[TldrawCanvas] Pending sync too large (${(totalSize / 1_000_000).toFixed(1)}MB) — skipping`
   233→                                );
   234→                                return;
   235→                        }
   236→
   237→                        // Batch all writes/deletes into a single Yjs transaction tagged
   238→                        // with our origin so the observer can skip re-applying them.
   239→                        ydoc.transact(() => {
   240→                                for (const [id, record] of writes) {
   241→                                        map.set(id, JSON.stringify(record));
   242→                                }
   243→                                for (const id of deletes) {
   244→                                        map.delete(id);
   245→                                }
   246→                        }, LOCAL_SYNC_ORIGIN);
   247→
   248→                        // Clear dirty flags for the records we just synced.
   249→                        for (const id of writes.keys()) {
   250→                                localDirtyRef.current.delete(id);
   251→                        }
   252→                        for (const id of deletes) {
   253→                                localDirtyRef.current.delete(id);
   254→                        }
   255→
   256→                        onStoreChangeRef.current?.();
   257→                };
   258→        }, [ydoc]);
   259→
   260→        // Keep flushPendingRef up-to-date.
   261→        flushPendingRef.current = buildFlush();
   262→
   263→        // ----------------------------------------------------------------
   264→        // Schedule a debounced flush
   265→        // ----------------------------------------------------------------
   266→
   267→        const buildScheduleSync = useCallback(() => {
   268→                return () => {
   269→                        if (debounceTimerRef.current) {
   270→                                clearTimeout(debounceTimerRef.current);
   271→                        }
   272→                        debounceTimerRef.current = setTimeout(() => {
   273→                                flushPendingRef.current();
   274→                        }, SYNC_DEBOUNCE_MS);
   275→                };
   276→        }, []);
   277→
   278→        // Keep scheduleSyncRef up-to-date.
   279→        scheduleSyncRef.current = buildScheduleSync();
   280→
   281→        // ----------------------------------------------------------------
   282→        // Load initial state from Yjs into the editor
   283→        // ----------------------------------------------------------------
   284→
   285→        const loadInitialState = useCallback(() => {
   286→                const editor = editorRef.current;
   287→                if (!editor) return;
   288→
   289→                // If no Yjs document (e.g. Hocuspocus unavailable), start fresh
   290→                if (!ydoc) {
   291→                        isInitialLoadRef.current = false;
   292→                        setIsLoadingSnapshot(false);
   293→                        console.log('[TldrawCanvas] No Yjs document — starting in standalone mode');
   294→                        return;
   295→                }
   296→
   297→                isInitialLoadRef.current = true;
   298→
   299→                const shapesMap = ydoc.getMap<string>(`${YJS_SHAPES_MAP_PREFIX}${pageIndex}`);
   300→                shapesMapRef.current = shapesMap;
   301→
   302→                // --- Try legacy full-snapshot format first ---
   303→                const legacyMap = ydoc.getMap<string>(YJS_LEGACY_MAP_KEY);
   304→                const legacyKey = `page-${pageIndex}`;
   305→                const legacyJson = legacyMap.get(legacyKey);
   306→
   307→                if (legacyJson && !migratedPagesRef.current.has(pageIndex)) {
   308→                        try {
   309→                                const storeSnapshot = JSON.parse(legacyJson) as TLStoreSnapshot;
   310→
   311→                                // editor.loadSnapshot accepts Partial<TLEditorSnapshot> | TLStoreSnapshot
   312→                                editor.loadSnapshot({ document: storeSnapshot });
   313→                                console.log(`[TldrawCanvas] Loaded legacy snapshot for page ${pageIndex}`);
   314→
   315→                                // Migrate the legacy snapshot into per-record Y.Map entries.
   316→                                // TLStoreSnapshot has { store: Record<string, TLRecord>, schema: ... }
   317→                                const records = (storeSnapshot as any).store;
   318→                                if (records) {
   319→                                        ydoc.transact(() => {
   320→                                                for (const [id, record] of Object.entries(records)) {
   321→                                                        shapesMap.set(id, JSON.stringify(record));
   322→                                                }
   323→                                        }, LOCAL_SYNC_ORIGIN);
   324→                                        console.log(
   325→                                                `[TldrawCanvas] Migrated ${Object.keys(records).length} records to per-record format for page ${pageIndex}`
   326→                                        );
   327→                                }
   328→
   329→                                migratedPagesRef.current.add(pageIndex);
   330→                        } catch (err) {
   331→                                console.error(
   332→                                        `[TldrawCanvas] Failed to load/parse legacy snapshot for page ${pageIndex}:`,
   333→                                        err
   334→                                );
   335→                        }
   336→                } else {
   337→                        // --- Load from per-record format ---
   338→                        const recordsToLoad: Record<string, unknown>[] = [];
   339→                        shapesMap.forEach((json, id) => {
   340→                                try {
   341→                                        recordsToLoad.push(JSON.parse(json));
   342→                                } catch (err) {
   343→                                        console.error(`[TldrawCanvas] Failed to parse record ${id}:`, err);
   344→                                }
   345→                        });
   346→
   347→                        if (recordsToLoad.length > 0) {
   348→                                // Use mergeRemoteChanges so these are tagged source='remote'
   349→                                // and won't trigger our local-sync listener.
   350→                                editor.store.mergeRemoteChanges(() => {
   351→                                        editor.store.put(recordsToLoad as any);
   352→                                });
   353→                                console.log(
   354→                                        `[TldrawCanvas] Loaded ${recordsToLoad.length} records from per-record format for page ${pageIndex}`
   355→                                );
   356→                        } else if (!legacyJson) {
   357→                                console.log(`[TldrawCanvas] No saved data for page ${pageIndex} — starting fresh`);
   358→                        }
   359→                }
   360→
   361→                isInitialLoadRef.current = false;
   362→                setIsLoadingSnapshot(false);
   363→        }, [ydoc, pageIndex]);
   364→
   365→        // ----------------------------------------------------------------
   366→        // Set up the Yjs remote-change observer for the shapes map
   367→        // ----------------------------------------------------------------
   368→
   369→        const setupRemoteObserver = useCallback(() => {
   370→                const shapesMap = shapesMapRef.current;
   371→                if (!shapesMap) return () => {};
   372→
   373→                const observer = (event: any, transaction: any) => {
   374→                        // Skip our own local writes (tagged with LOCAL_SYNC_ORIGIN).
   375→                        if (transaction.origin === LOCAL_SYNC_ORIGIN) return;
   376→
   377→                        const editor = editorRef.current;
   378→                        if (!editor) return;
   379→                        if (isInitialLoadRef.current) return;
   380→
   381→                        const toPut: Record<string, unknown>[] = [];
   382→                        const toRemove: string[] = [];
   383→
   384→                        // Determine which keys changed.
   385→                        // YMapEvent extends YEvent which has `changes.keys` (Map<string, {action, oldValue}>).
   386→                        // It also has `keysChanged` (Set<string>) which is a simpler set of changed keys.
   387→                        const changedKeys: Set<string> | undefined = event.keysChanged;
   388→                        if (!changedKeys) return;
   389→
   390→                        for (const id of changedKeys) {
   391→                                // Skip records with pending local changes — our local edit wins.
   392→                                if (localDirtyRef.current.has(id)) {
   393→                                        continue;
   394→                                }
   395→
   396→                                // Check if the key still exists in the map (add/update) or was deleted.
   397→                                if (shapesMap.has(id)) {
   398→                                        try {
   399→                                                toPut.push(JSON.parse(shapesMap.get(id)!));
   400→                                        } catch (err) {
   401→                                                console.error(`[TldrawCanvas] Failed to parse remote record ${id}:`, err);
   402→                                        }
   403→                                } else {
   404→                                        toRemove.push(id);
   405→                                }
   406→                        }
   407→
   408→                        if (toPut.length > 0 || toRemove.length > 0) {
   409→                                editor.store.mergeRemoteChanges(() => {
   410→                                        if (toPut.length > 0) {
   411→                                                editor.store.put(toPut as any);
   412→                                        }
   413→                                        if (toRemove.length > 0) {
   414→                                                editor.store.remove(toRemove as any);
   415→                                        }
   416→                                });
   417→                                console.log(
   418→                                        `[TldrawCanvas] Applied remote changes for page ${pageIndex}: ${toPut.length} puts, ${toRemove.length} removes`
   419→                                );
   420→                        }
   421→                };
   422→
   423→                shapesMap.observe(observer);
   424→                return () => {
   425→                        shapesMap.unobserve(observer);
   426→                };
   427→        }, [pageIndex]);
   428→
   429→        // ----------------------------------------------------------------
   430→        // Register the store listener with { source: 'user', scope: 'document' }
   431→        // so it only fires for local user edits (not for remote-applied changes).
   432→        // ----------------------------------------------------------------
   433→
   434→        const registerStoreListener = useCallback(() => {
   435→                const editor = editorRef.current;
   436→                if (!editor) return;
   437→
   438→                // Clean up previous listener if any.
   439→                unlistenStoreRef.current?.();
   440→
   441→                unlistenStoreRef.current = (editor.store as any).listen(
   442→                        (entry: HistoryEntryLike) => {
   443→                                processLocalDiff(entry);
   444→                        },
   445→                        { source: 'user', scope: 'document' } as any
   446→                );
   447→        }, [processLocalDiff]);
   448→
   449→        // ----------------------------------------------------------------
   450→        // Handle editor mount (called once by Tldraw)
   451→        // ----------------------------------------------------------------
   452→
   453→        const handleMount = useCallback(
   454→                (editor: Editor) => {
   455→                        editorRef.current = editor;
   456→                        prevPageIndexRef.current = pageIndex;
   457→
   458→                        // Notify parent with the editor ref.
   459→                        onEditorReady?.(editor);
   460→
   461→                        // Short delay lets the editor finish its internal initialization
   462→                        // before we start loading external data.
   463→                        setTimeout(() => {
   464→                                // 1. Load initial state (legacy snapshot or per-record format).
   465→                                loadInitialState();
   466→
   467→                                // 2. Register the store listener for user-initiated changes.
   468→                                registerStoreListener();
   469→
   470→                                // 3. Set up the Yjs remote observer.
   471→                                unobserveYjsRef.current = setupRemoteObserver();
   472→
   473→                                // 4. Mark as ready — enables the store listener's isReadyRef check.
   474→                                isReadyRef.current = true;
   475→                                setIsReady(true);
   476→                        }, 100);
   477→                },
   478→                [onEditorReady, pageIndex, loadInitialState, registerStoreListener, setupRemoteObserver]
   479→        );
   480→
   481→        // ----------------------------------------------------------------
   482→        // Handle page changes: flush old page, load new page
   483→        // ----------------------------------------------------------------
   484→
   485→        useEffect(() => {
   486→                if (!editorRef.current || !ydoc) return;
   487→                if (prevPageIndexRef.current === pageIndex) return;
   488→
   489→                // 1. Cancel any pending debounce.
   490→                if (debounceTimerRef.current) {
   491→                        clearTimeout(debounceTimerRef.current);
   492→                        debounceTimerRef.current = null;
   493→                }
   494→
   495→                // 2. Mark as not-ready to suppress store listener during transition.
   496→                isReadyRef.current = false;
   497→
   498→                // 3. Flush pending changes for the OLD page (shapesMapRef still points to old map).
   499→                flushPendingRef.current();
   500→
   501→                // 4. Clean up old Yjs observer.
   502→                unobserveYjsRef.current?.();
   503→                unobserveYjsRef.current = null;
   504→
   505→                // 5. Reset all pending/dirty state for the new page.
   506→                pendingWritesRef.current = new Map();
   507→                pendingDeletesRef.current = new Set();
   508→                localDirtyRef.current = new Set();
   509→
   510→                // 6. Update page tracking.
   511→                prevPageIndexRef.current = pageIndex;
   512→
   513→                // 7. Load the new page's state.
   514→                setIsLoadingSnapshot(true);
   515→                loadInitialState();
   516→
   517→                // 8. Set up new Yjs observer for the new page's map.
   518→                unobserveYjsRef.current = setupRemoteObserver();
   519→
   520→                // 9. Re-enable the store listener.
   521→                //    (registerStoreListener is idempotent — it cleans up the old one first.)
   522→                registerStoreListener();
   523→
   524→                // 10. Mark as ready again.
   525→                isReadyRef.current = true;
   526→                setIsReady(true);
   527→        }, [pageIndex, ydoc, loadInitialState, setupRemoteObserver, registerStoreListener]);
   528→
   529→        // ----------------------------------------------------------------
   530→        // License Gate Mitigation: Detect when tldraw unmounts due to
   531→        // missing production license and force re-mount.
   532→        // ----------------------------------------------------------------
   533→        const containerRef = useRef<HTMLDivElement>(null);
   534→        useEffect(() => {
   535→                const container = containerRef.current;
   536→                if (!container) return;
   537→
   538→                const observer = new MutationObserver(() => {
   539→                        // Check if the license-gate div has appeared (Tldraw unmounted itself)
   540→                        if (container.querySelector('[data-testid="tl-license-expired"]')) {
   541→                                // Force re-mount by incrementing key
   542→                                setTldrawKey((k) => k + 1);
   543→                                console.warn(
   544→                                        '[TldrawCanvas] tldraw license gate detected — re-mounting editor. ' +
   545→                                        'Obtain a production license from sales@tldraw.com to remove this workaround.'
   546→                                );
   547→                        }
   548→                });
   549→
   550→                observer.observe(container, { childList: true, subtree: true });
   551→                return () => observer.disconnect();
   552→        }, []);
   553→
   554→        // ----------------------------------------------------------------
   555→        // Cleanup on unmount
   556→        // ----------------------------------------------------------------
   557→
   558→        useEffect(() => {
   559→                return () => {
   560→                        // Cancel pending debounce.
   561→                        if (debounceTimerRef.current) {
   562→                                clearTimeout(debounceTimerRef.current);
   563→                                debounceTimerRef.current = null;
   564→                        }
   565→
   566→                        // Flush any remaining pending changes immediately.
   567→                        flushPendingRef.current();
   568→
   569→                        // Clean up listeners.
   570→                        unlistenStoreRef.current?.();
   571→                        unobserveYjsRef.current?.();
   572→
   573→                        editorRef.current = null;
   574→                };
   575→        }, []); // eslint-disable-line react-hooks/exhaustive-deps
   576→
   577→        // ----------------------------------------------------------------
   578→        // Render
   579→        // ----------------------------------------------------------------
   580→
   581→        return (
   582→                <div className="w-full h-full relative" ref={containerRef}>
   583→                        {/* Loading overlay while snapshot is being loaded */}
   584→                        {isLoadingSnapshot && (
