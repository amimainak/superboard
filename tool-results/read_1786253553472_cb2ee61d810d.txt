   731→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   732→                  <GoogleIcon className="w-4 h-4" />
   733→                  Sign up with Google
   734→                </Button>
   735→                <div className="flex items-center gap-3">
   736→                  <div className="flex-1 h-px bg-gray-200" />
   737→                  <span className="text-xs text-gray-400 font-medium">or register with email</span>
   738→                  <div className="flex-1 h-px bg-gray-200" />
   739→                </div>
   740→                <div className="space-y-2">
   741→                  <Label htmlFor="register-name" className="text-sm font-medium">Full Name</Label>
   742→                  <Input id="register-name" type="text" placeholder="Your name" value={registerName} onChange={(e) => setRegisterName(e.target.value)} className="rounded-xl" />

   751→                  {registerPassword && (
   752→                    <div className="space-y-1">
   753→                      <div className="flex gap-1">
   754→                        {[1,2,3,4,5].map((i) => (
   755→                          <div key={i} className={`h-1 flex-1 rounded-full transition-colors ${i <= passwordStrength.score ? passwordStrength.color : 'bg-gray-200'}`} />

   757→                      </div>
   758→                      <p className={`text-xs ${passwordStrength.score <= 1 ? 'text-red-500' : passwordStrength.score <= 3 ? 'text-amber-500' : 'text-emerald-600'}`}>
   759→                        {passwordStrength.label}
   760→                      </p>
   761→                    </div>
   762→                  )}
   763→                </div>
   764→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   765→                {authMessage && <p className="text-sm text-emerald-600 bg-emerald-50 rounded-lg px-3 py-2">{authMessage}</p>}
   766→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Create Account</Button>
   767→                <p className="text-center text-sm text-muted-foreground">
   768→                  Already have an account?{' '}
   769→                  <button type="button" className="text-emerald-600 font-medium hover:underline" onClick={() => setShowAuth('login')}>Sign in</button>
   770→                </p>
   771→              </form>
   772→            </>
   773→          ) : null}
   774→        </DialogContent>
   775→      </Dialog>
   776→
   777→      {/* ===== EMAIL VERIFICATION SCREEN ===== */}
   778→      {showVerification && (
   779→        <div className="fixed inset-0 z-50 bg-white flex items-center justify-center px-6">
   780→          <div className="max-w-md w-full text-center space-y-6">

   783→            </div>
   784→            <h2 className="text-2xl font-bold text-gray-900">Check your email</h2>

   786→              We&apos;ve sent a confirmation link to <strong>{registerEmail}</strong>.

   788→            </p>

   791→            </p>
   792→            <Button variant="outline" className="rounded-xl" onClick={() => setShowVerification(false)}>
   793→              Back to Sign In
   794→            </Button>
   795→          </div>
   796→        </div>
   797→      )}
   798→    </main>
   799→  );
