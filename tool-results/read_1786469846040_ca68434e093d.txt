   223→// RoomParticipant — Student Join Tracking
   224→// ============================================================
   225→// Tracks every time a student joins a room.
   226→// Used for agency billing and attendance tracking.
   227→//
   228→// studentId links to the Student model (registered agency students).
   229→// studentIdentity is the display identifier (Student.name or fallback).
   230→// ============================================================
   231→
   232→model RoomParticipant {
   233→  id               String   @id @default(uuid())
   234→  roomId           String
   235→  room             Room     @relation(fields: [roomId], references: [id], onDelete: Cascade)
   236→  studentId        String?  // FK to Student model (nullable for backward compat)
   237→  student          Student? @relation(fields: [studentId], references: [id], onDelete: SetNull)
   238→  studentIdentity  String   // Display name or identifier
   239→  studentName      String?  // Display name provided during join
   240→  joinedAt         DateTime @default(now())
   241→  lastActiveAt     DateTime @default(now())
   242→
   243→  // A student can join a room multiple times, but we track
   244→  // the latest visit via lastActiveAt (upserted on rejoin).
   245→  @@unique([roomId, studentIdentity]) // One record per student per room
   246→  @@index([roomId, lastActiveAt])
   247→  @@index([studentIdentity, lastActiveAt])
   248→  @@index([studentId])
   249→}

   308→  recipient      User?    @relation("InviteRecipient", fields: [recipientId], references: [id], onDelete: SetNull)

   310→  status         InviteStatus @default(PENDING)
   311→  expiresAt      DateTime // Invite expires at this time
   312→  acceptedAt     DateTime? // Set when accepted
   313→  createdAt      DateTime @default(now())
   314→  updatedAt      DateTime @updatedAt
   315→
   316→  @@index([invitedEmail, status])
