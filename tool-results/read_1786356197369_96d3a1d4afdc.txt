     1→// ============================================================
     2→// TldrawCanvas — Tldraw Editor with Yjs Real-Time Sync
     3→// ============================================================
     4→// Wraps the Tldraw editor and synchronizes state with Yjs
     5→// via the Hocuspocus CRDT provider for multi-user collaboration.
     6→//
     7→// Sync strategy: Incremental per-record CRDT sync.
     8→// - Each record (shape, camera, page-state, etc.) is stored as a
     9→//   separate entry in a per-page Y.Map, keyed by record ID.
    10→// - Local user edits are debounced (200ms) and batch-written to Yjs.
    11→// - Remote Yjs changes are merged record-by-record into the editor
    12→//   using store.mergeRemoteChanges(), which marks them source='remote'
    13→//   so they don't trigger the local-sync listener (no echo loop).
    14→// - A dirty-set tracks record IDs with unsynced local changes;
    15→//   remote updates for dirty records are skipped (last-local-writer-wins).
    16→// - A monotonic localChangeVersion counter is maintained per record
    17→//   for diagnostics and future conflict-resolution extensions.
    18→// - Backward compatibility: legacy full-snapshot format is detected
    19→//   on first load and migrated to the per-record format.
    20→// ============================================================
    21→
    22→'use client';
    23→
    24→import React, { useCallback, useEffect, useRef, useState } from 'react';
    25→import { Tldraw, Editor, TLStoreSnapshot } from 'tldraw';
    26→import type { Map as YMap } from 'yjs';
    27→import 'tldraw/tldraw.css';
    28→
    29→// ---- Types ----
    30→
    31→/** Shape of a RecordsDiff as delivered by store.listen in tldraw v5. */
    32→interface StoreChangeEvent {
    33→        changes: {
    34→                added: Record<string, Record<string, unknown>>;
    35→                updated: Record<string, [from: Record<string, unknown>, to: Record<string, unknown>]>;
    36→                removed: Record<string, Record<string, unknown>>;
    37→        };
    38→        source: 'user' | 'remote';
    39→}
    40→
    41→export interface TldrawCanvasProps {
    42→        /** Yjs document (from useYjsProvider) */
    43→        ydoc: import('yjs').Doc | null;
    44→        /** Called when the editor mounts — provides the Editor ref */
    45→        onEditorReady?: (editor: Editor) => void;
    46→        /** Called when editor store changes (for external sync triggers) */
    47→        onStoreChange?: () => void;
    48→        /** Current page index (for multi-page support) */
    49→        pageIndex: number;
    50→        /** Whether the user is a tutor (controls UI permissions) */
    51→        isTutor: boolean;
    52→        /** Whether the canvas is read-only (e.g., student in focus mode) */
    53→        readOnly?: boolean;
    54→}
    55→
    56→// ---- Constants ----
    57→
    58→/** Debounce Yjs writes to batch rapid edits into one transaction. */
    59→const SYNC_DEBOUNCE_MS = 200;
    60→
    61→/** Key of the legacy Y.Map that stored full page snapshots as JSON blobs. */
    62→const YJS_LEGACY_MAP_KEY = 'page-snapshot';
    63→
    64→/**
    65→ * Prefix for per-page Y.Maps that store individual records.
    66→ * E.g. pageIndex=3 → ydoc.getMap('page-shapes-3')
    67→ */
    68→const YJS_SHAPES_MAP_PREFIX = 'page-shapes-';
    69→
    70→/**
    71→ * Transaction origin symbol used to tag our own local writes.
    72→ * The Yjs observer checks this to skip re-applying our own changes.
    73→ */
    74→const LOCAL_SYNC_ORIGIN = Symbol('tldraw-local-sync');
    75→
    76→// ============================================================
    77→// Component
    78→// ============================================================
    79→
    80→export default function TldrawCanvas({
    81→        ydoc,
    82→        onEditorReady,
    83→        onStoreChange,
    84→        pageIndex,
    85→        isTutor,
    86→        readOnly = false,
    87→}: TldrawCanvasProps) {
    88→        // ---- Core refs ----
    89→        const editorRef = useRef<Editor | null>(null);
    90→        const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    91→        const [isReady, setIsReady] = useState(false);
    92→        const [isLoadingSnapshot, setIsLoadingSnapshot] = useState(true);
    93→
    94→        // ---- Incremental sync refs (stable across renders) ----
    95→
    96→        /** Records that changed locally and are awaiting debounce flush to Yjs. */
    97→        const pendingWritesRef = useRef<Map<string, Record<string, unknown>>>(new Map());
    98→
    99→        /** Record IDs that were deleted locally and are awaiting debounce flush. */
   100→        const pendingDeletesRef = useRef<Set<string>>(new Set());
   101→
   102→        /**
   103→         * Record IDs with unsynced local changes.
   104→         * Remote updates for these IDs are skipped to prevent overwriting
   105→         * concurrent local edits (last-local-writer-wins).  IDs are removed
   106→         * after the corresponding write/delete is flushed to Yjs.
   107→         */
   108→        const localDirtyRef = useRef<Set<string>>(new Set());
   109→
   110→        /** Monotonically increasing version counter for local edits. */
   111→        const localChangeVersionRef = useRef(0);
   112→
   113→        /** Per-record version: the localChangeVersion at which it was last edited locally. */
   114→        const recordVersionsRef = useRef<Map<string, number>>(new Map());
   115→
   116→        /** True while the initial snapshot is being loaded (suppresses sync-to-Yjs). */
   117→        const isInitialLoadRef = useRef(false);
   118→
   119→        /** Set of page indices that have already been migrated from legacy format. */
   120→        const migratedPagesRef = useRef<Set<number>>(new Set());
   121→
   122→        /** Cleanup function for the tldraw store listener. */
   123→        const unlistenStoreRef = useRef<(() => void) | null>(null);
   124→
   125→        /** Cleanup function for the Yjs shapes-map observer. */
   126→        const unobserveYjsRef = useRef<(() => void) | null>(null);
   127→
   128→        /**
   129→         * Ref to the Y.Map currently being used for this page's shapes.
   130→         * Updated explicitly on page switch so we can flush the OLD map
   131→         * before swapping to the new one.
   132→         */
   133→        const shapesMapRef = useRef<YMap<string> | null>(null);
   134→
   135→        /** Keep callback refs up-to-date without re-registering listeners. */
   136→        const onStoreChangeRef = useRef(onStoreChange);
   137→        onStoreChangeRef.current = onStoreChange;
   138→
   139→        // ----------------------------------------------------------------
   140→        // Flush pending changes to the current shapes Y.Map
   141→        // ----------------------------------------------------------------
   142→
   143→        const flushPendingToYjs = useCallback(() => {
   144→                const map = shapesMapRef.current;
   145→                if (!map || !ydoc) return;
   146→
   147→                // Nothing to flush?
   148→                if (pendingWritesRef.current.size === 0 && pendingDeletesRef.current.size === 0) return;
   149→
   150→                // Snapshot the pending state so new changes accumulate independently.
   151→                const writes = new Map(pendingWritesRef.current);
   152→                const deletes = new Set(pendingDeletesRef.current);
   153→                pendingWritesRef.current = new Map();
   154→                pendingDeletesRef.current = new Set();
   155→
   156→                // Check total size before writing (5 MB limit matches DB constraint)
   157→                let totalSize = 0;
   158→                for (const record of writes.values()) {
   159→                        totalSize += JSON.stringify(record).length;
   160→                }
   161→                if (totalSize > 5_000_000) {
   162→                        console.warn(
   163→                                `[TldrawCanvas] Pending sync too large (${(totalSize / 1_000_000).toFixed(1)}MB) — skipping`
   164→                        );
   165→                        return;
   166→                }
   167→
   168→                // Batch all writes/deletes into a single Yjs transaction tagged with
   169→                // our origin so the observer can skip re-applying them.
   170→                ydoc.transact(() => {
   171→                        for (const [id, record] of writes) {
   172→                                map.set(id, JSON.stringify(record));
   173→                        }
   174→                        for (const id of deletes) {
   175→                                map.delete(id);
   176→                        }
   177→                }, LOCAL_SYNC_ORIGIN);
   178→
   179→                // Clear dirty flags for the records we just synced.
   180→                for (const id of writes.keys()) {
   181→                        localDirtyRef.current.delete(id);
   182→                }
   183→                for (const id of deletes) {
   184→                        localDirtyRef.current.delete(id);
   185→                }
   186→
   187→                onStoreChangeRef.current?.();
   188→        }, [ydoc]);
   189→
   190→        // ----------------------------------------------------------------
   191→        // Schedule a debounced flush
   192→        // ----------------------------------------------------------------
   193→
   194→        const scheduleSync = useCallback(() => {
   195→                if (debounceTimerRef.current) {
   196→                        clearTimeout(debounceTimerRef.current);
   197→                }
   198→                debounceTimerRef.current = setTimeout(() => {
   199→                        flushPendingToYjs();
   200→                }, SYNC_DEBOUNCE_MS);
   201→        }, [flushPendingToYjs]);
   202→
   203→        // ----------------------------------------------------------------
   204→        // Load initial state from Yjs into the editor
   205→        // ----------------------------------------------------------------
   206→
   207→        const loadInitialState = useCallback(() => {
   208→                const editor = editorRef.current;
   209→                if (!editor || !ydoc) return;
   210→
   211→                isInitialLoadRef.current = true;
   212→
   213→                const shapesMap = ydoc.getMap<string>(`${YJS_SHAPES_MAP_PREFIX}${pageIndex}`);
   214→                shapesMapRef.current = shapesMap;
   215→
   216→                // --- Try legacy full-snapshot format first ---
   217→                const legacyMap = ydoc.getMap<string>(YJS_LEGACY_MAP_KEY);
   218→                const legacyKey = `page-${pageIndex}`;
   219→                const legacyJson = legacyMap.get(legacyKey);
   220→
   221→                if (legacyJson && !migratedPagesRef.current.has(pageIndex)) {
   222→                        try {
   223→                                const storeSnapshot = JSON.parse(legacyJson) as TLStoreSnapshot;
   224→
   225→                                // TLStoreSnapshot has { store: Record<string, TLRecord>, schema: ... }
   226→                                // editor.loadSnapshot accepts Partial<TLEditorSnapshot> | TLStoreSnapshot
   227→                                editor.loadSnapshot({ document: storeSnapshot });
   228→                                console.log(
   229→                                        `[TldrawCanvas] Loaded legacy snapshot for page ${pageIndex}`
   230→                                );
   231→
   232→                                // Migrate the legacy snapshot into per-record Y.Map entries.
   233→                                // We iterate storeSnapshot.store (the actual record records).
   234→                                const records = storeSnapshot.store;
   235→                                if (records) {
   236→                                        ydoc.transact(() => {
   237→                                                for (const [id, record] of Object.entries(records)) {
   238→                                                        shapesMap.set(id, JSON.stringify(record));
   239→                                                }
   240→                                        }, LOCAL_SYNC_ORIGIN);
   241→                                        console.log(
   242→                                                `[TldrawCanvas] Migrated ${Object.keys(records).length} records to per-record format for page ${pageIndex}`
   243→                                        );
   244→                                }
   245→
   246→                                migratedPagesRef.current.add(pageIndex);
   247→                        } catch (err) {
   248→                                console.error(
   249→                                        `[TldrawCanvas] Failed to load/parse legacy snapshot for page ${pageIndex}:`,
   250→                                        err
   251→                                );
   252→                        }
   253→                } else {
   254→                        // --- Load from per-record format ---
   255→                        const recordsToLoad: Record<string, unknown>[] = [];
   256→                        shapesMap.forEach((json, id) => {
   257→                                try {
   258→                                        recordsToLoad.push(JSON.parse(json));
   259→                                } catch (err) {
   260→                                        console.error(`[TldrawCanvas] Failed to parse record ${id}:`, err);
   261→                                }
   262→                        });
   263→
   264→                        if (recordsToLoad.length > 0) {
   265→                                // Use mergeRemoteChanges so these are tagged source='remote'
   266→                                // and won't trigger our local-sync listener.
   267→                                editor.store.mergeRemoteChanges(() => {
   268→                                        editor.store.put(recordsToLoad as any);
   269→                                });
   270→                                console.log(
   271→                                        `[TldrawCanvas] Loaded ${recordsToLoad.length} records from per-record format for page ${pageIndex}`
   272→                                );
   273→                        } else if (!legacyJson) {
   274→                                console.log(
   275→                                        `[TldrawCanvas] No saved data for page ${pageIndex} — starting fresh`
   276→                                );
   277→                        }
   278→                }
   279→
   280→                isInitialLoadRef.current = false;
   281→                setIsLoadingSnapshot(false);
   282→        }, [ydoc, pageIndex]);
   283→
   284→        // ----------------------------------------------------------------
   285→        // Set up the Yjs remote-change observer for the shapes map
   286→        // ----------------------------------------------------------------
   287→
   288→        const setupRemoteObserver = useCallback(() => {
   289→                const shapesMap = shapesMapRef.current;
   290→                if (!shapesMap) return;
   291→
   292→                const observer = (
   293→                        event: { target: YMap<string>; transaction: { origin: unknown } },
   294→                        transaction: { origin: unknown }
   295→                ) => {
   296→                        // Skip our own local writes (tagged with LOCAL_SYNC_ORIGIN).
   297→                        if (transaction.origin === LOCAL_SYNC_ORIGIN) return;
   298→
   299→                        const editor = editorRef.current;
   300→                        if (!editor) return;
   301→                        if (isInitialLoadRef.current) return;
   302→
   303→                        const toPut: Record<string, unknown>[] = [];
   304→                        const toRemove: string[] = [];
   305→
   306→                        // event.keysChanged (Set) tells us which keys changed in this transaction.
   307→                        // We need to check the current state of the Y.Map for each changed key.
   308→                        // The YMapEvent parent class provides `changes.keys` with action info,
   309→                        // but the actual event object passed to the observer may vary.
   310→                        // We inspect the target map directly for add/update vs checking if the key
   311→                        // still exists.
   312→                        const ymapEvent = event as any;
   313→                        const changedKeys: Map<
   314→                                string,
   315→                                { action: 'add' | 'update' | 'delete'; oldValue: any }
   316→                        > | null = ymapEvent.keysChanged
   317→                                ? // YMapEvent has a `keysChanged` set; we need the richer `changes.keys` from YEvent
   318→                                  null
   319→                                : ymapEvent.changes?.keys || null;
   320→
   321→                        // Determine which keys to process.
   322→                        // In Yjs, the observer fires with the full event. We use keysChanged
   323→                        // (a Set on YMapEvent) if available, otherwise iterate the changes map.
   324→                        const keysToProcess = new Set<string>();
   325→
   326→                        if (changedKeys) {
   327→                                for (const [key] of changedKeys) {
   328→                                        keysToProcess.add(key);
   329→                                }
   330→                        }
   331→
   332→                        // Fallback: iterate keysChanged if available on YMapEvent
   333→                        if (keysToProcess.size === 0 && ymapEvent.keysChanged) {
   334→                                for (const key of ymapEvent.keysChanged as Set<string>) {
   335→                                        keysToProcess.add(key);
   336→                                }
   337→                        }
   338→
   339→                        for (const id of keysToProcess) {
   340→                                // Skip records with pending local changes — our local edit wins.
   341→                                if (localDirtyRef.current.has(id)) {
   342→                                        continue;
   343→                                }
   344→
   345→                                const currentJson = shapesMap.get(id);
   346→                                if (currentJson !== undefined) {
   347→                                        // Record exists (add or update)
   348→                                        try {
   349→                                                toPut.push(JSON.parse(currentJson));
   350→                                        } catch (err) {
   351→                                                console.error(
   352→                                                        `[TldrawCanvas] Failed to parse remote record ${id}:`,
   353→                                                        err
   354→                                                );
   355→                                        }
   356→                                } else {
   357→                                        // Record was deleted
   358→                                        toRemove.push(id);
   359→                                }
   360→                        }
   361→
   362→                        if (toPut.length > 0 || toRemove.length > 0) {
   363→                                editor.store.mergeRemoteChanges(() => {
   364→                                        if (toPut.length > 0) {
   365→                                                editor.store.put(toPut as any);
   366→                                        }
   367→                                        if (toRemove.length > 0) {
   368→                                                editor.store.remove(toRemove as any);
   369→                                        }
   370→                                });
   371→                                console.log(
   372→                                        `[TldrawCanvas] Applied remote changes for page ${pageIndex}: ${toPut.length} puts, ${toRemove.length} removes`
   373→                                );
   374→                        }
   375→                };
   376→
   377→                shapesMap.observe(observer as any);
   378→                return () => {
   379→                        shapesMap.unobserve(observer as any);
   380→                };
   381→        }, [pageIndex]);
   382→
   383→        // ----------------------------------------------------------------
   384→        // Handle editor mount
   385→        // ----------------------------------------------------------------
   386→
   387→        const handleMount = useCallback(
   388→                (editor: Editor) => {
   389→                        editorRef.current = editor;
   390→
   391→                        // Notify parent with the editor ref.
   392→                        onEditorReady?.(editor);
   393→
   394→                        // Short delay lets the editor finish its internal initialization
   395→                        // before we start loading external data.
   396→                        setTimeout(() => {
   397→                                // 1. Load initial state (legacy snapshot or per-record format)
   398→                                loadInitialState();
   399→
   400→                                // 2. Start listening for user-initiated document-scoped store changes.
   401→                        //    Using { source: 'user', scope: 'document' } means:
   402→                        //    - We only see changes the local user made (not remote-applied ones)
   403→                        //    - We only see document-scoped records (not session/presence)
   404→                        //    This eliminates the need for manual echo-prevention flags.
   405→                                unlistenStoreRef.current = (editor.store as any).listen(
   406→                                        (entry: StoreChangeEvent) => {
   407→                                                if (isInitialLoadRef.current) return;
   408→                                                if (!isReady) return;
   409→
   410→                                                const { changes } = entry;
   411→
   412→                                                // Increment the global local version counter.
   413→                                                localChangeVersionRef.current++;
   414→                                                const version = localChangeVersionRef.current;
   415→
   416→                                                // Collect added records.
   417→                                                for (const [id, record] of Object.entries(changes.added)) {
   418→                                                        pendingWritesRef.current.set(id, record);
   419→                                                        pendingDeletesRef.current.delete(id);
   420→                                                        localDirtyRef.current.add(id);
   421→                                                        recordVersionsRef.current.set(id, version);
   422→                                                }
   423→
   424→                                                // Collect updated records (use the "to" value).
   425→                                                for (const [id, pair] of Object.entries(changes.updated)) {
   426→                                                        const to = pair[1];
   427→                                                        pendingWritesRef.current.set(id, to);
   428→                                                        pendingDeletesRef.current.delete(id);
   429→                                                        localDirtyRef.current.add(id);
   430→                                                        recordVersionsRef.current.set(id, version);
   431→                                                }
   432→
   433→                                                // Collect removed records.
   434→                                                for (const id of Object.keys(changes.removed)) {
   435→                                                        pendingDeletesRef.current.add(id);
   436→                                                        pendingWritesRef.current.delete(id);
   437→                                                        localDirtyRef.current.add(id);
   438→                                                        recordVersionsRef.current.set(id, version);
   439→                                                }
   440→
   441→                                                scheduleSync();
   442→                                        },
   443→                                        { source: 'user', scope: 'document' } as any
   444→                                );
   445→
   446→                                // 3. Set up the Yjs remote observer for per-record changes.
   447→                                unobserveYjsRef.current = setupRemoteObserver();
   448→
   449→                                setIsReady(true);
   450→                        }, 100);
   451→                },
   452→                [onEditorReady, loadInitialState, setupRemoteObserver, isReady, scheduleSync]
   453→        );
   454→
   455→        // ----------------------------------------------------------------
   456→        // Handle page changes: flush old page, load new page
   457→        // ----------------------------------------------------------------
   458→
   459→        const previousPageIndexRef = useRef(pageIndex);
   460→
   461→        useEffect(() => {
   462→                if (!editorRef.current || !ydoc) return;
   463→
   464→                // Only act on actual page changes (not initial mount).
   465→                if (previousPageIndexRef.current === pageIndex) return;
   466→                previousPageIndexRef.current = pageIndex;
   467→
   468→                // 1. Cancel any pending debounce for the old page.
   469→                if (debounceTimerRef.current) {
   470→                        clearTimeout(debounceTimerRef.current);
   471→                        debounceTimerRef.current = null;
   472→                }
   473→
   474→                // 2. Flush pending changes for the OLD page (shapesMapRef still points to old map).
   475→                flushPendingToYjs();
   476→
   477→                // 3. Clean up old listeners.
   478→                unlistenStoreRef.current?.();
   479→                unlistenStoreRef.current = null;
   480→                unobserveYjsRef.current?.();
   481→                unobserveYjsRef.current = null;
   482→
   483→                // 4. Reset all pending/dirty state for the new page.
   484→                pendingWritesRef.current = new Map();
   485→                pendingDeletesRef.current = new Set();
   486→                localDirtyRef.current = new Set();
   487→
   488→                // 5. Load the new page's state.
   489→                setIsLoadingSnapshot(true);
   490→                loadInitialState();
   491→
   492→                // 6. Re-register listeners for the new page.
   493→                setIsReady(false);
   494→                setTimeout(() => {
   495→                        const editor = editorRef.current;
   496→                        if (!editor) return;
   497→
   498→                        unlistenStoreRef.current = (editor.store as any).listen(
   499→                                (entry: StoreChangeEvent) => {
   500→                                        if (isInitialLoadRef.current) return;
   501→
   502→                                        const { changes } = entry;
   503→                                        localChangeVersionRef.current++;
   504→                                        const version = localChangeVersionRef.current;
   505→
   506→                                        for (const [id, record] of Object.entries(changes.added)) {
   507→                                                pendingWritesRef.current.set(id, record);
   508→                                                pendingDeletesRef.current.delete(id);
   509→                                                localDirtyRef.current.add(id);
   510→                                                recordVersionsRef.current.set(id, version);
   511→                                        }
   512→                                        for (const [id, pair] of Object.entries(changes.updated)) {
   513→                                                pendingWritesRef.current.set(id, pair[1]);
   514→                                                pendingDeletesRef.current.delete(id);
   515→                                                localDirtyRef.current.add(id);
   516→                                                recordVersionsRef.current.set(id, version);
   517→                                        }
   518→                                        for (const id of Object.keys(changes.removed)) {
   519→                                                pendingDeletesRef.current.add(id);
   520→                                                pendingWritesRef.current.delete(id);
   521→                                                localDirtyRef.current.add(id);
   522→                                                recordVersionsRef.current.set(id, version);
   523→                                        }
   524→
   525→                                        scheduleSync();
   526→                                },
   527→                                { source: 'user', scope: 'document' } as any
   528→                        );
   529→
   530→                        unobserveYjsRef.current = setupRemoteObserver();
   531→                        setIsReady(true);
   532→                }, 50);
   533→        }, [pageIndex, ydoc, flushPendingToYjs, loadInitialState, setupRemoteObserver, scheduleSync]);
   534→
   535→        // ----------------------------------------------------------------
   536→        // Cleanup on unmount
   537→        // ----------------------------------------------------------------
   538→
   539→        useEffect(() => {
   540→                return () => {
   541→                        // Cancel pending debounce.
   542→                        if (debounceTimerRef.current) {
   543→                                clearTimeout(debounceTimerRef.current);
   544→                                debounceTimerRef.current = null;
   545→                        }
   546→
   547→                        // Flush any remaining pending changes immediately.
   548→                        flushPendingToYjs();
   549→
   550→                        // Clean up listeners.
   551→                        unlistenStoreRef.current?.();
   552→                        unobserveYjsRef.current?.();
   553→
   554→                        editorRef.current = null;
   555→                };
   556→        }, [flushPendingToYjs]);
   557→
   558→        // ----------------------------------------------------------------
   559→        // Render
   560→        // ----------------------------------------------------------------
   561→
   562→        return (
   563→                <div className="w-full h-full relative">
   564→                        {/* Loading overlay while snapshot is being loaded */}
   565→                        {isLoadingSnapshot && (
   566→                                <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur-sm transition-opacity">
   567→                                        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   568→                                                <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   569→                                                <span>Loading canvas...</span>
   570→                                        </div>
   571→                                </div>
   572→                        )}
   573→
   574→                        {/* Tldraw Editor */}
   575→                        <Tldraw
   576→                                onMount={handleMount}
   577→                                options={{
   578→                                        maxPages: 50,
