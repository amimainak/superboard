   154→   154→    { id: 'billing', label: 'Billing', icon: CreditCard, description: 'Subscription & payments' },

   159→   159→    { group: 'Main', items: common },
   160→   160→    ...(workspace.length ? [{ group: 'Workspace', items: workspace }] : []),

   262→   262→export function AuthenticatedDashboard({ user, userName, tierLoading, isAdmin }: { user: User; userName: string | null; tierLoading: boolean; isAdmin?: boolean }) {
   263→   263→  const { tier, setTier, setRoom } = useAppStore();
   264→   264→  const { brandColor, agencyName, setBrandColor } = useTheme();
   265→   265→  const {
   266→   266→    aiCreditsUsed,
   267→   267→    aiCreditsLimit,
   268→   268→    videoMinutesUsed,
   269→   269→    videoMinutesLimit,
   270→   270→    recordingsUsed,
   271→   271→    recordingsLimit,
   272→   272→    loading: usageLoading,
   273→   273→  } = useCredits(user.id);
   274→   274→
   275→   275→  // ---- State ----
   276→   276→  const [activeView, setActiveView] = useState<DashboardView>('overview');
   277→   277→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   278→   278→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   279→   279→
   280→   280→  // Dialog state
   281→   281→  const [showNewLesson, setShowNewLesson] = useState(false);
   282→   282→  const [showTemplateGallery, setShowTemplateGallery] = useState(false);
   283→   283→  const [selectedSubject, setSelectedSubject] = useState<Subject>('GENERAL');
   284→   284→  const [creating, setCreating] = useState(false);
   285→   285→  const [showSignOutConfirm, setShowSignOutConfirm] = useState(false);
   286→   286→
   287→   287→  // Settings state
   288→   288→  const [saveLoading, setSaveLoading] = useState(false);
   289→   289→  const [saveSuccess, setSaveSuccess] = useState(false);
   290→   290→
   291→   291→  // Onboarding state — show wizard to first-time users (no rooms yet)
   292→   292→  const [showOnboarding, setShowOnboarding] = useState(false);
   293→   293→  const [roomsFetched, setRoomsFetched] = useState(false);
   294→   294→  const [hasRooms, setHasRooms] = useState(false);
   295→   295→
   296→   296→  // Student mode toggle
   297→   297→  const [isStudentMode, setIsStudentMode] = useState(false);
   298→   298→
   299→   299→  const { toast } = useToast();
   300→   300→
   301→   301→  // ---- Check if user has any rooms (for onboarding) ----
   302→   302→  useEffect(() => {
   303→   303→    if (!user?.id) return;
   304→   304→    const checkRooms = async () => {
   305→   305→      try {
   306→   306→        const { authFetch } = await import('@/lib/auth-fetch');
   307→   307→        const res = await authFetch(`/api/room/list?tutorId=${user.id}`);
   308→   308→        if (res.ok) {
   309→   309→          const data = await res.json();
   310→   310→          const rooms = data.rooms ?? [];
   311→   311→          setHasRooms(rooms.length > 0);
   312→   312→          if (rooms.length === 0) {
   313→   313→            setShowOnboarding(true);
   314→   314→          }
   315→   315→        }
   316→   316→      } catch { /* silent — proceed to dashboard */ }
   317→   317→      setRoomsFetched(true);
   318→   318→    };
   319→   319→    checkRooms();
   320→   320→  }, [user?.id]);
   321→   321→
   322→   322→  // ---- Dark mode ----
   323→   323→  const [isDark, setIsDark] = useState(false);
   324→   324→  useEffect(() => {
   325→   325→    const saved = localStorage.getItem('superboard-theme');
   326→   326→    if (saved === 'dark') {
   327→   327→      document.documentElement.classList.add('dark');
   328→   328→      setIsDark(true);
   329→   329→    }
   330→   330→  }, []);
   331→   331→
   332→   332→  const toggleDark = useCallback(() => {
   333→   333→    const next = !isDark;
   334→   334→    setIsDark(next);
   335→   335→    if (next) {
   336→   336→      document.documentElement.classList.add('dark');
   337→   337→    } else {
   338→   338→      document.documentElement.classList.remove('dark');
   339→   339→    }
   340→   340→    localStorage.setItem('superboard-theme', next ? 'dark' : 'light');
   341→   341→  }, [isDark]);
   342→   342→
   343→   343→  // ---- Fingerprint reporting ----
   344→   344→  useEffect(() => {
   345→   345→    import('@/lib/fingerprint').then(({ reportFingerprint }) => {
   346→   346→      if (user?.id) reportFingerprint(user.id).catch(console.error);
   347→   347→    });
   348→   348→  }, [user?.id]);
   349→   349→
   350→   350→  // ---- Create lesson handler ----
   351→   351→  const handleCreateLesson = useCallback(async () => {
   352→   352→    setCreating(true);
   353→   353→    try {
   354→   354→      const supabase = createClient();
   355→   355→      const { data: { session } } = await supabase!.auth.getSession();
   356→   356→      const token = session?.access_token;
   357→   357→      if (!token) { setCreating(false); return; }
   358→   358→      const response = await fetch('/api/room', {
   359→   359→        method: 'POST',
   360→   360→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   361→   361→        body: JSON.stringify({ tutorId: user?.id, subject: selectedSubject }),
   362→   362→      });
   363→   363→      if (!response.ok) {
   364→   364→        const errBody = await response.text().catch(() => '');
   365→   365→        throw new Error(`Failed to create room (${response.status}): ${errBody}`);
   366→   366→      }
   367→   367→      const data = await response.json();
   368→   368→      window.location.href = `/room/${data.roomId}`;
   369→   369→    } catch (error: any) {
   370→   370→      toast({ title: 'Failed to create lesson', description: error?.message || 'Please try again.', variant: 'destructive' });
   371→   371→      setCreating(false);
   372→   372→    }
   373→   373→  }, [selectedSubject, user?.id, toast]);
   374→   374→
   375→   375→  // ---- Save brand color handler ----
   376→   376→  const handleSaveBrandColor = useCallback(async () => {
   377→   377→    setSaveLoading(true);
   378→   378→    setSaveSuccess(false);
   379→   379→    try {
   380→   380→      const supabase = createClient();
   381→   381→      const { data: { session } } = await supabase!.auth.getSession();
   382→   382→      const token = session?.access_token;
   383→   383→      if (!token) return;
   384→   384→      await fetch('/api/auth/profile', {
   385→   385→        method: 'PATCH',
   386→   386→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   387→   387→        body: JSON.stringify({ brandingColor: brandColor || null }),
   388→   388→      });
   389→   389→      setSaveSuccess(true);
   390→   390→      setTimeout(() => setSaveSuccess(false), 2000);
   391→   391→    } catch { /* silent */ }
   392→   392→    setSaveLoading(false);
   393→   393→  }, [brandColor]);
   394→   394→
   395→   395→  // ---- Logout handler ----
   396→   396→  const handleLogout = useCallback(async () => {
   397→   397→    const supabase = createClient();
   398→   398→    if (supabase) await supabase.auth.signOut();
   399→   399→    setTier('FREE');
   400→   400→    window.location.href = '/';
   401→   401→  }, [setTier]);
   402→   402→
   403→   403→  // ---- Onboarding complete handler ----
   404→   404→  const handleOnboardingComplete = useCallback((roomId: string) => {
   405→   405→    setShowOnboarding(false);
   406→   406→    setHasRooms(true);
   407→   407→    window.location.href = `/room/${roomId}`;
   408→   408→  }, []);
   409→   409→
   410→   410→  const handleOnboardingSkip = useCallback(() => {
   411→   411→    setShowOnboarding(false);
   412→   412→    setHasRooms(false);
   413→   413→  }, []);
   414→   414→
   415→   415→  // ---- Template gallery handler ----
   416→   416→  const handleTemplateSelect = useCallback((template: TemplateInfo) => {
   417→   417→    setShowTemplateGallery(false);
   418→   418→    setSelectedSubject(template.subject);
   419→   419→    setShowNewLesson(true);
   420→   420→  }, []);
   421→   421→
   422→   422→  // ---- Derived values ----
   423→   423→  const tierLabel = isAgencyTier(tier)
   424→   424→    ? tier === 'AGENCY_PREMIUM' ? 'Agency Premium'
   425→   425→    : tier === 'AGENCY_STANDARD' ? 'Agency Standard'
   426→   426→    : 'Agency'
   427→   427→    : tier === 'PRO' ? 'Pro' : 'Free';
   428→   428→  const tierColor =
   429→   429→    tier === 'AGENCY_PREMIUM' ? 'bg-purple-100 text-purple-800'
   430→   430→    : tier === 'AGENCY_STANDARD' || tier === 'AGENCY' ? 'bg-amber-100 text-amber-800'
   431→   431→    : tier === 'PRO' ? 'bg-emerald-100 text-emerald-800'
   432→   432→    : 'bg-teal-50 text-teal-700';
   433→   433→
   434→   434→  const navGroups = getNavItems(tier, !!isAdmin);
   435→   435→
   436→   436→  // ---- Show onboarding for first-time users ----
   437→   437→  if (showOnboarding && roomsFetched) {
   438→   438→    return (
   439→   439→      <OnboardingWizard
   440→   440→        userId={user.id}
   441→   441→        userEmail={user.email || ''}
   442→   442→        userName={userName}
   443→   443→        onComplete={handleOnboardingComplete}
   444→   444→        onSkip={handleOnboardingSkip}
   445→   445→      />
   446→   446→    );
   447→   447→  }
   448→   448→
   449→   449→  // ---- Show student dashboard if in student mode ----
   450→   450→  if (isStudentMode) {
   451→   451→    return (
   452→   452→      <StudentDashboard
   453→   453→        userId={user.id}
   454→   454→        userName={userName}
   455→   455→        userEmail={user.email || null}
   456→   456→      />
   457→   457→    );
   458→   458→  }
   459→   459→
   460→   460→  // ============================================================
   461→   461→  // RENDER — Dashboard with Sidebar
   462→   462→  // ============================================================
   463→   463→  return (
   464→   464→    <TooltipProvider delayDuration={0}>
   465→   465→      <div className="min-h-screen bg-background flex">
   466→   466→        {/* ============================================================
   467→   467→            SIDEBAR — Collapsible on desktop, sheet on mobile
   468→   468→            ============================================================ */}
   469→   469→        {/* Mobile overlay */}
   470→   470→        {mobileMenuOpen && (
   471→   471→          <div
   472→   472→            className="fixed inset-0 z-40 bg-black/40 backdrop-blur-sm lg:hidden"
   473→   473→            onClick={() => setMobileMenuOpen(false)}
   474→   474→          />
   475→   475→        )}
   476→   476→
   477→   477→        {/* Sidebar */}
   478→   478→        <aside
   479→   479→          className={`
   480→   480→            fixed lg:sticky top-0 left-0 z-50 lg:z-30
