    78→function getInitials(name: string | null, email: string) {
    79→  if (name) return name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2);

    82→function tierBadgeColor(tier: string) {
    83→  const m: Record<string, string> = { FREE: 'bg-gray-100 text-gray-700', PRO: 'bg-cyan-50 text-cyan-700', AGENCY: 'bg-amber-50 text-amber-700' };
    84→  return m[tier] || m.FREE;

    86→function statusBadge(s: string) {
    87→  const m: Record<string, { bg: string; text: string; icon: any }> = {
    88→    ACTIVE: { bg: 'bg-emerald-50', text: 'text-emerald-700', icon: UserCheck },
    89→    SUSPENDED: { bg: 'bg-amber-50', text: 'text-amber-700', icon: Clock },
    90→    BANNED: { bg: 'bg-red-50', text: 'text-red-700', icon: UserX },
    91→  };
    92→  const v = m[s] || m.ACTIVE;
    93→  return <Badge variant="secondary" className={`${v.bg} ${v.text} border-0`}><v.icon className="w-3 h-3 mr-1" />{s}</Badge>;
    94→}
    95→function Pagination({ page, totalPages, setPage }: { page: number; totalPages: number; setPage: (p: number | ((prev: number) => number)) => void }) {
    96→  return (
    97→    <div className="flex items-center justify-between px-4 py-3 border-t">
    98→      <p className="text-sm text-gray-500">Page {page} of {totalPages}</p>
    99→      <div className="flex items-center gap-1">
   100→        <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage((p: number) => Math.max(1, p - 1))} disabled={page === 1}><ChevronLeft className="w-4 h-4" /></Button>
   101→        <Button variant="outline" size="icon" className="h-8 w-8" onClick={() => setPage((p: number) => Math.min(totalPages, p + 1))} disabled={page === totalPages}><ChevronRight className="w-4 h-4" /></Button>
   102→      </div>
   103→    </div>);
   104→}

   201→function DistributionBars({ items, total, colors }: { items: { key: string; count: number }[]; total: number; colors: Record<string, string> }) {
   202→  if (!items.length) return <p className="text-sm text-gray-400 text-center py-4">No data</p>;
   203→  return <div className="space-y-3">{items.map(item => { const pct = total > 0 ? (item.count / total) * 100 : 0; return (<div key={item.key}><div className="flex items-center justify-between mb-1"><span className="text-sm font-medium text-gray-700 capitalize">{item.key}</span><span className="text-sm text-gray-500">{item.count} ({pct.toFixed(1)}%)</span></div><div className="h-2.5 bg-gray-100 rounded-full overflow-hidden"><div className={`h-full rounded-full transition-all duration-500 ${colors[item.key] || 'bg-gray-400'}`} style={{ width: `${pct}%` }} /></div></div>); })}</div>;
   204→}
   205→
   206→function KPICard({ title, value, subtitle, icon, gradient }: { title: string; value: string | number; subtitle: string; icon: React.ReactNode; gradient: string }) {
   207→  return <Card className="card-hover overflow-hidden"><CardContent className="p-0"><div className="p-5"><div className="flex items-start justify-between"><div><p className="text-xs font-medium text-gray-500 mb-1">{title}</p><p className="text-2xl font-bold text-gray-900">{value}</p><p className="text-xs text-gray-400 mt-0.5">{subtitle}</p></div><div className={`w-9 h-9 rounded-lg ${gradient} flex items-center justify-center text-white`}>{icon}</div></div></div></CardContent></Card>;
   208→}

   213→function UsersTab() {
   214→  const { toast } = useToast();
   215→  const [users, setUsers] = useState<AdminUser[]>([]);
   216→  const [loading, setLoading] = useState(true);
   217→  const [page, setPage] = useState(1); const [totalPages, setTotalPages] = useState(1);
   218→  const [search, setSearch] = useState(''); const [tierFilter, setTierFilter] = useState('');
   219→  const [editUser, setEditUser] = useState<AdminUser | null>(null);
   220→  const [deleteTarget, setDeleteTarget] = useState<AdminUser | null>(null);
   221→  const [showAdd, setShowAdd] = useState(false);
   222→  const [newUser, setNewUser] = useState({ email: '', name: '', tier: 'FREE' });
   223→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   224→
   225→  const loadUsers = useCallback(async () => {
   226→    setLoading(true);
   227→    try {
   228→      const p = new URLSearchParams({ page: page.toString(), limit: '15', search, tier: tierFilter });
   229→      const r = await authFetch(`/api/admin/users?${p}`);
   230→      if (r.ok) { const d = await r.json(); setUsers(d.users); setTotalPages(d.pagination.totalPages); }
   231→    } catch (e) { console.error(e); } finally { setLoading(false); }
   232→  }, [page, search, tierFilter]);
   233→  useEffect(() => { loadUsers(); }, [loadUsers]);
   234→
   235→  async function addUser() {
   236→    const r = await authFetch('/api/admin/users', { method: 'POST', body: JSON.stringify(newUser) });
   237→    if (r.ok) { toast({ title: 'Created', description: `${newUser.email} added.` }); setShowAdd(false); setNewUser({ email: '', name: '', tier: 'FREE' }); loadUsers(); }
   238→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   239→  }
   240→  async function saveUser() {
   241→    if (!editUser) return;
   242→    const r = await authFetch(`/api/admin/users/${editUser.id}`, { method: 'PATCH', body: JSON.stringify({ tier: editUser.tier, name: editUser.name, isAdmin: editUser.isAdmin }) });
   243→    if (r.ok) { toast({ title: 'Updated', description: `${editUser.email} saved.` }); setEditUser(null); loadUsers(); }
   244→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   245→  }
   246→  async function confirmDeleteUser() {
   247→    if (!deleteTarget) return;
   248→    const r = await authFetch(`/api/admin/users/${deleteTarget.id}`, { method: 'DELETE' });
   249→    if (r.ok) { toast({ title: 'Deleted', description: `${deleteTarget.email} removed.` }); setDeleteTarget(null); loadUsers(); }
   250→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   251→  }
   252→  async function changeStatus(userId: string, status: string, reason?: string) {
   253→    const r = await authFetch(`/api/admin/users/${userId}/ban`, { method: 'POST', body: JSON.stringify({ status, reason }) });
   254→    if (r.ok) { toast({ title: status === 'ACTIVE' ? 'Activated' : status === 'SUSPENDED' ? 'Suspended' : 'Banned', description: 'User status updated.' }); loadUsers(); }
   255→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   256→  }
   257→  async function resetPassword(userId: string, email: string) {
   258→    const r = await authFetch(`/api/admin/users/${userId}/reset-password`, { method: 'POST' });
   259→    if (r.ok) toast({ title: 'Password Reset Sent', description: `Reset email sent to ${email}` });
   260→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   261→  }
   262→  async function bulkAction(action: string, tier?: string) {
   263→    const ids = Array.from(selectedIds);
   264→    if (!ids.length) { toast({ title: 'No users selected', variant: 'destructive' }); return; }
   265→    const r = await authFetch('/api/admin/users/bulk', { method: 'POST', body: JSON.stringify({ userIds: ids, action, tier }) });
   266→    if (r.ok) { const d = await r.json(); toast({ title: 'Bulk Action Done', description: d.message }); setSelectedIds(new Set()); loadUsers(); }
   267→    else { const d = await r.json(); toast({ title: 'Error', description: d.error, variant: 'destructive' }); }
   268→  }
   269→
   270→  return (
   271→    <div className="space-y-4">
   272→      {/* Toolbar */}
   273→      <Card><CardContent className="p-4">
   274→        <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
   275→          <div className="flex flex-1 flex-col sm:flex-row gap-2 w-full sm:w-auto">
   276→            <div className="relative flex-1 max-w-xs"><Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" /><Input placeholder="Search users..." className="pl-9" value={search} onChange={e => { setSearch(e.target.value); setPage(1); }} /></div>
   277→            <Select value={tierFilter} onValueChange={v => { setTierFilter(v === 'ALL' ? '' : v); setPage(1); }}><SelectTrigger className="w-32"><SelectValue placeholder="All Tiers" /></SelectTrigger><SelectContent><SelectItem value="ALL">All</SelectItem><SelectItem value="FREE">Free</SelectItem><SelectItem value="PRO">Pro</SelectItem><SelectItem value="AGENCY">Agency</SelectItem></SelectContent></Select>
   278→            {selectedIds.size > 0 && (
   279→              <div className="flex items-center gap-2">
   280→                <Badge variant="secondary">{selectedIds.size} selected</Badge>
   281→                <Select onValueChange={v => bulkAction('changeTier', v)}><SelectTrigger className="w-28 h-8 text-xs"><SelectValue placeholder="Set Tier" /></SelectTrigger><SelectContent><SelectItem value="FREE">Free</SelectItem><SelectItem value="PRO">Pro</SelectItem><SelectItem value="AGENCY">Agency</SelectItem></SelectContent></Select>
   282→                <Button size="sm" variant="outline" className="h-8 text-xs" onClick={() => bulkAction('suspend')}>Suspend</Button>
   283→                <Button size="sm" variant="outline" className="h-8 text-xs" onClick={() => bulkAction('activate')}>Activate</Button>
   284→              </div>
   285→            )}
   286→          </div>
   287→          <Dialog open={showAdd} onOpenChange={setShowAdd}><DialogTrigger asChild><Button className="gradient-primary text-white"><Plus className="w-4 h-4 mr-2" />Add User</Button></DialogTrigger><DialogContent><DialogHeader><DialogTitle>Add User</DialogTitle><DialogDescription>Create a new account.</DialogDescription></DialogHeader><div className="space-y-3 py-2"><div><Label>Email</Label><Input type="email" placeholder="user@example.com" value={newUser.email} onChange={e => setNewUser({ ...newUser, email: e.target.value })} /></div><div><Label>Name</Label><Input placeholder="John Doe" value={newUser.name} onChange={e => setNewUser({ ...newUser, name: e.target.value })} /></div><div><Label>Tier</Label><Select value={newUser.tier} onValueChange={v => setNewUser({ ...newUser, tier: v })}><SelectTrigger><SelectValue /></SelectTrigger><SelectContent><SelectItem value="FREE">Free</SelectItem><SelectItem value="PRO">Pro</SelectItem><SelectItem value="AGENCY">Agency</SelectItem></SelectContent></Select></div></div><DialogFooter><Button variant="outline" onClick={() => setShowAdd(false)}>Cancel</Button><Button className="gradient-primary text-white" onClick={addUser} disabled={!newUser.email}>Create</Button></DialogFooter></DialogContent></Dialog>
   288→        </div>
   289→      </CardContent></Card>
   290→      {/* Table */}
   291→      <Card><CardContent className="p-0">
   292→        {loading ? <div className="p-6 space-y-3">{Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}</div> :
   293→        users.length === 0 ? <div className="text-center py-12 text-gray-500"><Users className="w-12 h-12 mx-auto mb-3 text-gray-300" /><p>No users found</p></div> : (
   294→          <>
   295→            <div className="overflow-x-auto"><Table><TableHeader><TableRow className="bg-gray-50/50"><TableHead className="w-8"></TableHead><TableHead className="w-10"></TableHead><TableHead>User</TableHead><TableHead>Tier</TableHead><TableHead>Status</TableHead><TableHead className="hidden md:table-cell">Rooms</TableHead><TableHead className="hidden lg:table-cell">Joined</TableHead><TableHead className="text-right">Actions</TableHead></TableRow></TableHeader>

   541→  const [saving, setSaving] = useState(false);

   545→      try { const r = await authFetch('/api/admin/config'); if (r.ok) { const d = await r.json(); setMaintenance(d.maintenanceMode); setAnnouncement(d.announcementText || ''); } } catch {} finally { setLoading(false); }

   556→  }
   557→
   558→  if (loading) return <Card><CardContent className="p-6"><Skeleton className="h-4 w-48" /><Skeleton className="h-8 w-24 mt-3" /></CardContent></Card>;

   561→    <div className="space-y-6 max-w-2xl">
   562→      <Card><CardHeader><CardTitle className="text-base flex items-center gap-2"><AlertTriangle className="w-4 h-4 text-amber-500" />Maintenance Mode</CardTitle></CardHeader><CardContent className="space-y-4">
   563→        <div className="flex items-center justify-between p-4 bg-gray-50 rounded-lg"><div><p className="text-sm font-medium">Enable Maintenance Mode</p><p className="text-xs text-gray-500">When on, all users see a maintenance page. Admins can still access the panel.</p></div><Switch checked={maintenance} onCheckedChange={setMaintenance} /></div>
   564→        {maintenance && <div className="p-3 bg-red-50 border border-red-200 rounded-lg"><p className="text-sm text-red-700 font-medium">Maintenance mode is ON. Users cannot access the platform.</p></div>}
   565→      </CardContent></Card>
   566→      <Card><CardHeader><CardTitle className="text-base flex items-center gap-2"><Radio className="w-4 h-4 text-cyan-500" />Announcement Banner</CardTitle></CardHeader><CardContent className="space-y-4">

   568→        {announcement && <div className="p-3 bg-cyan-50 border border-cyan-200 rounded-lg"><p className="text-sm text-cyan-800 font-medium">Preview: {announcement}</p></div>}
   569→      </CardContent></Card>
   570→      <Button className="gradient-primary text-white" onClick={save} disabled={saving}>{saving ? 'Saving...' : 'Save Settings'}</Button>
   571→    </div>
