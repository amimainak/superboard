    89→export default function Dashboard() {
    90→  const { tier, setTier, setRoom } = useAppStore();
    91→  const { brandColor, agencyName, setBrandColor } = useTheme();
    92→  const {
    93→    aiCreditsUsed,
    94→    aiCreditsLimit,
    95→    videoMinutesUsed,
    96→    videoMinutesLimit,
    97→    loading: usageLoading,
    98→  } = useCredits();
    99→
   100→  const [user, setUser] = useState<User | null>(null);
   101→  const [authLoading, setAuthLoading] = useState(true);
   102→
   103→  // --- Auth check on mount ---
   104→  useEffect(() => {
   105→    const supabase = createClient();
   106→    if (!supabase) { setAuthLoading(false); return; }
   107→    supabase.auth.getUser().then(async ({ data: { user } }) => {
   108→      setUser(user);
   109→      if (user) {
   110→        try {
   111→          const tierRes = await fetch(`/api/auth/profile?userId=${user.id}`);
   112→          if (tierRes.ok) {
   113→            const profileData = await tierRes.json();
   114→            if (profileData.tier) setTier(profileData.tier as Tier);
   115→          }
   116→        } catch { /* ignore */ }
   117→      }
   118→      setAuthLoading(false);
   119→    });
   120→    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
   121→      setUser(session?.user ?? null);
   122→    });
   123→    return () => subscription.unsubscribe();
   124→  }, []);
   125→
   126→  // --- Loading state ---
   127→  if (authLoading) {
   128→    return (
   129→      <div className="min-h-screen gradient-auth flex items-center justify-center">
   130→        <div className="flex flex-col items-center gap-4 animate-fade-in-up">
   131→          <div className="w-16 h-16 rounded-2xl gradient-primary flex items-center justify-center shadow-lg shadow-emerald-500/25 animate-pulse-glow">
   132→            <GraduationCap className="w-9 h-9 text-white" />
   133→          </div>
   134→          <p className="text-sm text-muted-foreground font-medium">Loading Superboard...</p>
   135→        </div>
   136→      </div>
   137→    );
   138→  }
   139→
   140→  // --- Not logged in → Show Landing Page ---
   141→  if (!user) {
   142→    return <LandingPage />;
   143→  }
   144→
   145→  // --- Logged in → Show Dashboard ---
   146→  return <AuthenticatedDashboard user={user} />;
   147→}
