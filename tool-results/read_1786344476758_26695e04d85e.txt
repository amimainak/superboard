   741→   741→
   742→   742→function snapToEdge(
   743→   743→  pos: PanelPosition,
   744→   744→  width: number,
   745→   745→  height: number
   746→   746→): PanelPosition {
   747→   747→  const vw = window.innerWidth;
   748→   748→  const vh = window.innerHeight;
   749→   749→
   750→   750→  const centerX = pos.x + width / 2;
   751→   751→  const centerY = pos.y + height / 2;
   752→   752→
   753→   753→  const snapX = centerX < vw / 2 ? EDGE_SNAP : vw - width - EDGE_SNAP;
   754→   754→  const snapY = centerY < vh / 2 ? EDGE_SNAP : vh - height - EDGE_SNAP;
   755→   755→
   756→   756→  return { x: snapX, y: snapY };
   757→   757→}
