   120→interface NavGroup {

   124→
   125→function getNavItems(tier: Tier, isAdmin: boolean): NavGroup[] {
   126→  const common: NavItem[] = [
   127→    { id: 'overview', label: 'Overview', icon: Home, description: 'Dashboard overview' },
   128→    { id: 'lessons', label: 'My Lessons', icon: BookOpen, description: 'Active and past lessons' },
   129→  ];
   130→
   131→  const resources: NavItem[] = isAgencyTier(tier) ? [
   132→    { id: 'resources', label: 'Resources', icon: FolderOpen, description: 'Saved boards & templates' },
   133→  ] : [];
   134→
   135→  const agency: NavItem[] = isAgencyTier(tier) ? [
   136→    { id: 'agency', label: 'Agency', icon: Users, description: 'Sub-tutors & students' },
   137→  ] : [];
   138→
   139→  const account: NavItem[] = [
   140→    { id: 'billing', label: 'Billing', icon: CreditCard, description: 'Subscription & payments' },
   141→    { id: 'settings', label: 'Settings', icon: Settings, description: 'Account preferences' },
   142→  ];
   143→
   144→  return [
   145→    { group: 'Main', items: common },
   146→    ...(resources.length ? [{ group: 'Workspace', items: resources }] : []),
   147→    ...(agency.length ? [{ group: 'Team', items: agency }] : []),
   148→    { group: 'Account', items: account },
   149→  ];
   150→}

   246→// ============================================================
   247→export function AuthenticatedDashboard({ user, userName, tierLoading, isAdmin }: { user: User; userName: string | null; tierLoading: boolean; isAdmin?: boolean }) {
   248→  const { tier, setTier, setRoom } = useAppStore();
   249→  const { brandColor, agencyName, setBrandColor } = useTheme();
   250→  const {
   251→    aiCreditsUsed,
   252→    aiCreditsLimit,
   253→    videoMinutesUsed,
   254→    videoMinutesLimit,
   255→    recordingsUsed,
   256→    recordingsLimit,
   257→    loading: usageLoading,
   258→  } = useCredits(user.id);
   259→
   260→  // ---- State ----
   261→  const [activeView, setActiveView] = useState<DashboardView>('overview');
   262→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   263→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   264→
   265→  // Dialog state
   266→  const [showNewLesson, setShowNewLesson] = useState(false);
   267→  const [selectedSubject, setSelectedSubject] = useState<Subject>('GENERAL');
   268→  const [creating, setCreating] = useState(false);
   269→  const [showSignOutConfirm, setShowSignOutConfirm] = useState(false);
   270→
   271→  // Settings state
   272→  const [saveLoading, setSaveLoading] = useState(false);
   273→  const [saveSuccess, setSaveSuccess] = useState(false);
   274→
   275→  // Onboarding state — show wizard to first-time users (no rooms yet)
   276→  const [showOnboarding, setShowOnboarding] = useState(false);
   277→  const [roomsFetched, setRoomsFetched] = useState(false);
   278→  const [hasRooms, setHasRooms] = useState(false);
   279→
   280→  // Student mode toggle
   281→  const [isStudentMode, setIsStudentMode] = useState(false);
   282→
   283→  const { toast } = useToast();
   284→
   285→  // ---- Check if user has any rooms (for onboarding) ----
   286→  useEffect(() => {
   287→    if (!user?.id) return;
   288→    const checkRooms = async () => {
   289→      try {
   290→        const { authFetch } = await import('@/lib/auth-fetch');
   291→        const res = await authFetch(`/api/room/list?tutorId=${user.id}`);
   292→        if (res.ok) {
   293→          const data = await res.json();
   294→          const rooms = data.rooms ?? [];
   295→          setHasRooms(rooms.length > 0);
   296→          if (rooms.length === 0) {
   297→            setShowOnboarding(true);
   298→          }
   299→        }
   300→      } catch { /* silent — proceed to dashboard */ }
   301→      setRoomsFetched(true);
   302→    };
   303→    checkRooms();
   304→  }, [user?.id]);
   305→
   306→  // ---- Dark mode ----
   307→  const [isDark, setIsDark] = useState(false);
   308→  useEffect(() => {
   309→    const saved = localStorage.getItem('superboard-theme');
   310→    if (saved === 'dark') {
   311→      document.documentElement.classList.add('dark');
   312→      setIsDark(true);
   313→    }
   314→  }, []);
   315→
   316→  const toggleDark = useCallback(() => {
   317→    const next = !isDark;
   318→    setIsDark(next);
   319→    if (next) {
   320→      document.documentElement.classList.add('dark');
   321→    } else {
   322→      document.documentElement.classList.remove('dark');
   323→    }
   324→    localStorage.setItem('superboard-theme', next ? 'dark' : 'light');
   325→  }, [isDark]);
   326→
   327→  // ---- Fingerprint reporting ----
   328→  useEffect(() => {
   329→    import('@/lib/fingerprint').then(({ reportFingerprint }) => {
   330→      if (user?.id) reportFingerprint(user.id).catch(console.error);
   331→    });
   332→  }, [user?.id]);
   333→
   334→  // ---- Create lesson handler ----
   335→  const handleCreateLesson = useCallback(async () => {
   336→    setCreating(true);
   337→    try {
   338→      const supabase = createClient();
   339→      const { data: { session } } = await supabase!.auth.getSession();
   340→      const token = session?.access_token;
   341→      if (!token) { setCreating(false); return; }
   342→      const response = await fetch('/api/room', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   345→        body: JSON.stringify({ tutorId: user?.id, subject: selectedSubject }),
   346→      });
   347→      if (!response.ok) {
   348→        const errBody = await response.text().catch(() => '');
   349→        throw new Error(`Failed to create room (${response.status}): ${errBody}`);
   350→      }
   351→      const data = await response.json();
   352→      window.location.href = `/room/${data.roomId}`;
   353→    } catch (error: any) {
   354→      toast({ title: 'Failed to create lesson', description: error?.message || 'Please try again.', variant: 'destructive' });
   355→      setCreating(false);
   356→    }
   357→  }, [selectedSubject, user?.id, toast]);
   358→
   359→  // ---- Save brand color handler ----
   360→  const handleSaveBrandColor = useCallback(async () => {
   361→    setSaveLoading(true);
   362→    setSaveSuccess(false);
   363→    try {
   364→      const supabase = createClient();
   365→      const { data: { session } } = await supabase!.auth.getSession();
   366→      const token = session?.access_token;
   367→      if (!token) return;
   368→      await fetch('/api/auth/profile', {
   369→        method: 'PATCH',
   370→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   371→        body: JSON.stringify({ brandingColor: brandColor || null }),
   372→      });
   373→      setSaveSuccess(true);
   374→      setTimeout(() => setSaveSuccess(false), 2000);
   375→    } catch { /* silent */ }
   376→    setSaveLoading(false);
   377→  }, [brandColor]);
   378→
   379→  // ---- Logout handler ----
   380→  const handleLogout = useCallback(async () => {
   381→    const supabase = createClient();
   382→    if (supabase) await supabase.auth.signOut();
   383→    setTier('FREE');
   384→    window.location.href = '/';
   385→  }, [setTier]);
   386→
   387→  // ---- Onboarding complete handler ----
   388→  const handleOnboardingComplete = useCallback((roomId: string) => {
   389→    setShowOnboarding(false);
   390→    setHasRooms(true);
   391→    window.location.href = `/room/${roomId}`;
   392→  }, []);
   393→
   394→  const handleOnboardingSkip = useCallback(() => {
   395→    setShowOnboarding(false);
   396→    setHasRooms(false);
   397→  }, []);
   398→
   399→  // ---- Derived values ----
   400→  const tierLabel = isAgencyTier(tier)
   401→    ? tier === 'AGENCY_PREMIUM' ? 'Agency Premium'
   402→    : tier === 'AGENCY_STANDARD' ? 'Agency Standard'
   403→    : 'Agency'
   404→    : tier === 'PRO' ? 'Pro' : 'Free';
   405→  const tierColor =
   406→    tier === 'AGENCY_PREMIUM' ? 'bg-purple-100 text-purple-800'
   407→    : tier === 'AGENCY_STANDARD' || tier === 'AGENCY' ? 'bg-amber-100 text-amber-800'
   408→    : tier === 'PRO' ? 'bg-emerald-100 text-emerald-800'
   409→    : 'bg-teal-50 text-teal-700';
   410→
   411→  const navGroups = getNavItems(tier, !!isAdmin);
   412→
   413→  // ---- Show onboarding for first-time users ----
   414→  if (showOnboarding && roomsFetched) {
   415→    return (
   416→      <OnboardingWizard
   417→        userId={user.id}
   418→        userEmail={user.email || ''}
   419→        userName={userName}
   420→        onComplete={handleOnboardingComplete}
   421→        onSkip={handleOnboardingSkip}
   422→      />
   423→    );
   424→  }
   425→
   426→  // ---- Show student dashboard if in student mode ----
   427→  if (isStudentMode) {
   428→    return (
   429→      <StudentDashboard
   430→        userId={user.id}
   431→        userName={userName}
   432→        userEmail={user.email || null}
   433→      />
   434→    );
   435→  }
   436→
   437→  // ============================================================
   438→  // RENDER — Dashboard with Sidebar
   439→  // ============================================================
   440→  return (
   441→    <TooltipProvider delayDuration={0}>
   442→      <div className="min-h-screen bg-background flex">
   443→        {/* ============================================================
   444→            SIDEBAR — Collapsible on desktop, sheet on mobile
   445→            ============================================================ */}
   446→        {/* Mobile overlay */}
   447→        {mobileMenuOpen && (
   448→          <div
   449→            className="fixed inset-0 z-40 bg-black/40 backdrop-blur-sm lg:hidden"
   450→            onClick={() => setMobileMenuOpen(false)}
   451→          />
   452→        )}
   453→
   454→        {/* Sidebar */}
   455→        <aside
   456→          className={`
   457→            fixed lg:sticky top-0 left-0 z-50 lg:z-30
   458→            h-screen flex flex-col
   459→            bg-white border-r border-gray-200
   460→            transition-all duration-300 ease-in-out
   461→            ${sidebarCollapsed ? 'w-[68px]' : 'w-64'}
   462→            ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full lg:translate-x-0'}
   463→          `}
   464→        >
   465→          {/* Sidebar Header / Logo */}
   466→          <div className="flex items-center justify-between px-4 h-16 border-b border-gray-100 shrink-0">
   467→            <div className="flex items-center gap-2.5 min-w-0">
   468→              <div className="w-9 h-9 rounded-xl gradient-primary flex items-center justify-center shadow-md shadow-emerald-500/20 shrink-0">
   469→                <GraduationCap className="w-5 h-5 text-white" />
   470→              </div>
   471→              {!sidebarCollapsed && (
   472→                <div className="min-w-0">
   473→                  <h1 className="text-sm font-bold tracking-tight truncate">{agencyName || 'Superboard'}</h1>
   474→                </div>
   475→              )}
   476→            </div>
   477→            {/* Mobile close */}
   478→            <Button
   479→              variant="ghost"
   480→              size="icon"
   481→              className="lg:hidden h-8 w-8 shrink-0"
   482→              onClick={() => setMobileMenuOpen(false)}
   483→            >
   484→              <PanelLeftClose className="w-4 h-4" />
   485→            </Button>
   486→          </div>
   487→
   488→          {/* Navigation */}
   489→          <nav className="flex-1 overflow-y-auto py-3 px-3 custom-scrollbar">
   490→            {navGroups.map((group, gi) => (
   491→              <div key={group.group} className={gi > 0 ? 'mt-4' : ''}>
   492→                {!sidebarCollapsed && (
   493→                  <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground px-2 mb-1.5">
   494→                    {group.group}
   495→                  </p>
   496→                )}
   497→                {group.items.map((item) => {
   498→                  const isActive = activeView === item.id;
   499→                  const Icon = item.icon;
   500→                  return (
   501→                    <Tooltip key={item.id}>
   502→                      <TooltipTrigger asChild>
   503→                        <button
   504→                          onClick={() => {
   505→                            setActiveView(item.id);
   506→                            setMobileMenuOpen(false);
   507→                          }}
   508→                          className={`
   509→                            w-full flex items-center gap-2.5 rounded-xl text-sm font-medium
   510→                            transition-all duration-150
   511→                            ${sidebarCollapsed ? 'justify-center px-2 py-2.5' : 'px-3 py-2.5'}
   512→                            ${isActive
   513→                              ? 'bg-emerald-50 text-emerald-700 shadow-sm'
   514→                              : 'text-gray-600 hover:bg-gray-50 hover:text-gray-900'
   515→                            }
   516→                          `}
   517→                        >
   518→                          <Icon className={`w-4.5 h-4.5 shrink-0 ${isActive ? 'text-emerald-600' : 'text-gray-400'}`} />
   519→                          {!sidebarCollapsed && (
   520→                            <span className="truncate">{item.label}</span>
   521→                          )}
   522→                          {isActive && !sidebarCollapsed && (
   523→                            <div className="ml-auto w-1.5 h-1.5 rounded-full bg-emerald-500" />
   524→                          )}
   525→                        </button>
   526→                      </TooltipTrigger>
   527→                      {sidebarCollapsed && (
   528→                        <TooltipContent side="right" className="text-xs font-medium">
   529→                          {item.description}
   530→                        </TooltipContent>
   531→                      )}
   532→                    </Tooltip>
   533→                  );
   534→                })}
   535→              </div>
   536→            ))}
   537→          </nav>
   538→
   539→          {/* Sidebar Footer — Collapse toggle + User info */}
   540→          <div className="border-t border-gray-100 px-3 py-3 shrink-0">
   541→            {/* Collapse toggle (desktop only) */}
   542→            <button
   543→              onClick={() => setSidebarCollapsed(!sidebarCollapsed)}
   544→              className="hidden lg:flex w-full items-center justify-center gap-1.5 px-2 py-2 rounded-lg text-xs text-muted-foreground hover:bg-gray-50 hover:text-gray-600 transition-colors mb-2"
   545→            >
   546→              {sidebarCollapsed ? <PanelLeft className="w-3.5 h-3.5" /> : <PanelLeftClose className="w-3.5 h-3.5" />}
   547→              {!sidebarCollapsed && <span>Collapse</span>}
   548→            </button>
   549→
   550→            {/* User email + tier */}
   551→            <div className={`flex items-center gap-2.5 rounded-xl px-2 py-2 bg-gray-50 ${sidebarCollapsed ? 'justify-center' : ''}`}>
   552→              <div className="w-8 h-8 rounded-full gradient-primary flex items-center justify-center text-white text-xs font-bold shrink-0">
   553→                {(userName || user.email || '?').charAt(0).toUpperCase()}
   554→              </div>
   555→              {!sidebarCollapsed && (
   556→                <div className="min-w-0 flex-1">
   557→                  <p className="text-xs font-medium truncate">{userName || user.email?.split('@')[0]}</p>
   558→                  <p className="text-[10px] text-muted-foreground truncate">{user.email}</p>
   559→                </div>
   560→              )}
   561→            </div>
   562→          </div>
   563→        </aside>
   564→
   565→        {/* ============================================================
