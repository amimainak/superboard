   753→                {/* Google OAuth */}
   754→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   755→                  <GoogleIcon className="w-4 h-4" />

   770→                    <Input id="login-password" type={showLoginPassword ? 'text' : 'password'} placeholder="Your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="rounded-xl pr-10" required />
   771→                    <button
   772→                      type="button"
   773→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   774→                      onClick={() => setShowLoginPassword(!showLoginPassword)}
   775→                      aria-label={showLoginPassword ? 'Hide password' : 'Show password'}

   777→                      {showLoginPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}

   779→                  </div>
   780→                </div>
   781→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   782→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Sign In</Button>
   783→                <div className="text-center space-y-2">
   784→                  <button type="button" className="text-xs text-emerald-600 hover:underline" onClick={() => { setShowForgotPassword(true); setForgotEmail(loginEmail); setAuthError(''); }}>Forgot password?</button>
   785→                  <p className="text-sm text-muted-foreground">

   788→                  </p>

   803→                {/* Google OAuth */}
   804→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   805→                  <GoogleIcon className="w-4 h-4" />
   806→                  Sign up with Google
   807→                </Button>
   808→                <div className="flex items-center gap-3">
   809→                  <div className="flex-1 h-px bg-gray-200" />

   812→                </div>
   813→                <div className="space-y-2">
   814→                  <Label htmlFor="register-name" className="text-sm font-medium">Full Name</Label>
   815→                  <Input id="register-name" type="text" placeholder="Your name" value={registerName} onChange={(e) => setRegisterName(e.target.value)} className="rounded-xl" />
   816→                </div>

   818→                  <Label htmlFor="register-email" className="text-sm font-medium">Email</Label>
   819→                  <Input id="register-email" type="email" placeholder="you@example.com" value={registerEmail} onChange={(e) => setRegisterEmail(e.target.value)} className="rounded-xl" required />
   820→                </div>
   821→                <div className="space-y-2">
   822→                  <Label htmlFor="register-password" className="text-sm font-medium">Password</Label>
   823→                  <div className="relative">
   824→                    <Input id="register-password" type={showRegisterPassword ? 'text' : 'password'} placeholder="Create a strong password" value={registerPassword} onChange={(e) => { setRegisterPassword(e.target.value); setPasswordStrength(evaluatePasswordStrength(e.target.value)); }} className="rounded-xl pr-10" required />
   825→                    <button
   826→                      type="button"
   827→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   828→                      onClick={() => setShowRegisterPassword(!showRegisterPassword)}
   829→                      aria-label={showRegisterPassword ? 'Hide password' : 'Show password'}
   830→                    >
   831→                      {showRegisterPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   832→                    </button>
   833→                  </div>
   834→                  {registerPassword && (
   835→                    <div className="space-y-1">
   836→                      <div className="flex gap-1">
   837→                        {[1,2,3,4,5].map((i) => (
   838→                          <div key={i} className={`h-1 flex-1 rounded-full transition-colors ${i <= passwordStrength.score ? passwordStrength.color : 'bg-gray-200'}`} />
   839→                        ))}
   840→                      </div>
   841→                      <p className={`text-xs ${passwordStrength.score <= 1 ? 'text-red-500' : passwordStrength.score <= 3 ? 'text-amber-500' : 'text-emerald-600'}`}>

   843→                      </p>
   844→                    </div>

   846→                </div>
   847→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   848→                {authMessage && <p className="text-sm text-emerald-600 bg-emerald-50 rounded-lg px-3 py-2">{authMessage}</p>}
   849→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Create Account</Button>
   850→                <p className="text-center text-sm text-muted-foreground">
   851→                  Already have an account?{' '}
   852→                  <button type="button" className="text-emerald-600 font-medium hover:underline" onClick={() => setShowAuth('login')}>Sign in</button>
   853→                </p>
   854→              </form>
   855→            </>
   856→          ) : null}
   857→        </DialogContent>
   858→      </Dialog>
   859→
   860→      {/* ===== EMAIL VERIFICATION SCREEN ===== */}
   861→      {showVerification && (
   862→        <div className="fixed inset-0 z-50 bg-white flex items-center justify-center px-6">

   866→            </div>

   868→            <p className="text-gray-600 leading-relaxed">
   869→              We&apos;ve sent a confirmation link to <strong>{registerEmail}</strong>. 
   870→              Click the link to verify your account and start teaching.
   871→            </p>
   872→            <p className="text-sm text-gray-500">
   873→              Didn&apos;t receive the email? Check your spam folder.
   874→            </p>

   876→              Back to Sign In
   877→            </Button>
   878→          </div>
   879→        </div>
   880→      )}
   881→    </main>
   882→  );
   883→}
