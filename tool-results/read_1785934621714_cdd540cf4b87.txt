    90→export default function Dashboard() {
    91→  const { tier, setTier, setRoom } = useAppStore();
    92→  const { brandColor, agencyName, setBrandColor } = useTheme();
    93→  const {
    94→    aiCreditsUsed,
    95→    aiCreditsLimit,
    96→    videoMinutesUsed,
    97→    videoMinutesLimit,
    98→    loading: usageLoading,
    99→  } = useCredits(null);
   100→
   101→  const [user, setUser] = useState<User | null>(null);
   102→  const [authLoading, setAuthLoading] = useState(true);
   103→
   104→  // --- Auth check on mount ---
   105→  useEffect(() => {
   106→    const supabase = createClient();
   107→    if (!supabase) { setAuthLoading(false); return; }
   108→
   109→    // Initialize auth fetch token caching
   110→    initAuthFetch();
   111→
   112→    let mounted = true;
   113→
   114→    // Listen for auth state changes FIRST to catch any race conditions
   115→    const { data: { subscription } } = supabase.auth.onAuthStateChange(async (_event, session) => {
   116→      if (!mounted) return;
   117→      const user = session?.user ?? null;
   118→      setUser(user);
   119→      if (user) {
   120→        try {
   121→          const tierRes = await authFetch(`/api/auth/profile?userId=${user.id}`);
   122→          if (tierRes.ok) {
   123→            const profileData = await tierRes.json();
   124→            if (profileData.tier) setTier(profileData.tier as Tier);
   125→          }
   126→        } catch { /* ignore */ }
   127→      }
   128→      setAuthLoading(false);
   129→    });
   130→
   131→    // Then check for existing session
   132→    supabase.auth.getUser().then(async ({ data: { user } }) => {
   133→      if (!mounted) return;
   134→      if (user) {
   135→        setUser(user);
   136→        try {
   137→          const tierRes = await authFetch(`/api/auth/profile?userId=${user.id}`);
   138→          if (tierRes.ok) {
   139→            const profileData = await tierRes.json();
   140→            if (profileData.tier) setTier(profileData.tier as Tier);
   141→          }
   142→        } catch { /* ignore */ }
   143→      }
   144→      setAuthLoading(false);
   145→    });
   146→
   147→    return () => {
   148→      mounted = false;
   149→      subscription.unsubscribe();
   150→    };
   151→  }, []);
   152→
   153→  // --- Loading state ---
   154→  if (authLoading) {
   155→    return (
   156→      <div className="min-h-screen bg-white flex items-center justify-center">
   157→        <div className="flex flex-col items-center gap-4 animate-fade-in-up">
   158→          <div className="w-16 h-16 rounded-2xl gradient-primary flex items-center justify-center shadow-lg shadow-emerald-500/25 animate-pulse-glow">
   159→            <GraduationCap className="w-9 h-9 text-white" />
   160→          </div>
   161→          <p className="text-sm text-gray-500 font-medium">Loading Superboard...</p>
   162→        </div>
   163→      </div>
   164→    );
   165→  }
   166→
   167→  // --- Not logged in → Show Landing Page ---
   168→  if (!user) {
   169→    return <LandingPage />;
   170→  }
   171→
   172→  // --- Logged in → Show Dashboard ---
   173→  return <AuthenticatedDashboard user={user} />;
   174→}
