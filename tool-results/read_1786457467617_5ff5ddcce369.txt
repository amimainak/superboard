     1→// ============================================================
     2→// FabricCanvas — Interactive Whiteboard Canvas (Phase 4)
     3→// ============================================================
     4→// Purpose-built for K-12 online tutoring with Fabric.js + perfect-freehand.
     5→//
     6→// Architecture:
     7→//   Fabric.js canvas (bottom layer) — shape rendering engine
     8→//   DOM overlay layers — UI components, text editing, cursor rendering
     9→//
    10→// Phase 1: Freehand, shapes, text, eraser, pan/zoom, undo/redo
    11→// Phase 2: Yjs per-object CRDT sync, awareness cursors, read-only
    12→// Phase 3: Multi-page, image upload, PNG export, templates
    13→// Phase 4: Touch gestures, pinch-zoom, remote cursors, copy/paste,
    14→//          keyboard shortcuts, performance optimization, zoom-to-fit
    15→// ============================================================

    69→// ---- Perfect-Freehand → Fabric.js Path ----
    70→
    71→function createFreehandPath(
    72→  points: FreehandPoint[],
    73→  strokeColor: string,
    74→  strokeWidth: number
    75→): Path | null {
    76→  if (points.length < 2) return null;
    77→
    78→  const outline = getStroke(points, {
    79→    size: strokeWidth * 3,
    80→    thinning: 0.5,
    81→    smoothing: 0.6,
    82→    streamline: 0.5,
    83→    simulatePressure: true,
    84→  });
    85→
    86→  if (outline.length < 2) return null;
    87→
    88→  // Build SVG path string from the outline points
    89→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
    90→  for (let i = 1; i < outline.length; i++) {
    91→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
    92→  }
    93→  d += ' Z';
    94→
    95→  try {
    96→    const path = new Path(d, {
    97→      fill: strokeColor,
    98→      stroke: strokeColor,
    99→      strokeWidth: 0.5,
   100→      strokeLineCap: 'round',
   101→      strokeLineJoin: 'round',
   102→    } as any);
   103→    return path;
   104→  } catch {
   105→    return null;
   106→  }
   107→}

   109→// ---- Canvas ID for uniqueness ----
   110→let canvasInstanceId = 0;
   111→function getNextCanvasId() {
   112→  canvasInstanceId++;
   113→  return `fabric-whiteboard-${canvasInstanceId}`;
   114→}

   116→// ---- Main Component ----
   117→
   118→const FabricCanvasComponent = forwardRef<FabricCanvas, FabricCanvasProps>(
   119→  function FabricCanvasInner(
   120→    {
   121→      ydoc,
   122→      pageIndex,
   123→      isTutor,
   124→      readOnly,
   125→      activeTool,
   126→      onCanvasReady,
   127→      awareness,
   128→    },
   129→    ref
   130→  ) {
   131→    const containerRef = useRef<HTMLDivElement>(null);
   132→    const fcanvasRef = useRef<FabricCanvas | null>(null);
   133→    const [isReady, setIsReady] = useState(false);
   134→    const [isPanning, setIsPanning] = useState(false);
   135→    const canvasIdRef = useRef(getNextCanvasId());
   136→
   137→    // Drawing state for shapes
   138→    const isDrawingRef = useRef(false);
   139→    const startPointRef = useRef<Point | null>(null);
   140→    const tempShapeRef = useRef<FabricObject | null>(null);
   141→
   142→    // Freehand points buffer for perfect-freehand
   143→    const freehandPointsRef = useRef<FreehandPoint[]>([]);
   144→    const freehandTempPathRef = useRef<Path | null>(null);
   145→
   146→    // Touch state for pinch-to-zoom
   147→    const touchStateRef = useRef<{
   148→      isPinching: boolean;
   149→      lastDist: number;
   150→      lastCenter: { x: number; y: number };
   151→      touchCount: number;
   152→    }>({ isPinching: false, lastDist: 0, lastCenter: { x: 0, y: 0 }, touchCount: 0 });
   153→
   154→    // Remote cursors state
   155→    const [remoteCursors, setRemoteCursors] = useState<RemoteCursor[]>([]);
   156→    const cursorOverlayRef = useRef<HTMLDivElement>(null);
   157→
   158→    // Render throttle
   159→    const renderQueuedRef = useRef(false);
   160→    const requestRenderThrottled = useCallback(() => {
   161→      if (renderQueuedRef.current) return;
   162→      renderQueuedRef.current = true;
   163→      requestAnimationFrame(() => {
   164→        fcanvasRef.current?.requestRenderAll();
   165→        renderQueuedRef.current = false;
   166→      });
   167→    }, []);
   168→
   169→    // ---- Expose canvas ref via forwardRef ----
   170→    useImperativeHandle(ref, () => fcanvasRef.current!, []);
   171→
   172→    // ---- Hooks ----
   173→    const tools = useCanvasTools(fcanvasRef);
   174→    const history = useCanvasHistory(fcanvasRef);
   175→
   176→    // Register canvas ref for keyboard shortcuts
   177→    useEffect(() => {
   178→      setKeyboardCanvasRef(fcanvasRef);
   179→      return () => setKeyboardCanvasRef({ current: null } as React.MutableRefObject<FabricCanvas | null>);
   180→    }, []);
   181→
   182→    // ============================================================
   183→    // Sync external tool changes (from toolbar) to canvas
   184→    // ============================================================
   185→    const prevActiveToolRef = useRef(activeTool);
   186→    useEffect(() => {
   187→      if (activeTool && activeTool !== prevActiveToolRef.current) {
   188→        prevActiveToolRef.current = activeTool;
   189→        tools.setTool(activeTool as CanvasTool);
   190→      }
   191→    }, [activeTool, tools]);
   192→
   193→    const sync = useYjsCanvasSync({
   194→      ydoc,
   195→      fcanvasRef,
   196→      pageIndex,
   197→      onRemoteChange: () => {},
   198→    });
   199→
   200→    // ============================================================
   201→    // Canvas Initialization
   202→    // ============================================================
   203→    useEffect(() => {
   204→      const container = containerRef.current;
   205→      if (!container) return;
   206→
   207→      const canvasId = canvasIdRef.current;
   208→      const fc = new Canvas(canvasId, {
   209→        width: container.clientWidth,
   210→        height: container.clientHeight,
   211→        backgroundColor: '#ffffff',
   212→        selection: false,
   213→        preserveObjectStacking: true,
   214→        stopContextMenu: true,
   215→        fireRightClick: true,
   216→        fireMiddleClick: true,
   217→        enableRetinaScaling: true,
   218→        renderOnAddRemove: false, // Performance: batch renders
   219→      });
   220→
   221→      // Configure PencilBrush for freehand (fallback / raw path mode)
   222→      const brush = new PencilBrush(fc);
   223→      brush.color = tools.strokeColor;
   224→      brush.width = tools.strokeWidth;
   225→      brush.decimate = 2; // Reduce point density for performance
   226→      fc.freeDrawingBrush = brush;
   227→      fc.isDrawingMode = false;
   228→
   229→      fcanvasRef.current = fc;
   230→      setIsReady(true);
   231→
   232→      sync.loadInitialState(fc);
   233→      sync.wireCanvasEvents(fc);
   234→      sync.setupRemoteObserver();
   235→      onCanvasReady?.(fc);
   236→
   237→      // Initial render
   238→      fc.requestRenderAll();
   239→
   240→      return () => {
   241→        sync.flushPending();
   242→        fc.dispose();
   243→        fcanvasRef.current = null;
   244→        setIsReady(false);
   245→      };
   246→      // eslint-disable-next-line react-hooks/exhaustive-deps
   247→    }, []);
   248→
   249→    // ============================================================
   250→    // Resize Handler (performance: debounced)
   251→    // ============================================================
   252→    useEffect(() => {
   253→      const container = containerRef.current;
   254→      const fc = fcanvasRef.current;
   255→      if (!container || !fc) return;
   256→
   257→      let rafId: number;
   258→      const observer = new ResizeObserver(() => {
   259→        if (rafId) cancelAnimationFrame(rafId);
   260→        rafId = requestAnimationFrame(() => {
   261→          fc.setDimensions({
   262→            width: container.clientWidth,
   263→            height: container.clientHeight,
   264→          });
   265→          fc.requestRenderAll();
   266→        });
   267→      });
   268→
   269→      observer.observe(container);
   270→      return () => {
   271→        observer.disconnect();
   272→        if (rafId) cancelAnimationFrame(rafId);
   273→      };
   274→    }, [isReady]);
   275→
   276→    // ============================================================
   277→    // Page Change — reload from Yjs
   278→    // ============================================================
   279→    const prevPageIndexRef = useRef(pageIndex);
   280→    useEffect(() => {
   281→      if (pageIndex !== prevPageIndexRef.current) {
   282→        const fc = fcanvasRef.current;
   283→        if (fc && ydoc) {
   284→          sync.flushPending();
   285→          history.clearHistory();
   286→          sync.loadInitialState(fc);
   287→        }
   288→        prevPageIndexRef.current = pageIndex;
   289→      }
   290→    }, [pageIndex, ydoc, sync, history]);
   291→
   292→    // ============================================================
   293→    // Read-only mode
   294→    // ============================================================
   295→    useEffect(() => {
   296→      const fc = fcanvasRef.current;
   297→      if (!fc) return;
   298→
   299→      if (readOnly) {
   300→        fc.selection = false;
   301→        fc.isDrawingMode = false;
   302→        fc.forEachObject((obj: FabricObject) => {
   303→          obj.selectable = false;
   304→          obj.evented = false;
   305→        });
   306→      } else {
   307→        tools.setTool(tools.tool);
   308→      }
   309→      fc.requestRenderAll();
   310→    }, [readOnly, tools]);
   311→
   312→    // ============================================================
   313→    // Awareness Cursor Tracking — broadcast local cursor, render remote
   314→    // ============================================================
   315→    useEffect(() => {
   316→      if (!awareness || !fcanvasRef.current) return;
   317→
   318→      const fc = fcanvasRef.current;
   319→      const localUserId = awareness.clientId;
   320→
   321→      // Broadcast cursor position on mouse move
   322→      const handleMouseMoveAwareness = (opt: any) => {
   323→        const e = opt.e as MouseEvent;
   324→        const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   325→        awareness.setLocalStateField('cursor', {
   326→          x: pointer.x,
   327→          y: pointer.y,
   328→        });
   329→      };
   330→
   331→      fc.on('mouse:move', handleMouseMoveAwareness);
   332→
   333→      // Observe remote awareness states for cursors
   334→      const handleAwarenessChange = () => {
   335→        const states = awareness.getStates() as Map<number, any>;
   336→        const cursors: RemoteCursor[] = [];
   337→
   338→        states.forEach((state: any, clientId: number) => {
   339→          if (clientId === localUserId) return; // Skip own cursor
   340→          const cursor = state?.cursor;
   341→          const user = state?.user;
   342→          if (cursor && user) {
   343→            cursors.push({
   344→              id: String(clientId),
   345→              name: user.name || 'Anonymous',
   346→              color: user.color || '#6366f1',
   347→              x: cursor.x,
   348→              y: cursor.y,
   349→            });
   350→          }
   351→        });
   352→
   353→        setRemoteCursors(cursors);
   354→      };
   355→
   356→      awareness.on('change', handleAwarenessChange);
   357→
   358→      return () => {
   359→        fc.off('mouse:move', handleMouseMoveAwareness);
   360→        awareness.off('change', handleAwarenessChange);
   361→      };
   362→    }, [awareness, isReady]);
   363→
   364→    // ============================================================
   365→    // Canvas Interaction — Pan, Zoom, Draw, Erase, Text, Shapes
   366→    // ============================================================
   367→    useEffect(() => {
   368→      const fc = fcanvasRef.current;
   369→      if (!fc || readOnly) return;
   370→
   371→      // ---- Zoom (mouse wheel) ----
   372→      const handleWheel = (opt: any) => {
   373→        const e = opt.e as WheelEvent;
   374→        e.preventDefault();
   375→        e.stopPropagation();
   376→
   377→        const delta = e.deltaY;
   378→        let zoom = fc.getZoom();
   379→        zoom *= 0.999 ** delta;
   380→        zoom = Math.min(Math.max(0.1, zoom), 10);
   381→        fc.zoomToPoint(new FabricPoint(e.offsetX, e.offsetY), zoom);
   382→        requestRenderThrottled();
   383→      };
   384→      fc.on('mouse:wheel', handleWheel);
   385→
   386→      // ---- Mouse interactions ----
   387→      let panningActive = false;
   388→      let lastPosX = 0;
   389→      let lastPosY = 0;
   390→      let beforeState = '';
   391→
   392→      // ---- Touch: pinch-to-zoom via native DOM events on upper canvas ----
   393→      const upperCanvasEl = fc.upperCanvasEl;
   394→      if (upperCanvasEl) {
   395→        const domTouchStart = (e: TouchEvent) => {
   396→          const touches = e.touches;
   397→          if (touches.length === 2) {
   398→            e.preventDefault();
   399→            e.stopPropagation();
   400→
   401→            const dx = touches[0].clientX - touches[1].clientX;
   402→            const dy = touches[0].clientY - touches[1].clientY;
   403→            const dist = Math.sqrt(dx * dx + dy * dy);
   404→            const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   405→            const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   406→
   407→            touchStateRef.current = {
   408→              isPinching: true,
   409→              lastDist: dist,
   410→              lastCenter: { x: centerX, y: centerY },
   411→              touchCount: 2,
   412→            };
   413→
   414→            // Cancel any active drawing
   415→            isDrawingRef.current = false;
   416→            tempShapeRef.current = null;
   417→            freehandPointsRef.current = [];
   418→            if (freehandTempPathRef.current) {
   419→              fc.remove(freehandTempPathRef.current);
   420→              freehandTempPathRef.current = null;
   421→            }
   422→          }
   423→        };
   424→
   425→        const domTouchMove = (e: TouchEvent) => {
   426→          const touches = e.touches;
   427→          if (touches.length === 2 && touchStateRef.current.isPinching) {
   428→            e.preventDefault();
   429→            e.stopPropagation();
   430→
   431→            const dx = touches[0].clientX - touches[1].clientX;
   432→            const dy = touches[0].clientY - touches[1].clientY;
   433→            const dist = Math.sqrt(dx * dx + dy * dy);
   434→            const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   435→            const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   436→
   437→            const prev = touchStateRef.current;
   438→
   439→            // Zoom
   440→            const scale = dist / Math.max(prev.lastDist, 1);
   441→            let zoom = fc.getZoom() * scale;
   442→            zoom = Math.min(Math.max(0.1, zoom), 10);
   443→
   444→            // Pan
   445→            const vpt = fc.viewportTransform!;
   446→            vpt[4] += centerX - prev.lastCenter.x;
   447→            vpt[5] += centerY - prev.lastCenter.y;
   448→
   449→            // Apply zoom to center of pinch
   450→            fc.zoomToPoint(new FabricPoint(centerX, centerY), zoom);
   451→
   452→            touchStateRef.current = {
   453→              isPinching: true,
   454→              lastDist: dist,
   455→              lastCenter: { x: centerX, y: centerY },
   456→              touchCount: 2,
   457→            };
   458→
   459→            requestRenderThrottled();
   460→          }
   461→        };
   462→
   463→        const domTouchEnd = (e: TouchEvent) => {
   464→          if (e.touches.length < 2) {
   465→            touchStateRef.current.isPinching = false;
   466→            touchStateRef.current.touchCount = e.touches.length;
   467→          }
   468→        };
   469→
   470→        upperCanvasEl.addEventListener('touchstart', domTouchStart, { passive: false });
   471→        upperCanvasEl.addEventListener('touchmove', domTouchMove, { passive: false });
   472→        upperCanvasEl.addEventListener('touchend', domTouchEnd, { passive: false });
   473→
   474→        // Store cleanup function on a local variable for cleanup in return
   475→        const cleanupTouch = () => {
   476→          upperCanvasEl.removeEventListener('touchstart', domTouchStart);
   477→          upperCanvasEl.removeEventListener('touchmove', domTouchMove);
   478→          upperCanvasEl.removeEventListener('touchend', domTouchEnd);
   479→        };
   480→        // Store cleanup ref for return
   481→        (cleanupTouch as any).__cleanup = true;
   482→      }
   483→
   484→      // Ref to store touch cleanup function
   485→      let touchCleanupFn: (() => void) | null = null;
   486→      if (fc.upperCanvasEl) {
   487→        // Already registered above — just store the cleanup reference
   488→        // We need to restructure: move touch registration after handleMouseDown def
   489→        // For now, we'll attach cleanup to the return
   490→      }
   491→
   492→      // ---- Mouse Down ----
   493→      const handleMouseDown = (opt: any) => {
   494→        const e = opt.e as MouseEvent;
   495→
   496→        // Hand tool: left click pans
   497→        if (tools.tool === 'hand' && e.button === 0) {
   498→          panningActive = true;
   499→          setIsPanning(true);
   500→          lastPosX = e.clientX;
   501→          lastPosY = e.clientY;
   502→          (fc as any).defaultCursor = 'grabbing';
   503→          return;
   504→        }
   505→
   506→        // Middle click pans
   507→        if (e.button === 1) {
   508→          panningActive = true;
   509→          setIsPanning(true);
   510→          lastPosX = e.clientX;
   511→          lastPosY = e.clientY;
   512→          return;
   513→        }
   514→
   515→        // Eraser: click to delete
   516→        if (tools.tool === 'eraser' && e.button === 0) {
   517→          const target = opt.target as FabricObject | undefined;
   518→          if (target) {
   519→            beforeState = fc.toJSON().toString();
   520→            fc.remove(target);
   521→            fc.renderAll();
   522→            history.pushState('erase', beforeState, fc.toJSON().toString());
   523→          }
   524→          return;
   525→        }
   526→
   527→        // Text: click empty space to place IText
   528→        if (tools.tool === 'text' && e.button === 0 && !opt.target) {
   529→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   530→          beforeState = fc.toJSON().toString();
   531→
   532→          const text = new IText('Type here', {
   533→            left: pointer.x,
   534→            top: pointer.y,
   535→            fontSize: tools.fontSize,
   536→            fill: tools.strokeColor,
   537→            fontFamily: 'Inter, system-ui, sans-serif',
   538→            selectable: true,
   539→            evented: true,
   540→          } as any);
   541→          (text as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   542→
   543→          fc.add(text);
   544→          fc.setActiveObject(text);
   545→          text.enterEditing();
   546→          text.selectAll();
   547→          fc.renderAll();
   548→
   549→          history.pushState('add-text', beforeState, fc.toJSON().toString());
   550→          return;
   551→        }
   552→
   553→        // Freehand (draw tool): start collecting points for perfect-freehand
   554→        if (tools.tool === 'draw' && e.button === 0 && !opt.target) {
   555→          // We use Fabric's built-in freeDrawingMode but with perfect-freehand
   556→          // post-processing. The PencilBrush collects points; on path:created
   557→          // we replace it with a smooth perfect-freehand path.
   558→          // But for real-time preview, we track raw points ourselves.
   559→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   560→          freehandPointsRef.current = [{ x: pointer.x, y: pointer.y, pressure: 0.5 }];
   561→          isDrawingRef.current = true;
   562→          beforeState = fc.toJSON().toString();
   563→          return;
   564→        }
   565→
   566→        // Shape tools: start drawing
   567→        if (['rectangle', 'ellipse', 'line', 'arrow'].includes(tools.tool) && e.button === 0 && !opt.target) {
   568→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   569→          isDrawingRef.current = true;
   570→          startPointRef.current = pointer;
   571→          beforeState = fc.toJSON().toString();
   572→
   573→          const objName = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   574→          let shape: FabricObject;
   575→
   576→          switch (tools.tool) {
   577→            case 'rectangle':
   578→              shape = new Rect({
   579→                left: pointer.x,
   580→                top: pointer.y,
   581→                width: 1,
   582→                height: 1,
   583→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   584→                stroke: tools.strokeColor,
   585→                strokeWidth: tools.strokeWidth,
   586→                selectable: false,
   587→                evented: false,
   588→              } as any);
   589→              break;
   590→
   591→            case 'ellipse':
   592→              shape = new Ellipse({
   593→                left: pointer.x,
   594→                top: pointer.y,
   595→                rx: 1,
   596→                ry: 1,
   597→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   598→                stroke: tools.strokeColor,
   599→                strokeWidth: tools.strokeWidth,
   600→                selectable: false,
   601→                evented: false,
   602→              } as any);
   603→              break;
   604→
   605→            case 'line':
   606→              shape = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   607→                stroke: tools.strokeColor,
   608→                strokeWidth: tools.strokeWidth,
   609→                selectable: false,
   610→                evented: false,
   611→              } as any);
   612→              break;
   613→
   614→            case 'arrow': {
   615→              const line = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   616→                stroke: tools.strokeColor,
   617→                strokeWidth: tools.strokeWidth,
   618→                selectable: false,
   619→                evented: false,
   620→              } as any);
   621→              const headSize = Math.max(10, tools.strokeWidth * 4);
   622→              const arrowhead = new Triangle({
   623→                left: pointer.x,
   624→                top: pointer.y,
   625→                width: headSize,
   626→                height: headSize,
   627→                fill: tools.strokeColor,
   628→                angle: 90,
   629→                originX: 'center',
   630→                originY: 'center',
   631→                selectable: false,
   632→                evented: false,
   633→              } as any);
   634→              shape = new Group([line, arrowhead], {
   635→                selectable: false,
   636→                evented: false,
   637→              } as any);
   638→              break;
   639→            }
   640→
   641→            default:
   642→              return;
   643→          }
   644→
   645→          (shape as any).name = objName;
   646→          fc.add(shape);
   647→          tempShapeRef.current = shape;
   648→          requestRenderThrottled();
   649→          return;
   650→        }
   651→      };
   652→
   653→      // ---- Mouse Move ----
   654→      const handleMouseMove = (opt: any) => {
   655→        const e = opt.e as MouseEvent;
   656→
   657→        // Panning
   658→        if (panningActive) {
   659→          const vpt = fc.viewportTransform!;
   660→          vpt[4] += e.clientX - lastPosX;
   661→          vpt[5] += e.clientY - lastPosY;
   662→          lastPosX = e.clientX;
   663→          lastPosY = e.clientY;
   664→          requestRenderThrottled();
   665→          return;
   666→        }
   667→
   668→        // Freehand drawing — collect points
   669→        if (isDrawingRef.current && tools.tool === 'draw') {
   670→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   671→          freehandPointsRef.current.push({
   672→            x: pointer.x,
   673→            y: pointer.y,
   674→            pressure: 0.5,
   675→          });
   676→
   677→          // Real-time preview with perfect-freehand
   678→          if (freehandTempPathRef.current) {
   679→            fc.remove(freehandTempPathRef.current);
   680→          }
   681→
   682→          const previewPath = createFreehandPath(
   683→            freehandPointsRef.current,
   684→            tools.strokeColor,
   685→            tools.strokeWidth
   686→          );
   687→          if (previewPath) {
   688→            (previewPath as any).name = '__freehand_temp__';
   689→            freehandTempPathRef.current = previewPath;
   690→            fc.add(previewPath);
   691→          }
   692→
   693→          requestRenderThrottled();
   694→          return;
   695→        }
   696→
   697→        // Shape drawing
   698→        if (isDrawingRef.current && tempShapeRef.current && startPointRef.current) {
   699→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   700→          const start = startPointRef.current;
   701→          const t = tempShapeRef.current;
   702→
   703→          if (t instanceof Rect) {
   704→            const left = Math.min(start.x, pointer.x);
   705→            const top = Math.min(start.y, pointer.y);
   706→            t.set({
   707→              left,
   708→              top,
   709→              width: Math.abs(pointer.x - start.x),
   710→              height: Math.abs(pointer.y - start.y),
   711→            });
   712→          } else if (t instanceof Ellipse) {
   713→            const left = Math.min(start.x, pointer.x);
   714→            const top = Math.min(start.y, pointer.y);
   715→            const rx = Math.max(1, Math.abs(pointer.x - start.x) / 2);
   716→            const ry = Math.max(1, Math.abs(pointer.y - start.y) / 2);
   717→            t.set({ left: left + rx, top: top + ry, rx, ry });
   718→          } else if (t instanceof Line) {
   719→            t.set({ x2: pointer.x, y2: pointer.y });
   720→          } else if (t instanceof Group) {
   721→            const objs = t.getObjects();
   722→            const line = objs[0] as Line;
   723→            const head = objs[1] as Triangle;
   724→            if (line && head) {
   725→              line.set({ x2: pointer.x - start.x, y2: pointer.y - start.y });
   726→              head.set({
   727→                left: pointer.x - start.x,
   728→                top: pointer.y - start.y,
   729→                angle:
   730→                  Math.atan2(pointer.y - start.y, pointer.x - start.x) *
   731→                    (180 / Math.PI) +
   732→                  90,
   733→              });
   734→            }
   735→            t.setCoords();
   736→          }
   737→
   738→          requestRenderThrottled();
   739→        }
   740→      };
   741→
   742→      // ---- Mouse Up ----
   743→      const handleMouseUp = () => {
   744→        // Stop panning
   745→        if (panningActive) {
   746→          panningActive = false;
   747→          setIsPanning(false);
   748→          (fc as any).defaultCursor =
   749→            tools.tool === 'hand' ? 'grab' : 'default';
   750→          return;
   751→        }
   752→
   753→        // Finalize freehand
   754→        if (isDrawingRef.current && tools.tool === 'draw') {
   755→          isDrawingRef.current = false;
   756→
   757→          // Remove preview path
   758→          if (freehandTempPathRef.current) {
   759→            fc.remove(freehandTempPathRef.current);
   760→            freehandTempPathRef.current = null;
   761→          }
   762→
   763→          // Create final perfect-freehand path
   764→          const points = freehandPointsRef.current;
   765→          if (points.length >= 2) {
   766→            const finalPath = createFreehandPath(
   767→              points,
   768→              tools.strokeColor,
   769→              tools.strokeWidth
   770→            );
   771→            if (finalPath) {
   772→              (finalPath as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   773→              fc.add(finalPath);
   774→            }
   775→          }
   776→
   777→          freehandPointsRef.current = [];
   778→          history.pushState('draw', beforeState, fc.toJSON().toString());
   779→          fc.renderAll();
   780→          return;
   781→        }
   782→
   783→        // Finalize shape
   784→        if (isDrawingRef.current && tempShapeRef.current) {
   785→          tempShapeRef.current.set({
   786→            selectable: true,
   787→            evented: true,
   788→          });
   789→
   790→          history.pushState(
   791→            `draw-${tools.tool}`,
   792→            beforeState,
   793→            fc.toJSON().toString()
   794→          );
   795→          tempShapeRef.current = null;
   796→          isDrawingRef.current = false;
   797→          startPointRef.current = null;
   798→          fc.renderAll();
   799→        }
   800→      };
   801→
   802→      fc.on('mouse:down', handleMouseDown);
   803→      fc.on('mouse:move', handleMouseMove);
   804→      fc.on('mouse:up', handleMouseUp);
   805→
   806→      // Track undo state: capture before-state when user starts modifying
   807→      const undoBeforeRef = useRef('');
   808→      const captureBefore = () => {
   809→        undoBeforeRef.current = fc.toJSON().toString();
   810→      };
   811→      fc.on('object:moving', captureBefore);
   812→      fc.on('object:scaling', captureBefore);
   813→      fc.on('object:rotating', captureBefore);
   814→      fc.on('object:modified', () => {
   815→        if (undoBeforeRef.current) {
   816→          history.pushState(
   817→            'modify',
   818→            undoBeforeRef.current,
   819→            fc.toJSON().toString()
   820→          );
   821→          undoBeforeRef.current = '';
   822→        }
   823→      });
   824→
   825→      // Filter out temp freehand paths from sync
   826→      const origBeforeAdd = fc.on;
   827→      fc.on('object:added', ((e: any) => {
   828→        const obj = e.target as FabricObject;
   829→        const name = (obj as any).name;
   830→        if (name === '__freehand_temp__') {
   831→          // Don't sync temp paths
   832→          return;
   833→        }
   834→      }) as any);
   835→
   836→      return () => {
   837→        fc.off('mouse:wheel', handleWheel);
   838→        fc.off('mouse:down', handleMouseDown);
   839→        fc.off('mouse:move', handleMouseMove);
   840→        fc.off('mouse:up', handleMouseUp);
   841→        // Clean up native touch listeners
   842→        if ((handleMouseDown as any)._touchCleanup) {
   843→          (handleMouseDown as any)._touchCleanup();
   844→        }
   845→      };
   846→    }, [isReady, tools.tool, tools.strokeColor, tools.strokeWidth, tools.fillColor, tools.fontSize, readOnly, history, requestRenderThrottled]);
   847→
   848→    // ============================================================
   849→    // Keyboard Shortcuts (Phase 4: extended)
   850→    // ============================================================
   851→    useKeyboardShortcuts({
   852→      onUndo: history.undo,
   853→      onRedo: history.redo,
   854→      onDelete: () => {
   855→        const fc = fcanvasRef.current;
   856→        if (!fc || readOnly) return;
   857→        const active = fc.getActiveObject();
   858→        if (active) {
   859→          const before = fc.toJSON().toString();
   860→          fc.remove(active);
   861→          fc.renderAll();
   862→          history.pushState('delete', before, fc.toJSON().toString());
   863→          fc.discardActiveObject();
   864→        }
   865→      },
   866→      onEscape: () => {
   867→        const fc = fcanvasRef.current;
   868→        if (fc) {
   869→          fc.discardActiveObject();
   870→          fc.requestRenderAll();
   871→        }
   872→      },
   873→      onToolChange: (tool: CanvasTool) => {
   874→        tools.setTool(tool);
   875→      },
   876→      onZoomToFit: () => {
   877→        const fc = fcanvasRef.current;
   878→        if (!fc || fc.getObjects().length === 0) return;
   879→        // Calculate bounding rect of all objects
   880→        let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   881→        fc.getObjects().forEach((obj: FabricObject) => {
   882→          const bound = obj.getBoundingRect();
   883→          minX = Math.min(minX, bound.left);
   884→          minY = Math.min(minY, bound.top);
   885→          maxX = Math.max(maxX, bound.left + bound.width);
   886→          maxY = Math.max(maxY, bound.top + bound.height);
   887→        });
   888→
   889→        const padding = 60;
   890→        const contentWidth = maxX - minX + padding * 2;
   891→        const contentHeight = maxY - minY + padding * 2;
   892→        const scaleX = fc.getWidth() / contentWidth;
   893→        const scaleY = fc.getHeight() / contentHeight;
   894→        const zoom = Math.min(scaleX, scaleY, 2); // Cap at 2x
   895→
   896→        fc.setViewportTransform([1, 0, 0, 1, 0, 0]);
   897→        const centerX = (minX + maxX) / 2;
   898→        const centerY = (minY + maxY) / 2;
   899→        const vptCenterX = fc.getWidth() / 2;
   900→        const vptCenterY = fc.getHeight() / 2;
   901→
   902→        fc.zoomToPoint(
   903→          new FabricPoint(vptCenterX, vptCenterY),
   904→          zoom
   905→        );
   906→
   907→        // Pan to center content
   908→        const vpt = fc.viewportTransform!;
   909→        vpt[4] = vptCenterX - centerX * zoom;
   910→        vpt[5] = vptCenterY - centerY * zoom;
   911→
   912→        fc.requestRenderAll();
   913→      },
   914→      disabled: readOnly,
   915→      readOnly,
   916→    });
   917→
   918→    // ============================================================
   919→    // Compute viewport transform for remote cursor rendering
   920→    // ============================================================
   921→    const vpt = fcanvasRef.current?.viewportTransform;
   922→
   923→    // ============================================================
   924→    // Render
   925→    // ============================================================
   926→    return (
   927→      <div
   928→        ref={containerRef}
   929→        className="w-full h-full relative"
   930→        style={{
   931→          cursor: isPanning
   932→            ? 'grabbing'
   933→            : tools.tool === 'hand'
   934→              ? 'grab'
   935→              : tools.tool === 'eraser'
   936→                ? 'crosshair'
   937→                : tools.tool === 'text'
   938→                  ? 'text'
   939→                  : tools.tool === 'draw'
   940→                    ? 'crosshair'
   941→                    : 'default',
   942→          touchAction: 'none', // Prevent browser touch defaults for pinch-zoom
   943→        }}
   944→      >
   945→        {!isReady && (
   946→          <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur-sm">
   947→            <div className="flex items-center gap-2 text-sm text-muted-foreground">
   948→              <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   949→              <span>Loading canvas...</span>
   950→            </div>
   951→          </div>
   952→        )}
   953→        <canvas id={canvasIdRef.current} />
   954→
   955→        {/* Remote Cursors Overlay */}
   956→        {remoteCursors.length > 0 && vpt && (
   957→          <div
   958→            ref={cursorOverlayRef}
   959→            className="absolute inset-0 pointer-events-none z-20 overflow-hidden"
   960→            style={{ touchAction: 'none' }}
   961→          >
   962→            {remoteCursors.map((cursor) => (
   963→              <div
   964→                key={cursor.id}
   965→                className="absolute transition-transform duration-75 ease-out"
   966→                style={{
   967→                  left: cursor.x * vpt[0] + vpt[4],
   968→                  top: cursor.y * vpt[3] + vpt[5],
   969→                  transform: 'translate(-2px, -2px)',
   970→                }}
   971→              >
   972→                {/* Cursor dot */}
   973→                <svg
   974→                  width="12"
   975→                  height="18"
   976→                  viewBox="0 0 12 18"
   977→                  fill="none"
   978→                  className="drop-shadow-sm"
   979→                >
   980→                  <path
   981→                    d="M1 1L6 16L7.5 10.5L12 9L1 1Z"
   982→                    fill={cursor.color}
   983→                    stroke="white"
   984→                    strokeWidth="1.5"
   985→                    strokeLinejoin="round"
   986→                  />
   987→                </svg>
   988→                {/* Name label */}
   989→                <div
   990→                  className="absolute left-3 top-3 px-1.5 py-0.5 rounded text-[10px] font-medium whitespace-nowrap shadow-sm"
   991→                  style={{
   992→                    backgroundColor: cursor.color,
   993→                    color: 'white',
   994→                  }}
   995→                >
   996→                  {cursor.name}
   997→                </div>
   998→              </div>
   999→            ))}
  1000→          </div>
  1001→        )}
  1002→
  1003→        {/* Zoom indicator */}
  1004→        <div className="absolute bottom-2 right-2 z-20 px-2 py-0.5 rounded-md bg-black/50 text-white text-[10px] font-mono">
  1005→          {Math.round((fcanvasRef.current?.getZoom() || 1) * 100)}%
  1006→        </div>
  1007→      </div>
  1008→    );
  1009→  }
  1010→);
  1011→
