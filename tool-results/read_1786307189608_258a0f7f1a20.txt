   717→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   718→                  <GoogleIcon className="w-4 h-4" />
   719→                  Sign up with Google
   720→                </Button>
   721→                <div className="flex items-center gap-3">
   722→                  <div className="flex-1 h-px bg-gray-200" />
   723→                  <span className="text-xs text-gray-400 font-medium">or register with email</span>
   724→                  <div className="flex-1 h-px bg-gray-200" />
   725→                </div>

   727→                  <Label htmlFor="register-name" className="text-sm font-medium">Full Name</Label>

   737→                  {registerPassword && (
   738→                    <div className="space-y-1">
   739→                      <div className="flex gap-1">
   740→                        {[1,2,3,4,5].map((i) => (
   741→                          <div key={i} className={`h-1 flex-1 rounded-full transition-colors ${i <= passwordStrength.score ? passwordStrength.color : 'bg-gray-200'}`} />
   742→                        ))}
   743→                      </div>
   744→                      <p className={`text-xs ${passwordStrength.score <= 1 ? 'text-red-500' : passwordStrength.score <= 3 ? 'text-amber-500' : 'text-emerald-600'}`}>
   745→                        {passwordStrength.label}
   746→                      </p>
   747→                    </div>
   748→                  )}
   749→                </div>
   750→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   751→                {authMessage && <p className="text-sm text-emerald-600 bg-emerald-50 rounded-lg px-3 py-2">{authMessage}</p>}
   752→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Create Account</Button>
   753→                <p className="text-center text-sm text-muted-foreground">
   754→                  Already have an account?{' '}

   756→                </p>
   757→              </form>
   758→            </>
   759→          ) : null}
   760→        </DialogContent>
   761→      </Dialog>
   762→
   763→      {/* ===== EMAIL VERIFICATION SCREEN ===== */}
   764→      {showVerification && (
   765→        <div className="fixed inset-0 z-50 bg-white flex items-center justify-center px-6">
   766→          <div className="max-w-md w-full text-center space-y-6">

   769→            </div>

   771→            <p className="text-gray-600 leading-relaxed">

   774→            </p>

   779→              Back to Sign In
   780→            </Button>
   781→          </div>
   782→        </div>
   783→      )}
   784→    </main>
   785→  );
   786→}
