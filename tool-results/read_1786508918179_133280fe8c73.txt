    40→function randInt(min: number, max: number): number {
    41→  return Math.floor(Math.random() * (max - min + 1)) + min;
    42→}
    43→
    44→function pick<T>(arr: T[]): T {
    45→  return arr[Math.floor(Math.random() * arr.length)];
    46→}
    47→
    48→function shuffle<T>(arr: T[]): T[] {
    49→  const result = [...arr];
    50→  for (let i = result.length - 1; i > 0; i--) {
    51→    const j = Math.floor(Math.random() * (i + 1));
    52→    [result[i], result[j]] = [result[j], result[i]];
    53→  }
    54→  return result;
    55→}
    56→
    57→function makeDistractors(correct: string, wrongPool: string[], count: number = 3): string {
    58→  const filtered = wrongPool.filter(w => w !== correct);
    59→  return JSON.stringify(shuffle(filtered).slice(0, count));
    60→}
    61→
    62→function makeSolutionSteps(steps: string[]): string {
    63→  return JSON.stringify(steps);
    64→}

   366→function genWordProblems35(): SeedQuestion {
   367→  const type = pick(['multi-step', 'comparison', 'extra-info']);
   368→  if (type === 'multi-step') {
   369→    const a = randInt(5, 20);
   370→    const b = randInt(3, 10);
   371→    const c = randInt(2, 8);
   372→    const answer = a + b - c;
   373→    return {
   374→      subject: 'MATH', gradeBand: '3-5', topic: 'Word Problems',
   375→      difficulty: 4,
   376→      curriculum: 'CCSS', standardCode: 'CCSS.MATH.CONTENT.4.OA.A.3',
   377→      stem: `A store had ${a} notebooks. They received ${b} more notebooks, then sold ${c} notebooks. How many notebooks are left?`,
   378→      answerKey: String(answer),
   379→      solutionSteps: makeSolutionSteps([`Start: ${a} notebooks`, `Add ${b} more: ${a} + ${b} = ${a + b}`, `Subtract ${c} sold: ${a + b} - ${c} = ${answer}`]),
   380→      distractors: makeDistractors(String(answer), [String(a + b), String(a - c), String(a + b + c)]),
   381→      questionType: 'MCQ', tags: 'word-problem,multi-step,operations', estimatedTimeSec: 90,
   382→    };
   383→  } else if (type === 'comparison') {
   384→    const a = randInt(10, 50);
   385→    const b = randInt(10, 50);
   386→    const diff = Math.abs(a - b);
   387→    return {
   388→      subject: 'MATH', gradeBand: '3-5', topic: 'Word Problems',
   389→      difficulty: 3,
   390→      curriculum: 'CCSS', standardCode: 'CCSS.MATH.CONTENT.4.OA.A.2',
   391→      stem: `Team A scored ${a} points. Team B scored ${b} points. How many more points did the winning team score?`,
   392→      answerKey: String(diff),
   393→      distractors: makeDistractors(String(diff), [String(a + b), String(a), String(b)]),
   394→      questionType: 'MCQ', tags: 'word-problem,comparison,difference', estimatedTimeSec: 60,
   395→    };
   396→  }
   397→  // extra-info
   398→  const a = randInt(3, 12);
   399→  const b = randInt(2, 8);
   400→  const extra = randInt(10, 30);
   401→  const answer = a * b;
   402→  return {
   403→    subject: 'MATH', gradeBand: '3-5', topic: 'Word Problems',
   404→    difficulty: 4,
   405→    stem: `Each box contains ${a} toy cars. There are ${b} boxes. The boxes cost $${extra} total. How many toy cars are there in all?`,
   406→    answerKey: String(answer),
   407→    solutionSteps: makeSolutionSteps([`Ignore the cost — we need total toy cars`, `Each box has ${a} cars`, `${b} boxes`, `${a} × ${b} = ${answer}`]),
   408→    distractors: makeDistractors(String(answer), [String(extra), String(a + b), String(extra + a)]),
   409→    questionType: 'MCQ', tags: 'word-problem,extra-information,multiplication', estimatedTimeSec: 90,
