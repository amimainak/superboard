   101→function statusBadge(status: HomeworkItem['status']) {
   102→  const map: Record<HomeworkItem['status'], { cls: string; label: string }> = {
   103→    PENDING: {
   104→      cls: 'bg-amber-100 text-amber-700 hover:bg-amber-100 text-[10px] px-2 py-0 rounded-full font-medium',
   105→      label: 'Pending',
   106→    },
   107→    SUBMITTED: {
   108→      cls: 'bg-blue-100 text-blue-700 hover:bg-blue-100 text-[10px] px-2 py-0 rounded-full font-medium',
   109→      label: 'Submitted',
   110→    },
   111→    GRADED: {
   112→      cls: 'bg-emerald-100 text-emerald-700 hover:bg-emerald-100 text-[10px] px-2 py-0 rounded-full font-medium',
   113→      label: 'Graded',
   114→    },
   115→    OVERDUE: {
   116→      cls: 'bg-red-100 text-red-600 hover:bg-red-100 text-[10px] px-2 py-0 rounded-full font-medium',
   117→      label: 'Overdue',
   118→    },
   119→  };
   120→  const { cls, label } = map[status];
   121→  return <Badge className={cls}>{label}</Badge>;
   122→}
   123→
   124→function formatDate(iso: string | null) {
   125→  if (!iso) return '—';
   126→  return new Date(iso).toLocaleDateString('en-US', {
   127→    month: 'short',
   128→    day: 'numeric',
   129→    year: 'numeric',
   130→  });
   131→}
   132→
   133→function isOverdue(dueDate: string | null, status: string) {
   134→  if (!dueDate || status === 'GRADED' || status === 'SUBMITTED') return false;
   135→  return new Date(dueDate) < new Date();
   136→}

   141→type Props = { userId: string; agencyId?: string; userTier?: string };
   142→
   143→export function HomeworkPanel({ userId, agencyId, userTier }: Props) {
   144→  const isAgency = userTier ? isAgencyTier(userTier as any) : false;
   145→  const { toast } = useToast();
   146→
   147→  // My homework
   148→  const [myHomework, setMyHomework] = useState<HomeworkItem[]>([]);
   149→  // Agency all homework
   150→  const [agencyHomework, setAgencyHomework] = useState<HomeworkItem[]>([]);
   151→  const [loading, setLoading] = useState(true);
   152→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   153→  const [students, setStudents] = useState<StudentOption[]>([]);
   154→
   155→  // Create dialog
   156→  const [createOpen, setCreateOpen] = useState(false);
   157→  const [createForm, setCreateForm] = useState<CreateForm>(EMPTY_CREATE);
   158→  const [creating, setCreating] = useState(false);
   159→
   160→  // Grade dialog
   161→  const [gradeOpen, setGradeOpen] = useState(false);
   162→  const [gradeForm, setGradeForm] = useState<GradeForm>(EMPTY_GRADE);
   163→  const [gradingItem, setGradingItem] = useState<HomeworkItem | null>(null);
   164→  const [grading, setGrading] = useState(false);
   165→
   166→  // Action loading
   167→  const [actionLoading, setActionLoading] = useState<string | null>(null);
   168→
   169→  // ---- Fetch homework ----
   170→  // The API automatically returns agency-wide data for agency owners,
   171→  // and only own homework for tutors. We split on the client side.
   172→  const fetchHomework = useCallback(async () => {
   173→    try {
   174→      const res = await authFetch('/api/homework?limit=100');
   175→      if (!res.ok) return;
   176→      const data = await res.json();
   177→      const items: HomeworkItem[] = (data.homeworks || []).map((h: any) => {
   178→        if (h.status === 'PENDING' && isOverdue(h.dueDate, h.status)) {
   179→          return { ...h, status: 'OVERDUE' as const };
   180→        }
   181→        return h;
   182→      });
   183→      // For agency view: split into mine vs all
   184→      if (isAgency) {
   185→        setMyHomework(items.filter((h) => h.tutor?.id === userId));
   186→        setAgencyHomework(items);
   187→      } else {
   188→        setMyHomework(items);
   189→      }
   190→    } catch {
   191→      // silent
   192→    }
   193→  }, [userId, isAgency]);
   194→
   195→  const fetchAll = useCallback(async () => {
   196→    setLoading(true);
   197→    await fetchHomework();
   198→    setLoading(false);
   199→  }, [fetchHomework]);
   200→
   201→  // ---- Fetch students ----
   202→  const fetchStudents = useCallback(async () => {
   203→    try {
   204→      const res = await authFetch('/api/agency/students?status=active&limit=200');
   205→      if (!res.ok) return;
   206→      const data = await res.json();
   207→      setStudents((data.students || []).map((s: any) => ({
   208→        id: s.id,
   209→        name: s.name,
   210→        email: s.email,
   211→      })));
   212→    } catch {
   213→      // silent
   214→    }
   215→  }, []);
   216→
   217→  useEffect(() => {
   218→    fetchAll();
   219→    fetchStudents();
   220→  }, [fetchAll, fetchStudents]);
   221→
   222→  // Re-filter when userId changes
   223→  useEffect(() => {
   224→    if (isAgency && agencyHomework.length > 0) {
   225→      setMyHomework(agencyHomework.filter((h) => h.tutor?.id === userId));
   226→    }
   227→  }, [userId, isAgency, agencyHomework]);
   228→
   229→  // ---- Get active list ----
   230→  const homework = isAgency ? agencyHomework : myHomework;
   231→  const filtered = statusFilter === 'ALL'
   232→    ? homework
   233→    : homework.filter((h) => h.status === statusFilter);
   234→
   235→  // ---- Create ----
   236→  const handleCreate = async () => {
   237→    if (!createForm.title.trim() || !createForm.studentEmail.trim() || !createForm.dueDate) {
   238→      toast({ title: 'Please fill in title, student email, and due date', variant: 'destructive' });
   239→      return;
   240→    }
   241→    setCreating(true);
   242→    try {
   243→      // Find student by email
   244→      const matchedStudent = students.find(
   245→        (s) => s.email.toLowerCase() === createForm.studentEmail.trim().toLowerCase()
   246→      );
   247→      const body: Record<string, unknown> = {
   248→        title: createForm.title,
   249→        description: createForm.description || null,
   250→        subject: createForm.subject,
   251→        dueDate: new Date(createForm.dueDate).toISOString(),
   252→        studentId: matchedStudent?.id || null,
   253→      };
   254→      const res = await authFetch('/api/homework', {
   255→        method: 'POST',
   256→        body: JSON.stringify(body),
   257→      });
   258→      if (!res.ok) {
   259→        const bodyText = await res.text().catch(() => '');
   260→        throw new Error(bodyText || 'Failed to create homework');
   261→      }
   262→      toast({ title: 'Homework assigned!', description: createForm.title });
   263→      setCreateForm(EMPTY_CREATE);
   264→      setCreateOpen(false);
   265→      fetchAll();
   266→    } catch (err: any) {
   267→      toast({ title: 'Failed to create homework', description: err?.message || 'Please try again.', variant: 'destructive' });
   268→    } finally {
   269→      setCreating(false);
   270→    }
   271→  };
   272→
   273→  // ---- Grade ----
   274→  const openGradeDialog = (item: HomeworkItem) => {
   275→    setGradingItem(item);
   276→    setGradeForm({
   277→      tutorFeedback: item.tutorFeedback || '',
   278→      grade: item.grade || '',
   279→    });
   280→    setGradeOpen(true);
   281→  };
   282→
   283→  const handleGrade = async () => {
   284→    if (!gradingItem) return;
   285→    setGrading(true);
   286→    try {
   287→      const res = await authFetch(`/api/homework/${gradingItem.id}`, {
   288→        method: 'PATCH',
   289→        body: JSON.stringify({ status: 'GRADED', ...gradeForm }),
   290→      });
   291→      if (!res.ok) throw new Error('Failed to grade');
   292→      toast({ title: 'Homework graded!', description: gradingItem.title });
   293→      setGradeOpen(false);
   294→      setGradingItem(null);
   295→      fetchAll();
   296→    } catch {
   297→      toast({ title: 'Failed to grade homework', variant: 'destructive' });
   298→    } finally {
   299→      setGrading(false);
   300→    }
   301→  };
   302→
   303→  // ---- Mark submitted ----
   304→  const handleMarkSubmitted = async (id: string) => {
   305→    setActionLoading(id);
   306→    try {
   307→      const res = await authFetch(`/api/homework/${id}`, {
   308→        method: 'PATCH',
   309→        body: JSON.stringify({ status: 'SUBMITTED' }),
   310→      });
   311→      if (!res.ok) throw new Error('Failed to update');
   312→      toast({ title: 'Marked as submitted' });
   313→      fetchAll();
   314→    } catch {
   315→      toast({ title: 'Failed to update status', variant: 'destructive' });
   316→    } finally {
   317→      setActionLoading(null);
   318→    }
   319→  };
   320→
   321→  // ---- Mark graded (quick action) ----
   322→  const handleMarkGraded = async (id: string) => {
   323→    setActionLoading(id);
   324→    try {
   325→      const res = await authFetch(`/api/homework/${id}`, {
   326→        method: 'PATCH',
   327→        body: JSON.stringify({ status: 'GRADED' }),
   328→      });
   329→      if (!res.ok) throw new Error('Failed to update');
   330→      toast({ title: 'Marked as graded' });
   331→      fetchAll();
   332→    } catch {
   333→      toast({ title: 'Failed to update status', variant: 'destructive' });
   334→    } finally {
   335→      setActionLoading(null);
   336→    }
   337→  };
   338→
   339→  // ---- Loading ----
   340→  if (loading) {
   341→    return (
   342→      <Card className="rounded-2xl border border-gray-200 bg-white shadow-sm">
   343→        <CardHeader className="pb-3">
   344→          <CardTitle className="text-lg font-bold flex items-center gap-2">
   345→            <BookOpen className="w-5 h-5 text-emerald-500" />
   346→            Homework
   347→          </CardTitle>
   348→        </CardHeader>
   349→        <CardContent>
   350→          <div className="space-y-3">
   351→            {[1, 2, 3].map((i) => (
   352→              <div key={i} className="flex items-center gap-3 p-3 rounded-xl bg-gray-50 animate-pulse">
   353→                <div className="w-9 h-9 rounded-lg bg-gray-200" />
   354→                <div className="flex-1 space-y-2">
   355→                  <div className="h-4 bg-gray-200 rounded w-40" />
   356→                  <div className="h-3 bg-gray-200 rounded w-24" />
   357→                </div>
   358→              </div>
   359→            ))}
   360→          </div>
   361→        </CardContent>
   362→      </Card>
   363→    );
   364→  }
   365→
   366→  // ---- Render table ----
   367→  const renderTable = (items: HomeworkItem[]) => {
   368→    if (items.length === 0) {
   369→      return (
   370→        <div className="text-center py-8">
   371→          <CheckCircle className="w-8 h-8 text-emerald-300 mx-auto mb-2" />
   372→          <p className="text-sm text-muted-foreground">No homework matches this filter.</p>
   373→        </div>
   374→      );
   375→    }
   376→    return (
   377→      <div className="rounded-xl border overflow-hidden max-h-[420px] overflow-y-auto">
   378→        <table className="w-full text-sm">
   379→          <thead className="bg-muted/50 border-b sticky top-0 z-10">
   380→            <tr>
   381→              <th className="text-left px-4 py-2.5 font-semibold text-xs">Title</th>
   382→              <th className="text-left px-4 py-2.5 font-semibold text-xs hidden sm:table-cell">Student</th>
   383→              <th className="text-left px-4 py-2.5 font-semibold text-xs hidden md:table-cell">Subject</th>
   384→              <th className="text-left px-4 py-2.5 font-semibold text-xs">Status</th>
   385→              <th className="text-left px-4 py-2.5 font-semibold text-xs hidden lg:table-cell">Due Date</th>
   386→              <th className="text-left px-4 py-2.5 font-semibold text-xs hidden lg:table-cell">Grade</th>
   387→              <th className="px-4 py-2.5" />
   388→            </tr>
   389→          </thead>
   390→          <tbody>
   391→            {items.map((hw) => {
   392→              const meta = subjectMeta[hw.subject] || subjectMeta.GENERAL;
   393→              return (
   394→                <tr key={hw.id} className="border-b last:border-0 hover:bg-muted/30 transition-colors">
   395→                  <td className="px-4 py-3">
   396→                    <div className="flex items-center gap-2">
   397→                      <div className={`w-7 h-7 rounded-md ${meta.gradient} flex items-center justify-center shrink-0`}>
   398→                        <meta.icon className="w-3.5 h-3.5 text-white" />
   399→                      </div>
   400→                      <div className="min-w-0">
   401→                        <p className="font-medium text-xs truncate max-w-[140px]">{hw.title}</p>
   402→                      </div>
   403→                    </div>
   404→                  </td>
   405→                  <td className="px-4 py-3 text-xs hidden sm:table-cell">
   406→                    <div>
   407→                      <p className="font-medium">{hw.student?.name || '—'}</p>
   408→                      <p className="text-[10px] text-muted-foreground">{hw.student?.email}</p>
   409→                    </div>
   410→                  </td>
   411→                  <td className="px-4 py-3 text-xs text-muted-foreground hidden md:table-cell">{meta.label}</td>
   412→                  <td className="px-4 py-3">{statusBadge(hw.status)}</td>
   413→                  <td className="px-4 py-3 text-xs text-muted-foreground hidden lg:table-cell">
   414→                    <span className="flex items-center gap-1">
   415→                      <Clock className="w-3 h-3" />
   416→                      {formatDate(hw.dueDate)}
   417→                    </span>
   418→                  </td>
   419→                  <td className="px-4 py-3 text-xs font-medium hidden lg:table-cell">
   420→                    {hw.grade ? (
   421→                      <span className="flex items-center gap-1 text-emerald-600">
   422→                        <Star className="w-3 h-3" />
   423→                        {hw.grade}
   424→                      </span>
   425→                    ) : (
   426→                      <span className="text-muted-foreground">—</span>
   427→                    )}
   428→                  </td>
   429→                  <td className="px-4 py-3">
   430→                    <div className="flex items-center gap-1">
   431→                      {hw.status === 'SUBMITTED' && (
   432→                        <Button
   433→                          size="sm"
   434→                          variant="ghost"
   435→                          className="h-7 w-7 p-0 rounded-lg hover:bg-emerald-100 hover:text-emerald-700"
   436→                          title="Grade"
   437→                          onClick={() => openGradeDialog(hw)}
   438→                        >
   439→                          <Star className="w-3.5 h-3.5" />
   440→                        </Button>
   441→                      )}
   442→                      {hw.status === 'PENDING' && (
   443→                        <Button
   444→                          size="sm"
   445→                          variant="ghost"
   446→                          className="h-7 w-7 p-0 rounded-lg hover:bg-blue-50 hover:text-blue-600"
   447→                          title="Mark Submitted"
   448→                          disabled={actionLoading === hw.id}
   449→                          onClick={() => handleMarkSubmitted(hw.id)}
   450→                        >
   451→                          {actionLoading === hw.id ? (
   452→                            <Loader2 className="w-3.5 h-3.5 animate-spin text-emerald-500" />
   453→                          ) : (
   454→                            <Send className="w-3.5 h-3.5" />
   455→                          )}
   456→                        </Button>
   457→                      )}
   458→                      {hw.status === 'OVERDUE' && (
   459→                        <Button
   460→                          size="sm"
   461→                          variant="ghost"
   462→                          className="h-7 w-7 p-0 rounded-lg hover:bg-blue-50 hover:text-blue-600"
   463→                          title="Mark Submitted"
   464→                          disabled={actionLoading === hw.id}
   465→                          onClick={() => handleMarkSubmitted(hw.id)}
   466→                        >
   467→                          {actionLoading === hw.id ? (
   468→                            <Loader2 className="w-3.5 h-3.5 animate-spin text-emerald-500" />
   469→                          ) : (
   470→                            <Send className="w-3.5 h-3.5" />
   471→                          )}
   472→                        </Button>
   473→                      )}
   474→                      {hw.status === 'SUBMITTED' && (
   475→                        <Button
   476→                          size="sm"
   477→                          variant="ghost"
   478→                          className="h-7 w-7 p-0 rounded-lg hover:bg-emerald-50 hover:text-emerald-700"
   479→                          title="Mark Graded"
   480→                          disabled={actionLoading === hw.id}
   481→                          onClick={() => handleMarkGraded(hw.id)}
   482→                        >
   483→                          {actionLoading === hw.id ? (
   484→                            <Loader2 className="w-3.5 h-3.5 animate-spin text-emerald-500" />
   485→                          ) : (
   486→                            <ClipboardCheck className="w-3.5 h-3.5" />
   487→                          )}
   488→                        </Button>
   489→                      )}
   490→                      <Button
   491→                        size="sm"
   492→                        variant="ghost"
   493→                        className="h-7 w-7 p-0 rounded-lg hover:bg-gray-100"
   494→                        title="View details"
   495→                        onClick={() => openGradeDialog(hw)}
   496→                      >
   497→                        <Eye className="w-3.5 h-3.5" />
   498→                      </Button>
   499→                    </div>
   500→                  </td>
   501→                </tr>
   502→              );
   503→            })}
   504→          </tbody>
   505→        </table>
   506→      </div>
   507→    );
   508→  };
   509→
   510→  // ---- Render ----
   511→  return (
   512→    <Card className="rounded-2xl border border-gray-200 bg-white shadow-sm">
   513→      <CardHeader className="pb-3">
   514→        <div className="flex items-center justify-between">
   515→          <CardTitle className="text-lg font-bold flex items-center gap-2">
   516→            <div className="w-8 h-8 rounded-lg bg-emerald-100 flex items-center justify-center">
   517→              <BookOpen className="w-4 h-4 text-emerald-600" />
   518→            </div>
   519→            Homework
   520→            {homework.length > 0 && (
   521→              <Badge variant="secondary" className="ml-1 text-xs font-medium">
   522→                {homework.length}
   523→              </Badge>
   524→            )}
   525→          </CardTitle>
   526→
   527→          <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   528→            <DialogTrigger asChild>
   529→              <Button
   530→                size="sm"
   531→                className="rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all hover:-translate-y-0.5 text-sm gap-2"
   532→              >
   533→                <Plus className="w-4 h-4" />
   534→                Assign
   535→              </Button>
   536→            </DialogTrigger>
   537→            <DialogContent className="rounded-2xl sm:max-w-md">
   538→              <DialogHeader>
   539→                <DialogTitle className="flex items-center gap-2">
   540→                  <Plus className="w-5 h-5 text-emerald-500" />
   541→                  Assign Homework
   542→                </DialogTitle>
   543→                <DialogDescription>Create a new homework assignment for a student.</DialogDescription>
   544→              </DialogHeader>
   545→              <div className="space-y-4 pt-2">
   546→                <div>
   547→                  <Label className="text-sm font-medium text-gray-700 mb-1 block">Title</Label>
   548→                  <Input
   549→                    placeholder="e.g. Chapter 5 Practice Problems"
   550→                    value={createForm.title}
   551→                    onChange={(e) => setCreateForm((p) => ({ ...p, title: e.target.value }))}
   552→                    className="rounded-xl"
   553→                  />
   554→                </div>
   555→                <div>
   556→                  <Label className="text-sm font-medium text-gray-700 mb-1 block">Description</Label>
   557→                  <Textarea
   558→                    placeholder="Instructions or details..."
   559→                    value={createForm.description}
   560→                    onChange={(e) => setCreateForm((p) => ({ ...p, description: e.target.value }))}
   561→                    className="rounded-xl min-h-[80px]"
   562→                  />
   563→                </div>
   564→                <div>
   565→                  <Label className="text-sm font-medium text-gray-700 mb-1 block">Student Email</Label>
   566→                  <Input
   567→                    type="email"
   568→                    placeholder="student@example.com"
   569→                    value={createForm.studentEmail}
   570→                    onChange={(e) => setCreateForm((p) => ({ ...p, studentEmail: e.target.value }))}
   571→                    className="rounded-xl"
   572→                  />
   573→                </div>
   574→                <div>
   575→                  <Label className="text-sm font-medium text-gray-700 mb-1 block">Subject</Label>
   576→                  <Select value={createForm.subject} onValueChange={(v) => setCreateForm((p) => ({ ...p, subject: v }))}>
   577→                    <SelectTrigger className="rounded-xl"><SelectValue /></SelectTrigger>
   578→                    <SelectContent>
   579→                      {Object.entries(subjectMeta).map(([key, meta]) => (
   580→                        <SelectItem key={key} value={key}>{meta.label}</SelectItem>
   581→                      ))}
   582→                    </SelectContent>
   583→                  </Select>
   584→                </div>
   585→                <div>
   586→                  <Label className="text-sm font-medium text-gray-700 mb-1 block">Due Date</Label>
   587→                  <Input
   588→                    type="date"
   589→                    value={createForm.dueDate}
   590→                    onChange={(e) => setCreateForm((p) => ({ ...p, dueDate: e.target.value }))}
   591→                    className="rounded-xl"
   592→                  />
   593→                </div>
   594→                <Button
   595→                  onClick={handleCreate}
   596→                  disabled={creating}
   597→                  className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all hover:-translate-y-0.5 text-sm gap-2"
   598→                >
   599→                  {creating && <Loader2 className="w-4 h-4 animate-spin" />}
   600→                  Assign Homework
   601→                </Button>
   602→              </div>
   603→            </DialogContent>
   604→          </Dialog>
   605→        </div>
   606→      </CardHeader>
   607→
   608→      <CardContent>
   609→        {homework.length === 0 && myHomework.length === 0 && (!isAgency || agencyHomework.length === 0) ? (
   610→          <div className="text-center py-10">
   611→            <div className="w-14 h-14 rounded-2xl bg-emerald-50 flex items-center justify-center mx-auto mb-4">
   612→              <FileText className="w-7 h-7 text-emerald-400" />
   613→            </div>
   614→            <h3 className="text-sm font-semibold text-gray-900">No homework assigned</h3>
   615→            <p className="text-sm text-muted-foreground mt-1 mb-4">Create your first homework assignment.</p>
   616→            <Button
   617→              onClick={() => setCreateOpen(true)}
   618→              className="rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all hover:-translate-y-0.5 text-sm gap-2"
   619→              size="sm"
   620→            >
   621→              <Plus className="w-4 h-4" />
   622→              Assign Homework
   623→            </Button>
   624→          </div>
   625→        ) : (
   626→          <>
   627→            {/* Tabs for agency view */}
   628→            {isAgency && (
   629→              <Tabs
   630→                defaultValue="mine"
   631→                className="w-full mb-4"
   632→                onValueChange={() => {
   633→                  setStatusFilter('ALL');
   634→                }}
   635→              >
   636→                <TabsList className="w-full grid grid-cols-2 rounded-xl">
   637→                  <TabsTrigger value="mine" className="rounded-lg text-xs font-medium">
   638→                    My Homework
   639→                  </TabsTrigger>
   640→                  <TabsTrigger value="agency" className="rounded-lg text-xs font-medium">
   641→                    Agency All
   642→                  </TabsTrigger>
   643→                </TabsList>
   644→                <TabsContent value="mine" className="mt-3">
   645→                  <StatusFilterButtons statusFilter={statusFilter} setStatusFilter={setStatusFilter} />
   646→                  {renderTable(
   647→                    (statusFilter === 'ALL' ? myHomework : myHomework.filter((h) => h.status === statusFilter))
   648→                  )}
   649→                </TabsContent>
   650→                <TabsContent value="agency" className="mt-3">
   651→                  <StatusFilterButtons statusFilter={statusFilter} setStatusFilter={setStatusFilter} />
   652→                  {renderTable(
   653→                    (statusFilter === 'ALL' ? agencyHomework : agencyHomework.filter((h) => h.status === statusFilter))
   654→                  )}
   655→                </TabsContent>
   656→              </Tabs>
   657→            )}
   658→
   659→            {/* Non-agency: simple filter + table */}
   660→            {!isAgency && (
   661→              <>
   662→                <StatusFilterButtons statusFilter={statusFilter} setStatusFilter={setStatusFilter} />
   663→                {renderTable(filtered)}
   664→              </>
   665→            )}
   666→          </>
   667→        )}
   668→      </CardContent>
   669→
   670→      {/* Grade Dialog */}
   671→      <Dialog open={gradeOpen} onOpenChange={setGradeOpen}>
   672→        <DialogContent className="rounded-2xl sm:max-w-md">
   673→          <DialogHeader>
   674→            <DialogTitle className="flex items-center gap-2">
   675→              <Star className="w-5 h-5 text-emerald-500" />
   676→              {gradingItem?.status === 'SUBMITTED' ? 'Grade Homework' : 'Homework Details'}
   677→            </DialogTitle>
   678→            <DialogDescription>
   679→              {gradingItem?.title} — {gradingItem?.student?.name || 'Unknown'}
   680→            </DialogDescription>
   681→          </DialogHeader>
   682→          <div className="space-y-4 pt-2">
   683→            {gradingItem?.description && (
   684→              <div className="rounded-xl bg-gray-50 p-3">
   685→                <p className="text-xs font-medium text-muted-foreground mb-1">Description</p>
   686→                <p className="text-sm">{gradingItem.description}</p>
   687→              </div>
   688→            )}
   689→            <div>
   690→              <Label className="text-sm font-medium text-gray-700 mb-1 block">Grade</Label>
   691→              <Input
   692→                placeholder="e.g. A, 95%, 8/10"
   693→                value={gradeForm.grade}
   694→                onChange={(e) => setGradeForm((p) => ({ ...p, grade: e.target.value }))}
   695→                className="rounded-xl"
   696→                disabled={gradingItem?.status === 'GRADED'}
   697→              />
   698→            </div>
   699→            <div>
   700→              <Label className="text-sm font-medium text-gray-700 mb-1 block">Feedback</Label>
   701→              <Textarea
   702→                placeholder="Your feedback for the student..."
   703→                value={gradeForm.tutorFeedback}
   704→                onChange={(e) => setGradeForm((p) => ({ ...p, tutorFeedback: e.target.value }))}
   705→                className="rounded-xl min-h-[100px]"
   706→                disabled={gradingItem?.status === 'GRADED'}
   707→              />
   708→            </div>
   709→            {gradingItem?.status !== 'GRADED' && (
   710→              <Button
   711→                onClick={handleGrade}
   712→                disabled={grading}
   713→                className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all hover:-translate-y-0.5 text-sm gap-2"
   714→              >
   715→                {grading && <Loader2 className="w-4 h-4 animate-spin" />}
   716→                Submit Grade
   717→              </Button>
   718→            )}
   719→          </div>
   720→        </DialogContent>
   721→      </Dialog>
   722→    </Card>
   723→  );
   724→}

   727→// Status Filter Buttons (extracted for reuse in tabs)
   728→// ------------------------------------------------------------------
   729→function StatusFilterButtons({ statusFilter, setStatusFilter }: { statusFilter: string; setStatusFilter: (v: string) => void }) {
   730→  return (
   731→    <div className="flex items-center gap-2 mb-4 flex-wrap">
   732→      {STATUS_OPTIONS.map((s) => (
   733→        <Button
   734→          key={s}
   735→          variant={statusFilter === s ? 'default' : 'outline'}
   736→          size="sm"
   737→          className={`rounded-lg text-xs h-7 px-2.5 ${statusFilter === s ? 'gradient-primary border-0 text-white' : ''}`}
   738→          onClick={() => setStatusFilter(s)}
   739→        >
   740→          {s === 'ALL' ? 'All' : s.charAt(0) + s.slice(1).toLowerCase()}
   741→        </Button>
   742→      ))}
   743→    </div>
   744→  );
   745→}
