    41→    41→def make_distractors_int(answer: int, count: int = 3, spread: int = 5, **_kw) -> list:
    42→    42→    """Generate plausible wrong answers for integer questions."""
    43→    43→    distractors = set()
    44→    44→    attempts = 0
    45→    45→    while len(distractors) < count and attempts < 50:
    46→    46→        offset = random.choice([1, 2, 3, 5, 10, -1, -2, -3, -5, -10])
    47→    47→        d = answer + offset
    48→    48→        if d != answer and d >= 0:
    49→    49→            distractors.add(str(d))
    50→    50→        attempts += 1
    51→    51→    while len(distractors) < count:
    52→    52→        distractors.add(str(answer + len(distractors) + 1))
    53→    53→    return list(distractors)[:count]
    54→    54→
    55→    55→def make_distractors_frac(num: int, den: int, count: int = 3, **_kw) -> list:
    56→    56→    """Generate plausible wrong answers for fraction questions."""
    57→    57→    fracs = []
    58→    58→    for d_num in [num + 1, num - 1, den, num * 2]:
    59→    59→        for d_den in [den, den + 1, den - 1, num]:
    60→    60→            if d_den > 0 and d_num != num:
    61→    61→                fracs.append(f"{d_num}/{d_den}")
    62→    62→    random.shuffle(fracs)
    63→    63→    return list(set(fracs))[:count]

   105→   105→    def _sample_params(self, spec):
   106→   106→        params = {}
   107→   107→        for key, config in spec.items():
   108→   108→            if isinstance(config, list):
   109→   109→                params[key] = random.choice(config)
   110→   110→            elif isinstance(config, range):
   111→   111→                params[key] = random.choice(list(config))
   112→   112→            elif isinstance(config, dict) and config.get("type") == "choice":
   113→   113→                params[key] = random.choice(config["values"])
   114→   114→            elif isinstance(config, dict) and config.get("type") == "randint":
   115→   115→                params[key] = random.randint(config["min"], config["max"])
   116→   116→            elif callable(config):
   117→   117→                params[key] = config()
   118→   118→        return params
