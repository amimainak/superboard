     1→     1→// ============================================================
     2→     2→// TldrawCanvas — Tldraw Editor with Yjs Real-Time Sync
     3→     3→// ============================================================
     4→     4→// Wraps the Tldraw editor and synchronizes state with Yjs
     5→     5→// via the Hocuspocus CRDT provider for multi-user collaboration.
     6→     6→//
     7→     7→// Sync strategy: Incremental per-record CRDT sync.
     8→     8→// - Each record (shape, camera, page-state, etc.) is stored as a
     9→     9→//   separate entry in a per-page Y.Map, keyed by record ID.
    10→    10→// - Local user edits are debounced (200ms) and batch-written to Yjs.
    11→    11→// - Remote Yjs changes are merged record-by-record into the editor
    12→    12→//   using store.mergeRemoteChanges(), which marks them source='remote'
    13→    13→//   so they don't trigger the local-sync listener (no echo loop).
    14→    14→// - A dirty-set tracks record IDs with unsynced local changes;
    15→    15→//   remote updates for dirty records are skipped (last-local-writer-wins).
    16→    16→// - A monotonic localChangeVersion counter is maintained per record
    17→    17→//   for diagnostics and future conflict-resolution extensions.
    18→    18→// - Backward compatibility: legacy full-snapshot format is detected
    19→    19→//   on first load and migrated to the per-record format.
    20→    20→// ============================================================
    21→    21→
    22→    22→'use client';
    23→    23→
    24→    24→import React, { useCallback, useEffect, useRef, useState } from 'react';
    25→    25→import { Tldraw, Editor, TLStoreSnapshot } from 'tldraw';
    26→    26→import type { Map as YMap } from 'yjs';
    27→    27→import 'tldraw/tldraw.css';
    28→    28→
    29→    29→// ---- Types ----
    30→    30→
    31→    31→/** Shape of the RecordsDiff changes as delivered by store.listen in tldraw v5. */
    32→    32→interface RecordsDiffLike {
    33→    33→        added: Record<string, Record<string, unknown>>;
    34→    34→        updated: Record<string, [from: Record<string, unknown>, to: Record<string, unknown>]>;
    35→    35→        removed: Record<string, Record<string, unknown>>;
    36→    36→}
    37→    37→
    38→    38→/** Shape of the history entry passed to store.listen callbacks in tldraw v5. */
    39→    39→interface HistoryEntryLike {
    40→    40→        changes: RecordsDiffLike;
    41→    41→        source: 'user' | 'remote';
    42→    42→}
    43→    43→
    44→    44→export interface TldrawCanvasProps {
    45→    45→        /** Yjs document (from useYjsProvider) */
    46→    46→        ydoc: import('yjs').Doc | null;
    47→    47→        /** Called when the editor mounts — provides the Editor ref */
    48→    48→        onEditorReady?: (editor: Editor) => void;
    49→    49→        /** Called when editor store changes (for external sync triggers) */
    50→    50→        onStoreChange?: () => void;
    51→    51→        /** Current page index (for multi-page support) */
    52→    52→        pageIndex: number;
    53→    53→        /** Whether the user is a tutor (controls UI permissions) */
    54→    54→        isTutor: boolean;
    55→    55→        /** Whether the canvas is read-only (e.g., student in focus mode) */
    56→    56→        readOnly?: boolean;
    57→    57→}
    58→    58→
    59→    59→// ---- Constants ----
    60→    60→
    61→    61→/** Debounce Yjs writes to batch rapid edits into one transaction. */
    62→    62→const SYNC_DEBOUNCE_MS = 200;
    63→    63→
    64→    64→/** Key of the legacy Y.Map that stored full page snapshots as JSON blobs. */
    65→    65→const YJS_LEGACY_MAP_KEY = 'page-snapshot';
    66→    66→
    67→    67→/**
    68→    68→ * Prefix for per-page Y.Maps that store individual records.
    69→    69→ * E.g. pageIndex=3 → ydoc.getMap('page-shapes-3')
    70→    70→ */
    71→    71→const YJS_SHAPES_MAP_PREFIX = 'page-shapes-';
    72→    72→
    73→    73→/**
    74→    74→ * Transaction origin symbol used to tag our own local writes.
    75→    75→ * The Yjs observer checks this to skip re-applying our own changes.
    76→    76→ */
    77→    77→const LOCAL_SYNC_ORIGIN = Symbol('tldraw-local-sync');
    78→    78→
    79→    79→// ============================================================
    80→    80→// Component
    81→    81→// ============================================================
    82→    82→
    83→    83→export default function TldrawCanvas({
    84→    84→        ydoc,
    85→    85→        onEditorReady,
    86→    86→        onStoreChange,
    87→    87→        pageIndex,
    88→    88→        isTutor,
    89→    89→        readOnly = false,
    90→    90→}: TldrawCanvasProps) {
    91→    91→        // ---- Core state & refs ----
    92→    92→        const editorRef = useRef<Editor | null>(null);
    93→    93→        const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    94→    94→        const [isReady, setIsReady] = useState(false);
    95→    95→        const [isLoadingSnapshot, setIsLoadingSnapshot] = useState(true);
    96→    96→
    97→    97→        /**
    98→    98→         * Mirror of isReady as a ref so that closures inside setTimeout/
    99→    99→         * store listeners always see the current value without being
   100→   100→         * captured at creation time.
   101→   101→         */
   102→   102→        const isReadyRef = useRef(false);
   103→   103→
   104→   104→        // ---- Incremental sync refs (stable across renders) ----
   105→   105→
   106→   106→        /** Records that changed locally and are awaiting debounce flush to Yjs. */
   107→   107→        const pendingWritesRef = useRef<Map<string, Record<string, unknown>>>(new Map());
   108→   108→
   109→   109→        /** Record IDs that were deleted locally and are awaiting debounce flush. */
   110→   110→        const pendingDeletesRef = useRef<Set<string>>(new Set());
   111→   111→
   112→   112→        /**
   113→   113→         * Record IDs with unsynced local changes.
   114→   114→         * Remote updates for these IDs are skipped to prevent overwriting
   115→   115→         * concurrent local edits (last-local-writer-wins).  IDs are removed
   116→   116→         * after the corresponding write/delete is flushed to Yjs.
   117→   117→         */
   118→   118→        const localDirtyRef = useRef<Set<string>>(new Set());
   119→   119→
   120→   120→        /** Monotonically increasing version counter for local edits. */
   121→   121→        const localChangeVersionRef = useRef(0);
   122→   122→
   123→   123→        /** Per-record version: the localChangeVersion at which it was last edited locally. */
   124→   124→        const recordVersionsRef = useRef<Map<string, number>>(new Map());
   125→   125→
   126→   126→        /** True while the initial snapshot is being loaded (suppresses sync-to-Yjs). */
   127→   127→        const isInitialLoadRef = useRef(false);
   128→   128→
   129→   129→        /** Set of page indices that have already been migrated from legacy format. */
   130→   130→        const migratedPagesRef = useRef<Set<number>>(new Set());
   131→   131→
   132→   132→        /** Cleanup function for the tldraw store listener. */
   133→   133→        const unlistenStoreRef = useRef<(() => void) | null>(null);
   134→   134→
   135→   135→        /** Cleanup function for the Yjs shapes-map observer. */
   136→   136→        const unobserveYjsRef = useRef<(() => void) | null>(null);
   137→   137→
   138→   138→        /**
   139→   139→         * Ref to the Y.Map currently being used for this page's shapes.
   140→   140→         * Updated explicitly on page switch so we can flush the OLD map
   141→   141→         * before swapping to the new one.
   142→   142→         */
   143→   143→        const shapesMapRef = useRef<YMap<string> | null>(null);
   144→   144→
   145→   145→        /** Track the previous page index to detect page switches. */
   146→   146→        const prevPageIndexRef = useRef(pageIndex);
   147→   147→
   148→   148→        /** Keep callback refs up-to-date without re-registering listeners. */
   149→   149→        const onStoreChangeRef = useRef(onStoreChange);
   150→   150→        onStoreChangeRef.current = onStoreChange;
   151→   151→
   152→   152→        // ----------------------------------------------------------------
   153→   153→        // Core: process a RecordsDiff into pending writes/deletes & schedule sync
   154→   154→        // ----------------------------------------------------------------
   155→   155→
   156→   156→        const processLocalDiff = useCallback((entry: HistoryEntryLike) => {
   157→   157→                if (isInitialLoadRef.current) return;
   158→   158→                if (!isReadyRef.current) return;
   159→   159→
   160→   160→                const { changes } = entry;
   161→   161→
   162→   162→                // Increment the global local version counter.
   163→   163→                localChangeVersionRef.current++;
   164→   164→                const version = localChangeVersionRef.current;
   165→   165→
   166→   166→                // Collect added records.
   167→   167→                for (const [id, record] of Object.entries(changes.added)) {
   168→   168→                        pendingWritesRef.current.set(id, record);
   169→   169→                        pendingDeletesRef.current.delete(id);
   170→   170→                        localDirtyRef.current.add(id);
   171→   171→                        recordVersionsRef.current.set(id, version);
   172→   172→                }
   173→   173→
   174→   174→                // Collect updated records (use the "to" value — index 1).
   175→   175→                for (const [id, pair] of Object.entries(changes.updated)) {
   176→   176→                        pendingWritesRef.current.set(id, pair[1]);
   177→   177→                        pendingDeletesRef.current.delete(id);
   178→   178→                        localDirtyRef.current.add(id);
   179→   179→                        recordVersionsRef.current.set(id, version);
   180→   180→                }
   181→   181→
   182→   182→                // Collect removed records.
   183→   183→                for (const id of Object.keys(changes.removed)) {
   184→   184→                        pendingDeletesRef.current.add(id);
   185→   185→                        pendingWritesRef.current.delete(id);
   186→   186→                        localDirtyRef.current.add(id);
   187→   187→                        recordVersionsRef.current.set(id, version);
   188→   188→                }
   189→   189→
   190→   190→                scheduleSyncRef.current();
   191→   191→        }, []);
   192→   192→
   193→   193→        // ---- Ref-stable scheduleSync (never recreated) ----
   194→   194→        const scheduleSyncRef = useRef<() => void>(() => {});
   195→   195→
   196→   196→        // ---- Ref-stable flushPendingToYjs (never recreated) ----
   197→   197→        const flushPendingRef = useRef<() => void>(() => {});
   198→   198→
   199→   199→        // ----------------------------------------------------------------
   200→   200→        // Flush pending changes to the current shapes Y.Map
   201→   201→        // ----------------------------------------------------------------
   202→   202→
   203→   203→        const buildFlush = useCallback(() => {
   204→   204→                return () => {
   205→   205→                        const map = shapesMapRef.current;
   206→   206→                        if (!map || !ydoc) return;
   207→   207→
   208→   208→                        // Nothing to flush?
   209→   209→                        if (pendingWritesRef.current.size === 0 && pendingDeletesRef.current.size === 0) return;
   210→   210→
   211→   211→                        // Snapshot the pending state so new changes accumulate independently.
   212→   212→                        const writes = new Map(pendingWritesRef.current);
   213→   213→                        const deletes = new Set(pendingDeletesRef.current);
   214→   214→                        pendingWritesRef.current = new Map();
   215→   215→                        pendingDeletesRef.current = new Set();
   216→   216→
   217→   217→                        // Check total serialized size (5 MB limit matches DB constraint).
   218→   218→                        let totalSize = 0;
   219→   219→                        for (const record of writes.values()) {
   220→   220→                                totalSize += JSON.stringify(record).length;
   221→   221→                        }
   222→   222→                        if (totalSize > 5_000_000) {
   223→   223→                                console.warn(
   224→   224→                                        `[TldrawCanvas] Pending sync too large (${(totalSize / 1_000_000).toFixed(1)}MB) — skipping`
   225→   225→                                );
   226→   226→                                return;
   227→   227→                        }
   228→   228→
   229→   229→                        // Batch all writes/deletes into a single Yjs transaction tagged
   230→   230→                        // with our origin so the observer can skip re-applying them.
   231→   231→                        ydoc.transact(() => {
   232→   232→                                for (const [id, record] of writes) {
   233→   233→                                        map.set(id, JSON.stringify(record));
   234→   234→                                }
   235→   235→                                for (const id of deletes) {
   236→   236→                                        map.delete(id);
   237→   237→                                }
   238→   238→                        }, LOCAL_SYNC_ORIGIN);
   239→   239→
   240→   240→                        // Clear dirty flags for the records we just synced.
   241→   241→                        for (const id of writes.keys()) {
   242→   242→                                localDirtyRef.current.delete(id);
   243→   243→                        }
   244→   244→                        for (const id of deletes) {
   245→   245→                                localDirtyRef.current.delete(id);
   246→   246→                        }
   247→   247→
   248→   248→                        onStoreChangeRef.current?.();
   249→   249→                };
   250→   250→        }, [ydoc]);
   251→   251→
   252→   252→        // Keep flushPendingRef up-to-date.
   253→   253→        flushPendingRef.current = buildFlush();
   254→   254→
   255→   255→        // ----------------------------------------------------------------
   256→   256→        // Schedule a debounced flush
   257→   257→        // ----------------------------------------------------------------
   258→   258→
   259→   259→        const buildScheduleSync = useCallback(() => {
   260→   260→                return () => {
   261→   261→                        if (debounceTimerRef.current) {
   262→   262→                                clearTimeout(debounceTimerRef.current);
   263→   263→                        }
   264→   264→                        debounceTimerRef.current = setTimeout(() => {
   265→   265→                                flushPendingRef.current();
   266→   266→                        }, SYNC_DEBOUNCE_MS);
   267→   267→                };
   268→   268→        }, []);
   269→   269→
   270→   270→        // Keep scheduleSyncRef up-to-date.
   271→   271→        scheduleSyncRef.current = buildScheduleSync();
   272→   272→
   273→   273→        // ----------------------------------------------------------------
   274→   274→        // Load initial state from Yjs into the editor
   275→   275→        // ----------------------------------------------------------------
   276→   276→
   277→   277→        const loadInitialState = useCallback(() => {
   278→   278→                const editor = editorRef.current;
   279→   279→                if (!editor || !ydoc) return;
   280→   280→
   281→   281→                isInitialLoadRef.current = true;
   282→   282→
   283→   283→                const shapesMap = ydoc.getMap<string>(`${YJS_SHAPES_MAP_PREFIX}${pageIndex}`);
   284→   284→                shapesMapRef.current = shapesMap;
   285→   285→
   286→   286→                // --- Try legacy full-snapshot format first ---
   287→   287→                const legacyMap = ydoc.getMap<string>(YJS_LEGACY_MAP_KEY);
   288→   288→                const legacyKey = `page-${pageIndex}`;
   289→   289→                const legacyJson = legacyMap.get(legacyKey);
   290→   290→
   291→   291→                if (legacyJson && !migratedPagesRef.current.has(pageIndex)) {
   292→   292→                        try {
   293→   293→                                const storeSnapshot = JSON.parse(legacyJson) as TLStoreSnapshot;
   294→   294→
   295→   295→                                // editor.loadSnapshot accepts Partial<TLEditorSnapshot> | TLStoreSnapshot
   296→   296→                                editor.loadSnapshot({ document: storeSnapshot });
   297→   297→                                console.log(`[TldrawCanvas] Loaded legacy snapshot for page ${pageIndex}`);
   298→   298→
   299→   299→                                // Migrate the legacy snapshot into per-record Y.Map entries.
   300→   300→                                // TLStoreSnapshot has { store: Record<string, TLRecord>, schema: ... }
   301→   301→                                const records = (storeSnapshot as any).store;
   302→   302→                                if (records) {
   303→   303→                                        ydoc.transact(() => {
   304→   304→                                                for (const [id, record] of Object.entries(records)) {
   305→   305→                                                        shapesMap.set(id, JSON.stringify(record));
   306→   306→                                                }
   307→   307→                                        }, LOCAL_SYNC_ORIGIN);
   308→   308→                                        console.log(
   309→   309→                                                `[TldrawCanvas] Migrated ${Object.keys(records).length} records to per-record format for page ${pageIndex}`
   310→   310→                                        );
   311→   311→                                }
   312→   312→
   313→   313→                                migratedPagesRef.current.add(pageIndex);
   314→   314→                        } catch (err) {
   315→   315→                                console.error(
   316→   316→                                        `[TldrawCanvas] Failed to load/parse legacy snapshot for page ${pageIndex}:`,
   317→   317→                                        err
   318→   318→                                );
   319→   319→                        }
   320→   320→                } else {
   321→   321→                        // --- Load from per-record format ---
   322→   322→                        const recordsToLoad: Record<string, unknown>[] = [];
   323→   323→                        shapesMap.forEach((json, id) => {
   324→   324→                                try {
   325→   325→                                        recordsToLoad.push(JSON.parse(json));
   326→   326→                                } catch (err) {
   327→   327→                                        console.error(`[TldrawCanvas] Failed to parse record ${id}:`, err);
   328→   328→                                }
   329→   329→                        });
   330→   330→
   331→   331→                        if (recordsToLoad.length > 0) {
   332→   332→                                // Use mergeRemoteChanges so these are tagged source='remote'
   333→   333→                                // and won't trigger our local-sync listener.
   334→   334→                                editor.store.mergeRemoteChanges(() => {
   335→   335→                                        editor.store.put(recordsToLoad as any);
   336→   336→                                });
   337→   337→                                console.log(
   338→   338→                                        `[TldrawCanvas] Loaded ${recordsToLoad.length} records from per-record format for page ${pageIndex}`
   339→   339→                                );
   340→   340→                        } else if (!legacyJson) {
   341→   341→                                console.log(`[TldrawCanvas] No saved data for page ${pageIndex} — starting fresh`);
   342→   342→                        }
   343→   343→                }
   344→   344→
   345→   345→                isInitialLoadRef.current = false;
   346→   346→                setIsLoadingSnapshot(false);
   347→   347→        }, [ydoc, pageIndex]);
   348→   348→
   349→   349→        // ----------------------------------------------------------------
   350→   350→        // Set up the Yjs remote-change observer for the shapes map
   351→   351→        // ----------------------------------------------------------------
   352→   352→
   353→   353→        const setupRemoteObserver = useCallback(() => {
   354→   354→                const shapesMap = shapesMapRef.current;
   355→   355→                if (!shapesMap) return () => {};
   356→   356→
   357→   357→                const observer = (event: any, transaction: any) => {
   358→   358→                        // Skip our own local writes (tagged with LOCAL_SYNC_ORIGIN).
   359→   359→                        if (transaction.origin === LOCAL_SYNC_ORIGIN) return;
   360→   360→
   361→   361→                        const editor = editorRef.current;
   362→   362→                        if (!editor) return;
   363→   363→                        if (isInitialLoadRef.current) return;
   364→   364→
   365→   365→                        const toPut: Record<string, unknown>[] = [];
   366→   366→                        const toRemove: string[] = [];
   367→   367→
   368→   368→                        // Determine which keys changed.
   369→   369→                        // YMapEvent extends YEvent which has `changes.keys` (Map<string, {action, oldValue}>).
   370→   370→                        // It also has `keysChanged` (Set<string>) which is a simpler set of changed keys.
   371→   371→                        const changedKeys: Set<string> | undefined = event.keysChanged;
   372→   372→                        if (!changedKeys) return;
   373→   373→
   374→   374→                        for (const id of changedKeys) {
   375→   375→                                // Skip records with pending local changes — our local edit wins.
   376→   376→                                if (localDirtyRef.current.has(id)) {
   377→   377→                                        continue;
   378→   378→                                }
   379→   379→
   380→   380→                                // Check if the key still exists in the map (add/update) or was deleted.
   381→   381→                                if (shapesMap.has(id)) {
   382→   382→                                        try {
   383→   383→                                                toPut.push(JSON.parse(shapesMap.get(id)!));
   384→   384→                                        } catch (err) {
   385→   385→                                                console.error(`[TldrawCanvas] Failed to parse remote record ${id}:`, err);
   386→   386→                                        }
   387→   387→                                } else {
   388→   388→                                        toRemove.push(id);
   389→   389→                                }
   390→   390→                        }
   391→   391→
   392→   392→                        if (toPut.length > 0 || toRemove.length > 0) {
   393→   393→                                editor.store.mergeRemoteChanges(() => {
   394→   394→                                        if (toPut.length > 0) {
   395→   395→                                                editor.store.put(toPut as any);
   396→   396→                                        }
   397→   397→                                        if (toRemove.length > 0) {
   398→   398→                                                editor.store.remove(toRemove as any);
   399→   399→                                        }
   400→   400→                                });
   401→   401→                                console.log(
   402→   402→                                        `[TldrawCanvas] Applied remote changes for page ${pageIndex}: ${toPut.length} puts, ${toRemove.length} removes`
   403→   403→                                );
   404→   404→                        }
   405→   405→                };
   406→   406→
   407→   407→                shapesMap.observe(observer);
   408→   408→                return () => {
   409→   409→                        shapesMap.unobserve(observer);
   410→   410→                };
   411→   411→        }, [pageIndex]);
   412→   412→
   413→   413→        // ----------------------------------------------------------------
   414→   414→        // Register the store listener with { source: 'user', scope: 'document' }
   415→   415→        // so it only fires for local user edits (not for remote-applied changes).
   416→   416→        // ----------------------------------------------------------------
   417→   417→
   418→   418→        const registerStoreListener = useCallback(() => {
   419→   419→                const editor = editorRef.current;
   420→   420→                if (!editor) return;
   421→   421→
   422→   422→                // Clean up previous listener if any.
   423→   423→                unlistenStoreRef.current?.();
   424→   424→
   425→   425→                unlistenStoreRef.current = (editor.store as any).listen(
   426→   426→                        (entry: HistoryEntryLike) => {
   427→   427→                                processLocalDiff(entry);
   428→   428→                        },
   429→   429→                        { source: 'user', scope: 'document' } as any
   430→   430→                );
   431→   431→        }, [processLocalDiff]);
   432→   432→
   433→   433→        // ----------------------------------------------------------------
   434→   434→        // Handle editor mount (called once by Tldraw)
   435→   435→        // ----------------------------------------------------------------
   436→   436→
   437→   437→        const handleMount = useCallback(
   438→   438→                (editor: Editor) => {
   439→   439→                        editorRef.current = editor;
   440→   440→                        prevPageIndexRef.current = pageIndex;
   441→   441→
   442→   442→                        // Notify parent with the editor ref.
   443→   443→                        onEditorReady?.(editor);
   444→   444→
   445→   445→                        // Short delay lets the editor finish its internal initialization
   446→   446→                        // before we start loading external data.
   447→   447→                        setTimeout(() => {
   448→   448→                                // 1. Load initial state (legacy snapshot or per-record format).
   449→   449→                                loadInitialState();
   450→   450→
   451→   451→                                // 2. Register the store listener for user-initiated changes.
   452→   452→                                registerStoreListener();
   453→   453→
   454→   454→                                // 3. Set up the Yjs remote observer.
   455→   455→                                unobserveYjsRef.current = setupRemoteObserver();
   456→   456→
   457→   457→                                // 4. Mark as ready — enables the store listener's isReadyRef check.
   458→   458→                                isReadyRef.current = true;
   459→   459→                                setIsReady(true);
   460→   460→                        }, 100);
   461→   461→                },
   462→   462→                [onEditorReady, pageIndex, loadInitialState, registerStoreListener, setupRemoteObserver]
   463→   463→        );
   464→   464→
   465→   465→        // ----------------------------------------------------------------
   466→   466→        // Handle page changes: flush old page, load new page
   467→   467→        // ----------------------------------------------------------------
   468→   468→
   469→   469→        useEffect(() => {
   470→   470→                if (!editorRef.current || !ydoc) return;
   471→   471→                if (prevPageIndexRef.current === pageIndex) return;
   472→   472→
   473→   473→                // 1. Cancel any pending debounce.
   474→   474→                if (debounceTimerRef.current) {
   475→   475→                        clearTimeout(debounceTimerRef.current);
   476→   476→                        debounceTimerRef.current = null;
   477→   477→                }
   478→   478→
   479→   479→                // 2. Mark as not-ready to suppress store listener during transition.
   480→   480→                isReadyRef.current = false;
   481→   481→
   482→   482→                // 3. Flush pending changes for the OLD page (shapesMapRef still points to old map).
   483→   483→                flushPendingRef.current();
   484→   484→
   485→   485→                // 4. Clean up old Yjs observer.
   486→   486→                unobserveYjsRef.current?.();
   487→   487→                unobserveYjsRef.current = null;
   488→   488→
   489→   489→                // 5. Reset all pending/dirty state for the new page.
   490→   490→                pendingWritesRef.current = new Map();
   491→   491→                pendingDeletesRef.current = new Set();
   492→   492→                localDirtyRef.current = new Set();
   493→   493→
   494→   494→                // 6. Update page tracking.
   495→   495→                prevPageIndexRef.current = pageIndex;
   496→   496→
   497→   497→                // 7. Load the new page's state.
   498→   498→                setIsLoadingSnapshot(true);
   499→   499→                loadInitialState();
   500→   500→
   501→   501→                // 8. Set up new Yjs observer for the new page's map.
   502→   502→                unobserveYjsRef.current = setupRemoteObserver();
   503→   503→
   504→   504→                // 9. Re-enable the store listener.
   505→   505→                //    (registerStoreListener is idempotent — it cleans up the old one first.)
   506→   506→                registerStoreListener();
   507→   507→
   508→   508→                // 10. Mark as ready again.
   509→   509→                isReadyRef.current = true;
   510→   510→                setIsReady(true);
   511→   511→        }, [pageIndex, ydoc, loadInitialState, setupRemoteObserver, registerStoreListener]);
   512→   512→
   513→   513→        // ----------------------------------------------------------------
   514→   514→        // Cleanup on unmount
   515→   515→        // ----------------------------------------------------------------
   516→   516→
   517→   517→        useEffect(() => {
   518→   518→                return () => {
   519→   519→                        // Cancel pending debounce.
   520→   520→                        if (debounceTimerRef.current) {
   521→   521→                                clearTimeout(debounceTimerRef.current);
   522→   522→                                debounceTimerRef.current = null;
   523→   523→                        }
   524→   524→
   525→   525→                        // Flush any remaining pending changes immediately.
   526→   526→                        flushPendingRef.current();
   527→   527→
   528→   528→                        // Clean up listeners.
   529→   529→                        unlistenStoreRef.current?.();
   530→   530→                        unobserveYjsRef.current?.();
   531→   531→
   532→   532→                        editorRef.current = null;
   533→   533→                };
   534→   534→        }, []); // eslint-disable-line react-hooks/exhaustive-deps
   535→   535→
   536→   536→        // ----------------------------------------------------------------
   537→   537→        // Render
   538→   538→        // ----------------------------------------------------------------
   539→   539→
   540→   540→        return (
   541→   541→                <div className="w-full h-full relative">
   542→   542→                        {/* Loading overlay while snapshot is being loaded */}
   543→   543→                        {isLoadingSnapshot && (
   544→   544→                                <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur-sm transition-opacity">
   545→   545→                                        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   546→   546→                                                <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   547→   547→                                                <span>Loading canvas...</span>
   548→   548→                                        </div>
   549→   549→                                </div>
   550→   550→                        )}
   551→   551→
   552→   552→                        {/* Tldraw Editor */}
   553→   553→                        <Tldraw
   554→   554→                                onMount={handleMount}
   555→   555→                                options={{
   556→   556→                                        maxPages: 50,
   557→   557→                                        maxShapesPerPage: 500,
   558→   558→                                }}
   559→   559→                        />
   560→   560→                </div>
   561→   561→        );
   562→   562→}
