   247→   247→export function AuthenticatedDashboard({ user, userName, tierLoading, isAdmin }: { user: User; userName: string | null; tierLoading: boolean; isAdmin?: boolean }) {
   248→   248→  const { tier, setTier, setRoom } = useAppStore();
   249→   249→  const { brandColor, agencyName, setBrandColor } = useTheme();
   250→   250→  const {
   251→   251→    aiCreditsUsed,
   252→   252→    aiCreditsLimit,
   253→   253→    videoMinutesUsed,
   254→   254→    videoMinutesLimit,
   255→   255→    recordingsUsed,
   256→   256→    recordingsLimit,
   257→   257→    loading: usageLoading,
   258→   258→  } = useCredits(user.id);
   259→   259→
   260→   260→  // ---- State ----
   261→   261→  const [activeView, setActiveView] = useState<DashboardView>('overview');
   262→   262→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   263→   263→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   264→   264→
   265→   265→  // Dialog state
   266→   266→  const [showNewLesson, setShowNewLesson] = useState(false);
   267→   267→  const [selectedSubject, setSelectedSubject] = useState<Subject>('GENERAL');
   268→   268→  const [creating, setCreating] = useState(false);
   269→   269→  const [showSignOutConfirm, setShowSignOutConfirm] = useState(false);
   270→   270→
   271→   271→  // Settings state
   272→   272→  const [saveLoading, setSaveLoading] = useState(false);
   273→   273→  const [saveSuccess, setSaveSuccess] = useState(false);
   274→   274→
   275→   275→  // Onboarding state — show wizard to first-time users (no rooms yet)
   276→   276→  const [showOnboarding, setShowOnboarding] = useState(false);
   277→   277→  const [roomsFetched, setRoomsFetched] = useState(false);
   278→   278→  const [hasRooms, setHasRooms] = useState(false);
   279→   279→
   280→   280→  // Student mode toggle
   281→   281→  const [isStudentMode, setIsStudentMode] = useState(false);
   282→   282→
   283→   283→  const { toast } = useToast();
   284→   284→
   285→   285→  // ---- Check if user has any rooms (for onboarding) ----
   286→   286→  useEffect(() => {
   287→   287→    if (!user?.id) return;
   288→   288→    const checkRooms = async () => {
   289→   289→      try {
   290→   290→        const { authFetch } = await import('@/lib/auth-fetch');
   291→   291→        const res = await authFetch(`/api/room/list?tutorId=${user.id}`);
   292→   292→        if (res.ok) {
   293→   293→          const data = await res.json();
   294→   294→          const rooms = data.rooms ?? [];
   295→   295→          setHasRooms(rooms.length > 0);
   296→   296→          if (rooms.length === 0) {
   297→   297→            setShowOnboarding(true);
   298→   298→          }
   299→   299→        }
   300→   300→      } catch { /* silent — proceed to dashboard */ }
   301→   301→      setRoomsFetched(true);
   302→   302→    };
   303→   303→    checkRooms();
   304→   304→  }, [user?.id]);
   305→   305→
   306→   306→  // ---- Dark mode ----
   307→   307→  const [isDark, setIsDark] = useState(false);
   308→   308→  useEffect(() => {
   309→   309→    const saved = localStorage.getItem('superboard-theme');
   310→   310→    if (saved === 'dark') {
   311→   311→      document.documentElement.classList.add('dark');
   312→   312→      setIsDark(true);
   313→   313→    }
   314→   314→  }, []);
   315→   315→
   316→   316→  const toggleDark = useCallback(() => {
   317→   317→    const next = !isDark;
   318→   318→    setIsDark(next);
   319→   319→    if (next) {
   320→   320→      document.documentElement.classList.add('dark');
   321→   321→    } else {
   322→   322→      document.documentElement.classList.remove('dark');
   323→   323→    }
   324→   324→    localStorage.setItem('superboard-theme', next ? 'dark' : 'light');
   325→   325→  }, [isDark]);
   326→   326→
   327→   327→  // ---- Fingerprint reporting ----
   328→   328→  useEffect(() => {
   329→   329→    import('@/lib/fingerprint').then(({ reportFingerprint }) => {
   330→   330→      if (user?.id) reportFingerprint(user.id).catch(console.error);
   331→   331→    });
   332→   332→  }, [user?.id]);
   333→   333→
   334→   334→  // ---- Create lesson handler ----
   335→   335→  const handleCreateLesson = useCallback(async () => {
   336→   336→    setCreating(true);
   337→   337→    try {
   338→   338→      const supabase = createClient();
   339→   339→      const { data: { session } } = await supabase!.auth.getSession();
   340→   340→      const token = session?.access_token;
   341→   341→      if (!token) { setCreating(false); return; }
   342→   342→      const response = await fetch('/api/room', {
   343→   343→        method: 'POST',
   344→   344→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   345→   345→        body: JSON.stringify({ tutorId: user?.id, subject: selectedSubject }),
   346→   346→      });
   347→   347→      if (!response.ok) {
   348→   348→        const errBody = await response.text().catch(() => '');
   349→   349→        throw new Error(`Failed to create room (${response.status}): ${errBody}`);
   350→   350→      }
   351→   351→      const data = await response.json();
   352→   352→      window.location.href = `/room/${data.roomId}`;
   353→   353→    } catch (error: any) {
   354→   354→      toast({ title: 'Failed to create lesson', description: error?.message || 'Please try again.', variant: 'destructive' });
   355→   355→      setCreating(false);
   356→   356→    }
   357→   357→  }, [selectedSubject, user?.id, toast]);
   358→   358→
   359→   359→  // ---- Save brand color handler ----
   360→   360→  const handleSaveBrandColor = useCallback(async () => {
   361→   361→    setSaveLoading(true);
   362→   362→    setSaveSuccess(false);
   363→   363→    try {
   364→   364→      const supabase = createClient();
   365→   365→      const { data: { session } } = await supabase!.auth.getSession();
   366→   366→      const token = session?.access_token;
   367→   367→      if (!token) return;
   368→   368→      await fetch('/api/auth/profile', {
   369→   369→        method: 'PATCH',
   370→   370→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   371→   371→        body: JSON.stringify({ brandingColor: brandColor || null }),
   372→   372→      });
   373→   373→      setSaveSuccess(true);
   374→   374→      setTimeout(() => setSaveSuccess(false), 2000);
   375→   375→    } catch { /* silent */ }
   376→   376→    setSaveLoading(false);
   377→   377→  }, [brandColor]);
   378→   378→
   379→   379→  // ---- Logout handler ----
   380→   380→  const handleLogout = useCallback(async () => {
   381→   381→    const supabase = createClient();
   382→   382→    if (supabase) await supabase.auth.signOut();
   383→   383→    setTier('FREE');
   384→   384→    window.location.href = '/';
   385→   385→  }, [setTier]);
   386→   386→
   387→   387→  // ---- Onboarding complete handler ----
   388→   388→  const handleOnboardingComplete = useCallback((roomId: string) => {
   389→   389→    setShowOnboarding(false);
   390→   390→    setHasRooms(true);
   391→   391→    window.location.href = `/room/${roomId}`;
   392→   392→  }, []);
   393→   393→
   394→   394→  const handleOnboardingSkip = useCallback(() => {
   395→   395→    setShowOnboarding(false);
   396→   396→    setHasRooms(false);
   397→   397→  }, []);
   398→   398→
   399→   399→  // ---- Derived values ----
   400→   400→  const tierLabel = isAgencyTier(tier)
   401→   401→    ? tier === 'AGENCY_PREMIUM' ? 'Agency Premium'
   402→   402→    : tier === 'AGENCY_STANDARD' ? 'Agency Standard'
   403→   403→    : 'Agency'
   404→   404→    : tier === 'PRO' ? 'Pro' : 'Free';
   405→   405→  const tierColor =
   406→   406→    tier === 'AGENCY_PREMIUM' ? 'bg-purple-100 text-purple-800'
   407→   407→    : tier === 'AGENCY_STANDARD' || tier === 'AGENCY' ? 'bg-amber-100 text-amber-800'
   408→   408→    : tier === 'PRO' ? 'bg-emerald-100 text-emerald-800'
   409→   409→    : 'bg-teal-50 text-teal-700';
   410→   410→
   411→   411→  const navGroups = getNavItems(tier, !!isAdmin);
   412→   412→
   413→   413→  // ---- Show onboarding for first-time users ----
   414→   414→  if (showOnboarding && roomsFetched) {
   415→   415→    return (
   416→   416→      <OnboardingWizard
   417→   417→        userId={user.id}
   418→   418→        userEmail={user.email || ''}
   419→   419→        userName={userName}
   420→   420→        onComplete={handleOnboardingComplete}
   421→   421→        onSkip={handleOnboardingSkip}
   422→   422→      />
   423→   423→    );
   424→   424→  }
   425→   425→
   426→   426→  // ---- Show student dashboard if in student mode ----
   427→   427→  if (isStudentMode) {
   428→   428→    return (
   429→   429→      <StudentDashboard
   430→   430→        userId={user.id}
   431→   431→        userName={userName}
   432→   432→        userEmail={user.email || null}
   433→   433→      />
   434→   434→    );
   435→   435→  }
   436→   436→
   437→   437→  // ============================================================
   438→   438→  // RENDER — Dashboard with Sidebar
   439→   439→  // ============================================================
   440→   440→  return (
   441→   441→    <TooltipProvider delayDuration={0}>
   442→   442→      <div className="min-h-screen bg-background flex">
   443→   443→        {/* ============================================================
   444→   444→            SIDEBAR — Collapsible on desktop, sheet on mobile
   445→   445→            ============================================================ */}
   446→   446→        {/* Mobile overlay */}
   447→   447→        {mobileMenuOpen && (
   448→   448→          <div
   449→   449→            className="fixed inset-0 z-40 bg-black/40 backdrop-blur-sm lg:hidden"
   450→   450→            onClick={() => setMobileMenuOpen(false)}
   451→   451→          />
   452→   452→        )}
   453→   453→
   454→   454→        {/* Sidebar */}
   455→   455→        <aside
   456→   456→          className={`
   457→   457→            fixed lg:sticky top-0 left-0 z-50 lg:z-30
   458→   458→            h-screen flex flex-col
   459→   459→            bg-white border-r border-gray-200
   460→   460→            transition-all duration-300 ease-in-out
   461→   461→            ${sidebarCollapsed ? 'w-[68px]' : 'w-64'}
   462→   462→            ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full lg:translate-x-0'}
   463→   463→          `}
   464→   464→        >
   465→   465→          {/* Sidebar Header / Logo */}
   466→   466→          <div className="flex items-center justify-between px-4 h-16 border-b border-gray-100 shrink-0">
   467→   467→            <div className="flex items-center gap-2.5 min-w-0">
   468→   468→              <div className="w-9 h-9 rounded-xl gradient-primary flex items-center justify-center shadow-md shadow-emerald-500/20 shrink-0">
   469→   469→                <GraduationCap className="w-5 h-5 text-white" />
   470→   470→              </div>
   471→   471→              {!sidebarCollapsed && (
   472→   472→                <div className="min-w-0">
   473→   473→                  <h1 className="text-sm font-bold tracking-tight truncate">{agencyName || 'Superboard'}</h1>
   474→   474→                </div>
   475→   475→              )}
   476→   476→            </div>
   477→   477→            {/* Mobile close */}
   478→   478→            <Button
