    58→    58→/** Remap a color name through the color-blind palette if applicable */
    59→    59→function getColorBlindSafeColor(colorName: string, mode: string): string {
    60→    60→  if (mode === 'none' || !COLOR_BLIND_PALETTES[mode]) return colorName;
    61→    61→  return COLOR_BLIND_PALETTES[mode][colorName.toLowerCase()] || colorName;
    62→    62→}

   101→   101→// ---- Perfect-Freehand → Fabric.js Path ----
   102→   102→
   103→   103→function createFreehandPath(
   104→   104→  points: FreehandPoint[],
   105→   105→  strokeColor: string,
   106→   106→  strokeWidth: number
   107→   107→): Path | null {
   108→   108→  if (points.length < 2) return null;
   109→   109→
   110→   110→  const outline = getStroke(points, {
   111→   111→    size: strokeWidth * 3,
   112→   112→    thinning: 0.5,
   113→   113→    smoothing: 0.6,
   114→   114→    streamline: 0.5,
   115→   115→    simulatePressure: true,
   116→   116→  });
   117→   117→
   118→   118→  if (outline.length < 2) return null;
   119→   119→
   120→   120→  // Build SVG path string from the outline points
   121→   121→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
   122→   122→  for (let i = 1; i < outline.length; i++) {
   123→   123→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
   124→   124→  }
   125→   125→  d += ' Z';
   126→   126→
   127→   127→  try {
   128→   128→    const path = new Path(d, {
   129→   129→      fill: strokeColor,
   130→   130→      stroke: strokeColor,
   131→   131→      strokeWidth: 0.5,
   132→   132→      strokeLineCap: 'round',
   133→   133→      strokeLineJoin: 'round',
   134→   134→    } as any);
   135→   135→    return path;
   136→   136→  } catch {
   137→   137→    return null;
   138→   138→  }
   139→   139→}

   146→   146→}

   816→   816→          history.pushState('draw', beforeState, fc.toJSON().toString());
   817→   817→          fc.renderAll();
   818→   818→          return;

   824→   824→            selectable: true,
   825→   825→            evented: true,
   826→   826→          });
   827→   827→
   828→   828→          history.pushState(
   829→   829→            `draw-${tools.tool}`,
   830→   830→            beforeState,
   831→   831→            fc.toJSON().toString()
   832→   832→          );
   833→   833→          tempShapeRef.current = null;
   834→   834→          isDrawingRef.current = false;
   835→   835→          startPointRef.current = null;
   836→   836→          fc.renderAll();

   838→   838→      };

   841→   841→      fc.on('mouse:move', handleMouseMove);
   842→   842→      fc.on('mouse:up', handleMouseUp);

   882→   882→    // ============================================================
   883→   883→    // Touch: Pinch-to-zoom via native DOM events on container
   884→   884→    // ============================================================
   885→   885→    useEffect(() => {
   886→   886→      const container = containerRef.current;
   887→   887→      const fc = fcanvasRef.current;
   888→   888→      if (!container || !fc) return;
   889→   889→
   890→   890→      const domTouchStart = (e: TouchEvent) => {
   891→   891→        const touches = e.touches;
   892→   892→        if (touches.length === 2) {
   893→   893→          e.preventDefault();
   894→   894→          e.stopPropagation();
   895→   895→
   896→   896→          const dx = touches[0].clientX - touches[1].clientX;
   897→   897→          const dy = touches[0].clientY - touches[1].clientY;
   898→   898→          const dist = Math.sqrt(dx * dx + dy * dy);
   899→   899→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   900→   900→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   901→   901→
   902→   902→          touchStateRef.current = {
   903→   903→            isPinching: true,
   904→   904→            lastDist: dist,
   905→   905→            lastCenter: { x: centerX, y: centerY },
   906→   906→            touchCount: 2,
   907→   907→          };
   908→   908→
   909→   909→          // Cancel any active drawing
   910→   910→          isDrawingRef.current = false;
   911→   911→          tempShapeRef.current = null;
   912→   912→          freehandPointsRef.current = [];
   913→   913→          if (freehandTempPathRef.current) {
   914→   914→            fc.remove(freehandTempPathRef.current);
   915→   915→            freehandTempPathRef.current = null;
   916→   916→          }
   917→   917→        }
   918→   918→      };
   919→   919→
   920→   920→      const domTouchMove = (e: TouchEvent) => {
   921→   921→        const touches = e.touches;
   922→   922→        if (touches.length === 2 && touchStateRef.current.isPinching) {
   923→   923→          e.preventDefault();
   924→   924→          e.stopPropagation();
   925→   925→
   926→   926→          const dx = touches[0].clientX - touches[1].clientX;
   927→   927→          const dy = touches[0].clientY - touches[1].clientY;
   928→   928→          const dist = Math.sqrt(dx * dx + dy * dy);
   929→   929→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   930→   930→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   931→   931→
   932→   932→          const prev = touchStateRef.current;
   933→   933→
   934→   934→          // Zoom
   935→   935→          const scale = dist / Math.max(prev.lastDist, 1);
   936→   936→          let zoom = fc.getZoom() * scale;
   937→   937→          zoom = Math.min(Math.max(0.1, zoom), 10);
   938→   938→
   939→   939→          // Pan
   940→   940→          const vpt = fc.viewportTransform!;
   941→   941→          vpt[4] += centerX - prev.lastCenter.x;
   942→   942→          vpt[5] += centerY - prev.lastCenter.y;
   943→   943→
   944→   944→          // Apply zoom to center of pinch
   945→   945→          fc.zoomToPoint(new FabricPoint(centerX, centerY), zoom);
   946→   946→
   947→   947→          touchStateRef.current = {
   948→   948→            isPinching: true,
   949→   949→            lastDist: dist,
   950→   950→            lastCenter: { x: centerX, y: centerY },
   951→   951→            touchCount: 2,
   952→   952→          };
   953→   953→
   954→   954→          requestRenderThrottled();
   955→   955→        }
   956→   956→      };
   957→   957→
   958→   958→      const domTouchEnd = (e: TouchEvent) => {
   959→   959→        if (e.touches.length < 2) {
   960→   960→          touchStateRef.current.isPinching = false;
   961→   961→          touchStateRef.current.touchCount = e.touches.length;
   962→   962→        }
   963→   963→      };
   964→   964→
   965→   965→      container.addEventListener('touchstart', domTouchStart, { passive: false });
   966→   966→      container.addEventListener('touchmove', domTouchMove, { passive: false });
   967→   967→      container.addEventListener('touchend', domTouchEnd, { passive: false });
   968→   968→
   969→   969→      return () => {
   970→   970→        container.removeEventListener('touchstart', domTouchStart);
