     1→     1→// ============================================================
     2→     2→// Manipulative Renderer — AI-Generated Math Manipulatives
     3→     3→// ============================================================
     4→     4→// Converts AI-generated manipulative descriptions into Fabric.js
     5→     5→// object groups. Each manipulative is a self-contained Group of
     6→     6→// objects that can be moved, resized, and interacted with.
     7→     7→//
     8→     8→// Used by AI actions like "create-fraction-bars", "create-number-line",
     9→     9→// "create-base-ten-blocks", etc.
    10→    10→// ============================================================
    11→    11→
    12→    12→import {
    13→    13→  Rect,
    14→    14→  Circle,
    15→    15→  Line,
    16→    16→  IText,
    17→    17→  Group,
    18→    18→  Triangle,
    19→    19→  Polygon,
    20→    20→  type Object as FabricObject,
    21→    21→} from 'fabric';
    22→    22→
    23→    23→// ---- Types ----
    24→    24→
    25→    25→export interface ManipulativeSpec {
    26→    26→  type: string;
    27→    27→  params: Record<string, any>;
    28→    28→}
    29→    29→
    30→    30→// ---- Color palette for manipulatives ----
    31→    31→
    32→    32→const MANIPULATIVE_COLORS = [
    33→    33→  '#ef4444', // red
    34→    34→  '#f97316', // orange
    35→    35→  '#eab308', // yellow
    36→    36→  '#22c55e', // green
    37→    37→  '#3b82f6', // blue
    38→    38→  '#8b5cf6', // purple
    39→    39→  '#ec4899', // pink
    40→    40→  '#14b8a6', // teal
    41→    41→];
    42→    42→
    43→    43→function getColor(index: number): string {
    44→    44→  return MANIPULATIVE_COLORS[index % MANIPULATIVE_COLORS.length];
    45→    45→}

    47→    47→// ---- Main entry point ----
    48→    48→
    49→    49→/**
    50→    50→ * Render a manipulative specification into an array of Fabric.js objects.
    51→    51→ * Each manipulative type has its own factory function.
    52→    52→ */
    53→    53→export function renderManipulative(spec: ManipulativeSpec): FabricObject[] {
    54→    54→  switch (spec.type) {
    55→    55→    case 'fraction-bar':
    56→    56→      return createFractionBars(spec.params);
    57→    57→    case 'number-line':
    58→    58→      return createNumberLine(spec.params);
    59→    59→    case 'base-ten-blocks':
    60→    60→      return createBaseTenBlocks(spec.params);
    61→    61→    case 'coordinate-grid':
    62→    62→      return createCoordinateGrid(spec.params);
    63→    63→    case 'angle-protractor':
    64→    64→      return createAngleProtractor(spec.params);
    65→    65→    case 'geometry-shape':
    66→    66→      return createGeometryShape(spec.params);
    67→    67→    case 'place-value-chart':
    68→    68→      return createPlaceValueChart(spec.params);
    69→    69→    case 'clock':
    70→    70→      return createClock(spec.params);
    71→    71→    case 'bar-chart':
    72→    72→      return createBarChart(spec.params);
    73→    73→    default:
    74→    74→      return createGenericManipulative(spec);
    75→    75→  }
    76→    76→}

    79→    79→// Fraction Bars

    81→    81→// Visual representation of fractions using colored bars.
    82→    82→// params: { numerator, denominator, label? }
    83→    83→
    84→    84→function createFractionBars(params: Record<string, any>): FabricObject[] {
    85→    85→  const {
    86→    86→    numerator = 1,
    87→    87→    denominator = 4,
    88→    88→    label,
    89→    89→    x = 100,
    90→    90→    y = 100,
    91→    91→    width = 300,
    92→    92→    height = 40,
    93→    93→  } = params;
    94→    94→
    95→    95→  const objects: FabricObject[] = [];
    96→    96→  const slotWidth = width / denominator;
    97→    97→
    98→    98→  // Background bar (outline)
    99→    99→  const bg = new Rect({
   100→   100→    left: x,
   101→   101→    top: y,
   102→   102→    width,
   103→   103→    height,
   104→   104→    fill: 'transparent',
   105→   105→    stroke: '#374151',
   106→   106→    strokeWidth: 2,
   107→   107→    rx: 4,
   108→   108→    ry: 4,
   109→   109→  } as any);
   110→   110→  objects.push(bg);
   111→   111→
   112→   112→  // Filled slots (numerator)
   113→   113→  for (let i = 0; i < numerator && i < denominator; i++) {
   114→   114→    const filled = new Rect({
   115→   115→      left: x + i * slotWidth + 1,
   116→   116→      top: y + 1,
   117→   117→      width: slotWidth - 2,
   118→   118→      height: height - 2,
   119→   119→      fill: getColor(i),
   120→   120→      stroke: '#374151',
   121→   121→      strokeWidth: 1,
   122→   122→      rx: 3,
   123→   123→      ry: 3,
   124→   124→    } as any);
   125→   125→    objects.push(filled);
   126→   126→  }
   127→   127→
   128→   128→  // Slot dividers
   129→   129→  for (let i = 1; i < denominator; i++) {
   130→   130→    const divider = new Line(
   131→   131→      [x + i * slotWidth, y, x + i * slotWidth, y + height],
   132→   132→      {
   133→   133→        stroke: '#374151',
   134→   134→        strokeWidth: 1.5,
   135→   135→      } as any
   136→   136→    );
   137→   137→    objects.push(divider);
   138→   138→  }
   139→   139→
   140→   140→  // Label
   141→   141→  const labelText = label || `${numerator}/${denominator}`;
   142→   142→  const labelObj = new IText(labelText, {
   143→   143→    left: x + width / 2,
   144→   144→    top: y + height + 12,
   145→   145→    fontSize: 18,
   146→   146→    fontFamily: 'Inter, system-ui, sans-serif',
   147→   147→    fill: '#374151',
   148→   148→    originX: 'center',
   149→   149→    originY: 'top',
   150→   150→    textAlign: 'center',
   151→   151→  } as any);
   152→   152→  objects.push(labelObj);
   153→   153→
   154→   154→  // Wrap in a group
   155→   155→  const group = new Group(objects, {
   156→   156→    left: x,
   157→   157→    top: y,
   158→   158→  } as any);
   159→   159→  (group as any).name = `manipulative-fraction-bar-${Date.now()}`;
   160→   160→  return [group];
   161→   161→}

   166→   166→// Horizontal number line with tick marks and labels.
   167→   167→// params: { min, max, step?, highlightValues?, label? }
   168→   168→
   169→   169→function createNumberLine(params: Record<string, any>): FabricObject[] {
   170→   170→  const {
   171→   171→    min = 0,
   172→   172→    max = 10,
   173→   173→    step = 1,
   174→   174→    highlightValues = [],
   175→   175→    label,
   176→   176→    x = 100,
   177→   177→    y = 200,
   178→   178→    length = 500,
   179→   179→  } = params;
   180→   180→
   181→   181→  const objects: FabricObject[] = [];
   182→   182→  const tickSpacing = length / ((max - min) / step);
   183→   183→
   184→   184→  // Main line
   185→   185→  const mainLine = new Line([x, y, x + length, y], {
   186→   186→    stroke: '#374151',
   187→   187→    strokeWidth: 2,
   188→   188→    strokeLineCap: 'round',
   189→   189→  } as any);
   190→   190→  objects.push(mainLine);
   191→   191→
   192→   192→  // Arrow at end
   193→   193→  const arrow = new Triangle({
   194→   194→    left: x + length,
   195→   195→    top: y,
   196→   196→    width: 12,
   197→   197→    height: 12,
   198→   198→    fill: '#374151',
   199→   199→    angle: 90,
   200→   200→    originX: 'center',
   201→   201→    originY: 'center',
   202→   202→  } as any);
   203→   203→  objects.push(arrow);
   204→   204→
   205→   205→  // Tick marks and labels
   206→   206→  const totalTicks = Math.round((max - min) / step);
   207→   207→  for (let i = 0; i <= totalTicks; i++) {
   208→   208→    const value = min + i * step;
   209→   209→    const tickX = x + i * tickSpacing;
   210→   210→    const isHighlighted = highlightValues.includes(value);
   211→   211→
   212→   212→    // Tick mark
   213→   213→    const tick = new Line([tickX, y - 8, tickX, y + 8], {
   214→   214→      stroke: isHighlighted ? '#059669' : '#374151',
   215→   215→      strokeWidth: isHighlighted ? 3 : 1.5,
   216→   216→    } as any);
   217→   217→    objects.push(tick);
   218→   218→
   219→   219→    // Highlight dot
   220→   220→    if (isHighlighted) {
   221→   221→      const dot = new Circle({
   222→   222→        left: tickX,
   223→   223→        top: y - 16,
   224→   224→        radius: 6,
   225→   225→        fill: '#059669',
   226→   226→        originX: 'center',
   227→   227→        originY: 'center',
   228→   228→      } as any);
   229→   229→      objects.push(dot);
   230→   230→    }
   231→   231→
   232→   232→    // Number label (show every step, or every other for small steps)
   233→   233→    const showLabel = step >= 1 || i % Math.max(1, Math.round(1 / step)) === 0;
   234→   234→    if (showLabel) {
   235→   235→      const numLabel = new IText(
   236→   236→        Number.isInteger(value) ? String(value) : value.toFixed(1),
   237→   237→        {
   238→   238→          left: tickX,
   239→   239→          top: y + 16,
   240→   240→          fontSize: 14,
   241→   241→          fontFamily: 'Inter, system-ui, sans-serif',
   242→   242→          fill: '#374151',
   243→   243→          originX: 'center',
   244→   244→          originY: 'top',
   245→   245→          textAlign: 'center',
   246→   246→        } as any
   247→   247→      );
   248→   248→      objects.push(numLabel);
   249→   249→    }
   250→   250→  }
   251→   251→
   252→   252→  // Optional top label
   253→   253→  if (label) {
   254→   254→    const titleLabel = new IText(label, {
   255→   255→      left: x + length / 2,
   256→   256→      top: y - 40,
   257→   257→      fontSize: 16,
   258→   258→      fontFamily: 'Inter, system-ui, sans-serif',
   259→   259→      fill: '#374151',
   260→   260→      originX: 'center',
   261→   261→      originY: 'top',
   262→   262→      fontWeight: 'bold',
   263→   263→    } as any);
   264→   264→    objects.push(titleLabel);
   265→   265→  }
   266→   266→
   267→   267→  const group = new Group(objects, {
   268→   268→    left: x,
   269→   269→    top: y - 40,
   270→   270→  } as any);
   271→   271→  (group as any).name = `manipulative-number-line-${Date.now()}`;
   272→   272→  return [group];
   273→   273→}

   278→   278→// Visual representation of base-10 place value.
   279→   279→// params: { hundreds, tens, ones, x?, y? }
   280→   280→
   281→   281→function createBaseTenBlocks(params: Record<string, any>): FabricObject[] {
   282→   282→  const {
   283→   283→    hundreds = 0,
   284→   284→    tens = 0,
   285→   285→    ones = 0,
   286→   286→    x = 100,
   287→   287→    y = 100,
   288→   288→    unitSize = 20,
   289→   289→  } = params;
   290→   290→
   291→   291→  const objects: FabricObject[] = [];
   292→   292→  let currentX = x;
   293→   293→  const gap = 10;
   294→   294→
   295→   295→  // Label
   296→   296→  if (hundreds > 0 || tens > 0 || ones > 0) {
   297→   297→    const titleLabel = new IText('Base-Ten Blocks', {
   298→   298→      left: x,
   299→   299→      top: y - 30,
   300→   300→      fontSize: 16,
   301→   301→      fontFamily: 'Inter, system-ui, sans-serif',
   302→   302→      fill: '#374151',
   303→   303→      fontWeight: 'bold',
   304→   304→    } as any);
   305→   305→    objects.push(titleLabel);
   306→   306→  }
   307→   307→
   308→   308→  // Hundreds (large 10x10 grid)
   309→   309→  for (let h = 0; h < Math.min(hundreds, 5); h++) {
   310→   310→    const block = new Rect({
   311→   311→      left: currentX,
   312→   312→      top: y,
   313→   313→      width: unitSize * 10,
   314→   314→      height: unitSize * 10,
   315→   315→      fill: '#fbbf24',
   316→   316→      stroke: '#374151',
   317→   317→      strokeWidth: 2,
   318→   318→    } as any);
   319→   319→    objects.push(block);
   320→   320→
   321→   321→    // Grid lines for hundreds block
   322→   322→    for (let i = 1; i < 10; i++) {
   323→   323→      const vLine = new Line(
   324→   324→        [currentX + i * unitSize, y, currentX + i * unitSize, y + unitSize * 10],
   325→   325→        { stroke: '#92400e', strokeWidth: 0.5 } as any
   326→   326→      );
   327→   327→      objects.push(vLine);
   328→   328→      const hLine = new Line(
   329→   329→        [currentX, y + i * unitSize, currentX + unitSize * 10, y + i * unitSize],
   330→   330→        { stroke: '#92400e', strokeWidth: 0.5 } as any
   331→   331→      );
   332→   332→      objects.push(hLine);
   333→   333→    }
   334→   334→
   335→   335→    // Count label
   336→   336→    const hLabel = new IText(`${hundreds}00`, {
   337→   337→      left: currentX + unitSize * 5,
   338→   338→      top: y + unitSize * 10 + 8,
   339→   339→      fontSize: 14,
   340→   340→      fontFamily: 'Inter, system-ui, sans-serif',
   341→   341→      fill: '#92400e',
   342→   342→      originX: 'center',
   343→   343→    } as any);
   344→   344→    objects.push(hLabel);
   345→   345→
   346→   346→    currentX += unitSize * 10 + gap;
   347→   347→  }
   348→   348→
   349→   349→  // Tens (1x10 rods)
   350→   350→  for (let t = 0; t < Math.min(tens, 5); t++) {
   351→   351→    const rod = new Rect({
   352→   352→      left: currentX,
   353→   353→      top: y,
   354→   354→      width: unitSize,
   355→   355→      height: unitSize * 10,
   356→   356→      fill: '#60a5fa',
   357→   357→      stroke: '#1e3a5f',
   358→   358→      strokeWidth: 1.5,
   359→   359→    } as any);
   360→   360→    objects.push(rod);
   361→   361→
   362→   362→    // Grid lines for tens rod
   363→   363→    for (let i = 1; i < 10; i++) {
   364→   364→      const hLine = new Line(
   365→   365→        [currentX, y + i * unitSize, currentX + unitSize, y + i * unitSize],
   366→   366→        { stroke: '#1e3a5f', strokeWidth: 0.5 } as any
   367→   367→      );
   368→   368→      objects.push(hLine);
   369→   369→    }
   370→   370→
   371→   371→    const tLabel = new IText(`${tens}x10`, {
   372→   372→      left: currentX + unitSize / 2,
   373→   373→      top: y + unitSize * 10 + 8,
   374→   374→      fontSize: 14,
   375→   375→      fontFamily: 'Inter, system-ui, sans-serif',
   376→   376→      fill: '#1e3a5f',
   377→   377→      originX: 'center',
   378→   378→    } as any);
   379→   379→    objects.push(tLabel);
   380→   380→
   381→   381→    currentX += unitSize + gap;
   382→   382→  }
   383→   383→
   384→   384→  // Ones (individual unit cubes)
   385→   385→  for (let o = 0; o < Math.min(ones, 10); o++) {
   386→   386→    const cube = new Rect({
   387→   387→      left: currentX + (o % 5) * (unitSize + gap),
   388→   388→      top: y + Math.floor(o / 5) * (unitSize + gap),
   389→   389→      width: unitSize,
   390→   390→      height: unitSize,
   391→   391→      fill: '#34d399',
   392→   392→      stroke: '#065f46',
   393→   393→      strokeWidth: 1.5,
   394→   394→    } as any);
   395→   395→    objects.push(cube);
   396→   396→  }
   397→   397→
   398→   398→  if (ones > 0) {
   399→   399→    const oLabel = new IText(`${ones}`, {
   400→   400→      left: currentX + Math.min(ones, 5) * (unitSize + gap) / 2 - gap / 2,
   401→   401→      top: y + (ones > 5 ? 2 : 1) * (unitSize + gap) + 8,
   402→   402→      fontSize: 14,
   403→   403→      fontFamily: 'Inter, system-ui, sans-serif',
   404→   404→      fill: '#065f46',
   405→   405→      originX: 'center',
   406→   406→    } as any);
   407→   407→    objects.push(oLabel);
   408→   408→  }
   409→   409→
   410→   410→  const group = new Group(objects, {
   411→   411→    left: x,
   412→   412→    top: y - 30,
   413→   413→  } as any);
   414→   414→  (group as any).name = `manipulative-base-ten-${Date.now()}`;
   415→   415→  return [group];
   416→   416→}

   418→   418→// ============================================================
   419→   419→// Coordinate Grid
   420→   420→// ============================================================
   421→   421→// X-Y coordinate plane with grid lines.
   422→   422→// params: { xMin, xMax, yMin, yMax, step?, points? }
   423→   423→
   424→   424→function createCoordinateGrid(params: Record<string, any>): FabricObject[] {
   425→   425→  const {
   426→   426→    xMin = -5,
   427→   427→    xMax = 5,
   428→   428→    yMin = -5,
   429→   429→    yMax = 5,
   430→   430→    step = 1,
   431→   431→    points = [],
   432→   432→    x = 200,
   433→   433→    y = 200,
   434→   434→    cellSize = 40,
   435→   435→  } = params;
   436→   436→
   437→   437→  const objects: FabricObject[] = [];
   438→   438→  const originX = (0 - xMin) * cellSize;
   439→   439→  const originY = (yMax - 0) * cellSize;
   440→   440→
   441→   441→  // Grid lines
   442→   442→  for (let gx = xMin; gx <= xMax; gx += step) {
   443→   443→    const px = (gx - xMin) * cellSize;
   444→   444→    const isAxis = gx === 0;
   445→   445→    const line = new Line([px, 0, px, (yMax - yMin) * cellSize], {
   446→   446→      stroke: isAxis ? '#374151' : '#e5e7eb',
   447→   447→      strokeWidth: isAxis ? 2 : 0.5,
   448→   448→    } as any);
   449→   449→    objects.push(line);
   450→   450→  }
   451→   451→
   452→   452→  for (let gy = yMin; gy <= yMax; gy += step) {
   453→   453→    const py = (yMax - gy) * cellSize;
   454→   454→    const isAxis = gy === 0;
   455→   455→    const line = new Line([0, py, (xMax - xMin) * cellSize, py], {
   456→   456→      stroke: isAxis ? '#374151' : '#e5e7eb',
   457→   457→      strokeWidth: isAxis ? 2 : 0.5,
   458→   458→    } as any);
   459→   459→    objects.push(line);
   460→   460→  }
   461→   461→
   462→   462→  // Axis labels
   463→   463→  for (let gx = xMin; gx <= xMax; gx += step) {
   464→   464→    if (gx === 0) continue;
   465→   465→    const px = (gx - xMin) * cellSize;
   466→   466→    const label = new IText(String(gx), {
   467→   467→      left: px,
   468→   468→      top: originY + 6,
   469→   469→      fontSize: 11,
   470→   470→      fontFamily: 'Inter, system-ui, sans-serif',
   471→   471→      fill: '#6b7280',
   472→   472→      originX: 'center',
   473→   473→      originY: 'top',
   474→   474→    } as any);
   475→   475→    objects.push(label);
   476→   476→  }
   477→   477→
   478→   478→  for (let gy = yMin; gy <= yMax; gy += step) {
   479→   479→    if (gy === 0) continue;
   480→   480→    const py = (yMax - gy) * cellSize;
   481→   481→    const label = new IText(String(gy), {
   482→   482→      left: originX - 8,
   483→   483→      top: py,
   484→   484→      fontSize: 11,
   485→   485→      fontFamily: 'Inter, system-ui, sans-serif',
   486→   486→      fill: '#6b7280',
   487→   487→      originX: 'right',
   488→   488→      originY: 'center',
   489→   489→    } as any);
   490→   490→    objects.push(label);
   491→   491→  }
   492→   492→
   493→   493→  // Origin label
   494→   494→  const originLabel = new IText('0', {
   495→   495→    left: originX - 8,
   496→   496→    top: originY + 6,
   497→   497→    fontSize: 11,
   498→   498→    fontFamily: 'Inter, system-ui, sans-serif',
   499→   499→    fill: '#6b7280',
   500→   500→    originX: 'right',
   501→   501→    originY: 'top',
   502→   502→  } as any);
   503→   503→  objects.push(originLabel);
   504→   504→
   505→   505→  // Plotted points
   506→   506→  points.forEach((pt: { x: number; y: number; label?: string; color?: string }, idx: number) => {
   507→   507→    const px = (pt.x - xMin) * cellSize;
   508→   508→    const py = (yMax - pt.y) * cellSize;
   509→   509→    const color = pt.color || getColor(idx);
   510→   510→
   511→   511→    const dot = new Circle({
   512→   512→      left: px,
   513→   513→      top: py,
   514→   514→      radius: 5,
   515→   515→      fill: color,
   516→   516→      stroke: '#ffffff',
   517→   517→      strokeWidth: 2,
   518→   518→      originX: 'center',
   519→   519→      originY: 'center',
   520→   520→    } as any);
   521→   521→    objects.push(dot);
   522→   522→
   523→   523→    if (pt.label) {

   549→   549→function createAngleProtractor(params: Record<string, any>): FabricObject[] {
   550→   550→  const {
   551→   551→    degrees = 90,
   552→   552→    radius = 80,
   553→   553→    label,
   554→   554→    x = 200,
   555→   555→    y = 250,
   556→   556→  } = params;
   557→   557→
   558→   558→  const objects: FabricObject[] = [];
   559→   559→  const rad = (degrees * Math.PI) / 180;
   560→   560→
   561→   561→  // Baseline arm
   562→   562→  const baseArm = new Line([x, y, x + radius + 20, y], {
   563→   563→    stroke: '#374151',
   564→   564→    strokeWidth: 2,
   565→   565→    strokeLineCap: 'round',
   566→   566→  } as any);
   567→   567→  objects.push(baseArm);
   568→   568→
   569→   569→  // Angle arm
   570→   570→  const endX = x + (radius + 20) * Math.cos(-rad);
   571→   571→  const endY = y + (radius + 20) * Math.sin(-rad);
   572→   572→  const angleArm = new Line([x, y, endX, endY], {
   573→   573→    stroke: '#374151',
   574→   574→    strokeWidth: 2,
   575→   575→    strokeLineCap: 'round',
   576→   576→  } as any);
   577→   577→  objects.push(angleArm);
   578→   578→
   579→   579→  // Arc (approximate with line segments)
   580→   580→  const arcSegments = 36;
   581→   581→  const arcRadius = radius * 0.4;
   582→   582→  for (let i = 0; i < arcSegments; i++) {
   583→   583→    const startAngle = -((i / arcSegments) * rad);
   584→   584→    const endAngle = -(((i + 1) / arcSegments) * rad);
   585→   585→    const x1 = x + arcRadius * Math.cos(startAngle);
   586→   586→    const y1 = y + arcRadius * Math.sin(startAngle);
   587→   587→    const x2 = x + arcRadius * Math.cos(endAngle);
   588→   588→    const y2 = y + arcRadius * Math.sin(endAngle);
   589→   589→
   590→   590→    const seg = new Line([x1, y1, x2, y2], {
   591→   591→      stroke: '#059669',
   592→   592→      strokeWidth: 2,
   593→   593→    } as any);
   594→   594→    objects.push(seg);
   595→   595→  }
   596→   596→
   597→   597→  // Angle label
   598→   598→  const labelAngle = -rad / 2;
   599→   599→  const labelRadius = radius * 0.55;
   600→   600→  const angleLabel = new IText(
   601→   601→    label || `${degrees}°`,
   602→   602→    {
   603→   603→      left: x + labelRadius * Math.cos(labelAngle),
   604→   604→      top: y + labelRadius * Math.sin(labelAngle),
   605→   605→      fontSize: 18,
   606→   606→      fontFamily: 'Inter, system-ui, sans-serif',
   607→   607→      fill: '#059669',
   608→   608→      originX: 'center',
   609→   609→      originY: 'center',
   610→   610→      fontWeight: 'bold',
   611→   611→    } as any
   612→   612→  );
   613→   613→  objects.push(angleLabel);
   614→   614→
   615→   615→  // Vertex dot
   616→   616→  const vertex = new Circle({
   617→   617→    left: x,
   618→   618→    top: y,
   619→   619→    radius: 4,
   620→   620→    fill: '#374151',
   621→   621→    originX: 'center',
   622→   622→    originY: 'center',
   623→   623→  } as any);
   624→   624→  objects.push(vertex);
   625→   625→
   626→   626→  // Right angle indicator for 90°
   627→   627→  if (degrees === 90) {
   628→   628→    const squareSize = 15;
   629→   629→    const square = new Line(
   630→   630→      [x + squareSize, y, x + squareSize, y - squareSize, x, y - squareSize],
   631→   631→      {
   632→   632→        stroke: '#059669',
   633→   633→        strokeWidth: 1.5,
   634→   634→      } as any
   635→   635→    );
   636→   636→    objects.push(square);
   637→   637→  }
   638→   638→
   639→   639→  const group = new Group(objects, {
   640→   640→    left: x - 20,
   641→   641→    top: y - radius - 20,
   642→   642→  } as any);
   643→   643→  (group as any).name = `manipulative-angle-${Date.now()}`;
   644→   644→  return [group];
   645→   645→}

   653→   653→function createGeometryShape(params: Record<string, any>): FabricObject[] {
   654→   654→  const {
   655→   655→    shape = 'square',

   660→   660→    label,

   662→   662→    y = 200,

   667→   667→  } = params;
   668→   668→
   669→   669→  const objects: FabricObject[] = [];
   670→   670→  let shapeObj: FabricObject;
   671→   671→  const w = width || sideLength;
   672→   672→  const h = height || sideLength;
   673→   673→
   674→   674→  switch (shape) {
   675→   675→    case 'rectangle':
   676→   676→      shapeObj = new Rect({
   677→   677→        left: x, top: y, width: w, height: h,
   678→   678→        fill, stroke, strokeWidth,
   679→   679→      } as any);

   681→   681→
   682→   682→    case 'square':
   683→   683→      shapeObj = new Rect({
   684→   684→        left: x, top: y, width: sideLength, height: sideLength,
   685→   685→        fill, stroke, strokeWidth,
   686→   686→      } as any);
   687→   687→      break;
   688→   688→
   689→   689→    case 'circle':
   690→   690→      shapeObj = new Circle({
   691→   691→        left: x + radius, top: y + radius, radius,
   692→   692→        fill, stroke, strokeWidth,
   693→   693→        originX: 'center', originY: 'center',
   694→   694→      } as any);
   695→   695→      break;
   696→   696→
   697→   697→    case 'triangle': {
   698→   698→      const triHeight = sideLength * (Math.sqrt(3) / 2);
   699→   699→      const triPoints = [
   700→   700→        { x: x + sideLength / 2, y: y },
   701→   701→        { x: x + sideLength, y: y + triHeight },
   702→   702→        { x: x, y: y + triHeight },
   703→   703→      ];
   704→   704→      shapeObj = new Polygon(triPoints, {
   705→   705→        fill, stroke, strokeWidth,
   706→   706→      } as any);
   707→   707→      break;
   708→   708→    }
   709→   709→
   710→   710→    case 'pentagon': {
   711→   711→      const pentPoints = generateRegularPolygonPoints(x + sideLength / 2, y + sideLength / 2, sideLength / 2, 5, -Math.PI / 2);
   712→   712→      shapeObj = new Polygon(pentPoints, {
   713→   713→        fill, stroke, strokeWidth,
   714→   714→      } as any);
   715→   715→      break;
   716→   716→    }
   717→   717→
   718→   718→    case 'hexagon': {
   719→   719→      const hexPoints = generateRegularPolygonPoints(x + sideLength / 2, y + sideLength / 2, sideLength / 2, 6, 0);
   720→   720→      shapeObj = new Polygon(hexPoints, {
   721→   721→        fill, stroke, strokeWidth,
   722→   722→      } as any);
   723→   723→      break;
   724→   724→    }
   725→   725→
   726→   726→    default:

   728→   728→        left: x, top: y, width: w, height: h,
   729→   729→        fill, stroke, strokeWidth,
   730→   730→      } as any);
   731→   731→  }
   732→   732→
   733→   733→  objects.push(shapeObj);
   734→   734→
   735→   735→  // Dimension labels
   736→   736→  if (showDimensions) {
   737→   737→    if (shape === 'circle') {

   743→   743→        fill: '#6b7280',
   744→   744→        originX: 'center',
   745→   745→      } as any);
   746→   746→      objects.push(dimLabel);
   747→   747→    } else if (shape === 'rectangle' || shape === 'square') {
   748→   748→      const actualW = shape === 'square' ? sideLength : w;
   749→   749→      const actualH = shape === 'square' ? sideLength : h;
   750→   750→      const wLabel = new IText(`${actualW}`, {
   751→   751→        left: x + actualW / 2,
   752→   752→        top: y + actualH + 10,
   753→   753→        fontSize: 14,
   754→   754→        fontFamily: 'Inter, system-ui, sans-serif',
   755→   755→        fill: '#6b7280',
   756→   756→        originX: 'center',
   757→   757→      } as any);
   758→   758→      objects.push(wLabel);
   759→   759→
   760→   760→      if (shape === 'rectangle' && w !== h) {
   761→   761→        const hLabel = new IText(`${actualH}`, {
   762→   762→          left: x + actualW + 10,
   763→   763→          top: y + actualH / 2,
   764→   764→          fontSize: 14,
   765→   765→          fontFamily: 'Inter, system-ui, sans-serif',
   766→   766→          fill: '#6b7280',
   767→   767→          originY: 'center',
   768→   768→        } as any);
   769→   769→        objects.push(hLabel);
   770→   770→      }
   771→   771→    }
   772→   772→  }
   773→   773→
   774→   774→  // Custom label
   775→   775→  if (label) {
   776→   776→    const customLabel = new IText(label, {
   777→   777→      left: x + (shape === 'circle' ? radius : (w || sideLength) / 2),
   778→   778→      top: shape === 'circle' ? y - 10 : y - 10,
   779→   779→      fontSize: 16,
   780→   780→      fontFamily: 'Inter, system-ui, sans-serif',
   781→   781→      fill: '#374151',
   782→   782→      originX: 'center',
   783→   783→      originY: 'bottom',
   784→   784→      fontWeight: 'bold',
   785→   785→    } as any);
   786→   786→    objects.push(customLabel);
   787→   787→  }
   788→   788→
   789→   789→  const group = new Group(objects, {
   790→   790→    left: x,
   791→   791→    top: y,
   792→   792→  } as any);
   793→   793→  (group as any).name = `manipulative-geometry-${shape}-${Date.now()}`;
   794→   794→  return [group];
   795→   795→}

   977→   977→  const group = new Group(objects, {
   978→   978→    left: x - radius - 10,
   979→   979→    top: y - radius - 10,
   980→   980→  } as any);
   981→   981→  (group as any).name = `manipulative-clock-${Date.now()}`;
