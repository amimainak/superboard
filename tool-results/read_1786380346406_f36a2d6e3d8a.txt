   145→function getNavItems(tier: Tier, isAdmin: boolean): NavGroup[] {
   146→  const common: NavItem[] = [
   147→    { id: 'overview', label: 'Overview', icon: Home, description: 'Dashboard overview' },
   148→    { id: 'analytics', label: 'Analytics', icon: BarChart3, description: 'Usage analytics & insights' },
   149→    { id: 'lessons', label: 'My Lessons', icon: BookOpen, description: 'Active and past lessons' },
   150→  ];
   151→
   152→  const workspace: NavItem[] = [
   153→    { id: 'schedule', label: 'Schedule', icon: Calendar, description: 'Upcoming & past lessons' },
   154→    { id: 'recordings', label: 'Recordings', icon: Video, description: 'Lesson recordings' },
   155→    { id: 'homework', label: 'Homework', icon: GraduationCap, description: 'Assignments & grading' },
   156→    { id: 'lesson-notes', label: 'Lesson Notes', icon: FileText, description: 'Post-lesson notes & feedback' },
   157→  ];
   158→
   159→  const resources: NavItem[] = isAgencyTier(tier) ? [
   160→    { id: 'resources', label: 'Resources', icon: FolderOpen, description: 'Saved boards, templates & library' },
   161→    { id: 'invoices', label: 'Invoices', icon: Receipt, description: 'Billing invoices & payments' },
   162→  ] : [
   163→    { id: 'resources', label: 'Resources', icon: FolderOpen, description: 'Saved boards & templates' },
   164→  ];
   165→
   166→  const agency: NavItem[] = isAgencyTier(tier) ? [
   167→    { id: 'agency', label: 'Agency', icon: Users, description: 'Sub-tutors & students' },
   168→  ] : [];
   169→
   170→  const account: NavItem[] = [
   171→    { id: 'billing', label: 'Billing', icon: CreditCard, description: 'Subscription & payments' },
   172→    { id: 'settings', label: 'Settings', icon: Settings, description: 'Account preferences' },
   173→  ];
   174→
   175→  return [
   176→    { group: 'Main', items: common },
   177→    ...(workspace.length ? [{ group: 'Workspace', items: workspace }] : []),
   178→    ...(resources.length ? [{ group: 'Resources', items: resources }] : []),
   179→    ...(agency.length ? [{ group: 'Team', items: agency }] : []),
   180→    { group: 'Account', items: account },
   181→  ];
   182→}

   279→export function AuthenticatedDashboard({ user, userName, tierLoading, isAdmin }: { user: User; userName: string | null; tierLoading: boolean; isAdmin?: boolean }) {
   280→  const { tier, setTier, setRoom } = useAppStore();
   281→  const { brandColor, agencyName, setBrandColor } = useTheme();
   282→  const {
   283→    aiCreditsUsed,
   284→    aiCreditsLimit,
   285→    videoMinutesUsed,
   286→    videoMinutesLimit,
   287→    recordingsUsed,
   288→    recordingsLimit,
   289→    loading: usageLoading,
   290→  } = useCredits(user.id);
   291→
   292→  // ---- State ----
   293→  const [activeView, setActiveView] = useState<DashboardView>('overview');
   294→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   295→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   296→
   297→  // Dialog state
   298→  const [showNewLesson, setShowNewLesson] = useState(false);
   299→  const [showTemplateGallery, setShowTemplateGallery] = useState(false);
   300→  const [selectedSubject, setSelectedSubject] = useState<Subject>('GENERAL');
   301→  const [creating, setCreating] = useState(false);
   302→  const [showSignOutConfirm, setShowSignOutConfirm] = useState(false);
   303→
   304→  // Settings state
   305→  const [saveLoading, setSaveLoading] = useState(false);
   306→  const [saveSuccess, setSaveSuccess] = useState(false);
   307→
   308→  // Onboarding state — show wizard to first-time users (no rooms yet)
   309→  const [showOnboarding, setShowOnboarding] = useState(false);
   310→  const [roomsFetched, setRoomsFetched] = useState(false);
   311→  const [hasRooms, setHasRooms] = useState(false);
   312→
   313→  // Student mode toggle
   314→  const [isStudentMode, setIsStudentMode] = useState(false);
   315→
   316→  // Student progress drill-down state
   317→  const [progressStudentId, setProgressStudentId] = useState<string | null>(null);
   318→  const [progressStudentName, setProgressStudentName] = useState<string | null>(null);
   319→
   320→  const { toast } = useToast();
   321→
   322→  // ---- Check if user has any rooms (for onboarding) ----
   323→  useEffect(() => {
   324→    if (!user?.id) return;
   325→    const checkRooms = async () => {
   326→      try {
   327→        const { authFetch } = await import('@/lib/auth-fetch');
   328→        const res = await authFetch(`/api/room/list?tutorId=${user.id}`);
   329→        if (res.ok) {
   330→          const data = await res.json();
   331→          const rooms = data.rooms ?? [];
   332→          setHasRooms(rooms.length > 0);
   333→          if (rooms.length === 0) {
   334→            setShowOnboarding(true);
   335→          }
   336→        }
   337→      } catch { /* silent — proceed to dashboard */ }
   338→      setRoomsFetched(true);
   339→    };
   340→    checkRooms();
   341→  }, [user?.id]);
   342→
   343→  // ---- Dark mode ----
   344→  const [isDark, setIsDark] = useState(false);
   345→  useEffect(() => {
   346→    const saved = localStorage.getItem('superboard-theme');
   347→    if (saved === 'dark') {
   348→      document.documentElement.classList.add('dark');
   349→      setIsDark(true);
   350→    }
   351→  }, []);
   352→
   353→  const toggleDark = useCallback(() => {
   354→    const next = !isDark;
   355→    setIsDark(next);
   356→    if (next) {
   357→      document.documentElement.classList.add('dark');
   358→    } else {
   359→      document.documentElement.classList.remove('dark');
   360→    }
   361→    localStorage.setItem('superboard-theme', next ? 'dark' : 'light');
   362→  }, [isDark]);
   363→
   364→  // ---- Fingerprint reporting ----
   365→  useEffect(() => {
   366→    import('@/lib/fingerprint').then(({ reportFingerprint }) => {
   367→      if (user?.id) reportFingerprint(user.id).catch(console.error);
   368→    });
   369→  }, [user?.id]);
   370→
   371→  // ---- Create lesson handler ----
   372→  const handleCreateLesson = useCallback(async () => {
   373→    setCreating(true);
   374→    try {
   375→      const supabase = createClient();
   376→      const { data: { session } } = await supabase!.auth.getSession();
   377→      const token = session?.access_token;
   378→      if (!token) { setCreating(false); return; }
   379→      const response = await fetch('/api/room', {
   380→        method: 'POST',
   381→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   382→        body: JSON.stringify({ tutorId: user?.id, subject: selectedSubject }),
   383→      });
   384→      if (!response.ok) {
   385→        const errBody = await response.text().catch(() => '');
   386→        throw new Error(`Failed to create room (${response.status}): ${errBody}`);
   387→      }
   388→      const data = await response.json();
   389→      window.location.href = `/room/${data.roomId}`;
   390→    } catch (error: any) {
   391→      toast({ title: 'Failed to create lesson', description: error?.message || 'Please try again.', variant: 'destructive' });
   392→      setCreating(false);
   393→    }
   394→  }, [selectedSubject, user?.id, toast]);
   395→
   396→  // ---- Save brand color handler ----
   397→  const handleSaveBrandColor = useCallback(async () => {
   398→    setSaveLoading(true);
   399→    setSaveSuccess(false);
   400→    try {
   401→      const supabase = createClient();
   402→      const { data: { session } } = await supabase!.auth.getSession();
   403→      const token = session?.access_token;
   404→      if (!token) return;
   405→      await fetch('/api/auth/profile', {
   406→        method: 'PATCH',
   407→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   408→        body: JSON.stringify({ brandingColor: brandColor || null }),
   409→      });
   410→      setSaveSuccess(true);
   411→      setTimeout(() => setSaveSuccess(false), 2000);
   412→    } catch { /* silent */ }
   413→    setSaveLoading(false);
   414→  }, [brandColor]);
   415→
   416→  // ---- Logout handler ----
   417→  const handleLogout = useCallback(async () => {
   418→    const supabase = createClient();
   419→    if (supabase) await supabase.auth.signOut();
   420→    setTier('FREE');
   421→    window.location.href = '/';
   422→  }, [setTier]);
   423→
   424→  // ---- Onboarding complete handler ----
   425→  const handleOnboardingComplete = useCallback((roomId: string) => {
   426→    setShowOnboarding(false);
   427→    setHasRooms(true);
   428→    window.location.href = `/room/${roomId}`;
   429→  }, []);
   430→
   431→  const handleOnboardingSkip = useCallback(() => {
   432→    setShowOnboarding(false);
   433→    setHasRooms(false);
   434→  }, []);
   435→
   436→  // ---- Template gallery handler ----
   437→  const handleTemplateSelect = useCallback((template: TemplateInfo) => {
   438→    setShowTemplateGallery(false);
   439→    setSelectedSubject(template.subject);
   440→    setShowNewLesson(true);
   441→  }, []);
   442→
   443→  // ---- Derived values ----
   444→  const tierLabel = isAgencyTier(tier)
   445→    ? tier === 'AGENCY_PREMIUM' ? 'Agency Premium'
   446→    : tier === 'AGENCY_STANDARD' ? 'Agency Standard'
   447→    : 'Agency'
   448→    : tier === 'PRO' ? 'Pro' : 'Free';
   449→  const tierColor =
   450→    tier === 'AGENCY_PREMIUM' ? 'bg-purple-100 text-purple-800'
   451→    : tier === 'AGENCY_STANDARD' || tier === 'AGENCY' ? 'bg-amber-100 text-amber-800'
   452→    : tier === 'PRO' ? 'bg-emerald-100 text-emerald-800'
   453→    : 'bg-teal-50 text-teal-700';
   454→
   455→  const navGroups = getNavItems(tier, !!isAdmin);
   456→
   457→  // ---- Show onboarding for first-time users ----
   458→  if (showOnboarding && roomsFetched) {
   459→    return (
   460→      <OnboardingWizard
   461→        userId={user.id}
   462→        userEmail={user.email || ''}
   463→        userName={userName}
   464→        onComplete={handleOnboardingComplete}
   465→        onSkip={handleOnboardingSkip}
   466→      />
   467→    );
   468→  }
   469→
   470→  // ---- Show student dashboard if in student mode ----
   471→  if (isStudentMode) {
   472→    return (
   473→      <StudentDashboard
   474→        userId={user.id}
   475→        userName={userName}
   476→        userEmail={user.email || null}
   477→      />
   478→    );
   479→  }
   480→
   481→  // ============================================================
   482→  // RENDER — Dashboard with Sidebar
   483→  // ============================================================
   484→  return (
   485→    <TooltipProvider delayDuration={0}>
   486→      <div className="min-h-screen bg-background flex">
   487→        {/* ============================================================
   488→            SIDEBAR — Collapsible on desktop, sheet on mobile
   489→            ============================================================ */}
   490→        {/* Mobile overlay */}
   491→        {mobileMenuOpen && (
   492→          <div
   493→            className="fixed inset-0 z-40 bg-black/40 backdrop-blur-sm lg:hidden"
   494→            onClick={() => setMobileMenuOpen(false)}
   495→          />
   496→        )}
   497→
   498→        {/* Sidebar */}
   499→        <aside
   500→          className={`
   501→            fixed lg:sticky top-0 left-0 z-50 lg:z-30
   502→            h-screen flex flex-col
   503→            bg-white border-r border-gray-200
   504→            transition-all duration-300 ease-in-out
   505→            ${sidebarCollapsed ? 'w-[68px]' : 'w-64'}
   506→            ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full lg:translate-x-0'}
   507→          `}
   508→        >
   509→          {/* Sidebar Header / Logo */}
   510→          <div className="flex items-center justify-between px-4 h-16 border-b border-gray-100 shrink-0">
   511→            <div className="flex items-center gap-2.5 min-w-0">
   512→              <div className="w-9 h-9 rounded-xl gradient-primary flex items-center justify-center shadow-md shadow-emerald-500/20 shrink-0">
   513→                <GraduationCap className="w-5 h-5 text-white" />
   514→              </div>
   515→              {!sidebarCollapsed && (
   516→                <div className="min-w-0">
   517→                  <h1 className="text-sm font-bold tracking-tight truncate">{agencyName || 'Superboard'}</h1>
   518→                </div>
   519→              )}
   520→            </div>
   521→            {/* Mobile close */}
   522→            <Button
   523→              variant="ghost"
   524→              size="icon"
   525→              className="lg:hidden h-8 w-8 shrink-0"
   526→              aria-label="Close sidebar"
   527→              onClick={() => setMobileMenuOpen(false)}
   528→            >
   529→              <PanelLeftClose className="w-4 h-4" />
   530→            </Button>
   531→          </div>
   532→
   533→          {/* Navigation */}
   534→          <nav className="flex-1 overflow-y-auto py-3 px-3 custom-scrollbar">
   535→            {navGroups.map((group, gi) => (
   536→              <div key={group.group} className={gi > 0 ? 'mt-4' : ''}>
   537→                {!sidebarCollapsed && (
   538→                  <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground px-2 mb-1.5">
   539→                    {group.group}
   540→                  </p>
   541→                )}
   542→                {group.items.map((item) => {
   543→                  const isActive = activeView === item.id;
   544→                  const Icon = item.icon;
   545→                  return (
   546→                    <Tooltip key={item.id}>
   547→                      <TooltipTrigger asChild>
   548→                        <button
   549→                          onClick={() => {
   550→                            setActiveView(item.id);
   551→                            setMobileMenuOpen(false);
   552→                          }}
   553→                          className={`
   554→                            w-full flex items-center gap-2.5 rounded-xl text-sm font-medium
   555→                            transition-all duration-150
   556→                            ${sidebarCollapsed ? 'justify-center px-2 py-2.5' : 'px-3 py-2.5'}
   557→                            ${isActive
   558→                              ? 'bg-emerald-50 text-emerald-700 shadow-sm'
   559→                              : 'text-gray-600 hover:bg-gray-50 hover:text-gray-900'
   560→                            }
   561→                          `}
   562→                        >
   563→                          <Icon className={`w-4.5 h-4.5 shrink-0 ${isActive ? 'text-emerald-600' : 'text-gray-400'}`} />
