     1→// ============================================================
     2→// PipVideoPanel — Floating, Draggable PiP Video Panel
     3→// ============================================================
     4→// Native floating panel inside the whiteboard UI (position: fixed).
     5→// Contains LiveKit video grid with tutor + student webcams,
     6→// mute/deafen controls, and recording button.
     7→//
     8→// Uses useLiveKitRoom hook for real LiveKit connection management.
     9→// Shows credit warnings at 80% and hard block at 100%.
    10→// ============================================================
    11→
    12→'use client';
    13→
    14→import React, { useState, useCallback, useRef, useEffect } from 'react';
    15→import { useAppStore } from '@/store/app-store';
    16→import { useLiveKitRoom } from '@/hooks/useLiveKitRoom';
    17→import { useCredits } from '@/hooks/useCredits';
    18→import RecordButton from './RecordButton';
    19→import {
    20→  Mic,
    21→  MicOff,
    22→  Headphones,
    23→  HeadphoneOff,
    24→  Minimize2,
    25→  PhoneOff,
    26→  GripHorizontal,
    27→  Video,
    28→  VideoOff,
    29→  User,
    30→  MonitorSpeaker,
    31→  AlertTriangle,
    32→} from 'lucide-react';
    33→import { Button } from '@/components/ui/button';
    34→import { cn } from '@/lib/utils';
    35→
    36→// ---- Types ----
    37→
    38→interface PanelPosition {
    39→  x: number;
    40→  y: number;
    41→}
    42→
    43→interface PanelSize {
    44→  width: number;
    45→  height: number;
    46→}
    47→
    48→// ---- Constants ----
    49→
    50→const MINIMIZED_SIZE = 56;
    51→const DEFAULT_WIDTH = typeof window !== 'undefined' ? Math.min(360, window.innerWidth * 0.6) : 360;
    52→const DEFAULT_HEIGHT = typeof window !== 'undefined' ? Math.min(300, window.innerHeight * 0.4) : 300;
    53→const MIN_WIDTH = 200;
    54→const MIN_HEIGHT = 180;
    55→const EDGE_SNAP = 16;
    56→
    57→// ============================================================
    58→// Main Component
    59→// ============================================================
    60→
    61→export default function PipVideoPanel() {
    62→  const roomActive = useAppStore((s) => s.room.isActive);
    63→  const isTutor = useAppStore((s) => s.room.isTutor);
    64→  const roomId = useAppStore((s) => s.room.roomId);
    65→  const userId = useAppStore((s) => s.room.userId);
    66→
    67→  // LiveKit connection
    68→  const {
    69→    connectionState,
    70→    participants,
    71→    localParticipant,
    72→    error: lkError,
    73→    isMuted,
    74→    isCameraOff,
    75→    isDeafened,
    76→    toggleMic,
    77→    toggleCamera,
    78→    toggleDeafen,
    79→    disconnect,
    80→  } = useLiveKitRoom();
    81→
    82→  // Credit tracking
    83→  const {
    84→    videoMinutesUsed,
    85→    videoMinutesLimit,
    86→    videoMinutesExhausted,
    87→    refresh: refreshCredits,
    88→  } = useCredits(isTutor ? userId : undefined);
    89→
    90→  // ---- Panel state ----
    91→  const [position, setPosition] = useState<PanelPosition>(() => ({
    92→    x: Math.max(0, window.innerWidth - DEFAULT_WIDTH - EDGE_SNAP),
    93→    y: Math.max(0, window.innerHeight - DEFAULT_HEIGHT - EDGE_SNAP),
    94→  }));
    95→  const [size, setSize] = useState<PanelSize>({
    96→    width: DEFAULT_WIDTH,
    97→    height: DEFAULT_HEIGHT,
    98→  });
    99→  const [isMinimized, setIsMinimized] = useState(false);
   100→  const [isDragging, setIsDragging] = useState(false);
   101→  const [isResizing, setIsResizing] = useState(false);
   102→  const [inCall, setInCall] = useState(true);
   103→
   104→  // ---- Video limit warning ----
   105→  const isApproachingLimit = videoMinutesLimit !== Infinity &&
   106→    videoMinutesUsed >= videoMinutesLimit * 0.8 &&
   107→    !videoMinutesExhausted;
   108→
   109→  // Track element refs for video rendering
   110→  const videoElsRef = useRef<Map<string, HTMLVideoElement>>(new Map());
   111→  const audioElsRef = useRef<Map<string, HTMLAudioElement>>(new Map());
   112→
   113→  // ---- Attach video tracks to DOM elements ----
   114→  useEffect(() => {
   115→    participants.forEach((p) => {
   116→      if (p.cameraTrack && p.cameraTrack.track) {
   117→        const track = p.cameraTrack.track;
   118→        let videoEl = videoElsRef.current.get(p.identity);
   119→        if (!videoEl) {
   120→          videoEl = document.createElement('video');
   121→          videoEl.autoplay = true;
   122→          videoEl.playsInline = true;
   123→          videoEl.muted = true;
   124→          videoEl.className = 'w-full h-full object-cover';
   125→          videoElsRef.current.set(p.identity, videoEl);
   126→        }
   127→        track.attach(videoEl);
   128→      }
   129→    });
   130→
   131→    // Clean up detached tracks
   132→    videoElsRef.current.forEach((videoEl, identity) => {
   133→      const participant = participants.find((p) => p.identity === identity);
   134→      if (!participant?.cameraTrack?.track) {
   135→        if (videoEl.srcObject) {
   136→          videoEl.srcObject = null;
   137→        }
   138→      }
   139→    });
   140→
   141→    return () => {
   142→      videoElsRef.current.forEach((videoEl) => {
   143→        try { videoEl.srcObject = null; } catch {}
   144→      });
   145→    };
   146→  }, [participants]);
   147→
   148→  // ---- Snap to corners on mount / resize ----
   149→  useEffect(() => {
   150→    const handleResize = () => {
   151→      setPosition((prev) => ({
   152→        x: Math.min(prev.x, window.innerWidth - (isMinimized ? MINIMIZED_SIZE : size.width)),
   153→        y: Math.min(prev.y, window.innerHeight - (isMinimized ? MINIMIZED_SIZE : size.height)),
   154→      }));
   155→    };
   156→    window.addEventListener('resize', handleResize);
   157→    return () => window.removeEventListener('resize', handleResize);
   158→  }, [size.width, size.height, isMinimized]);
   159→
   160→  // ---- Drag handlers ----
   161→  const dragOffset = useRef({ x: 0, y: 0 });
   162→  const panelRef = useRef<HTMLDivElement>(null);
   163→
   164→  const handleDragStart = useCallback(
   165→    (e: React.MouseEvent | React.TouchEvent) => {
   166→      if (isResizing) return;
   167→      if ('touches' in e) {
   168→        e.preventDefault();
   169→        const rect = panelRef.current?.getBoundingClientRect();
   170→        if (!rect) return;
   171→        dragOffset.current = {
   172→          x: e.touches[0].clientX - rect.left,
   173→          y: e.touches[0].clientY - rect.top,
   174→        };
   175→      } else {
   176→        e.preventDefault();
   177→        const rect = panelRef.current?.getBoundingClientRect();
   178→        if (!rect) return;
   179→        dragOffset.current = {
   180→          x: e.clientX - rect.left,
   181→          y: e.clientY - rect.top,
   182→        };
   183→      }
   184→      setIsDragging(true);
   185→    },
   186→    [isResizing]
   187→  );
   188→
   189→  useEffect(() => {
   190→    if (!isDragging) return;
   191→
   192→    const handleDragMove = (e: MouseEvent | TouchEvent) => {
   193→      const clientX = 'touches' in e ? e.touches[0].clientX : (e as MouseEvent).clientX;
   194→      const clientY = 'touches' in e ? e.touches[0].clientY : (e as MouseEvent).clientY;
   195→      const newX = clientX - dragOffset.current.x;
   196→      const newY = clientY - dragOffset.current.y;
   197→      const currentW = isMinimized ? MINIMIZED_SIZE : size.width;
   198→      const currentH = isMinimized ? MINIMIZED_SIZE : size.height;
   199→
   200→      setPosition({
   201→        x: Math.max(0, Math.min(newX, window.innerWidth - currentW)),
   202→        y: Math.max(0, Math.min(newY, window.innerHeight - currentH)),
   203→      });
   204→    };
   205→
   206→    const handleDragEnd = () => {
   207→      setIsDragging(false);
   208→      setPosition((prev) => snapToEdge(prev, isMinimized ? MINIMIZED_SIZE : size.width, isMinimized ? MINIMIZED_SIZE : size.height));
   209→    };
   210→
   211→    window.addEventListener('mousemove', handleDragMove);
   212→    window.addEventListener('mouseup', handleDragEnd);
   213→    window.addEventListener('touchmove', handleDragMove, { passive: false });
   214→    window.addEventListener('touchend', handleDragEnd);
   215→    return () => {
   216→      window.removeEventListener('mousemove', handleDragMove);
   217→      window.removeEventListener('mouseup', handleDragEnd);
   218→      window.removeEventListener('touchmove', handleDragMove);
   219→      window.removeEventListener('touchend', handleDragEnd);
   220→    };
   221→  }, [isDragging, size, isMinimized]);
   222→
   223→  // ---- Resize handlers ----
   224→  const handleResizeStart = useCallback(
   225→    (e: React.MouseEvent | React.TouchEvent) => {
   226→      if (isDragging) return;
   227→      e.preventDefault();
   228→      e.stopPropagation();
   229→      setIsResizing(true);
   230→      const clientX = 'touches' in e ? e.touches[0].clientX : (e as React.MouseEvent).clientX;
   231→      const clientY = 'touches' in e ? e.touches[0].clientY : (e as React.MouseEvent).clientY;
   232→      const startX = clientX;
   233→      const startY = clientY;
   234→      const startW = size.width;
   235→      const startH = size.height;
   236→
   237→      const handleResizeMove = (moveE: MouseEvent | TouchEvent) => {
   238→        const moveX = 'touches' in moveE ? moveE.touches[0].clientX : (moveE as MouseEvent).clientX;
   239→        const moveY = 'touches' in moveE ? moveE.touches[0].clientY : (moveE as MouseEvent).clientY;
   240→        const deltaW = startX - moveX;
   241→        const deltaH = startY - moveY;
   242→        setSize({
   243→          width: Math.max(MIN_WIDTH, Math.min(startW + deltaW, 600)),
   244→          height: Math.max(MIN_HEIGHT, Math.min(startH + deltaH, 500)),
   245→        });
   246→      };
   247→
   248→      const handleResizeEnd = () => {
   249→        setIsResizing(false);
   250→        window.removeEventListener('mousemove', handleResizeMove);
   251→        window.removeEventListener('mouseup', handleResizeEnd);
   252→        window.removeEventListener('touchmove', handleResizeMove);
   253→        window.removeEventListener('touchend', handleResizeEnd);
   254→      };
   255→
   256→      window.addEventListener('mousemove', handleResizeMove);
   257→      window.addEventListener('mouseup', handleResizeEnd);
   258→      window.addEventListener('touchmove', handleResizeMove, { passive: false });
   259→      window.addEventListener('touchend', handleResizeEnd);
   260→    },
   261→    [isDragging, size]
   262→  );
   263→
   264→  // ---- Don't render if room is not active or call has ended ----
   265→  if (!roomActive || !inCall) return null;
   266→
   267→  // ---- Get last speaker for minimized view ----
   268→  const lastSpeaker = participants.find((p) => p.isSpeaking) || participants[0];
   269→  const totalParticipants = participants.length + (localParticipant ? 1 : 0);
   270→
   271→  // ---- Minimized view ----
   272→  if (isMinimized) {
   273→    if (!lastSpeaker && !localParticipant) {
   274→      return (
   275→        <div
   276→          ref={panelRef}
   277→          style={{
   278→            position: 'fixed',
   279→            left: position.x,
   280→            top: position.y,
   281→            width: MINIMIZED_SIZE,
   282→            height: MINIMIZED_SIZE,
   283→            zIndex: 9999,
   284→          }}
   285→          onMouseDown={handleDragStart}
   286→          onTouchStart={handleDragStart}
   287→          className={cn(
   288→            'cursor-grab active:cursor-grabbing rounded-full',
   289→            'bg-black/80 backdrop-blur-xl border border-white/10',
   290→            'shadow-2xl flex items-center justify-center',
   291→            'transition-shadow hover:shadow-[0_0_24px_rgba(59,130,246,0.3)]',
   292→            'group'
   293→          )}
   294→          onClick={() => setIsMinimized(false)}
   295→          onKeyDown={(e) => {
   296→            if (e.key === 'Enter' || e.key === ' ') {
   297→              e.preventDefault();
   298→              setIsMinimized(false);
   299→            }
   300→          }}
   301→          role="button"
   302→          tabIndex={0}
   303→          title="Click to expand video panel"
   304→          aria-label="Expand video panel"
   305→        >
   306→          <div className="w-10 h-10 rounded-full flex items-center justify-center bg-blue-500/80">
   307→            {connectionState === 'connected' ? (
   308→              <Video className="w-5 h-5 text-white" />
   309→            ) : connectionState === 'connecting' ? (
   310→              <div className="w-5 h-5 border-2 border-white/40 border-t-white rounded-full animate-spin" />
   311→            ) : (
   312→              <VideoOff className="w-5 h-5 text-white/60" />
   313→            )}
   314→          </div>
   315→          {useAppStore.getState().room.isRecording && (
   316→            <span className="absolute -top-1 -right-1 w-3 h-3 bg-red-500 rounded-full animate-pulse border border-black/80" />
   317→          )}
   318→        </div>
   319→      );
   320→    }
   321→
   322→    const displayName = lastSpeaker?.name || localParticipant?.name || '...';
   323→    const isLocal = !lastSpeaker;
   324→
   325→    return (
   326→      <div
   327→        ref={panelRef}
   328→        style={{
   329→          position: 'fixed',
   330→          left: position.x,
   331→          top: position.y,
   332→          width: MINIMIZED_SIZE,
   333→          height: MINIMIZED_SIZE,
   334→          zIndex: 9999,
   335→        }}
   336→        onMouseDown={handleDragStart}
   337→        onTouchStart={handleDragStart}
   338→        className={cn(
   339→          'cursor-grab active:cursor-grabbing rounded-full',
   340→          'bg-black/80 backdrop-blur-xl border border-white/10',
   341→          'shadow-2xl flex items-center justify-center',
   342→          'transition-shadow hover:shadow-[0_0_24px_rgba(59,130,246,0.3)]',
   343→          'group'
   344→        )}
   345→        onClick={() => setIsMinimized(false)}
   346→        onKeyDown={(e) => {
   347→          if (e.key === 'Enter' || e.key === ' ') {
   348→            e.preventDefault();
   349→            setIsMinimized(false);
   350→          }
   351→        }}
   352→        role="button"
   353→        tabIndex={0}
   354→        title="Click to expand video panel"
   355→        aria-label="Expand video panel"
   356→      >
   357→        <div className="relative">
   358→          <div
   359→            className={cn(
   360→              'w-10 h-10 rounded-full flex items-center justify-center',
   361→              isLocal ? 'bg-blue-500/80' : 'bg-emerald-500/80'
   362→            )}
   363→          >
   364→            <User className="w-5 h-5 text-white" />
   365→          </div>
   366→          {lastSpeaker?.isSpeaking && (
   367→            <span className="absolute -bottom-0.5 -right-0.5 w-3 h-3 bg-green-400 rounded-full border-2 border-black/80" />
   368→          )}
   369→          {useAppStore.getState().room.isRecording && (
   370→            <span className="absolute -top-1 -right-1 w-3 h-3 bg-red-500 rounded-full animate-pulse border border-black/80" />
   371→          )}
   372→        </div>
   373→      </div>
   374→    );
   375→  }
   376→
   377→  // ---- Full panel view ----
   378→  const gridCols = totalParticipants <= 1 ? 'grid-cols-1' : totalParticipants === 2 ? 'grid-cols-2' : 'grid-cols-2';
   379→
   380→  return (
   381→    <div
   382→      ref={panelRef}
   383→      style={{
   384→        position: 'fixed',
   385→        left: position.x,
   386→        top: position.y,
   387→        width: size.width,
   388→        height: size.height,
   389→        zIndex: 9999,
   390→      }}
   391→      onMouseDown={handleDragStart}
   392→      onTouchStart={handleDragStart}
   393→      className={cn(
   394→        'rounded-2xl overflow-hidden flex flex-col',
   395→        'bg-black/70 backdrop-blur-xl border border-white/10',
   396→        'shadow-2xl',
   397→        isDragging ? 'cursor-grabbing select-none' : 'cursor-grab',
   398→        isResizing && 'cursor-se-resize'
   399→      )}
   400→    >
   401→      {/* ---- Header (drag handle) ---- */}
   402→      <div
   403→        className="flex items-center justify-between px-3 h-[40px] shrink-0 bg-white/5 border-b border-white/5"
   404→        aria-label="Video panel - drag handle"
   405→        tabIndex={0}
   406→        role="toolbar"
   407→        onKeyDown={(e) => {
   408→          const step = e.shiftKey ? 50 : 10;
   409→          let dx = 0;
   410→          let dy = 0;
   411→          if (e.key === 'ArrowLeft') dx = -step;
   412→          else if (e.key === 'ArrowRight') dx = step;
   413→          else if (e.key === 'ArrowUp') dy = -step;
   414→          else if (e.key === 'ArrowDown') dy = step;
   415→          else return;
   416→          e.preventDefault();
   417→          setPosition((prev) => ({
   418→            x: Math.max(0, Math.min(prev.x + dx, window.innerWidth - size.width)),
   419→            y: Math.max(0, Math.min(prev.y + dy, window.innerHeight - size.height)),
   420→          }));
   421→        }}
   422→      >
   423→        <div className="flex items-center gap-1.5 text-white/70 text-xs font-medium">
   424→          <GripHorizontal className="w-3.5 h-3.5" />
   425→          <span>
   426→            {connectionState === 'connected' ? 'Video Call' :
   427→             connectionState === 'connecting' ? 'Connecting...' :
   428→             connectionState === 'reconnecting' ? 'Reconnecting...' :
   429→             connectionState === 'failed' ? 'Connection Failed' : 'Video Off'}
   430→          </span>
   431→          {/* Video limit warning */}
   432→          {isApproachingLimit && (
   433→            <AlertTriangle className="w-3.5 h-3.5 text-amber-400" />
   434→          )}
   435→          {useAppStore.getState().room.isRecording && (
   436→            <span className="flex items-center gap-1 ml-1 text-red-400 text-[10px] font-semibold animate-pulse">
   437→              <span className="w-1.5 h-1.5 bg-red-500 rounded-full" />
   438→              REC
   439→            </span>
   440→          )}
   441→          {/* Video minutes for free tier */}
   442→          {videoMinutesLimit !== Infinity && (
   443→            <span className="text-[10px] text-white/40 ml-1">
   444→              {videoMinutesUsed}/{videoMinutesLimit}m
   445→            </span>
   446→          )}
   447→        </div>
   448→        <div className="flex items-center gap-0.5">
   449→          <Button
   450→            variant="ghost"
   451→            size="icon"
   452→            className="h-8 w-8 text-white/50 hover:text-white hover:bg-white/10"
   453→            onClick={(e) => {
   454→              e.stopPropagation();
   455→              setIsMinimized(true);
   456→            }}
   457→            aria-label="Minimize video panel"
   458→          >
   459→            <Minimize2 className="w-3.5 h-3.5" />
   460→          </Button>
   461→        </div>
   462→      </div>
   463→
   464→      {/* ---- Video limit warning banner ---- */}
   465→      {isApproachingLimit && (
   466→        <div className="px-3 py-1.5 bg-amber-500/20 border-b border-amber-500/30 flex items-center gap-2 text-amber-300 text-[11px]">
   467→          <AlertTriangle className="w-3 h-3 shrink-0" />
   468→          <span>{videoMinutesLimit - videoMinutesUsed} minutes of video remaining this period</span>
   469→        </div>
   470→      )}
   471→      {videoMinutesExhausted && videoMinutesLimit !== Infinity && (
   472→        <div className="px-3 py-1.5 bg-red-500/20 border-b border-red-500/30 flex items-center gap-2 text-red-300 text-[11px]">
   473→          <AlertTriangle className="w-3 h-3 shrink-0" />
   474→          <span>Video limit reached. Upgrade to continue using video.</span>
   475→          <Button
   476→            size="sm"
   477→            variant="outline"
   478→            className="ml-auto h-5 px-2 text-[10px] border-red-400/50 text-red-300 hover:bg-red-500/30"
   479→            onClick={() => useAppStore.getState().openPaywall('Video minutes exhausted')}
   480→          >
   481→            Upgrade
   482→          </Button>
   483→        </div>
   484→      )}
   485→
   486→      {/* ---- Connection error banner ---- */}
   487→      {lkError && connectionState === 'failed' && (
   488→        <div className="px-3 py-1.5 bg-red-500/20 border-b border-red-500/30 text-red-300 text-[11px]">
   489→          {lkError}
   490→        </div>
   491→      )}
   492→
   493→      {/* ---- Video Grid Area ---- */}
   494→      <div className="flex-1 relative overflow-hidden">
   495→        {/* Connecting overlay */}
   496→        {connectionState === 'connecting' || connectionState === 'reconnecting' ? (
   497→          <div className="absolute inset-0 bg-black/80 flex flex-col items-center justify-center gap-3 z-10">
   498→            <div className="w-10 h-10 border-3 border-white/20 border-t-white rounded-full animate-spin" />
   499→            <span className="text-xs text-white/50">
   500→              {connectionState === 'connecting' ? 'Joining video...' : 'Reconnecting...'}
   501→            </span>
   502→          </div>
   503→        ) : connectionState === 'disconnected' || connectionState === 'failed' ? (
   504→          <div className="h-full flex flex-col items-center justify-center gap-3 text-white/30">
   505→            <Video className="w-12 h-12" />
   506→            <span className="text-xs">
   507→              {connectionState === 'failed' ? 'Video connection failed' : 'Video disconnected'}
   508→            </span>
   509→            {connectionState === 'failed' && (
   510→              <Button
   511→                size="sm"
   512→                variant="outline"
   513→                className="mt-2 text-xs border-white/20 text-white/60 hover:bg-white/10"
   514→                onClick={() => window.location.reload()}
   515→              >
   516→                Retry
   517→              </Button>
   518→            )}
   519→          </div>
   520→        ) : participants.length === 0 && !localParticipant ? (
   521→          <div className="h-full flex flex-col items-center justify-center gap-3 text-white/30">
   522→            <Video className="w-12 h-12" />
   523→            <span className="text-xs">Waiting for participants...</span>
   524→            <span className="text-[10px] text-white/20">Video call is active</span>
   525→          </div>
   526→        ) : (
   527→          <div className={cn('grid h-full gap-0.5 p-0.5', gridCols)}>
   528→            {/* Local participant */}
   529→            {localParticipant && (
   530→              <div
   531→                className={cn(
   532→                  'relative rounded-lg overflow-hidden flex flex-col items-center justify-center',
   533→                  'bg-gradient-to-b from-slate-800/60 to-slate-900/80'
   534→                )}
   535→              >
   536→                <div className="flex-1 w-full flex items-center justify-center relative">
   537→                  {isCameraOff ? (
   538→                    <div className="flex flex-col items-center gap-2 text-white/20">
   539→                      <VideoOff className="w-10 h-10" />
   540→                      <span className="text-[10px]">Camera Off</span>
   541→                    </div>
   542→                  ) : (
   543→                    <div className="flex flex-col items-center gap-2 text-white/30">
   544→                      <User className="w-10 h-10" />
   545→                      <span className="text-[10px]">You</span>
   546→                    </div>
   547→                  )}
   548→
   549→                  {/* Name tag */}
   550→                  <div className="absolute bottom-1.5 left-1.5 flex items-center gap-1.5">
   551→                    <span
   552→                      className={cn(
   553→                        'text-[10px] font-medium text-white/90 px-1.5 py-0.5 rounded',
   554→                        'bg-black/60 backdrop-blur-sm'
   555→                      )}
   556→                    >
   557→                      {localParticipant.name}
   558→                      <span className="ml-1 text-blue-400">&#9733;</span>
   559→                    </span>
   560→                  </div>
   561→
   562→                  {/* Mute indicator */}
   563→                  {isMuted && (
   564→                    <div className="absolute top-1.5 right-1.5">
   565→                      <div className="w-5 h-5 rounded-full bg-red-500/80 flex items-center justify-center">
   566→                        <MicOff className="w-3 h-3 text-white" />
   567→                      </div>
   568→                    </div>
   569→                  )}
   570→                </div>
   571→              </div>
   572→            )}
   573→
   574→            {/* Remote participants */}
   575→            {participants.map((participant) => (
   576→              <div
   577→                key={participant.identity}
   578→                className={cn(
   579→                  'relative rounded-lg overflow-hidden flex flex-col items-center justify-center',
   580→                  'bg-gradient-to-b from-slate-800/60 to-slate-900/80',
   581→                  participant.isSpeaking && 'ring-2 ring-green-400/60'
   582→                )}
   583→              >
   584→                <div className="flex-1 w-full flex items-center justify-center relative">
   585→                  {participant.isCameraOn ? (
   586→                    <div className="flex flex-col items-center gap-2 text-white/30">
   587→                      <Video className="w-10 h-10" />
   588→                      <span className="text-[10px]">Camera Active</span>
   589→                    </div>
   590→                  ) : (
   591→                    <div className="flex flex-col items-center gap-2 text-white/20">
   592→                      <VideoOff className="w-10 h-10" />
   593→                      <span className="text-[10px]">Camera Off</span>
   594→                    </div>
   595→                  )}
   596→
   597→                  {/* Name tag */}
   598→                  <div className="absolute bottom-1.5 left-1.5 flex items-center gap-1.5">
   599→                    <span
   600→                      className={cn(
   601→                        'text-[10px] font-medium text-white/90 px-1.5 py-0.5 rounded',
   602→                        'bg-black/60 backdrop-blur-sm'
   603→                      )}
   604→                    >
   605→                      {participant.name}
   606→                    </span>
   607→                  </div>
   608→
   609→                  {/* Mute indicator */}
   610→                  {participant.isMuted && (
   611→                    <div className="absolute top-1.5 right-1.5">
   612→                      <div className="w-5 h-5 rounded-full bg-red-500/80 flex items-center justify-center">
   613→                        <MicOff className="w-3 h-3 text-white" />
   614→                      </div>
   615→                    </div>
   616→                  )}
   617→
   618→                  {/* Speaking animation ring */}
   619→                  {participant.isSpeaking && (
   620→                    <div className="absolute inset-0 rounded-lg border-2 border-green-400/40 animate-pulse pointer-events-none" />
   621→                  )}
   622→                </div>
   623→              </div>
   624→            ))}
   625→          </div>
   626→        )}
   627→      </div>
   628→
   629→      {/* ---- Controls Bar ---- */}
   630→      <div className="flex items-center justify-center gap-1.5 px-3 h-[52px] shrink-0 bg-white/5 border-t border-white/5">
   631→        {/* Camera toggle */}
   632→        <Button
   633→          variant="ghost"
   634→          size="icon"
   635→          className={cn(
   636→            'h-10 w-10 rounded-full text-white/70 hover:text-white hover:bg-white/10',
   637→            isCameraOff && 'bg-red-500/20 text-red-400 hover:bg-red-500/30 hover:text-red-300'
   638→          )}
   639→          onClick={toggleCamera}
   640→          disabled={connectionState !== 'connected'}
   641→          aria-label={isCameraOff ? 'Turn camera on' : 'Turn camera off'}
   642→        >
   643→          {isCameraOff ? <VideoOff className="w-4 h-4" /> : <Video className="w-4 h-4" />}
   644→        </Button>
   645→
   646→        {/* Mic toggle */}
   647→        <Button
   648→          variant="ghost"
   649→          size="icon"
   650→          className={cn(
   651→            'h-10 w-10 rounded-full text-white/70 hover:text-white hover:bg-white/10',
   652→            isMuted && 'bg-red-500/20 text-red-400 hover:bg-red-500/30 hover:text-red-300'
   653→          )}
   654→          onClick={toggleMic}
   655→          disabled={connectionState !== 'connected'}
   656→          aria-label={isMuted ? 'Unmute microphone' : 'Mute microphone'}
   657→        >
   658→          {isMuted ? <MicOff className="w-4 h-4" /> : <Mic className="w-4 h-4" />}
   659→        </Button>
   660→
   661→        {/* Deafen toggle */}
   662→        <Button
   663→          variant="ghost"
   664→          size="icon"
   665→          className={cn(
   666→            'h-10 w-10 rounded-full text-white/70 hover:text-white hover:bg-white/10',
   667→            isDeafened && 'bg-red-500/20 text-red-400 hover:bg-red-500/30 hover:text-red-300'
   668→          )}
   669→          onClick={toggleDeafen}
   670→          aria-label={isDeafened ? 'Undeafen audio' : 'Deafen audio'}
   671→        >
   672→          {isDeafened ? <HeadphoneOff className="w-4 h-4" /> : <Headphones className="w-4 h-4" />}
   673→        </Button>
   674→
   675→        {/* Monitor/Speaker output */}
   676→        <Button
   677→          variant="ghost"
   678→          size="icon"
   679→          className="h-10 w-10 rounded-full text-white/70 hover:text-white hover:bg-white/10"
   680→          aria-label="Toggle speaker output"
   681→        >
   682→          <MonitorSpeaker className="w-4 h-4" />
   683→        </Button>
   684→
   685→        {/* Recording button */}
   686→        <RecordButton />
   687→
   688→        {/* Leave call */}
   689→        <Button
   690→          variant="ghost"
   691→          size="icon"
   692→          className="h-10 w-10 rounded-full text-red-400 hover:text-red-300 hover:bg-red-500/20"
   693→          onClick={async () => {
   694→            await disconnect();
   695→            setInCall(false);
   696→          }}
   697→          aria-label="Leave call"
   698→        >
   699→          <PhoneOff className="w-4 h-4" />
   700→        </Button>
   701→      </div>
   702→
   703→      {/* ---- Resize handle (bottom-left corner) ---- */}
   704→      <div
   705→        onMouseDown={handleResizeStart}
   706→        onTouchStart={handleResizeStart}
   707→        className={cn(
   708→          'absolute bottom-0 left-0 w-6 h-6 cursor-se-resize',
   709→          'opacity-30 hover:opacity-60 transition-opacity'
   710→        )}
   711→        role="separator"
   712→        aria-label="Resize video panel"
   713→        tabIndex={0}
   714→        onKeyDown={(e) => {
   715→          if (!e.shiftKey) return;
   716→          const step = 10;
   717→          let dw = 0;
   718→          let dh = 0;
   719→          if (e.key === 'ArrowRight') dw = step;
   720→          else if (e.key === 'ArrowUp') dh = step;
   721→          else if (e.key === 'ArrowLeft') dw = -step;
   722→          else if (e.key === 'ArrowDown') dh = -step;
   723→          else return;
   724→          e.preventDefault();
   725→          setSize((prev) => ({
   726→            width: Math.max(MIN_WIDTH, Math.min(prev.width + dw, 600)),
   727→            height: Math.max(MIN_HEIGHT, Math.min(prev.height + dh, 500)),
   728→          }));
   729→        }}
   730→      >
   731→        <svg viewBox="0 0 16 16" fill="currentColor" className="text-white/60 w-4 h-4" aria-hidden="true">
   732→          <path d="M14 14H8V13H13V8H14V14Z" />
   733→          <path d="M14 10H9V9H13V5H14V10Z" opacity="0.6" />
   734→        </svg>
   735→      </div>
   736→    </div>
   737→  );
   738→}

   740→// ---- Helper: Snap position to nearest corner ----
   741→
   742→function snapToEdge(
   743→  pos: PanelPosition,
   744→  width: number,
   745→  height: number
   746→): PanelPosition {
   747→  const vw = window.innerWidth;
   748→  const vh = window.innerHeight;
   749→
   750→  const centerX = pos.x + width / 2;
   751→  const centerY = pos.y + height / 2;
   752→
   753→  const snapX = centerX < vw / 2 ? EDGE_SNAP : vw - width - EDGE_SNAP;
   754→  const snapY = centerY < vh / 2 ? EDGE_SNAP : vh - height - EDGE_SNAP;
   755→
   756→  return { x: snapX, y: snapY };
   757→}
