   133→function getNavItems(tier: Tier, isAdmin: boolean): NavGroup[] {
   134→  const common: NavItem[] = [
   135→    { id: 'overview', label: 'Overview', icon: Home, description: 'Dashboard overview' },
   136→    { id: 'analytics', label: 'Analytics', icon: BarChart3, description: 'Usage analytics & insights' },
   137→    { id: 'lessons', label: 'My Lessons', icon: BookOpen, description: 'Active and past lessons' },
   138→  ];
   139→
   140→  const workspace: NavItem[] = [
   141→    { id: 'schedule', label: 'Schedule', icon: Calendar, description: 'Upcoming & past lessons' },
   142→    { id: 'recordings', label: 'Recordings', icon: Video, description: 'Lesson recordings' },
   143→  ];
   144→
   145→  const resources: NavItem[] = isAgencyTier(tier) ? [
   146→    { id: 'resources', label: 'Resources', icon: FolderOpen, description: 'Saved boards & templates' },
   147→  ] : [];
   148→
   149→  const agency: NavItem[] = isAgencyTier(tier) ? [
   150→    { id: 'agency', label: 'Agency', icon: Users, description: 'Sub-tutors & students' },
   151→  ] : [];
   152→
   153→  const account: NavItem[] = [
   154→    { id: 'billing', label: 'Billing', icon: CreditCard, description: 'Subscription & payments' },
   155→    { id: 'settings', label: 'Settings', icon: Settings, description: 'Account preferences' },
   156→  ];
   157→
   158→  return [
   159→    { group: 'Main', items: common },
   160→    ...(workspace.length ? [{ group: 'Workspace', items: workspace }] : []),
   161→    ...(resources.length ? [{ group: 'Resources', items: resources }] : []),
   162→    ...(agency.length ? [{ group: 'Team', items: agency }] : []),
   163→    { group: 'Account', items: account },
   164→  ];
   165→}

   260→// Main Export — AuthenticatedDashboard
   261→// ============================================================
   262→export function AuthenticatedDashboard({ user, userName, tierLoading, isAdmin }: { user: User; userName: string | null; tierLoading: boolean; isAdmin?: boolean }) {
   263→  const { tier, setTier, setRoom } = useAppStore();
   264→  const { brandColor, agencyName, setBrandColor } = useTheme();
   265→  const {
   266→    aiCreditsUsed,
   267→    aiCreditsLimit,
   268→    videoMinutesUsed,
   269→    videoMinutesLimit,
   270→    recordingsUsed,
   271→    recordingsLimit,
   272→    loading: usageLoading,
   273→  } = useCredits(user.id);
   274→
   275→  // ---- State ----
   276→  const [activeView, setActiveView] = useState<DashboardView>('overview');
   277→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   278→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   279→
   280→  // Dialog state
   281→  const [showNewLesson, setShowNewLesson] = useState(false);
   282→  const [showTemplateGallery, setShowTemplateGallery] = useState(false);
   283→  const [selectedSubject, setSelectedSubject] = useState<Subject>('GENERAL');
   284→  const [creating, setCreating] = useState(false);
   285→  const [showSignOutConfirm, setShowSignOutConfirm] = useState(false);
   286→
   287→  // Settings state
   288→  const [saveLoading, setSaveLoading] = useState(false);
   289→  const [saveSuccess, setSaveSuccess] = useState(false);
   290→
   291→  // Onboarding state — show wizard to first-time users (no rooms yet)
   292→  const [showOnboarding, setShowOnboarding] = useState(false);
   293→  const [roomsFetched, setRoomsFetched] = useState(false);
   294→  const [hasRooms, setHasRooms] = useState(false);
   295→
   296→  // Student mode toggle
   297→  const [isStudentMode, setIsStudentMode] = useState(false);
   298→
   299→  const { toast } = useToast();
   300→
   301→  // ---- Check if user has any rooms (for onboarding) ----
   302→  useEffect(() => {
   303→    if (!user?.id) return;
   304→    const checkRooms = async () => {
   305→      try {
   306→        const { authFetch } = await import('@/lib/auth-fetch');
   307→        const res = await authFetch(`/api/room/list?tutorId=${user.id}`);
   308→        if (res.ok) {
   309→          const data = await res.json();
   310→          const rooms = data.rooms ?? [];
   311→          setHasRooms(rooms.length > 0);
   312→          if (rooms.length === 0) {
   313→            setShowOnboarding(true);
   314→          }
   315→        }
   316→      } catch { /* silent — proceed to dashboard */ }
   317→      setRoomsFetched(true);
   318→    };
   319→    checkRooms();
   320→  }, [user?.id]);
   321→
   322→  // ---- Dark mode ----
   323→  const [isDark, setIsDark] = useState(false);
   324→  useEffect(() => {
   325→    const saved = localStorage.getItem('superboard-theme');
   326→    if (saved === 'dark') {
   327→      document.documentElement.classList.add('dark');
   328→      setIsDark(true);
   329→    }
   330→  }, []);
   331→
   332→  const toggleDark = useCallback(() => {
   333→    const next = !isDark;
   334→    setIsDark(next);
   335→    if (next) {
   336→      document.documentElement.classList.add('dark');
   337→    } else {
   338→      document.documentElement.classList.remove('dark');
   339→    }
   340→    localStorage.setItem('superboard-theme', next ? 'dark' : 'light');
   341→  }, [isDark]);
   342→
   343→  // ---- Fingerprint reporting ----
   344→  useEffect(() => {
   345→    import('@/lib/fingerprint').then(({ reportFingerprint }) => {
   346→      if (user?.id) reportFingerprint(user.id).catch(console.error);
   347→    });
   348→  }, [user?.id]);
   349→
   350→  // ---- Create lesson handler ----
   351→  const handleCreateLesson = useCallback(async () => {
   352→    setCreating(true);
   353→    try {
   354→      const supabase = createClient();
   355→      const { data: { session } } = await supabase!.auth.getSession();
   356→      const token = session?.access_token;
   357→      if (!token) { setCreating(false); return; }
   358→      const response = await fetch('/api/room', {
   359→        method: 'POST',
   360→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   361→        body: JSON.stringify({ tutorId: user?.id, subject: selectedSubject }),
   362→      });
   363→      if (!response.ok) {
   364→        const errBody = await response.text().catch(() => '');
   365→        throw new Error(`Failed to create room (${response.status}): ${errBody}`);
   366→      }
   367→      const data = await response.json();
   368→      window.location.href = `/room/${data.roomId}`;
   369→    } catch (error: any) {
   370→      toast({ title: 'Failed to create lesson', description: error?.message || 'Please try again.', variant: 'destructive' });
   371→      setCreating(false);
   372→    }
   373→  }, [selectedSubject, user?.id, toast]);
   374→
   375→  // ---- Save brand color handler ----
   376→  const handleSaveBrandColor = useCallback(async () => {
   377→    setSaveLoading(true);
   378→    setSaveSuccess(false);
   379→    try {
   380→      const supabase = createClient();
   381→      const { data: { session } } = await supabase!.auth.getSession();
   382→      const token = session?.access_token;
   383→      if (!token) return;
   384→      await fetch('/api/auth/profile', {
   385→        method: 'PATCH',
   386→        headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
   387→        body: JSON.stringify({ brandingColor: brandColor || null }),
   388→      });
   389→      setSaveSuccess(true);
   390→      setTimeout(() => setSaveSuccess(false), 2000);
   391→    } catch { /* silent */ }
   392→    setSaveLoading(false);
   393→  }, [brandColor]);
   394→
   395→  // ---- Logout handler ----
   396→  const handleLogout = useCallback(async () => {
   397→    const supabase = createClient();
   398→    if (supabase) await supabase.auth.signOut();
   399→    setTier('FREE');
   400→    window.location.href = '/';
   401→  }, [setTier]);
   402→
   403→  // ---- Onboarding complete handler ----
   404→  const handleOnboardingComplete = useCallback((roomId: string) => {
   405→    setShowOnboarding(false);
   406→    setHasRooms(true);
   407→    window.location.href = `/room/${roomId}`;
   408→  }, []);
   409→
   410→  const handleOnboardingSkip = useCallback(() => {
   411→    setShowOnboarding(false);
   412→    setHasRooms(false);
   413→  }, []);
   414→
   415→  // ---- Template gallery handler ----
   416→  const handleTemplateSelect = useCallback((template: TemplateInfo) => {
   417→    setShowTemplateGallery(false);
   418→    setSelectedSubject(template.subject);
   419→    setShowNewLesson(true);
   420→  }, []);
   421→
   422→  // ---- Derived values ----
   423→  const tierLabel = isAgencyTier(tier)
   424→    ? tier === 'AGENCY_PREMIUM' ? 'Agency Premium'
   425→    : tier === 'AGENCY_STANDARD' ? 'Agency Standard'
   426→    : 'Agency'
   427→    : tier === 'PRO' ? 'Pro' : 'Free';
   428→  const tierColor =
   429→    tier === 'AGENCY_PREMIUM' ? 'bg-purple-100 text-purple-800'
   430→    : tier === 'AGENCY_STANDARD' || tier === 'AGENCY' ? 'bg-amber-100 text-amber-800'
   431→    : tier === 'PRO' ? 'bg-emerald-100 text-emerald-800'
   432→    : 'bg-teal-50 text-teal-700';
   433→
   434→  const navGroups = getNavItems(tier, !!isAdmin);
   435→
   436→  // ---- Show onboarding for first-time users ----
   437→  if (showOnboarding && roomsFetched) {
   438→    return (
   439→      <OnboardingWizard
   440→        userId={user.id}
   441→        userEmail={user.email || ''}
   442→        userName={userName}
   443→        onComplete={handleOnboardingComplete}
   444→        onSkip={handleOnboardingSkip}
   445→      />
   446→    );
   447→  }
   448→
   449→  // ---- Show student dashboard if in student mode ----
   450→  if (isStudentMode) {
   451→    return (
   452→      <StudentDashboard
   453→        userId={user.id}
   454→        userName={userName}
   455→        userEmail={user.email || null}
   456→      />
   457→    );
   458→  }
   459→
   460→  // ============================================================
   461→  // RENDER — Dashboard with Sidebar
   462→  // ============================================================
   463→  return (
   464→    <TooltipProvider delayDuration={0}>
   465→      <div className="min-h-screen bg-background flex">
   466→        {/* ============================================================
   467→            SIDEBAR — Collapsible on desktop, sheet on mobile
   468→            ============================================================ */}
   469→        {/* Mobile overlay */}
   470→        {mobileMenuOpen && (
   471→          <div
   472→            className="fixed inset-0 z-40 bg-black/40 backdrop-blur-sm lg:hidden"
   473→            onClick={() => setMobileMenuOpen(false)}
   474→          />
   475→        )}
   476→
   477→        {/* Sidebar */}
   478→        <aside
   479→          className={`
   480→            fixed lg:sticky top-0 left-0 z-50 lg:z-30
   481→            h-screen flex flex-col
   482→            bg-white border-r border-gray-200
   483→            transition-all duration-300 ease-in-out
   484→            ${sidebarCollapsed ? 'w-[68px]' : 'w-64'}
   485→            ${mobileMenuOpen ? 'translate-x-0' : '-translate-x-full lg:translate-x-0'}
   486→          `}
   487→        >
   488→          {/* Sidebar Header / Logo */}
   489→          <div className="flex items-center justify-between px-4 h-16 border-b border-gray-100 shrink-0">
   490→            <div className="flex items-center gap-2.5 min-w-0">
   491→              <div className="w-9 h-9 rounded-xl gradient-primary flex items-center justify-center shadow-md shadow-emerald-500/20 shrink-0">
   492→                <GraduationCap className="w-5 h-5 text-white" />
   493→              </div>
   494→              {!sidebarCollapsed && (
   495→                <div className="min-w-0">
   496→                  <h1 className="text-sm font-bold tracking-tight truncate">{agencyName || 'Superboard'}</h1>
   497→                </div>
   498→              )}
   499→            </div>
   500→            {/* Mobile close */}
   501→            <Button
   502→              variant="ghost"
   503→              size="icon"
   504→              className="lg:hidden h-8 w-8 shrink-0"
   505→              onClick={() => setMobileMenuOpen(false)}
   506→            >
   507→              <PanelLeftClose className="w-4 h-4" />
   508→            </Button>
   509→          </div>
   510→
   511→          {/* Navigation */}
   512→          <nav className="flex-1 overflow-y-auto py-3 px-3 custom-scrollbar">
   513→            {navGroups.map((group, gi) => (
   514→              <div key={group.group} className={gi > 0 ? 'mt-4' : ''}>
   515→                {!sidebarCollapsed && (
   516→                  <p className="text-[10px] font-semibold uppercase tracking-wider text-muted-foreground px-2 mb-1.5">
   517→                    {group.group}
   518→                  </p>
   519→                )}
   520→                {group.items.map((item) => {
   521→                  const isActive = activeView === item.id;
   522→                  const Icon = item.icon;
   523→                  return (
   524→                    <Tooltip key={item.id}>
   525→                      <TooltipTrigger asChild>
   526→                        <button
   527→                          onClick={() => {
   528→                            setActiveView(item.id);
   529→                            setMobileMenuOpen(false);
   530→                          }}
   531→                          className={`
   532→                            w-full flex items-center gap-2.5 rounded-xl text-sm font-medium
   533→                            transition-all duration-150
   534→                            ${sidebarCollapsed ? 'justify-center px-2 py-2.5' : 'px-3 py-2.5'}
   535→                            ${isActive
   536→                              ? 'bg-emerald-50 text-emerald-700 shadow-sm'
   537→                              : 'text-gray-600 hover:bg-gray-50 hover:text-gray-900'
   538→                            }
   539→                          `}
   540→                        >
   541→                          <Icon className={`w-4.5 h-4.5 shrink-0 ${isActive ? 'text-emerald-600' : 'text-gray-400'}`} />
   542→                          {!sidebarCollapsed && (
   543→                            <span className="truncate">{item.label}</span>
   544→                          )}
   545→                          {isActive && !sidebarCollapsed && (
   546→                            <div className="ml-auto w-1.5 h-1.5 rounded-full bg-emerald-500" />
   547→                          )}
   548→                        </button>
   549→                      </TooltipTrigger>
   550→                      {sidebarCollapsed && (
   551→                        <TooltipContent side="right" className="text-xs font-medium">
   552→                          {item.description}
   553→                        </TooltipContent>
   554→                      )}
   555→                    </Tooltip>
   556→                  );
   557→                })}
   558→              </div>
   559→            ))}
   560→          </nav>
   561→
   562→          {/* Templates Button — opens Template Gallery dialog */}
   563→          <div className="px-3 pb-1">
   564→            <Tooltip>
   565→              <TooltipTrigger asChild>
   566→                <button
   567→                  onClick={() => setShowTemplateGallery(true)}
   568→                  className={`
   569→                    w-full flex items-center gap-2.5 rounded-xl text-sm font-medium
   570→                    transition-all duration-150
   571→                    ${sidebarCollapsed ? 'justify-center px-2 py-2.5' : 'px-3 py-2.5'}
