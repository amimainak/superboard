    75→    75→// ---- Perfect-Freehand → Fabric.js Path ----
    76→    76→
    77→    77→function createFreehandPath(
    78→    78→  points: FreehandPoint[],
    79→    79→  strokeColor: string,
    80→    80→  strokeWidth: number
    81→    81→): Path | null {
    82→    82→  if (points.length < 2) return null;
    83→    83→
    84→    84→  const outline = getStroke(points, {
    85→    85→    size: strokeWidth * 3,
    86→    86→    thinning: 0.5,
    87→    87→    smoothing: 0.6,
    88→    88→    streamline: 0.5,
    89→    89→    simulatePressure: true,
    90→    90→  });
    91→    91→
    92→    92→  if (outline.length < 2) return null;
    93→    93→
    94→    94→  // Build SVG path string from the outline points
    95→    95→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
    96→    96→  for (let i = 1; i < outline.length; i++) {
    97→    97→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
    98→    98→  }
    99→    99→  d += ' Z';
   100→   100→
   101→   101→  try {
   102→   102→    const path = new Path(d, {
   103→   103→      fill: strokeColor,
   104→   104→      stroke: strokeColor,
   105→   105→      strokeWidth: 0.5,
   106→   106→      strokeLineCap: 'round',
   107→   107→      strokeLineJoin: 'round',
   108→   108→    } as any);
   109→   109→    return path;
   110→   110→  } catch {
   111→   111→    return null;
   112→   112→  }
   113→   113→}

   119→   119→  return `fabric-whiteboard-${canvasInstanceId}`;
   120→   120→}

   260→   260→    // Resize Handler (performance: debounced)
   261→   261→    // ============================================================
   262→   262→    useEffect(() => {
   263→   263→      const container = containerRef.current;
   264→   264→      const fc = fcanvasRef.current;
   265→   265→      if (!container || !fc) return;

   271→   271→          fc.setDimensions({
   272→   272→            width: container.clientWidth,

   276→   276→        });
   277→   277→      });
   278→   278→
   279→   279→      observer.observe(container);
   280→   280→      return () => {
   281→   281→        observer.disconnect();
   282→   282→        if (rafId) cancelAnimationFrame(rafId);
   283→   283→      };
   284→   284→    }, [isReady]);
   285→   285→

   297→   297→        }
   298→   298→        prevPageIndexRef.current = pageIndex;

   300→   300→    }, [pageIndex, ydoc, sync, history]);
   301→   301→

   303→   303→    // Read-only mode
   304→   304→    // ============================================================
   305→   305→    useEffect(() => {
   306→   306→      const fc = fcanvasRef.current;
   307→   307→      if (!fc) return;
   308→   308→
   309→   309→      if (readOnly) {
   310→   310→        fc.selection = false;
   311→   311→        fc.isDrawingMode = false;
   312→   312→        fc.forEachObject((obj: FabricObject) => {
   313→   313→          obj.selectable = false;
   314→   314→          obj.evented = false;

   317→   317→        tools.setTool(tools.tool);

   321→   321→

   323→   323→    // Awareness Cursor Tracking — broadcast local cursor, render remote
   324→   324→    // ============================================================
   325→   325→    useEffect(() => {
   326→   326→      if (!awareness || !fcanvasRef.current) return;
   327→   327→
   328→   328→      const fc = fcanvasRef.current;
   329→   329→      const localUserId = awareness.clientId;
   330→   330→

   332→   332→      const handleMouseMoveAwareness = (opt: any) => {
   333→   333→        const e = opt.e as MouseEvent;
   334→   334→        const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   335→   335→        awareness.setLocalStateField('cursor', {
   336→   336→          x: pointer.x,
   337→   337→          y: pointer.y,
   338→   338→        });
   339→   339→      };
   340→   340→
   341→   341→      fc.on('mouse:move', handleMouseMoveAwareness);
   342→   342→
   343→   343→      // Observe remote awareness states for cursors
   344→   344→      const handleAwarenessChange = () => {
   345→   345→        const states = awareness.getStates() as Map<number, any>;
   346→   346→        const cursors: RemoteCursor[] = [];
   347→   347→
   348→   348→        states.forEach((state: any, clientId: number) => {
   349→   349→          if (clientId === localUserId) return; // Skip own cursor
   350→   350→          const cursor = state?.cursor;
   351→   351→          const user = state?.user;

   362→   362→

   365→   365→
   366→   366→      awareness.on('change', handleAwarenessChange);
   367→   367→
   368→   368→      return () => {
   369→   369→        fc.off('mouse:move', handleMouseMoveAwareness);
   370→   370→        awareness.off('change', handleAwarenessChange);
   371→   371→      };

   373→   373→
   374→   374→    // ============================================================
   375→   375→    // Sprint 1: Viewport Broadcast (Tutor Side — Focus Mode)
   376→   376→    // Broadcasts viewport (x, y, zoom) via awareness when focus mode is on.
   377→   377→    // ============================================================
   378→   378→    useEffect(() => {
   379→   379→      const fc = fcanvasRef.current;
   380→   380→      if (!fc || !awareness || !isTutor || !focusMode) return;
   381→   381→
   382→   382→      let lastKey = '';
   383→   383→      let debounceTimer: ReturnType<typeof setTimeout> | null = null;
   384→   384→

   386→   386→        const vpt = fc.viewportTransform;

   389→   389→        const x = Math.round(vpt[4] || 0);
   390→   390→        const y = Math.round(vpt[5] || 0);
   391→   391→        const key = `${x},${y},${zoom}`;
   392→   392→
   393→   393→        if (key !== lastKey) {
   394→   394→          lastKey = key;
   395→   395→          if (debounceTimer) clearTimeout(debounceTimer);
   396→   396→          debounceTimer = setTimeout(() => {
   397→   397→            awareness.setLocalStateField('viewport', { x, y, zoom });
   398→   398→          }, 60);
   399→   399→        }
   400→   400→      };
   401→   401→
   402→   402→      // Broadcast on zoom (wheel) and pan (after:render when panning)

   404→   404→        if (isPanning) broadcastCurrentViewport();
   405→   405→      };
   406→   406→
   407→   407→      const handleZoom = () => {
   408→   408→        broadcastCurrentViewport();
   409→   409→      };
   410→   410→
   411→   411→      fc.on('after:render', handleAfterRender);
   412→   412→      fc.on('mouse:wheel', handleZoom);
   413→   413→
   414→   414→      // Initial broadcast
   415→   415→      broadcastCurrentViewport();
   416→   416→
   417→   417→      return () => {
   418→   418→        fc.off('after:render', handleAfterRender);
   419→   419→        fc.off('mouse:wheel', handleZoom);
   420→   420→        if (debounceTimer) clearTimeout(debounceTimer);
   421→   421→        // Clear viewport when focus mode changes or component unmounts
   422→   422→        awareness.setLocalStateField('viewport', null);
   423→   423→      };
   424→   424→    }, [awareness, isTutor, focusMode, isReady, isPanning]);
   425→   425→

   431→   431→      const fc = fcanvasRef.current;

   435→   435→      vpt[0] = appliedViewport.zoom;
   436→   436→      vpt[3] = appliedViewport.zoom;
   437→   437→      vpt[4] = appliedViewport.x;
   438→   438→      vpt[5] = appliedViewport.y;
   439→   439→      fc.setViewportTransform(vpt as any);
   440→   440→      requestRenderThrottled();
   441→   441→    }, [appliedViewport, isTutor, isReady]); // eslint-disable-line react-hooks/exhaustive-deps
   442→   442→
   443→   443→    // ============================================================
   444→   444→    // Canvas Interaction — Pan, Zoom, Draw, Erase, Text, Shapes
   445→   445→    // ============================================================
   446→   446→    useEffect(() => {
   447→   447→      const fc = fcanvasRef.current;
   448→   448→      if (!fc || readOnly) return;
   449→   449→
   450→   450→      // ---- Zoom (mouse wheel) ----
   451→   451→      const handleWheel = (opt: any) => {
   452→   452→        const e = opt.e as WheelEvent;
   453→   453→        e.preventDefault();
   454→   454→        e.stopPropagation();
   455→   455→
   456→   456→        const delta = e.deltaY;
   457→   457→        let zoom = fc.getZoom();
   458→   458→        zoom *= 0.999 ** delta;
   459→   459→        zoom = Math.min(Math.max(0.1, zoom), 10);
   460→   460→        fc.zoomToPoint(new FabricPoint(e.offsetX, e.offsetY), zoom);
   461→   461→        requestRenderThrottled();
   462→   462→      };
   463→   463→      fc.on('mouse:wheel', handleWheel);
   464→   464→
   465→   465→      // ---- Mouse interactions ----
   466→   466→      let panningActive = false;
   467→   467→      let lastPosX = 0;
   468→   468→      let lastPosY = 0;
   469→   469→      let beforeState = '';
   470→   470→
   471→   471→      // ---- Mouse Down ----
   472→   472→      const handleMouseDown = (opt: any) => {
   473→   473→        const e = opt.e as MouseEvent;
   474→   474→
   475→   475→        // Hand tool: left click pans
   476→   476→        if (tools.tool === 'hand' && e.button === 0) {
   477→   477→          panningActive = true;
   478→   478→          setIsPanning(true);
   479→   479→          lastPosX = e.clientX;
   480→   480→          lastPosY = e.clientY;
   481→   481→          (fc as any).defaultCursor = 'grabbing';
   482→   482→          return;
   483→   483→        }
   484→   484→
   485→   485→        // Middle click pans
   486→   486→        if (e.button === 1) {
   487→   487→          panningActive = true;
   488→   488→          setIsPanning(true);
   489→   489→          lastPosX = e.clientX;
   490→   490→          lastPosY = e.clientY;
   491→   491→          return;
   492→   492→        }
   493→   493→
   494→   494→        // Eraser: click to delete
   495→   495→        if (tools.tool === 'eraser' && e.button === 0) {
   496→   496→          const target = opt.target as FabricObject | undefined;
   497→   497→          if (target) {
   498→   498→            beforeState = fc.toJSON().toString();
   499→   499→            fc.remove(target);
   500→   500→            fc.renderAll();
   501→   501→            history.pushState('erase', beforeState, fc.toJSON().toString());
   502→   502→          }
   503→   503→          return;
   504→   504→        }
   505→   505→
   506→   506→        // Text: click empty space to place IText
   507→   507→        if (tools.tool === 'text' && e.button === 0 && !opt.target) {
   508→   508→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   509→   509→          beforeState = fc.toJSON().toString();
   510→   510→
   511→   511→          const text = new IText('Type here', {
   512→   512→            left: pointer.x,

   732→   732→        // Finalize freehand
   733→   733→        if (isDrawingRef.current && tools.tool === 'draw') {
   734→   734→          isDrawingRef.current = false;
   735→   735→

   737→   737→          if (freehandTempPathRef.current) {
   738→   738→            fc.remove(freehandTempPathRef.current);
   739→   739→            freehandTempPathRef.current = null;

   752→   752→              fc.add(finalPath);
   753→   753→            }
   754→   754→          }
   755→   755→
   756→   756→          freehandPointsRef.current = [];
   757→   757→          history.pushState('draw', beforeState, fc.toJSON().toString());
   758→   758→          fc.renderAll();
   759→   759→          return;
   760→   760→        }
   761→   761→
   762→   762→        // Finalize shape
   763→   763→        if (isDrawingRef.current && tempShapeRef.current) {
   764→   764→          tempShapeRef.current.set({
   765→   765→            selectable: true,
   766→   766→            evented: true,
   767→   767→          });
   768→   768→
   769→   769→          history.pushState(
   770→   770→            `draw-${tools.tool}`,
   771→   771→            beforeState,
   772→   772→            fc.toJSON().toString()
   773→   773→          );
   774→   774→          tempShapeRef.current = null;
   775→   775→          isDrawingRef.current = false;
   776→   776→          startPointRef.current = null;
   777→   777→          fc.renderAll();
   778→   778→        }
   779→   779→      };
   780→   780→
   781→   781→      fc.on('mouse:down', handleMouseDown);
   782→   782→      fc.on('mouse:move', handleMouseMove);
   783→   783→      fc.on('mouse:up', handleMouseUp);

   810→   810→          // Don't sync temp paths

   814→   814→

   824→   824→    // Touch: Pinch-to-zoom via native DOM events on container
   825→   825→    // ============================================================
   826→   826→    useEffect(() => {
   827→   827→      const container = containerRef.current;
   828→   828→      const fc = fcanvasRef.current;
   829→   829→      if (!container || !fc) return;
   830→   830→
   831→   831→      const domTouchStart = (e: TouchEvent) => {
   832→   832→        const touches = e.touches;
   833→   833→        if (touches.length === 2) {
   834→   834→          e.preventDefault();
   835→   835→          e.stopPropagation();
   836→   836→
   837→   837→          const dx = touches[0].clientX - touches[1].clientX;
   838→   838→          const dy = touches[0].clientY - touches[1].clientY;
   839→   839→          const dist = Math.sqrt(dx * dx + dy * dy);
   840→   840→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   841→   841→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   842→   842→
   843→   843→          touchStateRef.current = {
   844→   844→            isPinching: true,
   845→   845→            lastDist: dist,
   846→   846→            lastCenter: { x: centerX, y: centerY },
   847→   847→            touchCount: 2,
   848→   848→          };
   849→   849→
   850→   850→          // Cancel any active drawing
   851→   851→          isDrawingRef.current = false;
   852→   852→          tempShapeRef.current = null;
   853→   853→          freehandPointsRef.current = [];
   854→   854→          if (freehandTempPathRef.current) {
   855→   855→            fc.remove(freehandTempPathRef.current);
   856→   856→            freehandTempPathRef.current = null;
   857→   857→          }
   858→   858→        }
   859→   859→      };
   860→   860→
   861→   861→      const domTouchMove = (e: TouchEvent) => {
   862→   862→        const touches = e.touches;
   863→   863→        if (touches.length === 2 && touchStateRef.current.isPinching) {
   864→   864→          e.preventDefault();
   865→   865→          e.stopPropagation();
   866→   866→
   867→   867→          const dx = touches[0].clientX - touches[1].clientX;
   868→   868→          const dy = touches[0].clientY - touches[1].clientY;
   869→   869→          const dist = Math.sqrt(dx * dx + dy * dy);
   870→   870→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   871→   871→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   872→   872→
   873→   873→          const prev = touchStateRef.current;
   874→   874→
   875→   875→          // Zoom
   876→   876→          const scale = dist / Math.max(prev.lastDist, 1);
   877→   877→          let zoom = fc.getZoom() * scale;
   878→   878→          zoom = Math.min(Math.max(0.1, zoom), 10);
   879→   879→
   880→   880→          // Pan
   881→   881→          const vpt = fc.viewportTransform!;
   882→   882→          vpt[4] += centerX - prev.lastCenter.x;
   883→   883→          vpt[5] += centerY - prev.lastCenter.y;
   884→   884→
   885→   885→          // Apply zoom to center of pinch
   886→   886→          fc.zoomToPoint(new FabricPoint(centerX, centerY), zoom);
   887→   887→
   888→   888→          touchStateRef.current = {
   889→   889→            isPinching: true,
   890→   890→            lastDist: dist,
   891→   891→            lastCenter: { x: centerX, y: centerY },
   892→   892→            touchCount: 2,
   893→   893→          };
   894→   894→
   895→   895→          requestRenderThrottled();
   896→   896→        }
   897→   897→      };
   898→   898→
   899→   899→      const domTouchEnd = (e: TouchEvent) => {
   900→   900→        if (e.touches.length < 2) {
   901→   901→          touchStateRef.current.isPinching = false;
   902→   902→          touchStateRef.current.touchCount = e.touches.length;
   903→   903→        }
   904→   904→      };
   905→   905→
   906→   906→      container.addEventListener('touchstart', domTouchStart, { passive: false });
   907→   907→      container.addEventListener('touchmove', domTouchMove, { passive: false });
   908→   908→      container.addEventListener('touchend', domTouchEnd, { passive: false });
   909→   909→
   910→   910→      return () => {
   911→   911→        container.removeEventListener('touchstart', domTouchStart);
   912→   912→        container.removeEventListener('touchmove', domTouchMove);
   913→   913→        container.removeEventListener('touchend', domTouchEnd);
   914→   914→      };
   915→   915→    }, [isReady, requestRenderThrottled]);

   948→   948→        // Calculate bounding rect of all objects

   950→   950→        fc.getObjects().forEach((obj: FabricObject) => {

   962→   962→        const scaleY = fc.getHeight() / contentHeight;
   963→   963→        const zoom = Math.min(scaleX, scaleY, 2); // Cap at 2x
   964→   964→

   966→   966→        const centerX = (minX + maxX) / 2;
   967→   967→        const centerY = (minY + maxY) / 2;
   968→   968→        const vptCenterX = fc.getWidth() / 2;
   969→   969→        const vptCenterY = fc.getHeight() / 2;
   970→   970→
   971→   971→        fc.zoomToPoint(
   972→   972→          new FabricPoint(vptCenterX, vptCenterY),
   973→   973→          zoom
   974→   974→        );
   975→   975→
   976→   976→        // Pan to center content
   977→   977→        const vpt = fc.viewportTransform!;
   978→   978→        vpt[4] = vptCenterX - centerX * zoom;
   979→   979→        vpt[5] = vptCenterY - centerY * zoom;
   980→   980→
   981→   981→        fc.requestRenderAll();
   982→   982→      },
   983→   983→      disabled: readOnly,
   984→   984→      readOnly,
   985→   985→    });
