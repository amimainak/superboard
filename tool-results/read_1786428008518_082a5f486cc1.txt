   864→              </form>
   865→            </>

   867→        </DialogContent>
   868→      </Dialog>

   870→      {/* ===== EMAIL VERIFICATION SCREEN ===== */}

   881→            </p>

   884→            </p>
   885→            <Button variant="outline" className="rounded-xl" onClick={() => setShowVerification(false)}>
   886→              Back to Sign In
   887→            </Button>
   888→          </div>

   890→      )}
   891→    </main>
   892→  );
