    50→  },
    51→  tritanopia: {
    52→    red: '#cc0000', orange: '#cc6600', green: '#009900',
    53→    blue: '#cc3300', purple: '#990099', pink: '#ff6666',

    56→
    57→/** Remap a color name through the color-blind palette if applicable */
    58→function getColorBlindSafeColor(colorName: string, mode: string): string {
    59→  if (mode === 'none' || !COLOR_BLIND_PALETTES[mode]) return colorName;
    60→  return COLOR_BLIND_PALETTES[mode][colorName.toLowerCase()] || colorName;
    61→}

   100→// ---- Perfect-Freehand → Fabric.js Path ----
   101→
   102→function createFreehandPath(
   103→  points: FreehandPoint[],
   104→  strokeColor: string,
   105→  strokeWidth: number
   106→): Path | null {
   107→  if (points.length < 2) return null;
   108→
   109→  const outline = getStroke(points, {
   110→    size: strokeWidth * 3,
   111→    thinning: 0.5,
   112→    smoothing: 0.6,
   113→    streamline: 0.5,
   114→    simulatePressure: true,
   115→  });
   116→
   117→  if (outline.length < 2) return null;
   118→
   119→  // Build SVG path string from the outline points
   120→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
   121→  for (let i = 1; i < outline.length; i++) {
   122→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
   123→  }
   124→  d += ' Z';
   125→
   126→  try {
   127→    const path = new Path(d, {
   128→      fill: strokeColor,
   129→      stroke: strokeColor,
   130→      strokeWidth: 0.5,
   131→      strokeLineCap: 'round',
   132→      strokeLineJoin: 'round',
   133→    } as any);
   134→    return path;
   135→  } catch {
   136→    return null;
   137→  }
   138→}

   145→}

   147→// ---- Main Component ----
   148→
   149→const FabricCanvasComponent = forwardRef<FabricCanvas, FabricCanvasProps>(
   150→  function FabricCanvasInner(
   151→    {
   152→      ydoc,
   153→      pageIndex,
   154→      isTutor,
   155→      readOnly,
   156→      activeTool,
   157→      onCanvasReady,
   158→      awareness,
   159→      appliedViewport,
   160→      focusMode,
   161→      isScratchpad,
   162→    },
   163→    ref
   164→  ) {
   165→    const containerRef = useRef<HTMLDivElement>(null);
   166→    const fcanvasRef = useRef<FabricCanvas | null>(null);
   167→    const [isReady, setIsReady] = useState(false);
   168→    const [isPanning, setIsPanning] = useState(false);
   169→    const canvasIdRef = useRef(getNextCanvasId());
   170→
   171→    // Drawing state for shapes
   172→    const isDrawingRef = useRef(false);
   173→    const startPointRef = useRef<Point | null>(null);
   174→    const tempShapeRef = useRef<FabricObject | null>(null);
   175→
   176→    // Freehand points buffer for perfect-freehand
   177→    const freehandPointsRef = useRef<FreehandPoint[]>([]);
   178→    const freehandTempPathRef = useRef<Path | null>(null);
   179→
   180→    // Touch state for pinch-to-zoom
   181→    const touchStateRef = useRef<{
   182→      isPinching: boolean;
   183→      lastDist: number;
   184→      lastCenter: { x: number; y: number };
   185→      touchCount: number;
   186→    }>({ isPinching: false, lastDist: 0, lastCenter: { x: 0, y: 0 }, touchCount: 0 });
   187→
   188→    // Remote cursors state
   189→    const [remoteCursors, setRemoteCursors] = useState<RemoteCursor[]>([]);
   190→    const cursorOverlayRef = useRef<HTMLDivElement>(null);
   191→
   192→    // Render throttle
   193→    const renderQueuedRef = useRef(false);
   194→    const requestRenderThrottled = useCallback(() => {
   195→      if (renderQueuedRef.current) return;
   196→      renderQueuedRef.current = true;
   197→      requestAnimationFrame(() => {
   198→        fcanvasRef.current?.requestRenderAll();
   199→        renderQueuedRef.current = false;
   200→      });
   201→    }, []);
   202→
   203→    // ---- Expose canvas ref via forwardRef ----
   204→    useImperativeHandle(ref, () => fcanvasRef.current!, []);
   205→
   206→    // ---- Hooks ----
   207→    const tools = useCanvasTools(fcanvasRef);
   208→    const history = useCanvasHistory(fcanvasRef);
   209→
   210→    // Sprint 1: Focus mode — use the hook for viewport broadcast/receive
   211→    const focusModeHook = useFocusMode({
   212→      awareness,
   213→      onViewportReceived: undefined, // Student-side receive handled by appliedViewport prop
   214→    });
   215→
   216→    // Register canvas ref for keyboard shortcuts
   217→    useEffect(() => {
   218→      setKeyboardCanvasRef(fcanvasRef);
   219→      return () => setKeyboardCanvasRef({ current: null } as React.MutableRefObject<FabricCanvas | null>);
   220→    }, []);
   221→
   222→    // ============================================================
   223→    // Sync external tool changes (from toolbar) to canvas
   224→    // ============================================================
   225→    const prevActiveToolRef = useRef(activeTool);
   226→    useEffect(() => {
   227→      if (activeTool && activeTool !== prevActiveToolRef.current) {
   228→        prevActiveToolRef.current = activeTool;
   229→        tools.setTool(activeTool as CanvasTool);
   230→      }
   231→    }, [activeTool, tools]);
   232→
   233→    const sync = useYjsCanvasSync({
   234→      ydoc,
   235→      fcanvasRef,
   236→      pageIndex: isScratchpad ? -1 : pageIndex,
   237→      mapKeyPrefix: isScratchpad ? 'scratchpad-shapes' : 'page-shapes',
   238→      onRemoteChange: () => {},
   239→    });
   240→
   241→    // ============================================================
   242→    // Canvas Initialization
   243→    // ============================================================
   244→    useEffect(() => {
   245→      const container = containerRef.current;
   246→      if (!container) return;
   247→
   248→      const canvasId = canvasIdRef.current;
   249→      const fc = new Canvas(canvasId, {
   250→        width: container.clientWidth,
   251→        height: container.clientHeight,
   252→        backgroundColor: '#ffffff',
   253→        selection: false,
   254→        preserveObjectStacking: true,
   255→        stopContextMenu: true,
   256→        fireRightClick: true,
   257→        fireMiddleClick: true,
   258→        enableRetinaScaling: true,
   259→        renderOnAddRemove: false, // Performance: batch renders
   260→      });
   261→
   262→      // Configure PencilBrush for freehand (fallback / raw path mode)
   263→      const brush = new PencilBrush(fc);
   264→      brush.color = tools.strokeColor;
   265→      brush.width = tools.strokeWidth;
   266→      brush.decimate = 2; // Reduce point density for performance
   267→      fc.freeDrawingBrush = brush;
   268→      fc.isDrawingMode = false;
   269→
   270→      fcanvasRef.current = fc;
   271→      setIsReady(true);
   272→
   273→      sync.loadInitialState(fc);
   274→      sync.wireCanvasEvents(fc);
   275→      sync.setupRemoteObserver();
   276→      onCanvasReady?.(fc);
   277→
   278→      // Initial render
   279→      fc.requestRenderAll();
   280→
   281→      return () => {
   282→        sync.flushPending();
   283→        fc.dispose();
   284→        fcanvasRef.current = null;
   285→        setIsReady(false);
   286→      };
   287→      // eslint-disable-next-line react-hooks/exhaustive-deps
   288→    }, []);
   289→
   290→    // ============================================================
   291→    // Resize Handler (performance: debounced)
   292→    // ============================================================
   293→    useEffect(() => {
   294→      const container = containerRef.current;
   295→      const fc = fcanvasRef.current;
   296→      if (!container || !fc) return;
   297→
   298→      let rafId: number;
   299→      const observer = new ResizeObserver(() => {
   300→        if (rafId) cancelAnimationFrame(rafId);
   301→        rafId = requestAnimationFrame(() => {
   302→          fc.setDimensions({
   303→            width: container.clientWidth,
   304→            height: container.clientHeight,
   305→          });
   306→          fc.requestRenderAll();
   307→        });
   308→      });
   309→
   310→      observer.observe(container);
   311→      return () => {
   312→        observer.disconnect();
   313→        if (rafId) cancelAnimationFrame(rafId);
   314→      };
   315→    }, [isReady]);
   316→
   317→    // ============================================================
   318→    // Page Change — reload from Yjs
   319→    // ============================================================
   320→    const prevPageIndexRef = useRef(pageIndex);
   321→    useEffect(() => {
   322→      if (pageIndex !== prevPageIndexRef.current) {
   323→        const fc = fcanvasRef.current;
   324→        if (fc && ydoc) {
   325→          sync.flushPending();
   326→          history.clearHistory();
   327→          sync.loadInitialState(fc);
   328→        }
   329→        prevPageIndexRef.current = pageIndex;
   330→      }
   331→    }, [pageIndex, ydoc, sync, history]);
   332→
   333→    // ============================================================
   334→    // Read-only mode
   335→    // ============================================================
   336→    useEffect(() => {
   337→      const fc = fcanvasRef.current;
   338→      if (!fc) return;
   339→
   340→      if (readOnly) {
   341→        fc.selection = false;
   342→        fc.isDrawingMode = false;
   343→        fc.forEachObject((obj: FabricObject) => {
   344→          obj.selectable = false;
   345→          obj.evented = false;
   346→        });
   347→      } else {
   348→        tools.setTool(tools.tool);
   349→      }
   350→      fc.requestRenderAll();
   351→    }, [readOnly, tools]);
   352→
   353→    // ============================================================
   354→    // Awareness Cursor Tracking — broadcast local cursor, render remote
   355→    // ============================================================
   356→    useEffect(() => {
   357→      if (!awareness || !fcanvasRef.current) return;
   358→
   359→      const fc = fcanvasRef.current;
   360→      const localUserId = awareness.clientId;
   361→
   362→      // Broadcast cursor position on mouse move
   363→      const handleMouseMoveAwareness = (opt: any) => {
   364→        const e = opt.e as MouseEvent;
   365→        const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   366→        awareness.setLocalStateField('cursor', {
   367→          x: pointer.x,
   368→          y: pointer.y,
   369→        });
   370→      };
   371→
   372→      fc.on('mouse:move', handleMouseMoveAwareness);
   373→
   374→      // Observe remote awareness states for cursors
   375→      const handleAwarenessChange = () => {
   376→        const states = awareness.getStates() as Map<number, any>;
   377→        const cursors: RemoteCursor[] = [];
   378→
   379→        states.forEach((state: any, clientId: number) => {
   380→          if (clientId === localUserId) return; // Skip own cursor
   381→          const cursor = state?.cursor;
   382→          const user = state?.user;
   383→          if (cursor && user) {
   384→            cursors.push({
   385→              id: String(clientId),
   386→              name: user.name || 'Anonymous',
   387→              color: user.color || '#6366f1',
   388→              x: cursor.x,
   389→              y: cursor.y,
   390→            });
   391→          }
   392→        });
   393→
   394→        setRemoteCursors(cursors);
   395→      };
   396→
   397→      awareness.on('change', handleAwarenessChange);
   398→
   399→      return () => {
   400→        fc.off('mouse:move', handleMouseMoveAwareness);
   401→        awareness.off('change', handleAwarenessChange);
   402→      };
   403→    }, [awareness, isReady]);
   404→
   405→    // ============================================================
   406→    // Sprint 1: Viewport Broadcast (Tutor Side — Focus Mode)
   407→    // Broadcasts viewport (x, y, zoom) via awareness when focus mode is on.
   408→    // ============================================================
   409→    useEffect(() => {
   410→      const fc = fcanvasRef.current;
   411→      if (!fc || !awareness || !isTutor || !focusMode) return;
   412→
   413→      let lastKey = '';
   414→      let debounceTimer: ReturnType<typeof setTimeout> | null = null;
   415→
   416→      const broadcastCurrentViewport = () => {
   417→        const vpt = fc.viewportTransform;
   418→        if (!vpt) return;
   419→        const zoom = parseFloat((vpt[0] || 1).toFixed(2));
   420→        const x = Math.round(vpt[4] || 0);
   421→        const y = Math.round(vpt[5] || 0);
   422→        const key = `${x},${y},${zoom}`;
   423→
   424→        if (key !== lastKey) {
   425→          lastKey = key;
   426→          if (debounceTimer) clearTimeout(debounceTimer);
   427→          debounceTimer = setTimeout(() => {
   428→            awareness.setLocalStateField('viewport', { x, y, zoom });
   429→          }, 60);
   430→        }
   431→      };
   432→
   433→      // Broadcast on zoom (wheel) and pan (after:render when panning)
   434→      const handleAfterRender = () => {
   435→        if (isPanning) broadcastCurrentViewport();
   436→      };
   437→
   438→      const handleZoom = () => {
   439→        broadcastCurrentViewport();
   440→      };
   441→
   442→      // Sprint 1: Also broadcast on mouse:up (when pan/zoom gesture ends)
   443→      const handleMouseUp = () => {
   444→        broadcastCurrentViewport();
   445→      };
   446→
   447→      fc.on('after:render', handleAfterRender);
   448→      fc.on('mouse:wheel', handleZoom);
   449→      fc.on('mouse:up', handleMouseUp);
   450→
   451→      // Initial broadcast
   452→      broadcastCurrentViewport();
   453→
   454→      return () => {
   455→        fc.off('after:render', handleAfterRender);
   456→        fc.off('mouse:wheel', handleZoom);
   457→        fc.off('mouse:up', handleMouseUp);
   458→        if (debounceTimer) clearTimeout(debounceTimer);
   459→        // Clear viewport when focus mode changes or component unmounts
   460→        awareness.setLocalStateField('viewport', null);
   461→      };
   462→    }, [awareness, isTutor, focusMode, isReady, isPanning]);
   463→
   464→    // ============================================================
   465→    // Sprint 1: Viewport Receive (Student Side — Focus Mode)
   466→    // Applies viewport broadcasted by tutor.
   467→    // ============================================================
   468→    useEffect(() => {
   469→      const fc = fcanvasRef.current;
   470→      if (!fc || !appliedViewport || isTutor) return;
   471→
   472→      const vpt = [...(fc.viewportTransform || [1, 0, 0, 1, 0, 0])];
   473→      vpt[0] = appliedViewport.zoom;
   474→      vpt[3] = appliedViewport.zoom;
   475→      vpt[4] = appliedViewport.x;
   476→      vpt[5] = appliedViewport.y;
   477→      fc.setViewportTransform(vpt as any);
   478→      requestRenderThrottled();
   479→    }, [appliedViewport, isTutor, isReady]); // eslint-disable-line react-hooks/exhaustive-deps
   480→
   481→    // ============================================================
   482→    // Canvas Interaction — Pan, Zoom, Draw, Erase, Text, Shapes
   483→    // ============================================================
   484→    useEffect(() => {
   485→      const fc = fcanvasRef.current;
   486→      if (!fc || readOnly) return;
   487→
   488→      // ---- Zoom (mouse wheel) ----
   489→      const handleWheel = (opt: any) => {
   490→        const e = opt.e as WheelEvent;
   491→        e.preventDefault();
   492→        e.stopPropagation();
   493→
   494→        const delta = e.deltaY;
   495→        let zoom = fc.getZoom();
   496→        zoom *= 0.999 ** delta;
   497→        zoom = Math.min(Math.max(0.1, zoom), 10);
   498→        fc.zoomToPoint(new FabricPoint(e.offsetX, e.offsetY), zoom);
   499→        requestRenderThrottled();
   500→      };
   501→      fc.on('mouse:wheel', handleWheel);
   502→
   503→      // ---- Mouse interactions ----
   504→      let panningActive = false;
   505→      let lastPosX = 0;
   506→      let lastPosY = 0;
   507→      let beforeState = '';
   508→
   509→      // ---- Mouse Down ----
   510→      const handleMouseDown = (opt: any) => {
   511→        const e = opt.e as MouseEvent;
   512→
   513→        // Hand tool: left click pans
   514→        if (tools.tool === 'hand' && e.button === 0) {
   515→          panningActive = true;
   516→          setIsPanning(true);
   517→          lastPosX = e.clientX;
   518→          lastPosY = e.clientY;
   519→          (fc as any).defaultCursor = 'grabbing';
   520→          return;
   521→        }
   522→
   523→        // Middle click pans
   524→        if (e.button === 1) {
   525→          panningActive = true;
   526→          setIsPanning(true);
   527→          lastPosX = e.clientX;
   528→          lastPosY = e.clientY;
   529→          return;
   530→        }
   531→
   532→        // Eraser: click to delete
   533→        if (tools.tool === 'eraser' && e.button === 0) {
   534→          const target = opt.target as FabricObject | undefined;
   535→          if (target) {
   536→            beforeState = fc.toJSON().toString();
   537→            fc.remove(target);
   538→            fc.renderAll();
   539→            history.pushState('erase', beforeState, fc.toJSON().toString());
   540→          }
   541→          return;
   542→        }
   543→
   544→        // Text: click empty space to place IText
   545→        if (tools.tool === 'text' && e.button === 0 && !opt.target) {
   546→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   547→          beforeState = fc.toJSON().toString();
   548→
   549→          const text = new IText('Type here', {
   550→            left: pointer.x,
   551→            top: pointer.y,
   552→            fontSize: tools.fontSize,
   553→            fill: tools.strokeColor,
   554→            fontFamily: 'Inter, system-ui, sans-serif',
   555→            selectable: true,
   556→            evented: true,
   557→          } as any);
   558→          (text as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   559→
   560→          fc.add(text);
   561→          fc.setActiveObject(text);
   562→          text.enterEditing();
   563→          text.selectAll();
   564→          fc.renderAll();
   565→
   566→          history.pushState('add-text', beforeState, fc.toJSON().toString());
   567→          return;
   568→        }
   569→
   570→        // Freehand (draw tool): start collecting points for perfect-freehand
   571→        if (tools.tool === 'draw' && e.button === 0 && !opt.target) {
   572→          // We use Fabric's built-in freeDrawingMode but with perfect-freehand
   573→          // post-processing. The PencilBrush collects points; on path:created
   574→          // we replace it with a smooth perfect-freehand path.
   575→          // But for real-time preview, we track raw points ourselves.
   576→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   577→          freehandPointsRef.current = [{ x: pointer.x, y: pointer.y, pressure: 0.5 }];
   578→          isDrawingRef.current = true;
   579→          beforeState = fc.toJSON().toString();
   580→          return;
   581→        }
   582→
   583→        // Shape tools: start drawing
   584→        if (['rectangle', 'ellipse', 'line', 'arrow'].includes(tools.tool) && e.button === 0 && !opt.target) {
   585→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   586→          isDrawingRef.current = true;
   587→          startPointRef.current = pointer;
   588→          beforeState = fc.toJSON().toString();
   589→
   590→          const objName = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   591→          let shape: FabricObject;
   592→
   593→          switch (tools.tool) {
   594→            case 'rectangle':
   595→              shape = new Rect({
   596→                left: pointer.x,
   597→                top: pointer.y,
   598→                width: 1,
   599→                height: 1,
   600→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   601→                stroke: tools.strokeColor,
   602→                strokeWidth: tools.strokeWidth,
   603→                selectable: false,
   604→                evented: false,
   605→              } as any);
   606→              break;
   607→
   608→            case 'ellipse':
   609→              shape = new Ellipse({
   610→                left: pointer.x,
   611→                top: pointer.y,
   612→                rx: 1,
   613→                ry: 1,
   614→                fill: tools.fillColor === 'transparent' ? 'transparent' : tools.fillColor,
   615→                stroke: tools.strokeColor,
   616→                strokeWidth: tools.strokeWidth,
   617→                selectable: false,
   618→                evented: false,
   619→              } as any);
   620→              break;
   621→
   622→            case 'line':
   623→              shape = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   624→                stroke: tools.strokeColor,
   625→                strokeWidth: tools.strokeWidth,
   626→                selectable: false,
   627→                evented: false,
   628→              } as any);
   629→              break;
   630→
   631→            case 'arrow': {
   632→              const line = new Line([pointer.x, pointer.y, pointer.x, pointer.y], {
   633→                stroke: tools.strokeColor,
   634→                strokeWidth: tools.strokeWidth,
   635→                selectable: false,
   636→                evented: false,
   637→              } as any);
   638→              const headSize = Math.max(10, tools.strokeWidth * 4);
   639→              const arrowhead = new Triangle({
   640→                left: pointer.x,
   641→                top: pointer.y,
   642→                width: headSize,
   643→                height: headSize,
   644→                fill: tools.strokeColor,
   645→                angle: 90,
   646→                originX: 'center',
   647→                originY: 'center',
   648→                selectable: false,
   649→                evented: false,
   650→              } as any);
   651→              shape = new Group([line, arrowhead], {
   652→                selectable: false,
   653→                evented: false,
   654→              } as any);
   655→              break;
   656→            }
   657→
   658→            default:
   659→              return;
   660→          }
   661→
   662→          (shape as any).name = objName;
   663→          fc.add(shape);
   664→          tempShapeRef.current = shape;
   665→          requestRenderThrottled();
   666→          return;
   667→        }
   668→      };
   669→
   670→      // ---- Mouse Move ----
   671→      const handleMouseMove = (opt: any) => {
   672→        const e = opt.e as MouseEvent;
   673→
   674→        // Panning
   675→        if (panningActive) {
   676→          const vpt = fc.viewportTransform!;
   677→          vpt[4] += e.clientX - lastPosX;
   678→          vpt[5] += e.clientY - lastPosY;
   679→          lastPosX = e.clientX;
   680→          lastPosY = e.clientY;
   681→          requestRenderThrottled();
   682→          return;
   683→        }
   684→
   685→        // Freehand drawing — collect points
   686→        if (isDrawingRef.current && tools.tool === 'draw') {
   687→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   688→          freehandPointsRef.current.push({
   689→            x: pointer.x,
   690→            y: pointer.y,
   691→            pressure: 0.5,
   692→          });
   693→
   694→          // Real-time preview with perfect-freehand
   695→          if (freehandTempPathRef.current) {
   696→            fc.remove(freehandTempPathRef.current);
   697→          }
   698→
   699→          const previewPath = createFreehandPath(
   700→            freehandPointsRef.current,
   701→            tools.strokeColor,
   702→            tools.strokeWidth
   703→          );
   704→          if (previewPath) {
   705→            (previewPath as any).name = '__freehand_temp__';
   706→            freehandTempPathRef.current = previewPath;
   707→            fc.add(previewPath);
   708→          }
   709→
   710→          requestRenderThrottled();
   711→          return;
   712→        }
   713→
   714→        // Shape drawing
   715→        if (isDrawingRef.current && tempShapeRef.current && startPointRef.current) {
   716→          const pointer = fc.getScenePoint(e) as unknown as FabricPoint;
   717→          const start = startPointRef.current;
   718→          const t = tempShapeRef.current;
   719→
   720→          if (t instanceof Rect) {
   721→            const left = Math.min(start.x, pointer.x);
   722→            const top = Math.min(start.y, pointer.y);
   723→            t.set({
   724→              left,
   725→              top,
   726→              width: Math.abs(pointer.x - start.x),
   727→              height: Math.abs(pointer.y - start.y),
   728→            });
   729→          } else if (t instanceof Ellipse) {
   730→            const left = Math.min(start.x, pointer.x);
   731→            const top = Math.min(start.y, pointer.y);
   732→            const rx = Math.max(1, Math.abs(pointer.x - start.x) / 2);
   733→            const ry = Math.max(1, Math.abs(pointer.y - start.y) / 2);
   734→            t.set({ left: left + rx, top: top + ry, rx, ry });
   735→          } else if (t instanceof Line) {
   736→            t.set({ x2: pointer.x, y2: pointer.y });
   737→          } else if (t instanceof Group) {
   738→            const objs = t.getObjects();
   739→            const line = objs[0] as Line;
   740→            const head = objs[1] as Triangle;
   741→            if (line && head) {
   742→              line.set({ x2: pointer.x - start.x, y2: pointer.y - start.y });
   743→              head.set({
   744→                left: pointer.x - start.x,
   745→                top: pointer.y - start.y,
   746→                angle:
   747→                  Math.atan2(pointer.y - start.y, pointer.x - start.x) *
   748→                    (180 / Math.PI) +
   749→                  90,
   750→              });
   751→            }
   752→            t.setCoords();
   753→          }
   754→
   755→          requestRenderThrottled();
   756→        }
   757→      };
   758→
   759→      // ---- Mouse Up ----
   760→      const handleMouseUp = () => {
   761→        // Stop panning
   762→        if (panningActive) {
   763→          panningActive = false;
   764→          setIsPanning(false);
   765→          (fc as any).defaultCursor =
   766→            tools.tool === 'hand' ? 'grab' : 'default';
   767→          return;
   768→        }
   769→
   770→        // Finalize freehand
   771→        if (isDrawingRef.current && tools.tool === 'draw') {
   772→          isDrawingRef.current = false;
   773→
   774→          // Remove preview path
   775→          if (freehandTempPathRef.current) {
   776→            fc.remove(freehandTempPathRef.current);
   777→            freehandTempPathRef.current = null;
   778→          }
   779→
   780→          // Create final perfect-freehand path
   781→          const points = freehandPointsRef.current;
   782→          if (points.length >= 2) {
   783→            const finalPath = createFreehandPath(
   784→              points,
   785→              tools.strokeColor,
   786→              tools.strokeWidth
   787→            );
   788→            if (finalPath) {
   789→              (finalPath as any).name = `obj-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   790→              fc.add(finalPath);
   791→            }
   792→          }
   793→
   794→          freehandPointsRef.current = [];
   795→          history.pushState('draw', beforeState, fc.toJSON().toString());
   796→          fc.renderAll();
   797→          return;
   798→        }
   799→
   800→        // Finalize shape
   801→        if (isDrawingRef.current && tempShapeRef.current) {
   802→          tempShapeRef.current.set({
   803→            selectable: true,
   804→            evented: true,
   805→          });
   806→
   807→          history.pushState(
   808→            `draw-${tools.tool}`,
   809→            beforeState,
   810→            fc.toJSON().toString()
   811→          );
   812→          tempShapeRef.current = null;
   813→          isDrawingRef.current = false;
   814→          startPointRef.current = null;
   815→          fc.renderAll();
   816→        }
   817→      };
   818→
   819→      fc.on('mouse:down', handleMouseDown);
   820→      fc.on('mouse:move', handleMouseMove);
   821→      fc.on('mouse:up', handleMouseUp);
   822→
   823→      // Track undo state: capture before-state when user starts modifying
   824→      const undoBeforeRef = useRef('');
   825→      const captureBefore = () => {
   826→        undoBeforeRef.current = fc.toJSON().toString();
   827→      };
   828→      fc.on('object:moving', captureBefore);
   829→      fc.on('object:scaling', captureBefore);
   830→      fc.on('object:rotating', captureBefore);
   831→      fc.on('object:modified', () => {
   832→        if (undoBeforeRef.current) {
   833→          history.pushState(
   834→            'modify',
   835→            undoBeforeRef.current,
   836→            fc.toJSON().toString()
   837→          );
   838→          undoBeforeRef.current = '';
   839→        }
   840→      });
   841→
   842→      // Filter out temp freehand paths from sync
   843→      const origBeforeAdd = fc.on;
   844→      fc.on('object:added', ((e: any) => {
   845→        const obj = e.target as FabricObject;
   846→        const name = (obj as any).name;
   847→        if (name === '__freehand_temp__') {
   848→          // Don't sync temp paths
   849→          return;
   850→        }
   851→      }) as any);
   852→
   853→      return () => {
   854→        fc.off('mouse:wheel', handleWheel);
   855→        fc.off('mouse:down', handleMouseDown);
   856→        fc.off('mouse:move', handleMouseMove);
   857→        fc.off('mouse:up', handleMouseUp);
   858→      };
   859→    }, [isReady, tools.tool, tools.strokeColor, tools.strokeWidth, tools.fillColor, tools.fontSize, readOnly, history, requestRenderThrottled]);
   860→
   861→    // ============================================================
   862→    // Touch: Pinch-to-zoom via native DOM events on container
   863→    // ============================================================
   864→    useEffect(() => {
   865→      const container = containerRef.current;
   866→      const fc = fcanvasRef.current;
   867→      if (!container || !fc) return;
   868→
   869→      const domTouchStart = (e: TouchEvent) => {
   870→        const touches = e.touches;
   871→        if (touches.length === 2) {
   872→          e.preventDefault();
   873→          e.stopPropagation();
   874→
   875→          const dx = touches[0].clientX - touches[1].clientX;
   876→          const dy = touches[0].clientY - touches[1].clientY;
   877→          const dist = Math.sqrt(dx * dx + dy * dy);
   878→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   879→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   880→
   881→          touchStateRef.current = {
   882→            isPinching: true,
   883→            lastDist: dist,
   884→            lastCenter: { x: centerX, y: centerY },
   885→            touchCount: 2,
   886→          };
   887→
   888→          // Cancel any active drawing
   889→          isDrawingRef.current = false;
   890→          tempShapeRef.current = null;
   891→          freehandPointsRef.current = [];
   892→          if (freehandTempPathRef.current) {
   893→            fc.remove(freehandTempPathRef.current);
   894→            freehandTempPathRef.current = null;
   895→          }
   896→        }
   897→      };
   898→
   899→      const domTouchMove = (e: TouchEvent) => {
   900→        const touches = e.touches;
   901→        if (touches.length === 2 && touchStateRef.current.isPinching) {
   902→          e.preventDefault();
   903→          e.stopPropagation();
   904→
   905→          const dx = touches[0].clientX - touches[1].clientX;
   906→          const dy = touches[0].clientY - touches[1].clientY;
   907→          const dist = Math.sqrt(dx * dx + dy * dy);
   908→          const centerX = (touches[0].clientX + touches[1].clientX) / 2;
   909→          const centerY = (touches[0].clientY + touches[1].clientY) / 2;
   910→
   911→          const prev = touchStateRef.current;
   912→
   913→          // Zoom
   914→          const scale = dist / Math.max(prev.lastDist, 1);
   915→          let zoom = fc.getZoom() * scale;
   916→          zoom = Math.min(Math.max(0.1, zoom), 10);
   917→
   918→          // Pan
   919→          const vpt = fc.viewportTransform!;
   920→          vpt[4] += centerX - prev.lastCenter.x;
   921→          vpt[5] += centerY - prev.lastCenter.y;
   922→
   923→          // Apply zoom to center of pinch
   924→          fc.zoomToPoint(new FabricPoint(centerX, centerY), zoom);
   925→
   926→          touchStateRef.current = {
   927→            isPinching: true,
   928→            lastDist: dist,
   929→            lastCenter: { x: centerX, y: centerY },
   930→            touchCount: 2,
   931→          };
   932→
   933→          requestRenderThrottled();
   934→        }
   935→      };
   936→
   937→      const domTouchEnd = (e: TouchEvent) => {
   938→        if (e.touches.length < 2) {
   939→          touchStateRef.current.isPinching = false;
   940→          touchStateRef.current.touchCount = e.touches.length;
   941→        }
   942→      };
   943→
   944→      container.addEventListener('touchstart', domTouchStart, { passive: false });
   945→      container.addEventListener('touchmove', domTouchMove, { passive: false });
   946→      container.addEventListener('touchend', domTouchEnd, { passive: false });
   947→
   948→      return () => {
   949→        container.removeEventListener('touchstart', domTouchStart);
   950→        container.removeEventListener('touchmove', domTouchMove);
   951→        container.removeEventListener('touchend', domTouchEnd);
   952→      };
   953→    }, [isReady, requestRenderThrottled]);
   954→
   955→    // ============================================================
   956→    // Keyboard Shortcuts (Phase 4: extended)
   957→    // ============================================================
   958→    useKeyboardShortcuts({
   959→      onUndo: history.undo,
   960→      onRedo: history.redo,
   961→      onDelete: () => {
   962→        const fc = fcanvasRef.current;
   963→        if (!fc || readOnly) return;
   964→        const active = fc.getActiveObject();
   965→        if (active) {
   966→          const before = fc.toJSON().toString();
   967→          fc.remove(active);
   968→          fc.renderAll();
   969→          history.pushState('delete', before, fc.toJSON().toString());
   970→          fc.discardActiveObject();
   971→        }
   972→      },
   973→      onEscape: () => {
   974→        const fc = fcanvasRef.current;
   975→        if (fc) {
   976→          fc.discardActiveObject();
   977→          fc.requestRenderAll();
   978→        }
   979→      },
   980→      onToolChange: (tool: CanvasTool) => {
   981→        tools.setTool(tool);
   982→      },
   983→      onZoomToFit: () => {
   984→        const fc = fcanvasRef.current;
   985→        if (!fc || fc.getObjects().length === 0) return;
   986→        // Calculate bounding rect of all objects
   987→        let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   988→        fc.getObjects().forEach((obj: FabricObject) => {
   989→          const bound = obj.getBoundingRect();
   990→          minX = Math.min(minX, bound.left);
   991→          minY = Math.min(minY, bound.top);
   992→          maxX = Math.max(maxX, bound.left + bound.width);
   993→          maxY = Math.max(maxY, bound.top + bound.height);
   994→        });
   995→
   996→        const padding = 60;
   997→        const contentWidth = maxX - minX + padding * 2;
   998→        const contentHeight = maxY - minY + padding * 2;
   999→        const scaleX = fc.getWidth() / contentWidth;
  1000→        const scaleY = fc.getHeight() / contentHeight;
  1001→        const zoom = Math.min(scaleX, scaleY, 2); // Cap at 2x
  1002→
  1003→        fc.setViewportTransform([1, 0, 0, 1, 0, 0]);
  1004→        const centerX = (minX + maxX) / 2;
  1005→        const centerY = (minY + maxY) / 2;
  1006→        const vptCenterX = fc.getWidth() / 2;
  1007→        const vptCenterY = fc.getHeight() / 2;
  1008→
  1009→        fc.zoomToPoint(
  1010→          new FabricPoint(vptCenterX, vptCenterY),
  1011→          zoom
  1012→        );
  1013→
  1014→        // Pan to center content
  1015→        const vpt = fc.viewportTransform!;
  1016→        vpt[4] = vptCenterX - centerX * zoom;
  1017→        vpt[5] = vptCenterY - centerY * zoom;
  1018→
  1019→        fc.requestRenderAll();
  1020→      },
  1021→      disabled: readOnly,
  1022→      readOnly,
  1023→    });
  1024→
  1025→    // ============================================================
  1026→    // Compute viewport transform for remote cursor rendering
  1027→    // ============================================================
  1028→    const vpt = fcanvasRef.current?.viewportTransform;
  1029→
  1030→    // ============================================================
  1031→    // Render
  1032→    // ============================================================
  1033→    return (
  1034→      <div
  1035→        ref={containerRef}
  1036→        className="w-full h-full relative"
  1037→        style={{
  1038→          cursor: isPanning
  1039→            ? 'grabbing'
  1040→            : tools.tool === 'hand'
  1041→              ? 'grab'
  1042→              : tools.tool === 'eraser'
  1043→                ? 'crosshair'
  1044→                : tools.tool === 'text'
  1045→                  ? 'text'
  1046→                  : tools.tool === 'draw'
  1047→                    ? 'crosshair'
  1048→                    : 'default',
  1049→          touchAction: 'none', // Prevent browser touch defaults for pinch-zoom
  1050→        }}
  1051→      >
  1052→        {!isReady && (
  1053→          <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur-sm">
  1054→            <div className="flex items-center gap-2 text-sm text-muted-foreground">
  1055→              <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
  1056→              <span>Loading canvas...</span>
  1057→            </div>
  1058→          </div>
  1059→        )}
  1060→        <canvas id={canvasIdRef.current} />
  1061→
  1062→        {/* Remote Cursors Overlay */}
  1063→        {remoteCursors.length > 0 && vpt && (
  1064→          <div
  1065→            ref={cursorOverlayRef}
  1066→            className="absolute inset-0 pointer-events-none z-20 overflow-hidden"
  1067→            style={{ touchAction: 'none' }}
  1068→          >
  1069→            {remoteCursors.map((cursor) => (
  1070→              <div
  1071→                key={cursor.id}
  1072→                className="absolute transition-transform duration-75 ease-out"
  1073→                style={{
  1074→                  left: cursor.x * vpt[0] + vpt[4],
  1075→                  top: cursor.y * vpt[3] + vpt[5],
  1076→                  transform: 'translate(-2px, -2px)',
  1077→                }}
  1078→              >
  1079→                {/* Cursor dot */}
  1080→                <svg
  1081→                  width="12"
  1082→                  height="18"
  1083→                  viewBox="0 0 12 18"
  1084→                  fill="none"
  1085→                  className="drop-shadow-sm"
  1086→                >
  1087→                  <path
  1088→                    d="M1 1L6 16L7.5 10.5L12 9L1 1Z"
  1089→                    fill={cursor.color}
  1090→                    stroke="white"
  1091→                    strokeWidth="1.5"
  1092→                    strokeLinejoin="round"
  1093→                  />
  1094→                </svg>
  1095→                {/* Name label */}
  1096→                <div
  1097→                  className="absolute left-3 top-3 px-1.5 py-0.5 rounded text-[10px] font-medium whitespace-nowrap shadow-sm"
  1098→                  style={{
  1099→                    backgroundColor: cursor.color,
  1100→                    color: 'white',
  1101→                  }}
  1102→                >
  1103→                  {cursor.name}
  1104→                </div>
  1105→              </div>
  1106→            ))}
  1107→          </div>
  1108→        )}
  1109→
  1110→        {/* Zoom indicator */}
  1111→        <div className="absolute bottom-2 right-2 z-20 px-2 py-0.5 rounded-md bg-black/50 text-white text-[10px] font-mono">
  1112→          {Math.round((fcanvasRef.current?.getZoom() || 1) * 100)}%
  1113→        </div>
  1114→      </div>
  1115→    );
  1116→  }
  1117→);
