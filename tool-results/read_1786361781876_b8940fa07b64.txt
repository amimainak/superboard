   143→    if (score <= 3) return { score, label: 'Fair', color: 'bg-amber-500' };

   743→                  <BookOpen className="w-7 h-7 text-emerald-600" />
   744→                </div>
   745→                <h2 className="text-xl font-bold text-white">Welcome Back</h2>
   746→                <p className="text-sm text-white/70 mt-1">Sign in to continue teaching</p>
   747→              </div>
   748→              <form onSubmit={handleLogin} className="px-6 pb-6 pt-4 space-y-4">
   749→                {/* Google OAuth */}
   750→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   751→                  <GoogleIcon className="w-4 h-4" />
   752→                  Continue with Google
   753→                </Button>
   754→                <div className="flex items-center gap-3">
   755→                  <div className="flex-1 h-px bg-gray-200" />
   756→                  <span className="text-xs text-gray-400 font-medium">or sign in with email</span>
   757→                  <div className="flex-1 h-px bg-gray-200" />
   758→                </div>
   759→                <div className="space-y-2">
   760→                  <Label htmlFor="login-email" className="text-sm font-medium">Email</Label>
   761→                  <Input id="login-email" type="email" placeholder="you@example.com" value={loginEmail} onChange={(e) => setLoginEmail(e.target.value)} className="rounded-xl" required />
   762→                </div>
   763→                <div className="space-y-2">
   764→                  <Label htmlFor="login-password" className="text-sm font-medium">Password</Label>
   765→                  <Input id="login-password" type="password" placeholder="Your password" value={loginPassword} onChange={(e) => setLoginPassword(e.target.value)} className="rounded-xl" required />
   766→                </div>
   767→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   768→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Sign In</Button>
   769→                <div className="text-center space-y-2">
   770→                  <button type="button" className="text-xs text-emerald-600 hover:underline" onClick={() => { setShowForgotPassword(true); setForgotEmail(loginEmail); setAuthError(''); }}>Forgot password?</button>
   771→                  <p className="text-sm text-muted-foreground">
   772→                    Don&apos;t have an account?{' '}
   773→                    <button type="button" className="text-emerald-600 font-medium hover:underline" onClick={() => setShowAuth('register')}>Sign up</button>
   774→                  </p>
   775→                </div>
   776→              </form>
   777→            </>
   778→          ) : showAuth === 'register' ? (
   779→            <>
   780→              <DialogTitle className="sr-only">Create Account</DialogTitle>
   781→              <div className="gradient-primary px-6 pt-8 pb-6 text-center">
   782→                <div className="w-14 h-14 rounded-2xl bg-white/90 flex items-center justify-center mx-auto mb-4">
   783→                  <Sparkles className="w-7 h-7 text-emerald-600" />
   784→                </div>
   785→                <h2 className="text-xl font-bold text-white">Create Your Account</h2>
   786→                <p className="text-sm text-white/70 mt-1">Start teaching in under a minute</p>
   787→              </div>
   788→              <form onSubmit={handleRegister} className="px-6 pb-6 pt-4 space-y-4">
   789→                {/* Google OAuth */}
   790→                <Button type="button" variant="outline" className="w-full rounded-xl font-medium text-sm h-11 gap-2 border-gray-200 hover:bg-gray-50" onClick={handleGoogleAuth}>
   791→                  <GoogleIcon className="w-4 h-4" />
   792→                  Sign up with Google
   793→                </Button>
   794→                <div className="flex items-center gap-3">
   795→                  <div className="flex-1 h-px bg-gray-200" />
   796→                  <span className="text-xs text-gray-400 font-medium">or register with email</span>
   797→                  <div className="flex-1 h-px bg-gray-200" />
   798→                </div>
   799→                <div className="space-y-2">
   800→                  <Label htmlFor="register-name" className="text-sm font-medium">Full Name</Label>
   801→                  <Input id="register-name" type="text" placeholder="Your name" value={registerName} onChange={(e) => setRegisterName(e.target.value)} className="rounded-xl" />
   802→                </div>
   803→                <div className="space-y-2">
   804→                  <Label htmlFor="register-email" className="text-sm font-medium">Email</Label>
   805→                  <Input id="register-email" type="email" placeholder="you@example.com" value={registerEmail} onChange={(e) => setRegisterEmail(e.target.value)} className="rounded-xl" required />
   806→                </div>
   807→                <div className="space-y-2">
   808→                  <Label htmlFor="register-password" className="text-sm font-medium">Password</Label>
   809→                  <Input id="register-password" type="password" placeholder="Create a strong password" value={registerPassword} onChange={(e) => { setRegisterPassword(e.target.value); setPasswordStrength(evaluatePasswordStrength(e.target.value)); }} className="rounded-xl" required />
   810→                  {registerPassword && (
   811→                    <div className="space-y-1">
   812→                      <div className="flex gap-1">
   813→                        {[1,2,3,4,5].map((i) => (
   814→                          <div key={i} className={`h-1 flex-1 rounded-full transition-colors ${i <= passwordStrength.score ? passwordStrength.color : 'bg-gray-200'}`} />
   815→                        ))}
   816→                      </div>
   817→                      <p className={`text-xs ${passwordStrength.score <= 1 ? 'text-red-500' : passwordStrength.score <= 3 ? 'text-amber-500' : 'text-emerald-600'}`}>
   818→                        {passwordStrength.label}
   819→                      </p>
   820→                    </div>
   821→                  )}
   822→                </div>
   823→                {authError && <p className="text-sm text-red-500 bg-red-50 rounded-lg px-3 py-2">{authError}</p>}
   824→                {authMessage && <p className="text-sm text-emerald-600 bg-emerald-50 rounded-lg px-3 py-2">{authMessage}</p>}
   825→                <Button type="submit" className="w-full rounded-xl gradient-primary border-0 text-white font-semibold shadow-lg shadow-emerald-500/25 hover:shadow-emerald-500/40 transition-all">Create Account</Button>
   826→                <p className="text-center text-sm text-muted-foreground">
   827→                  Already have an account?{' '}
   828→                  <button type="button" className="text-emerald-600 font-medium hover:underline" onClick={() => setShowAuth('login')}>Sign in</button>
   829→                </p>
   830→              </form>
   831→            </>
   832→          ) : null}
   833→        </DialogContent>
   834→      </Dialog>
   835→
   836→      {/* ===== EMAIL VERIFICATION SCREEN ===== */}
   837→      {showVerification && (
   838→        <div className="fixed inset-0 z-50 bg-white flex items-center justify-center px-6">
   839→          <div className="max-w-md w-full text-center space-y-6">
   840→            <div className="w-20 h-20 rounded-2xl gradient-primary flex items-center justify-center mx-auto shadow-lg shadow-emerald-500/25">
   841→              <Mail className="w-10 h-10 text-white" />
   842→            </div>
   843→            <h2 className="text-2xl font-bold text-gray-900">Check your email</h2>
   844→            <p className="text-gray-600 leading-relaxed">
   845→              We&apos;ve sent a confirmation link to <strong>{registerEmail}</strong>. 
   846→              Click the link to verify your account and start teaching.
   847→            </p>
   848→            <p className="text-sm text-gray-500">
   849→              Didn&apos;t receive the email? Check your spam folder.
   850→            </p>
   851→            <Button variant="outline" className="rounded-xl" onClick={() => setShowVerification(false)}>
   852→              Back to Sign In
   853→            </Button>
   854→          </div>
   855→        </div>
   856→      )}
   857→    </main>
   858→  );
   859→}
