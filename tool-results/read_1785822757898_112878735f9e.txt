    89→    89→export default function Dashboard() {
    90→    90→  const { tier, setTier, setRoom } = useAppStore();
    91→    91→  const { brandColor, agencyName, setBrandColor } = useTheme();
    92→    92→  const {
    93→    93→    aiCreditsUsed,
    94→    94→    aiCreditsLimit,
    95→    95→    videoMinutesUsed,
    96→    96→    videoMinutesLimit,
    97→    97→    loading: usageLoading,
    98→    98→  } = useCredits();
    99→    99→
   100→   100→  const [user, setUser] = useState<User | null>(null);
   101→   101→  const [authLoading, setAuthLoading] = useState(true);
   102→   102→
   103→   103→  // --- Auth check on mount ---
   104→   104→  useEffect(() => {
   105→   105→    const supabase = createClient();
   106→   106→    if (!supabase) { setAuthLoading(false); return; }
   107→   107→    supabase.auth.getUser().then(async ({ data: { user } }) => {
   108→   108→      setUser(user);
   109→   109→      if (user) {
   110→   110→        try {
   111→   111→          const tierRes = await fetch(`/api/auth/profile?userId=${user.id}`);
   112→   112→          if (tierRes.ok) {
   113→   113→            const profileData = await tierRes.json();
   114→   114→            if (profileData.tier) setTier(profileData.tier as Tier);
   115→   115→          }
   116→   116→        } catch { /* ignore */ }
   117→   117→      }
   118→   118→      setAuthLoading(false);
   119→   119→    });
   120→   120→    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
   121→   121→      setUser(session?.user ?? null);
   122→   122→    });
   123→   123→    return () => subscription.unsubscribe();
   124→   124→  }, []);
   125→   125→
   126→   126→  // --- Loading state ---
   127→   127→  if (authLoading) {
   128→   128→    return (
   129→   129→      <div className="min-h-screen gradient-auth flex items-center justify-center">
   130→   130→        <div className="flex flex-col items-center gap-4 animate-fade-in-up">
   131→   131→          <div className="w-16 h-16 rounded-2xl gradient-primary flex items-center justify-center shadow-lg shadow-violet-500/25">
   132→   132→            <GraduationCap className="w-9 h-9 text-white" />
   133→   133→          </div>
   134→   134→          <p className="text-sm text-muted-foreground font-medium">Loading Superboard...</p>
   135→   135→        </div>
   136→   136→      </div>
   137→   137→    );
   138→   138→  }
   139→   139→
   140→   140→  // --- Not logged in → Show Landing Page ---
   141→   141→  if (!user) {
   142→   142→    return <LandingPage />;
   143→   143→  }
   144→   144→
   145→   145→  // --- Logged in → Show Dashboard ---
   146→   146→  return <AuthenticatedDashboard user={user} />;
   147→   147→}
