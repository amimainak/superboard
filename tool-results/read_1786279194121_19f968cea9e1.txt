   348→
   349→        {/* Category Filters */}

   352→            onClick={() => setActiveCategory(null)}
   353→            className={`px-4 py-2 rounded-full text-sm font-medium transition-all ${
   354→              activeCategory === null
   355→                ? 'bg-gray-900 text-white shadow-md'
   356→                : 'bg-white text-gray-600 hover:bg-gray-100 border border-gray-200'
   357→            }`}
   358→          >
   359→            All Questions
   360→          </button>
   361→          {categories.map((cat) => (
   362→            <button
   363→              key={cat}
   364→              onClick={() =>
   365→                setActiveCategory(activeCategory === cat ? null : cat)
   366→              }
   367→              className={`px-4 py-2 rounded-full text-sm font-medium transition-all ${
   368→                activeCategory === cat
   369→                  ? 'bg-gray-900 text-white shadow-md'
   370→                  : 'bg-white text-gray-600 hover:bg-gray-100 border border-gray-200'
   371→              }`}
   372→            >

   375→          ))}

   377→
   378→        {/* FAQ Accordion */}

   380→          {filteredItems.map((item) => {
   381→            const IconComp = item.icon;
   382→            return (
   383→              <AccordionItem
   384→                key={item.id}
   385→                value={item.id}
   386→                className="rounded-2xl border border-gray-200 bg-white data-[state=open]:border-emerald-200 data-[state=open]:shadow-lg data-[state=open]:shadow-emerald-500/5 transition-all duration-300 overflow-hidden px-6"
   387→              >
   388→                <AccordionTrigger className="hover:no-underline py-5">
   389→                  <div className="flex items-center gap-4 text-left">
   390→                    <div className="flex-shrink-0 w-10 h-10 rounded-xl bg-gradient-to-br from-emerald-50 to-teal-50 flex items-center justify-center">
   391→                      <IconComp className="w-5 h-5 text-emerald-600" />
   392→                    </div>
   393→                    <div className="flex-1">
   394→                      <div className="flex items-center gap-2 mb-0.5">
   395→                        <span className="text-[11px] font-medium uppercase tracking-wider text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded-full">
   396→                          {item.category}
