     1→// ============================================================
     2→// TldrawCanvas — Tldraw Editor with Yjs Real-Time Sync
     3→// ============================================================
     4→// Wraps the Tldraw editor and synchronizes state with Yjs
     5→// via the Hocuspocus CRDT provider for multi-user collaboration.
     6→//
     7→// Sync strategy: Incremental per-record CRDT sync.
     8→// - Each record (shape, camera, page-state, etc.) is stored as a
     9→//   separate entry in a per-page Y.Map, keyed by record ID.
    10→// - Local user edits are debounced (200ms) and batch-written to Yjs.
    11→// - Remote Yjs changes are merged record-by-record into the editor
    12→//   using store.mergeRemoteChanges(), which marks them source='remote'
    13→//   so they don't trigger the local-sync listener (no echo loop).
    14→// - A dirty-set tracks record IDs with unsynced local changes;
    15→//   remote updates for dirty records are skipped (last-local-writer-wins).
    16→// - A monotonic localChangeVersion counter is maintained per record
    17→//   for diagnostics and future conflict-resolution extensions.
    18→// - Backward compatibility: legacy full-snapshot format is detected
    19→//   on first load and migrated to the per-record format.
    20→// ============================================================
    21→
    22→'use client';
    23→
    24→import React, { useCallback, useEffect, useRef, useState } from 'react';
    25→import { Tldraw, Editor, TLStoreSnapshot } from 'tldraw';
    26→import type { Map as YMap } from 'yjs';
    27→import 'tldraw/tldraw.css';
    28→
    29→// ---- Types ----
    30→
    31→/** Shape of the RecordsDiff changes as delivered by store.listen in tldraw v5. */
    32→interface RecordsDiffLike {
    33→        added: Record<string, Record<string, unknown>>;
    34→        updated: Record<string, [from: Record<string, unknown>, to: Record<string, unknown>]>;
    35→        removed: Record<string, Record<string, unknown>>;
    36→}
    37→
    38→/** Shape of the history entry passed to store.listen callbacks in tldraw v5. */
    39→interface HistoryEntryLike {
    40→        changes: RecordsDiffLike;
    41→        source: 'user' | 'remote';
    42→}
    43→
    44→export interface TldrawCanvasProps {
    45→        /** Yjs document (from useYjsProvider) */
    46→        ydoc: import('yjs').Doc | null;
    47→        /** Called when the editor mounts — provides the Editor ref */
    48→        onEditorReady?: (editor: Editor) => void;
    49→        /** Called when editor store changes (for external sync triggers) */
    50→        onStoreChange?: () => void;
    51→        /** Current page index (for multi-page support) */
    52→        pageIndex: number;
    53→        /** Whether the user is a tutor (controls UI permissions) */
    54→        isTutor: boolean;
    55→        /** Whether the canvas is read-only (e.g., student in focus mode) */
    56→        readOnly?: boolean;
    57→}
    58→
    59→// ---- Constants ----
    60→
    61→/** Debounce Yjs writes to batch rapid edits into one transaction. */
    62→const SYNC_DEBOUNCE_MS = 200;
    63→
    64→/** Key of the legacy Y.Map that stored full page snapshots as JSON blobs. */
    65→const YJS_LEGACY_MAP_KEY = 'page-snapshot';
    66→
    67→/**
    68→ * Prefix for per-page Y.Maps that store individual records.
    69→ * E.g. pageIndex=3 → ydoc.getMap('page-shapes-3')
    70→ */
    71→const YJS_SHAPES_MAP_PREFIX = 'page-shapes-';
    72→
    73→/**
    74→ * Transaction origin symbol used to tag our own local writes.
    75→ * The Yjs observer checks this to skip re-applying our own changes.
    76→ */
    77→const LOCAL_SYNC_ORIGIN = Symbol('tldraw-local-sync');
    78→
    79→// ============================================================
    80→// Component
    81→// ============================================================
    82→
    83→export default function TldrawCanvas({
    84→        ydoc,
    85→        onEditorReady,
    86→        onStoreChange,
    87→        pageIndex,
    88→        isTutor,
    89→        readOnly = false,
    90→}: TldrawCanvasProps) {
    91→        // ---- Core state & refs ----
    92→        const editorRef = useRef<Editor | null>(null);
    93→        const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    94→        const [isReady, setIsReady] = useState(false);
    95→        const [isLoadingSnapshot, setIsLoadingSnapshot] = useState(true);
    96→
    97→        /**
    98→         * Mirror of isReady as a ref so that closures inside setTimeout/
    99→         * store listeners always see the current value without being
   100→         * captured at creation time.
   101→         */
   102→        const isReadyRef = useRef(false);
   103→
   104→        // ---- Incremental sync refs (stable across renders) ----
   105→
   106→        /** Records that changed locally and are awaiting debounce flush to Yjs. */
   107→        const pendingWritesRef = useRef<Map<string, Record<string, unknown>>>(new Map());
   108→
   109→        /** Record IDs that were deleted locally and are awaiting debounce flush. */
   110→        const pendingDeletesRef = useRef<Set<string>>(new Set());
   111→
   112→        /**
   113→         * Record IDs with unsynced local changes.
   114→         * Remote updates for these IDs are skipped to prevent overwriting
   115→         * concurrent local edits (last-local-writer-wins).  IDs are removed
   116→         * after the corresponding write/delete is flushed to Yjs.
   117→         */
   118→        const localDirtyRef = useRef<Set<string>>(new Set());
   119→
   120→        /** Monotonically increasing version counter for local edits. */
   121→        const localChangeVersionRef = useRef(0);
   122→
   123→        /** Per-record version: the localChangeVersion at which it was last edited locally. */
   124→        const recordVersionsRef = useRef<Map<string, number>>(new Map());
   125→
   126→        /** True while the initial snapshot is being loaded (suppresses sync-to-Yjs). */
   127→        const isInitialLoadRef = useRef(false);
   128→
   129→        /** Set of page indices that have already been migrated from legacy format. */
   130→        const migratedPagesRef = useRef<Set<number>>(new Set());
   131→
   132→        /** Cleanup function for the tldraw store listener. */
   133→        const unlistenStoreRef = useRef<(() => void) | null>(null);
   134→
   135→        /** Cleanup function for the Yjs shapes-map observer. */
   136→        const unobserveYjsRef = useRef<(() => void) | null>(null);
   137→
   138→        /**
   139→         * Ref to the Y.Map currently being used for this page's shapes.
   140→         * Updated explicitly on page switch so we can flush the OLD map
   141→         * before swapping to the new one.
   142→         */
   143→        const shapesMapRef = useRef<YMap<string> | null>(null);
   144→
   145→        /** Track the previous page index to detect page switches. */
   146→        const prevPageIndexRef = useRef(pageIndex);
   147→
   148→        /** Keep callback refs up-to-date without re-registering listeners. */
   149→        const onStoreChangeRef = useRef(onStoreChange);
   150→        onStoreChangeRef.current = onStoreChange;
   151→
   152→        // ----------------------------------------------------------------
   153→        // Core: process a RecordsDiff into pending writes/deletes & schedule sync
   154→        // ----------------------------------------------------------------
   155→
   156→        const processLocalDiff = useCallback((entry: HistoryEntryLike) => {
   157→                if (isInitialLoadRef.current) return;
   158→                if (!isReadyRef.current) return;
   159→
   160→                const { changes } = entry;
   161→
   162→                // Increment the global local version counter.
   163→                localChangeVersionRef.current++;
   164→                const version = localChangeVersionRef.current;
   165→
   166→                // Collect added records.
   167→                for (const [id, record] of Object.entries(changes.added)) {
   168→                        pendingWritesRef.current.set(id, record);
   169→                        pendingDeletesRef.current.delete(id);
   170→                        localDirtyRef.current.add(id);
   171→                        recordVersionsRef.current.set(id, version);
   172→                }
   173→
   174→                // Collect updated records (use the "to" value — index 1).
   175→                for (const [id, pair] of Object.entries(changes.updated)) {
   176→                        pendingWritesRef.current.set(id, pair[1]);
   177→                        pendingDeletesRef.current.delete(id);
   178→                        localDirtyRef.current.add(id);
   179→                        recordVersionsRef.current.set(id, version);
   180→                }
   181→
   182→                // Collect removed records.
   183→                for (const id of Object.keys(changes.removed)) {
   184→                        pendingDeletesRef.current.add(id);
   185→                        pendingWritesRef.current.delete(id);
   186→                        localDirtyRef.current.add(id);
   187→                        recordVersionsRef.current.set(id, version);
   188→                }
   189→
   190→                scheduleSyncRef.current();
   191→        }, []);
   192→
   193→        // ---- Ref-stable scheduleSync (never recreated) ----
   194→        const scheduleSyncRef = useRef<() => void>(() => {});
   195→
   196→        // ---- Ref-stable flushPendingToYjs (never recreated) ----
   197→        const flushPendingRef = useRef<() => void>(() => {});
   198→
   199→        // ----------------------------------------------------------------
   200→        // Flush pending changes to the current shapes Y.Map
   201→        // ----------------------------------------------------------------
   202→
   203→        const buildFlush = useCallback(() => {
   204→                return () => {
   205→                        const map = shapesMapRef.current;
   206→                        if (!map || !ydoc) return;
   207→
   208→                        // Nothing to flush?
   209→                        if (pendingWritesRef.current.size === 0 && pendingDeletesRef.current.size === 0) return;
   210→
   211→                        // Snapshot the pending state so new changes accumulate independently.
   212→                        const writes = new Map(pendingWritesRef.current);
   213→                        const deletes = new Set(pendingDeletesRef.current);
   214→                        pendingWritesRef.current = new Map();
   215→                        pendingDeletesRef.current = new Set();
   216→
   217→                        // Check total serialized size (5 MB limit matches DB constraint).
   218→                        let totalSize = 0;
   219→                        for (const record of writes.values()) {
   220→                                totalSize += JSON.stringify(record).length;
   221→                        }
   222→                        if (totalSize > 5_000_000) {
   223→                                console.warn(
   224→                                        `[TldrawCanvas] Pending sync too large (${(totalSize / 1_000_000).toFixed(1)}MB) — skipping`
   225→                                );
   226→                                return;
   227→                        }
   228→
   229→                        // Batch all writes/deletes into a single Yjs transaction tagged
   230→                        // with our origin so the observer can skip re-applying them.
   231→                        ydoc.transact(() => {
   232→                                for (const [id, record] of writes) {
   233→                                        map.set(id, JSON.stringify(record));
   234→                                }
   235→                                for (const id of deletes) {
   236→                                        map.delete(id);
   237→                                }
   238→                        }, LOCAL_SYNC_ORIGIN);
   239→
   240→                        // Clear dirty flags for the records we just synced.
   241→                        for (const id of writes.keys()) {
   242→                                localDirtyRef.current.delete(id);
   243→                        }
   244→                        for (const id of deletes) {
   245→                                localDirtyRef.current.delete(id);
   246→                        }
   247→
   248→                        onStoreChangeRef.current?.();
   249→                };
   250→        }, [ydoc]);
   251→
   252→        // Keep flushPendingRef up-to-date.
   253→        flushPendingRef.current = buildFlush();
   254→
   255→        // ----------------------------------------------------------------
   256→        // Schedule a debounced flush
   257→        // ----------------------------------------------------------------
   258→
   259→        const buildScheduleSync = useCallback(() => {
   260→                return () => {
   261→                        if (debounceTimerRef.current) {
   262→                                clearTimeout(debounceTimerRef.current);
   263→                        }
   264→                        debounceTimerRef.current = setTimeout(() => {
   265→                                flushPendingRef.current();
   266→                        }, SYNC_DEBOUNCE_MS);
   267→                };
   268→        }, []);
   269→
   270→        // Keep scheduleSyncRef up-to-date.
   271→        scheduleSyncRef.current = buildScheduleSync();
   272→
   273→        // ----------------------------------------------------------------
   274→        // Load initial state from Yjs into the editor
   275→        // ----------------------------------------------------------------
   276→
   277→        const loadInitialState = useCallback(() => {
   278→                const editor = editorRef.current;
   279→                if (!editor || !ydoc) return;
   280→
   281→                isInitialLoadRef.current = true;
   282→
   283→                const shapesMap = ydoc.getMap<string>(`${YJS_SHAPES_MAP_PREFIX}${pageIndex}`);
   284→                shapesMapRef.current = shapesMap;
   285→
   286→                // --- Try legacy full-snapshot format first ---
   287→                const legacyMap = ydoc.getMap<string>(YJS_LEGACY_MAP_KEY);
   288→                const legacyKey = `page-${pageIndex}`;
   289→                const legacyJson = legacyMap.get(legacyKey);
   290→
   291→                if (legacyJson && !migratedPagesRef.current.has(pageIndex)) {
   292→                        try {
   293→                                const storeSnapshot = JSON.parse(legacyJson) as TLStoreSnapshot;
   294→
   295→                                // editor.loadSnapshot accepts Partial<TLEditorSnapshot> | TLStoreSnapshot
   296→                                editor.loadSnapshot({ document: storeSnapshot });
   297→                                console.log(`[TldrawCanvas] Loaded legacy snapshot for page ${pageIndex}`);
   298→
   299→                                // Migrate the legacy snapshot into per-record Y.Map entries.
   300→                                // TLStoreSnapshot has { store: Record<string, TLRecord>, schema: ... }
   301→                                const records = (storeSnapshot as any).store;
   302→                                if (records) {
   303→                                        ydoc.transact(() => {
   304→                                                for (const [id, record] of Object.entries(records)) {
   305→                                                        shapesMap.set(id, JSON.stringify(record));
   306→                                                }
   307→                                        }, LOCAL_SYNC_ORIGIN);
   308→                                        console.log(
   309→                                                `[TldrawCanvas] Migrated ${Object.keys(records).length} records to per-record format for page ${pageIndex}`
   310→                                        );
   311→                                }
   312→
   313→                                migratedPagesRef.current.add(pageIndex);
   314→                        } catch (err) {
   315→                                console.error(
   316→                                        `[TldrawCanvas] Failed to load/parse legacy snapshot for page ${pageIndex}:`,
   317→                                        err
   318→                                );
   319→                        }
   320→                } else {
   321→                        // --- Load from per-record format ---
   322→                        const recordsToLoad: Record<string, unknown>[] = [];
   323→                        shapesMap.forEach((json, id) => {
   324→                                try {
   325→                                        recordsToLoad.push(JSON.parse(json));
   326→                                } catch (err) {
   327→                                        console.error(`[TldrawCanvas] Failed to parse record ${id}:`, err);
   328→                                }
   329→                        });
   330→
   331→                        if (recordsToLoad.length > 0) {
   332→                                // Use mergeRemoteChanges so these are tagged source='remote'
   333→                                // and won't trigger our local-sync listener.
   334→                                editor.store.mergeRemoteChanges(() => {
   335→                                        editor.store.put(recordsToLoad as any);
   336→                                });
   337→                                console.log(
   338→                                        `[TldrawCanvas] Loaded ${recordsToLoad.length} records from per-record format for page ${pageIndex}`
   339→                                );
   340→                        } else if (!legacyJson) {
   341→                                console.log(`[TldrawCanvas] No saved data for page ${pageIndex} — starting fresh`);
   342→                        }
   343→                }
   344→
   345→                isInitialLoadRef.current = false;
   346→                setIsLoadingSnapshot(false);
   347→        }, [ydoc, pageIndex]);
   348→
   349→        // ----------------------------------------------------------------
   350→        // Set up the Yjs remote-change observer for the shapes map
   351→        // ----------------------------------------------------------------
   352→
   353→        const setupRemoteObserver = useCallback(() => {
   354→                const shapesMap = shapesMapRef.current;
   355→                if (!shapesMap) return () => {};
   356→
   357→                const observer = (event: any, transaction: any) => {
   358→                        // Skip our own local writes (tagged with LOCAL_SYNC_ORIGIN).
   359→                        if (transaction.origin === LOCAL_SYNC_ORIGIN) return;
   360→
   361→                        const editor = editorRef.current;
   362→                        if (!editor) return;
   363→                        if (isInitialLoadRef.current) return;
   364→
   365→                        const toPut: Record<string, unknown>[] = [];
   366→                        const toRemove: string[] = [];
   367→
   368→                        // Determine which keys changed.
   369→                        // YMapEvent extends YEvent which has `changes.keys` (Map<string, {action, oldValue}>).
   370→                        // It also has `keysChanged` (Set<string>) which is a simpler set of changed keys.
   371→                        const changedKeys: Set<string> | undefined = event.keysChanged;
   372→                        if (!changedKeys) return;
   373→
   374→                        for (const id of changedKeys) {
   375→                                // Skip records with pending local changes — our local edit wins.
   376→                                if (localDirtyRef.current.has(id)) {
   377→                                        continue;
   378→                                }
   379→
   380→                                // Check if the key still exists in the map (add/update) or was deleted.
   381→                                if (shapesMap.has(id)) {
   382→                                        try {
   383→                                                toPut.push(JSON.parse(shapesMap.get(id)!));
   384→                                        } catch (err) {
   385→                                                console.error(`[TldrawCanvas] Failed to parse remote record ${id}:`, err);
   386→                                        }
   387→                                } else {
   388→                                        toRemove.push(id);
   389→                                }
   390→                        }
   391→
   392→                        if (toPut.length > 0 || toRemove.length > 0) {
   393→                                editor.store.mergeRemoteChanges(() => {
   394→                                        if (toPut.length > 0) {
   395→                                                editor.store.put(toPut as any);
   396→                                        }
   397→                                        if (toRemove.length > 0) {
   398→                                                editor.store.remove(toRemove as any);
   399→                                        }
   400→                                });
   401→                                console.log(
   402→                                        `[TldrawCanvas] Applied remote changes for page ${pageIndex}: ${toPut.length} puts, ${toRemove.length} removes`
   403→                                );
   404→                        }
   405→                };
   406→
   407→                shapesMap.observe(observer);
   408→                return () => {
   409→                        shapesMap.unobserve(observer);
   410→                };
   411→        }, [pageIndex]);
   412→
   413→        // ----------------------------------------------------------------
   414→        // Register the store listener with { source: 'user', scope: 'document' }
   415→        // so it only fires for local user edits (not for remote-applied changes).
   416→        // ----------------------------------------------------------------
   417→
   418→        const registerStoreListener = useCallback(() => {
   419→                const editor = editorRef.current;
   420→                if (!editor) return;
   421→
   422→                // Clean up previous listener if any.
   423→                unlistenStoreRef.current?.();
   424→
   425→                unlistenStoreRef.current = (editor.store as any).listen(
   426→                        (entry: HistoryEntryLike) => {
   427→                                processLocalDiff(entry);
   428→                        },
   429→                        { source: 'user', scope: 'document' } as any
   430→                );
   431→        }, [processLocalDiff]);
   432→
   433→        // ----------------------------------------------------------------
   434→        // Handle editor mount (called once by Tldraw)
   435→        // ----------------------------------------------------------------
   436→
   437→        const handleMount = useCallback(
   438→                (editor: Editor) => {
   439→                        editorRef.current = editor;
   440→                        prevPageIndexRef.current = pageIndex;
   441→
   442→                        // Notify parent with the editor ref.
   443→                        onEditorReady?.(editor);
   444→
   445→                        // Short delay lets the editor finish its internal initialization
   446→                        // before we start loading external data.
   447→                        setTimeout(() => {
   448→                                // 1. Load initial state (legacy snapshot or per-record format).
   449→                                loadInitialState();
   450→
   451→                                // 2. Register the store listener for user-initiated changes.
   452→                                registerStoreListener();
   453→
   454→                                // 3. Set up the Yjs remote observer.
   455→                                unobserveYjsRef.current = setupRemoteObserver();
   456→
   457→                                // 4. Mark as ready — enables the store listener's isReadyRef check.
   458→                                isReadyRef.current = true;
   459→                                setIsReady(true);
   460→                        }, 100);
   461→                },
   462→                [onEditorReady, pageIndex, loadInitialState, registerStoreListener, setupRemoteObserver]
   463→        );
   464→
   465→        // ----------------------------------------------------------------
   466→        // Handle page changes: flush old page, load new page
   467→        // ----------------------------------------------------------------
   468→
   469→        useEffect(() => {
   470→                if (!editorRef.current || !ydoc) return;
   471→                if (prevPageIndexRef.current === pageIndex) return;
   472→
   473→                // 1. Cancel any pending debounce.
   474→                if (debounceTimerRef.current) {
   475→                        clearTimeout(debounceTimerRef.current);
   476→                        debounceTimerRef.current = null;
   477→                }
   478→
   479→                // 2. Mark as not-ready to suppress store listener during transition.
   480→                isReadyRef.current = false;
   481→
   482→                // 3. Flush pending changes for the OLD page (shapesMapRef still points to old map).
   483→                flushPendingRef.current();
   484→
   485→                // 4. Clean up old Yjs observer.
   486→                unobserveYjsRef.current?.();
   487→                unobserveYjsRef.current = null;
   488→
   489→                // 5. Reset all pending/dirty state for the new page.
   490→                pendingWritesRef.current = new Map();
   491→                pendingDeletesRef.current = new Set();
   492→                localDirtyRef.current = new Set();
   493→
   494→                // 6. Update page tracking.
   495→                prevPageIndexRef.current = pageIndex;
   496→
   497→                // 7. Load the new page's state.
   498→                setIsLoadingSnapshot(true);
   499→                loadInitialState();
   500→
   501→                // 8. Set up new Yjs observer for the new page's map.
   502→                unobserveYjsRef.current = setupRemoteObserver();
   503→
   504→                // 9. Re-enable the store listener.
   505→                //    (registerStoreListener is idempotent — it cleans up the old one first.)
   506→                registerStoreListener();
   507→
   508→                // 10. Mark as ready again.
   509→                isReadyRef.current = true;
   510→                setIsReady(true);
   511→        }, [pageIndex, ydoc, loadInitialState, setupRemoteObserver, registerStoreListener]);
   512→
   513→        // ----------------------------------------------------------------
   514→        // Cleanup on unmount
   515→        // ----------------------------------------------------------------
   516→
   517→        useEffect(() => {
   518→                return () => {
   519→                        // Cancel pending debounce.
   520→                        if (debounceTimerRef.current) {
   521→                                clearTimeout(debounceTimerRef.current);
   522→                                debounceTimerRef.current = null;
   523→                        }
   524→
   525→                        // Flush any remaining pending changes immediately.
   526→                        flushPendingRef.current();
   527→
   528→                        // Clean up listeners.
   529→                        unlistenStoreRef.current?.();
   530→                        unobserveYjsRef.current?.();
   531→
   532→                        editorRef.current = null;
   533→                };
   534→        }, []); // eslint-disable-line react-hooks/exhaustive-deps
   535→
   536→        // ----------------------------------------------------------------
   537→        // Render
   538→        // ----------------------------------------------------------------
   539→
   540→        return (
   541→                <div className="w-full h-full relative">
   542→                        {/* Loading overlay while snapshot is being loaded */}
   543→                        {isLoadingSnapshot && (
   544→                                <div className="absolute inset-0 z-10 flex items-center justify-center bg-background/80 backdrop-blur-sm transition-opacity">
   545→                                        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   546→                                                <div className="w-4 h-4 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   547→                                                <span>Loading canvas...</span>
   548→                                        </div>
   549→                                </div>
   550→                        )}
   551→
   552→                        {/* Tldraw Editor */}
   553→                        <Tldraw
   554→                                onMount={handleMount}
   555→                                options={{
   556→                                        maxPages: 50,
   557→                                        maxShapesPerPage: 500,
   558→                                }}
   559→                        />
   560→                </div>
   561→        );
   562→}
   563→
