    20→    69→// ---- Perfect-Freehand → Fabric.js Path ----
    21→    70→
    22→    71→function createFreehandPath(
    23→    72→  points: FreehandPoint[],
    24→    73→  strokeColor: string,
    25→    74→  strokeWidth: number
    26→    75→): Path | null {
    27→    76→  if (points.length < 2) return null;
    28→    77→
    29→    78→  const outline = getStroke(points, {
    30→    79→    size: strokeWidth * 3,
    31→    80→    thinning: 0.5,
    32→    81→    smoothing: 0.6,
    33→    82→    streamline: 0.5,
    34→    83→    simulatePressure: true,
    35→    84→  });
    36→    85→
    37→    86→  if (outline.length < 2) return null;
    38→    87→
    39→    88→  // Build SVG path string from the outline points
    40→    89→  let d = `M ${outline[0][0].toFixed(1)} ${outline[0][1].toFixed(1)}`;
    41→    90→  for (let i = 1; i < outline.length; i++) {
    42→    91→    d += ` L ${outline[i][0].toFixed(1)} ${outline[i][1].toFixed(1)}`;
    43→    92→  }
    44→    93→  d += ' Z';
    45→    94→
    46→    95→  try {
    47→    96→    const path = new Path(d, {
    48→    97→      fill: strokeColor,
    49→    98→      stroke: strokeColor,
    50→    99→      strokeWidth: 0.5,
    51→   100→      strokeLineCap: 'round',
    52→   101→      strokeLineJoin: 'round',
    53→   102→    } as any);
    54→   103→    return path;
    55→   104→  } catch {
    56→   105→    return null;
    57→   106→  }
    58→   107→}

   490→   539→            }

   492→   541→            default:
   493→   542→              return;
   494→   543→          }
   495→   544→
   496→   545→          (shape as any).name = objName;
   497→   546→          fc.add(shape);
   498→   547→          tempShapeRef.current = shape;
   499→   548→          requestRenderThrottled();
   500→   549→          return;
   501→   550→        }
   502→   551→      };
   503→   552→
   504→   553→      // ---- Mouse Move ----
   505→   554→      const handleMouseMove = (opt: any) => {
   506→   555→        const e = opt.e as MouseEvent;
   507→   556→
   508→   557→        // Panning

   511→   560→          vpt[4] += e.clientX - lastPosX;
   512→   561→          vpt[5] += e.clientY - lastPosY;

   518→   567→

   565→   614→            const top = Math.min(start.y, pointer.y);

   570→   619→            t.set({ x2: pointer.x, y2: pointer.y });

   572→   621→            const objs = t.getObjects();

   574→   623→            const head = objs[1] as Triangle;

   578→   627→                left: pointer.x - start.x,
   579→   628→                top: pointer.y - start.y,
   580→   629→                angle:
   581→   630→                  Math.atan2(pointer.y - start.y, pointer.x - start.x) *
   582→   631→                    (180 / Math.PI) +
   583→   632→                  90,
   584→   633→              });
   585→   634→            }
   586→   635→            t.setCoords();
   587→   636→          }
   588→   637→
   589→   638→          requestRenderThrottled();
   590→   639→        }
   591→   640→      };
   592→   641→
   593→   642→      // ---- Mouse Up ----

   601→   650→          return;
   602→   651→        }
   603→   652→
   604→   653→        // Finalize freehand
   605→   654→        if (isDrawingRef.current && tools.tool === 'draw') {

   607→   656→

   609→   658→          if (freehandTempPathRef.current) {

   615→   664→          const points = freehandPointsRef.current;
   616→   665→          if (points.length >= 2) {

   620→   669→              tools.strokeWidth

   624→   673→              fc.add(finalPath);
   625→   674→            }
   626→   675→          }
   627→   676→
   628→   677→          freehandPointsRef.current = [];
   629→   678→          history.pushState('draw', beforeState, fc.toJSON().toString());
   630→   679→          fc.renderAll();
   631→   680→          return;
   632→   681→        }
   633→   682→
   634→   683→        // Finalize shape

   638→   687→            evented: true,
   639→   688→          });

   642→   691→            `draw-${tools.tool}`,
   643→   692→            beforeState,
   644→   693→            fc.toJSON().toString()
   645→   694→          );
   646→   695→          tempShapeRef.current = null;
   647→   696→          isDrawingRef.current = false;
   648→   697→          startPointRef.current = null;
   649→   698→          fc.renderAll();
   650→   699→        }
   651→   700→      };
   652→   701→
   653→   702→      fc.on('mouse:down', handleMouseDown);
   654→   703→      fc.on('mouse:move', handleMouseMove);
   655→   704→      fc.on('mouse:up', handleMouseUp);

   684→   733→        }

   686→   735→

   689→   738→        fc.off('mouse:down', handleMouseDown);
   690→   739→        fc.off('mouse:move', handleMouseMove);
   691→   740→        fc.off('mouse:up', handleMouseUp);
   692→   741→      };

   851→   900→        vpt[5] = vptCenterY - centerY * zoom;
