    91→export default function Dashboard() {
    92→  const { tier, setTier, setRoom } = useAppStore();
    93→  const { brandColor, agencyName, setBrandColor } = useTheme();
    94→  const {
    95→    aiCreditsUsed,
    96→    aiCreditsLimit,
    97→    videoMinutesUsed,
    98→    videoMinutesLimit,
    99→    loading: usageLoading,
   100→  } = useCredits(null);
   101→
   102→  const [user, setUser] = useState<User | null>(null);
   103→  const [userName, setUserName] = useState<string | null>(null);
   104→  const [authLoading, setAuthLoading] = useState(true);
   105→  const [tierLoading, setTierLoading] = useState(true);
   106→
   107→  // --- Auth check on mount ---
   108→  useEffect(() => {
   109→    const supabase = createClient();
   110→    if (!supabase) { setAuthLoading(false); return; }
   111→
   112→    // Initialize auth fetch token caching
   113→    initAuthFetch();
   114→
   115→    let mounted = true;
   116→
   117→    // Listen for auth state changes FIRST to catch any race conditions
   118→    const { data: { subscription } } = supabase.auth.onAuthStateChange(async (_event: any, session: any) => {
   119→      if (!mounted) return;
   120→      const user = session?.user ?? null;
   121→      setUser(user);
   122→      if (user) {
   123→        try {
   124→          const tierRes = await authFetch(`/api/auth/profile?userId=${user.id}`);
   125→          if (tierRes.ok) {
   126→            const profileData = await tierRes.json();
   127→            if (profileData.tier) setTier(profileData.tier as Tier);
   128→            if (profileData.name) setUserName(profileData.name);
   129→          }
   130→        } catch { /* ignore */ }
   131→      }
   132→      setAuthLoading(false);
   133→      setTierLoading(false);
   134→    });
   135→
   136→    // Then check for existing session
   137→    supabase.auth.getUser().then(async ({ data: { user } }: any) => {
   138→      if (!mounted) return;
   139→      if (user) {
   140→        setUser(user);
   141→        try {
   142→          const tierRes = await authFetch(`/api/auth/profile?userId=${user.id}`);
   143→          if (tierRes.ok) {
   144→            const profileData = await tierRes.json();
   145→            if (profileData.tier) setTier(profileData.tier as Tier);
   146→            if (profileData.name) setUserName(profileData.name);
   147→          }
   148→        } catch { /* ignore */ }
   149→      }
   150→      setAuthLoading(false);
   151→      setTierLoading(false);
   152→    });
   153→
   154→    return () => {
   155→      mounted = false;
   156→      subscription.unsubscribe();
   157→    };
   158→  }, []);
   159→
   160→  // --- Loading state ---
   161→  if (authLoading) {
   162→    return (
   163→      <div className="min-h-screen bg-white flex items-center justify-center">
   164→        <div className="flex flex-col items-center gap-4 animate-fade-in-up">
   165→          <div className="w-16 h-16 rounded-2xl gradient-primary flex items-center justify-center shadow-lg shadow-emerald-500/25 animate-pulse-glow">
   166→            <GraduationCap className="w-9 h-9 text-white" />
   167→          </div>
   168→          <p className="text-sm text-gray-500 font-medium">Loading Superboard...</p>
   169→        </div>
   170→      </div>
   171→    );
   172→  }
   173→
   174→  // --- Not logged in → Show Landing Page ---
   175→  if (!user) {
   176→    return <LandingPage />;
   177→  }
   178→
   179→  // --- Logged in → Show Dashboard ---
   180→  return <AuthenticatedDashboard user={user} userName={userName} tierLoading={tierLoading} />;
   181→}
